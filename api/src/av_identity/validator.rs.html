9/// Verification key material, bound to a `kid`.
10#[derive(Clone)]
11pub enum KeyMaterial {
12    /// Ed25519 public key, SPKI PEM (`-----BEGIN PUBLIC KEY-----`). EdDSA.
13    Ed25519Pem(String),
14    /// Ed25519 JWK `x` coordinate, base64url without padding.
15    Ed25519Jwk(String),
16    /// HMAC shared secret. HS256 (dev / shared-secret IdP integrations).
17    HmacSecret(Vec<u8>),
18}
19
20impl std::fmt::Debug for KeyMaterial {
21    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
22        match self {
23            Self::Ed25519Pem(_) => f.write_str("KeyMaterial::Ed25519Pem(..)"),
24            Self::Ed25519Jwk(_) => f.write_str("KeyMaterial::Ed25519Jwk(..)"),
25            Self::HmacSecret(_) => f.write_str("KeyMaterial::HmacSecret(..)"), // never print secrets
26        }
27    }
28}

30/// Identity validation failures.
31#[derive(Debug, thiserror::Error, PartialEq, Eq)]
32#[non_exhaustive]
33pub enum IdentityError {
34    /// Token structurally malformed.
35    #[error("malformed token: {0}")]
36    Malformed(String),
37    /// Header lacks a `kid`.
38    #[error("token header has no kid")]
39    MissingKid,
40    /// No key registered for this `kid`.
41    #[error("unknown kid {0:?}")]
42    UnknownKid(String),
43    /// Token `alg` is not accepted, or does not match the key type for its
44    /// `kid` (algorithm-confusion defense).
45    #[error("algorithm {alg:?} not permitted for kid {kid:?}")]
46    AlgorithmRejected {
47        /// Stated algorithm.
48        alg: String,
49        /// Key id.
50        kid: String,
51    },
52    /// Signature invalid / expired / nbf future / wrong audience — collapsed
53    /// by jsonwebtoken; the detail string preserves the cause.
54    #[error("verification failed: {0}")]
55    Verification(String),
56    /// `exp - iat` exceeds the 15-minute NHI cap.
57    #[error("ttl {0}s exceeds the {MAX_TTL_SECS}s NHI cap")]
58    TtlTooLong(u64),
59    /// `exp ≤ iat` or other timestamp nonsense.
60    #[error("inconsistent timestamps (iat {iat}, exp {exp})")]
61    BadTimestamps {
62        /// Issued-at.
63        iat: u64,
64        /// Expiry.
65        exp: u64,
66    },
67    /// `iat` is unreasonably far in the future.
68    #[error("issued-at timestamp {iat} is in the future (now {now})")]
69    FutureIat {
70        /// Issued-at claim.
71        iat: u64,
72        /// Validator wall clock.
73        now: u64,
74    },
75    /// Required identity field empty.
76    #[error("empty identity field {0}")]
77    EmptyField(&'static str),
78    /// Identity field longer than its documented cap.
79    #[error("identity field {field} exceeds {max} characters")]
80    FieldTooLong {
81        /// Field name.
82        field: &'static str,
83        /// Documented cap in Unicode code points.
84        max: usize,
85    },
86    /// Field carries a bidi override or zero-width character that would
87    /// spoof how the identity renders in a log or audit view.
88    #[error("identity field {0} carries a bidi/zero-width spoofing character")]
89    SpoofingCharacter(&'static str),
90    /// Issuer not in the allowlist.
91    #[error("issuer {0:?} not allowed")]
92    BadIssuer(String),
93    /// JWKS document was malformed or contained no supported signing keys.
94    #[error("invalid JWKS: {0}")]
95    Jwks(String),
96    /// Child scopes exceed parent scopes.
97    #[error("scope escalation: {scope:?} not granted by parent")]
98    ScopeEscalation {
99        /// The offending scope.
100        scope: String,
101    },
102    /// Child outlives parent.
103    #[error("child exp {child} outlives parent exp {parent}")]
104    ExpEscalation {
105        /// Child expiry.
106        child: u64,
107        /// Parent expiry.
108        parent: u64,
109    },
110    /// Child claims to be issued before its parent (the delegator).
111    /// A backdated child forges audit-trail causality: consumers that
112    /// treat `claims.iat` as "when this identity became authorized"
113    /// see the child asserting authority before the parent that
114    /// granted it existed.
115    #[error("child iat {child} predates parent iat {parent}")]
116    IatEscalation {
117        /// Child issued-at.
118        child: u64,
119        /// Parent issued-at.
120        parent: u64,
121    },
122    /// Child's effective start (`nbf` when set, else `iat`) is
123    /// earlier than the parent's effective start (`nbf` when set,
124    /// else `iat`) — same forgery class as `IatEscalation`. The
125    /// carried `child` and `parent` values are the compared
126    /// EFFECTIVE starts, not the raw `nbf` fields (which may be
127    /// absent). See `IdentityValidator::validate` for the full
128    /// case matrix.
129    #[error("child effective-start {child} predates parent effective-start {parent}")]
130    NbfEscalation {
131        /// Child effective start: `child.nbf` when set, else
132        /// `child.iat`. The fallback closes an input case where the
133        /// child omits `nbf` but its `iat` predates the parent's
134        /// explicit `nbf` — a temporal-inversion forgery an earlier
135        /// version of this guard silently accepted.
136        child: u64,
137        /// Parent effective start: `parent.nbf` when set, else
138        /// `parent.iat`.
139        parent: u64,
140    },
141    /// Delegation chain deeper than permitted.
142    #[error("delegation chain deeper than {0}")]
143    ChainTooDeep(usize),
144}

183    pub fn new(audience: impl Into<String>) -> Self {
184        Self {

194    /// Register key material under a `kid`.
195    ///
196    /// Refuse to shadow a JWKS-tracked kid. Without
197    /// this guard, an ordering hazard silently discarded operator
198    /// intent: if `add_key("X", …)` was called for a kid `X` that
199    /// a prior `add_jwks` had installed, the manual entry would
200    /// overwrite the JWKS one — but `jwks_kids` still contained
201    /// `X`, so the *next* `add_jwks` drain would remove `X` and
202    /// reinstall the JWKS version, silently discarding the
203    /// operator's manual key. `add_key` is normally a startup
204    /// call, but nothing in the API constrained late/admin use.
205    pub fn add_key(&self, kid: impl Into<String>, key: KeyMaterial) -> Result<(), IdentityError> {
206        let kid = kid.into();
207        // Lock order matches `add_jwks` (keys before jwks_kids) and the
208        // keys lock is held across the check AND the insert: releasing
209        // between them let a concurrent `add_jwks` install kid X in the
210        // gap, after which this insert silently shadowed the JWKS entry —
211        // exactly the ordering hazard this guard exists to refuse.
212        let mut loaded = self.keys.write();
213        let prior = self.jwks_kids.read();
214        if prior.contains(&kid) {
215            return Err(IdentityError::Jwks(format!(
216                "manual kid {kid:?} conflicts with a JWKS-tracked kid; rotate JWKS first"
217            )));
218        }
219        drop(prior);
220        loaded.insert(kid, key);
221        Ok(())
222    }

224    /// Add all supported Ed25519 keys from a standard JWKS document. Keys
225    /// loaded by a *previous* `add_jwks` call are retired (replace
226    /// semantics, so IdP rotation drops superseded JWKS keys); keys
227    /// registered manually via `add_key` are left untouched, and a JWKS
228    /// entry whose `kid` collides with a manually-registered key is
229    /// refused so the manual key stays authoritative.
230    pub fn add_jwks(&self, document: &serde_json::Value) -> Result<usize, IdentityError> {
231        // Cap the total number of
232        // entries iterated (parsed OR skipped), so a hostile JWKS
233        // full of RSA/EC decoys with a handful of legitimate OKP
234        // keys cannot stall the parse loop just by inflating the
235        // `keys` array to tens of thousands of entries. An
236        // earlier fix only capped parsed OKP entries, so a JWKS
237        // with 40k `kty=RSA` decoys still walked the whole array.
238        // Cap the outer array up front at the same threshold, and
239        // keep the inner cap as defense-in-depth against a future
240        // parser that stops short-circuiting on non-Ed25519 entries.
241        const MAX_JWKS_KEYS: usize = 256;
242        let keys = document
243            .get("keys")
244            .and_then(serde_json::Value::as_array)
245            .ok_or_else(|| IdentityError::Jwks("missing keys array".to_owned()))?;
246        if keys.len() > MAX_JWKS_KEYS {
247            return Err(IdentityError::Jwks(format!(
248                "JWKS keys array carries {} entries; refusing to walk more than {MAX_JWKS_KEYS} (this cap fires before the inner parser regardless of `kty`)",
249                keys.len()
250            )));
251        }
252        let mut parsed = Vec::new();
253        // Refuse duplicate `kid` within a single JWKS.
254        // HashMap's insert-with-overwrite semantics would otherwise
255        // silently accept a poisoned refresh where an attacker mixes
256        // an alien public key with the same kid as a legitimate one —
257        // the array's LAST entry silently wins verification with no
258        // log line, no counter, no error.
259        let mut seen: std::collections::HashSet<String> = std::collections::HashSet::new();
260        for key in keys {
261            if key.get("kty").and_then(serde_json::Value::as_str) != Some("OKP")
262                || key.get("crv").and_then(serde_json::Value::as_str) != Some("Ed25519")
263            {
264                continue;
265            }
266            // Respect RFC 7517 §4.2/§4.4. A JWK's
267            // `use` (public key use) member — when present — MUST
268            // be "sig" for verification material; "enc" keys are
269            // encryption-only and MUST NOT be installed for
270            // signature verification. Similarly `alg` — when
271            // present — MUST identify the algorithm intended for
272            // this key. For OKP/Ed25519 that's exclusively
273            // "EdDSA" (RFC 8037 §3.1). An IdP that ships an
274            // encryption-only key with a signing-domain kid
275            // (misconfig or partial compromise) would otherwise
276            // be silently installed as a verifier. Signature
277            // correctness is still protected by the alg/kty
278            // check at verify time, so this is defence-in-depth
279            // rather than a direct forgery close, but it aligns
280            // with the IdP's stated policy so audits can rely on
281            // it.
282            let kid_for_diag = key
283                .get("kid")
284                .and_then(serde_json::Value::as_str)
285                .unwrap_or("<missing>");
286            if let Some(use_) = key.get("use").and_then(serde_json::Value::as_str) {
287                if use_ != "sig" {
288                    return Err(IdentityError::Jwks(format!(
289                        "kid {kid_for_diag:?} declares use={use_:?}; only \"sig\" is accepted"
290                    )));
291                }
292            }
293            if let Some(alg) = key.get("alg").and_then(serde_json::Value::as_str) {
294                if alg != "EdDSA" {
295                    return Err(IdentityError::Jwks(format!(
296                        "kid {kid_for_diag:?} declares alg={alg:?}; only \"EdDSA\" is accepted for OKP/Ed25519"
297                    )));
298                }
299            }
300            let kid = key
301                .get("kid")
302                .and_then(serde_json::Value::as_str)
303                .filter(|value| !value.is_empty())
304                .ok_or_else(|| IdentityError::Jwks("Ed25519 key missing kid".to_owned()))?;
305            if !seen.insert(kid.to_owned()) {
306                return Err(IdentityError::Jwks(format!(
307                    "duplicate kid {kid:?} in JWKS document; refusing to accept a poisoned key set"
308                )));
309            }
310            let x = key
311                .get("x")
312                .and_then(serde_json::Value::as_str)
313                .filter(|value| !value.is_empty())
314                .ok_or_else(|| IdentityError::Jwks(format!("key {kid:?} missing x")))?;
315            // Shape-check the key at load time. Without
316            // this, a malformed base64url `x` (e.g., "!!!") or a
317            // wrong-length pubkey installs successfully, then every
318            // JWT with that kid fails at verify time with a confusing
319            // "bad JWK for kid" error. Rejecting during `add_jwks`
320            // surfaces operator config errors at boot/JWKS refresh
321            // instead of silently poisoning every downstream request.
322            let decoded = base64::engine::general_purpose::URL_SAFE_NO_PAD
323                .decode(x)
324                .map_err(|error| {
325                    IdentityError::Jwks(format!("key {kid:?} has malformed base64url `x`: {error}"))
326                })?;
327            if decoded.len() != 32 {
328                return Err(IdentityError::Jwks(format!(
329                    "key {kid:?} `x` decodes to {} bytes; Ed25519 public keys must be exactly 32",
330                    decoded.len()
331                )));
332            }
333            parsed.push((kid.to_owned(), KeyMaterial::Ed25519Jwk(x.to_owned())));
334            // The outer `keys.len() > MAX_JWKS_KEYS`
335            // guard at the top of the function already bounds
336            // `parsed.len()` (parsed is a subset of the outer
337            // array). This inner check is therefore unreachable in
338            // practice — kept as defense-in-depth so a future
339            // refactor that drops the outer guard cannot silently
340            // reopen the write-lock stall vector.
341            debug_assert!(
342                parsed.len() <= MAX_JWKS_KEYS,
343                "outer keys.len() cap should have already refused this document"
344            );
345            if parsed.len() > MAX_JWKS_KEYS {
346                return Err(IdentityError::Jwks(format!(
347                    "JWKS declares more than {MAX_JWKS_KEYS} Ed25519 OKP keys; refusing to install"
348                )));
349            }
350        }
351        if parsed.is_empty() {
352            return Err(IdentityError::Jwks("no Ed25519 OKP keys found".to_owned()));
353        }
354        let mut loaded = self.keys.write();
355        let mut prior = self.jwks_kids.write();
356        // A manually-added key must not be silently converted to a
357        // JWKS-tracked entry: without this refusal, the next JWKS refresh
358        // that no longer carries the colliding kid would retire (delete)
359        // an operator-configured key. Report the conflict — the operator
360        // can rename either side.
361        for (kid, _) in &parsed {
362            if loaded.contains_key(kid) && !prior.contains(kid) {
363                return Err(IdentityError::Jwks(format!(
364                    "JWKS kid {kid:?} conflicts with a manually-registered key; rename one"
365                )));
366            }
367        }
368        for kid in prior.drain() {
369            loaded.remove(&kid);
370        }
371        for (kid, material) in &parsed {
372            loaded.insert(kid.clone(), material.clone());
373            prior.insert(kid.clone());
374        }
375        Ok(parsed.len())
376    }

393    /// Validate a token and its full delegation chain.
394    pub fn validate(&self, token: &str) -> Result<ValidatedIdentity, IdentityError> {
395        let leaf = self.validate_single(token)?;
396        let mut depth = 0usize;
397        let mut child = leaf.clone();
398        let mut parent_token = leaf.parent_token.clone();
399        while let Some(pt) = parent_token {
400            depth += 1;
401            if depth > self.max_chain_depth {
402                return Err(IdentityError::ChainTooDeep(self.max_chain_depth));
403            }
404            let parent = self.validate_single(&pt)?;
405            // Scope inheritance: child ⊆ parent, with the SAME wildcard
406            // semantics the harness's runtime authorization uses
407            // (`av-harness::pipeline::scope_allows`). Do not use
408            // exact-string equality here: it rejected legitimately-delegated
409            // narrower scopes when the parent held a wildcard — e.g.,
410            // parent `["tool:*"]`, child `["tool:db_write"]` was refused
411            // as `ScopeEscalation("tool:db_write")` even though the
412            // runtime gate would authorize it, so the delegation
413            // machinery was strictly stricter than the authorization
414            // machinery. That asymmetry made wildcard-parent tokens
415            // effectively unusable for narrowing delegation, the
416            // canonical use case for scope subsetting.
417            if let Some(escalated) = child
418                .scopes
419                .iter()
420                .find(|scope| !scope_covered_by(scope, &parent.scopes))
421            {
422                return Err(IdentityError::ScopeEscalation {
423                    scope: escalated.clone(),
424                });
425            }
426            // Child must not outlive parent.
427            if child.exp > parent.exp {
428                return Err(IdentityError::ExpEscalation {
429                    child: child.exp,
430                    parent: parent.exp,
431                });
432            }
433            // Child must not be backdated before its parent. A
434            // hostile HMAC-shared-secret holder could otherwise mint
435            // a child with `iat` up to MAX_TTL_SECS BEFORE the parent
436            // that "delegated" it — asserting authorization causality
437            // that never happened. Consumers that treat `claims.iat`
438            // as "when this identity became authorized" (audit
439            // pipelines, revocation-cache pruning that keys eviction
440            // horizons on `iat + MAX_TTL_SECS`) are then lied to.
441            if child.iat < parent.iat {
442                return Err(IdentityError::IatEscalation {
443                    child: child.iat,
444                    parent: parent.iat,
445                });
446            }
447            // Same posture for `nbf`: a child cannot become usable
448            // before the parent that granted it did. Compare
449            // EFFECTIVE start times uniformly — fall back to `iat`
450            // on either side that omits `nbf`. An earlier version
451            // of this block was guarded by
452            // `if let Some(child_nbf) = child.nbf`, so a child that
453            // OMITTED `nbf` sailed past the check even when its raw
454            // `iat` predated the parent's explicit `nbf`: e.g.
455            // `parent{iat=t0, nbf=t0+15}` +
456            // `child{iat=t0+10, nbf=None}` was accepted, though the
457            // child asserts authority 5 seconds before the parent
458            // grant became active. The `IatEscalation` check above
459            // fires only for `child.iat < parent.iat` and does not
460            // cover this gap. Consumers treating `claims.iat` (or
461            // `nbf`) as "when this identity became authorized" saw
462            // an inverted causal ordering.
463            let parent_start = parent.nbf.unwrap_or(parent.iat);
464            let child_start = child.nbf.unwrap_or(child.iat);
465            if child_start < parent_start {
466                return Err(IdentityError::NbfEscalation {
467                    child: child_start,
468                    parent: parent_start,
469                });
470            }
471            parent_token = parent.parent_token.clone();
472            child = parent;
473        }
474        let now_s = av_core::time::now_ms() / av_core::units::MS_PER_SEC;
475        Ok(ValidatedIdentity {
476            ttl_remaining_s: leaf.exp.saturating_sub(now_s),
477            chain_depth: depth,
478            claims: leaf,
479        })
480    }

482    /// Validate one JWT, in order: pre-auth 8 KiB size cap, header sanity,
483    /// kid lookup, alg/key-type match, signature,
484    /// exp/aud/sub/iss required (+ `nbf` when present), `exp > iat`
485    /// consistency, future-iat, TTL cap, field presence,
486    /// bidi/zero-width spoofing guard, issuer allowlist (when configured).
487    fn validate_single(&self, token: &str) -> Result<NhiClaims, IdentityError> {
488        // Reject oversized tokens up front so an unauthenticated caller
489        // cannot amplify their pre-auth memory footprint through
490        // `jsonwebtoken::decode_header`, which base64-decodes the
491        // header segment before signature verification. RFC-realistic
492        // NHI JWTs are at most a few KiB; 8 KiB is a comfortable
493        // ceiling that blocks the amplification while accepting real
494        // tokens with generous claim sets.
495        const MAX_JWT_BYTES: usize = 8 * 1024;
496        if token.len() > MAX_JWT_BYTES {
497            return Err(IdentityError::Malformed(format!(
498                "token is {} bytes, exceeds pre-auth cap of {MAX_JWT_BYTES}",
499                token.len()
500            )));
501        }
502        let header =
503            jsonwebtoken::decode_header(token).map_err(|e| IdentityError::Malformed(e.to_string()))?;
504        let kid = header.kid.ok_or(IdentityError::MissingKid)?;
505        let keys = self.keys.read();
506        let key = keys
507            .get(&kid)
508            .ok_or_else(|| IdentityError::UnknownKid(kid.clone()))?;
509
510        // Algorithm-confusion defense: the key's type dictates the only
511        // acceptable alg; the token's stated alg must equal it exactly.
512        let (expected_alg, decoding_key) = match key {
513            KeyMaterial::Ed25519Pem(pem) => (
514                Algorithm::EdDSA,
515                DecodingKey::from_ed_pem(pem.as_bytes())
516                    .map_err(|e| IdentityError::Malformed(format!("bad key for kid {kid}: {e}")))?,
517            ),
518            KeyMaterial::Ed25519Jwk(x) => (
519                Algorithm::EdDSA,
520                DecodingKey::from_ed_components(x)
521                    .map_err(|e| IdentityError::Malformed(format!("bad JWK for kid {kid}: {e}")))?,
522            ),
523            KeyMaterial::HmacSecret(secret) => (Algorithm::HS256, DecodingKey::from_secret(secret)),
524        };
525        if header.alg != expected_alg {
526            return Err(IdentityError::AlgorithmRejected {
527                alg: format!("{:?}", header.alg),
528                kid,
529            });
530        }
531
532        let mut validation = Validation::new(expected_alg);
533        validation.set_audience(std::slice::from_ref(&self.audience));
534        validation.set_required_spec_claims(&["exp", "aud", "sub", "iss"]);
535        validation.leeway = self.leeway_secs;
536        validation.validate_nbf = true;
537
538        let data = jsonwebtoken::decode::<NhiClaims>(token, &decoding_key, &validation)
539            .map_err(|e| IdentityError::Verification(e.to_string()))?;
540        let claims = data.claims;
541
542        if claims.exp <= claims.iat {
543            return Err(IdentityError::BadTimestamps {
544                iat: claims.iat,
545                exp: claims.exp,
546            });
547        }
548        let now_s = av_core::time::now_ms() / av_core::units::MS_PER_SEC;
549        if claims.iat > now_s.saturating_add(self.leeway_secs) {
550            return Err(IdentityError::FutureIat {
551                iat: claims.iat,
552                now: now_s,
553            });
554        }
555        let ttl = claims.exp - claims.iat;
556        if ttl > MAX_TTL_SECS {
557            return Err(IdentityError::TtlTooLong(ttl));
558        }
559        if claims.instance_uid.is_empty() {
560            return Err(IdentityError::EmptyField("instance_uid"));
561        }
562        if claims.charter.is_empty() {
563            return Err(IdentityError::EmptyField("charter"));
564        }
565        if claims.version.is_empty() {
566            return Err(IdentityError::EmptyField("version"));
567        }
568        // docs/reference/LIMITS.md documents a 256-code-point charter cap
569        // ("longer are refused with 400"); the same reasoning applies to
570        // EVERY identity string that flows into logs, receipts, and
571        // event chains — an unbounded (up to the ~7 KiB per-claim JWT
572        // budget) attacker-chosen field is log-spam surface and — since
573        // instance_uid/version bind into every SIGNED receipt — bloats
574        // the JCS-canonicalized signing input on every request. Cap all
575        // identity strings at the same limit for one consistent rule.
576        // Threat model: an HMAC-shared-secret deployment where multiple
577        // principals hold the identity signing key. Any of them can
578        // construct a valid JWT with hostile-length claims.
579        const MAX_IDENTITY_STRING_CHARS: usize = 256;
580        for (name, value) in [
581            ("instance_uid", claims.instance_uid.as_str()),
582            ("charter", claims.charter.as_str()),
583            ("version", claims.version.as_str()),
584            ("sub", claims.sub.as_str()),
585            ("iss", claims.iss.as_str()),
586            ("jti", claims.jti.as_str()),
587        ] {
588            if value.chars().count() > MAX_IDENTITY_STRING_CHARS {
589                return Err(IdentityError::FieldTooLong {
590                    field: name,
591                    max: MAX_IDENTITY_STRING_CHARS,
592                });
593            }
594        }
595        // Bound the scopes array too: an unbounded list (or an
596        // individually oversized scope) is the same log-spam / receipt-
597        // bloat vector as the strings above, and the delegation-chain
598        // subset check runs a per-element comparison so a 10000-entry
599        // scopes[] amplifies delegation-verification cost per request.
600        const MAX_SCOPES: usize = 64;
601        if claims.scopes.len() > MAX_SCOPES {
602            return Err(IdentityError::FieldTooLong {
603                field: "scopes",
604                max: MAX_SCOPES,
605            });
606        }
607        for scope in &claims.scopes {
608            if scope.chars().count() > MAX_IDENTITY_STRING_CHARS {
609                return Err(IdentityError::FieldTooLong {
610                    field: "scopes[]",
611                    max: MAX_IDENTITY_STRING_CHARS,
612                });
613            }
614        }
615        // Trojan-Source guard: any bidi override or zero-width character in
616        // a rendered identity field would spoof how it looks in operator
617        // logs, receipts, and event chains while remaining part of the raw
618        // bytes on the wire. Scopes must be guarded too — they are the
619        // most audit-prominent identity strings and were the only one
620        // omitted from the original list, an inconsistency that let a
621        // scope like `payout\u{202E}elbast` render as visually-corrupt
622        // junk in operator logs while still surviving the length cap and
623        // the scope-subset check on raw bytes.
624        for (name, value) in [
625            ("instance_uid", claims.instance_uid.as_str()),
626            ("charter", claims.charter.as_str()),
627            ("version", claims.version.as_str()),
628            ("sub", claims.sub.as_str()),
629            ("iss", claims.iss.as_str()),
630            ("jti", claims.jti.as_str()),
631        ] {
632            if av_core::text::contains_bidi_or_zero_width(value) {
633                return Err(IdentityError::SpoofingCharacter(name));
634            }
635        }
636        for scope in &claims.scopes {
637            if av_core::text::contains_bidi_or_zero_width(scope) {
638                return Err(IdentityError::SpoofingCharacter("scopes[]"));
639            }
640        }
641        if let Some(allowed) = &self.allowed_issuers {
642            if !allowed.contains(&claims.iss) {
643                return Err(IdentityError::BadIssuer(claims.iss));
644            }
645        }
646        Ok(claims)
647    }
648}

650/// Return true if `scope` is authorized by any of the `parent_scopes`,
651/// using the SAME wildcard semantics
652/// (`av-harness::pipeline::scope_allows`) applied at runtime
653/// authorization. Duplicating (rather than re-exporting) the check
654/// keeps `av-identity` free of an `av-harness` dependency; if these
655/// two ever diverge, the delegation gate becomes strictly stricter or
656/// looser than the runtime gate — exact-string matching here once
657/// rejected legitimately-delegated narrower scopes under a wildcard
658/// parent, which is the class of bug such divergence produces.
659fn scope_covered_by(scope: &str, parent_scopes: &[String]) -> bool {
660    parent_scopes.iter().any(|parent| {
661        parent == "*"
662            || parent == scope
663            || parent
664                .strip_suffix(":*")
665                .is_some_and(|prefix| scope.starts_with(&format!("{prefix}:")))
666    })
667}
