9/// Verification key material, bound to a `kid`.
10#[derive(Clone)]
11pub enum KeyMaterial {
12    /// Ed25519 public key, SPKI PEM (`-----BEGIN PUBLIC KEY-----`). EdDSA.
13    Ed25519Pem(String),
14    /// Ed25519 JWK `x` coordinate, base64url without padding.
15    Ed25519Jwk(String),
16    /// HMAC shared secret. HS256 (dev / shared-secret IdP integrations).
17    HmacSecret(Vec<u8>),
18}
19
20impl std::fmt::Debug for KeyMaterial {
21    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
22        match self {
23            Self::Ed25519Pem(_) => f.write_str("KeyMaterial::Ed25519Pem(..)"),
24            Self::Ed25519Jwk(_) => f.write_str("KeyMaterial::Ed25519Jwk(..)"),
25            Self::HmacSecret(_) => f.write_str("KeyMaterial::HmacSecret(..)"), // never print secrets
26        }
27    }
28}

30/// Identity validation failures.
31#[derive(Debug, thiserror::Error, PartialEq, Eq)]
32#[non_exhaustive]
33pub enum IdentityError {
34    /// Token structurally malformed.
35    #[error("malformed token: {0}")]
36    Malformed(String),
37    /// Header lacks a `kid`.
38    #[error("token header has no kid")]
39    MissingKid,
40    /// No key registered for this `kid`.
41    #[error("unknown kid {0:?}")]
42    UnknownKid(String),
43    /// Token `alg` is not accepted, or does not match the key type for its
44    /// `kid` (algorithm-confusion defense).
45    #[error("algorithm {alg:?} not permitted for kid {kid:?}")]
46    AlgorithmRejected {
47        /// Stated algorithm.
48        alg: String,
49        /// Key id.
50        kid: String,
51    },
52    /// Signature invalid / expired / nbf future / wrong audience — collapsed
53    /// by jsonwebtoken; the detail string preserves the cause.
54    #[error("verification failed: {0}")]
55    Verification(String),
56    /// `exp - iat` exceeds the 15-minute NHI cap.
57    #[error("ttl {0}s exceeds the {MAX_TTL_SECS}s NHI cap")]
58    TtlTooLong(u64),
59    /// `exp ≤ iat` or other timestamp nonsense.
60    #[error("inconsistent timestamps (iat {iat}, exp {exp})")]
61    BadTimestamps {
62        /// Issued-at.
63        iat: u64,
64        /// Expiry.
65        exp: u64,
66    },
67    /// `iat` is unreasonably far in the future.
68    #[error("issued-at timestamp {iat} is in the future (now {now})")]
69    FutureIat {
70        /// Issued-at claim.
71        iat: u64,
72        /// Validator wall clock.
73        now: u64,
74    },
75    /// Token expired — with enough context to tell a stale token from a
76    /// skewed validator clock. Round-106/107 drills: a +6h host rejects
77    /// every honest client with a bare "ExpiredSignature", sending the
78    /// operator hunting client TTLs instead of their own NTP. An
79    /// "expiry" further in the past than the maximum issuable TTL can
80    /// only mean the validating host's clock is ahead.
81    #[error(
82        "token expired: exp {exp}, validator now {now} ({past_s}s past expiry{hint})",
83        past_s = now.saturating_sub(*exp),
84        hint = if now.saturating_sub(*exp) > MAX_TTL_SECS {
85            " — longer than the maximum token TTL; this host's clock is likely ahead (check NTP)"
86        } else {
87            ""
88        }
89    )]
90    Expired {
91        /// Expiry claim (signature-verified before extraction).
92        exp: u64,
93        /// Validator wall clock.
94        now: u64,
95    },
96    /// Required identity field empty.
97    #[error("empty identity field {0}")]
98    EmptyField(&'static str),
99    /// Identity field longer than its documented cap.
100    #[error("identity field {field} exceeds {max} characters")]
101    FieldTooLong {
102        /// Field name.
103        field: &'static str,
104        /// Documented cap in Unicode code points.
105        max: usize,
106    },
107    /// Field carries a bidi override or zero-width character that would
108    /// spoof how the identity renders in a log or audit view.
109    #[error("identity field {0} carries a bidi/zero-width spoofing character")]
110    SpoofingCharacter(&'static str),
111    /// Issuer not in the allowlist.
112    #[error("issuer {0:?} not allowed")]
113    BadIssuer(String),
114    /// JWKS document was malformed or contained no supported signing keys.
115    #[error("invalid JWKS: {0}")]
116    Jwks(String),
117    /// Child scopes exceed parent scopes.
118    #[error("scope escalation: {scope:?} not granted by parent")]
119    ScopeEscalation {
120        /// The offending scope.
121        scope: String,
122    },
123    /// Child outlives parent.
124    #[error("child exp {child} outlives parent exp {parent}")]
125    ExpEscalation {
126        /// Child expiry.
127        child: u64,
128        /// Parent expiry.
129        parent: u64,
130    },
131    /// Child claims to be issued before its parent (the delegator).
132    /// A backdated child forges audit-trail causality: consumers that
133    /// treat `claims.iat` as "when this identity became authorized"
134    /// see the child asserting authority before the parent that
135    /// granted it existed.
136    #[error("child iat {child} predates parent iat {parent}")]
137    IatEscalation {
138        /// Child issued-at.
139        child: u64,
140        /// Parent issued-at.
141        parent: u64,
142    },
143    /// Child's effective start (`nbf` when set, else `iat`) is
144    /// earlier than the parent's effective start (`nbf` when set,
145    /// else `iat`) — same forgery class as `IatEscalation`. The
146    /// carried `child` and `parent` values are the compared
147    /// EFFECTIVE starts, not the raw `nbf` fields (which may be
148    /// absent). See `IdentityValidator::validate` for the full
149    /// case matrix.
150    #[error("child effective-start {child} predates parent effective-start {parent}")]
151    NbfEscalation {
152        /// Child effective start: `child.nbf` when set, else
153        /// `child.iat`. The fallback closes an input case where the
154        /// child omits `nbf` but its `iat` predates the parent's
155        /// explicit `nbf` — a temporal-inversion forgery an earlier
156        /// version of this guard silently accepted.
157        child: u64,
158        /// Parent effective start: `parent.nbf` when set, else
159        /// `parent.iat`.
160        parent: u64,
161    },
162    /// Delegation chain deeper than permitted.
163    #[error("delegation chain deeper than {0}")]
164    ChainTooDeep(usize),
165}

202impl IdentityValidator {
203    /// Create a validator for `audience`.
204    pub fn new(audience: impl Into<String>) -> Self {
205        Self {
206            keys: RwLock::new(HashMap::new()),
207            jwks_kids: RwLock::new(HashSet::new()),

209            allowed_issuers: None,
210            max_chain_depth: 4,
211            leeway_secs: 30,

215    /// Register key material under a `kid`.
216    ///
217    /// Refuse to shadow a JWKS-tracked kid. Without
218    /// this guard, an ordering hazard silently discarded operator
219    /// intent: if `add_key("X", …)` was called for a kid `X` that
220    /// a prior `add_jwks` had installed, the manual entry would
221    /// overwrite the JWKS one — but `jwks_kids` still contained
222    /// `X`, so the *next* `add_jwks` drain would remove `X` and
223    /// reinstall the JWKS version, silently discarding the
224    /// operator's manual key. `add_key` is normally a startup
225    /// call, but nothing in the API constrained late/admin use.
226    pub fn add_key(&self, kid: impl Into<String>, key: KeyMaterial) -> Result<(), IdentityError> {
227        let kid = kid.into();
228        // Lock order matches `add_jwks` (keys before jwks_kids) and the
229        // keys lock is held across the check AND the insert: releasing
230        // between them let a concurrent `add_jwks` install kid X in the
231        // gap, after which this insert silently shadowed the JWKS entry —
232        // exactly the ordering hazard this guard exists to refuse.
233        let mut loaded = self.keys.write();
234        let prior = self.jwks_kids.read();
235        if prior.contains(&kid) {
236            return Err(IdentityError::Jwks(format!(
237                "manual kid {kid:?} conflicts with a JWKS-tracked kid; rotate JWKS first"
238            )));
239        }
240        drop(prior);
241        loaded.insert(kid, key);
242        Ok(())
243    }

245    /// Add all supported Ed25519 keys from a standard JWKS document. Keys
246    /// loaded by a *previous* `add_jwks` call are retired (replace
247    /// semantics, so IdP rotation drops superseded JWKS keys); keys
248    /// registered manually via `add_key` are left untouched, and a JWKS
249    /// entry whose `kid` collides with a manually-registered key is
250    /// refused so the manual key stays authoritative.
251    pub fn add_jwks(&self, document: &serde_json::Value) -> Result<usize, IdentityError> {
252        // Cap the total number of
253        // entries iterated (parsed OR skipped), so a hostile JWKS
254        // full of RSA/EC decoys with a handful of legitimate OKP
255        // keys cannot stall the parse loop just by inflating the
256        // `keys` array to tens of thousands of entries. An
257        // earlier fix only capped parsed OKP entries, so a JWKS
258        // with 40k `kty=RSA` decoys still walked the whole array.
259        // Cap the outer array up front at the same threshold, and
260        // keep the inner cap as defense-in-depth against a future
261        // parser that stops short-circuiting on non-Ed25519 entries.
262        const MAX_JWKS_KEYS: usize = 256;
263        let keys = document
264            .get("keys")
265            .and_then(serde_json::Value::as_array)
266            .ok_or_else(|| IdentityError::Jwks("missing keys array".to_owned()))?;
267        if keys.len() > MAX_JWKS_KEYS {
268            return Err(IdentityError::Jwks(format!(
269                "JWKS keys array carries {} entries; refusing to walk more than {MAX_JWKS_KEYS} (this cap fires before the inner parser regardless of `kty`)",
270                keys.len()
271            )));
272        }
273        let mut parsed = Vec::new();
274        // Refuse duplicate `kid` within a single JWKS.
275        // HashMap's insert-with-overwrite semantics would otherwise
276        // silently accept a poisoned refresh where an attacker mixes
277        // an alien public key with the same kid as a legitimate one —
278        // the array's LAST entry silently wins verification with no
279        // log line, no counter, no error.
280        let mut seen: std::collections::HashSet<String> = std::collections::HashSet::new();
281        for key in keys {
282            if key.get("kty").and_then(serde_json::Value::as_str) != Some("OKP")
283                || key.get("crv").and_then(serde_json::Value::as_str) != Some("Ed25519")
284            {
285                continue;
286            }
287            // Respect RFC 7517 §4.2/§4.4. A JWK's
288            // `use` (public key use) member — when present — MUST
289            // be "sig" for verification material; "enc" keys are
290            // encryption-only and MUST NOT be installed for
291            // signature verification. Similarly `alg` — when
292            // present — MUST identify the algorithm intended for
293            // this key. For OKP/Ed25519 that's exclusively
294            // "EdDSA" (RFC 8037 §3.1). An IdP that ships an
295            // encryption-only key with a signing-domain kid
296            // (misconfig or partial compromise) would otherwise
297            // be silently installed as a verifier. Signature
298            // correctness is still protected by the alg/kty
299            // check at verify time, so this is defence-in-depth
300            // rather than a direct forgery close, but it aligns
301            // with the IdP's stated policy so audits can rely on
302            // it.
303            let kid_for_diag = key
304                .get("kid")
305                .and_then(serde_json::Value::as_str)
306                .unwrap_or("<missing>");
307            if let Some(use_) = key.get("use").and_then(serde_json::Value::as_str) {
308                if use_ != "sig" {
309                    return Err(IdentityError::Jwks(format!(
310                        "kid {kid_for_diag:?} declares use={use_:?}; only \"sig\" is accepted"
311                    )));
312                }
313            }
314            if let Some(alg) = key.get("alg").and_then(serde_json::Value::as_str) {
315                if alg != "EdDSA" {
316                    return Err(IdentityError::Jwks(format!(
317                        "kid {kid_for_diag:?} declares alg={alg:?}; only \"EdDSA\" is accepted for OKP/Ed25519"
318                    )));
319                }
320            }
321            let kid = key
322                .get("kid")
323                .and_then(serde_json::Value::as_str)
324                .filter(|value| !value.is_empty())
325                .ok_or_else(|| IdentityError::Jwks("Ed25519 key missing kid".to_owned()))?;
326            if !seen.insert(kid.to_owned()) {
327                return Err(IdentityError::Jwks(format!(
328                    "duplicate kid {kid:?} in JWKS document; refusing to accept a poisoned key set"
329                )));
330            }
331            let x = key
332                .get("x")
333                .and_then(serde_json::Value::as_str)
334                .filter(|value| !value.is_empty())
335                .ok_or_else(|| IdentityError::Jwks(format!("key {kid:?} missing x")))?;
336            // Shape-check the key at load time. Without
337            // this, a malformed base64url `x` (e.g., "!!!") or a
338            // wrong-length pubkey installs successfully, then every
339            // JWT with that kid fails at verify time with a confusing
340            // "bad JWK for kid" error. Rejecting during `add_jwks`
341            // surfaces operator config errors at boot/JWKS refresh
342            // instead of silently poisoning every downstream request.
343            let decoded = base64::engine::general_purpose::URL_SAFE_NO_PAD
344                .decode(x)
345                .map_err(|error| {
346                    IdentityError::Jwks(format!("key {kid:?} has malformed base64url `x`: {error}"))
347                })?;
348            if decoded.len() != 32 {
349                return Err(IdentityError::Jwks(format!(
350                    "key {kid:?} `x` decodes to {} bytes; Ed25519 public keys must be exactly 32",
351                    decoded.len()
352                )));
353            }
354            parsed.push((kid.to_owned(), KeyMaterial::Ed25519Jwk(x.to_owned())));
355            // The outer `keys.len() > MAX_JWKS_KEYS`
356            // guard at the top of the function already bounds
357            // `parsed.len()` (parsed is a subset of the outer
358            // array). This inner check is therefore unreachable in
359            // practice — kept as defense-in-depth so a future
360            // refactor that drops the outer guard cannot silently
361            // reopen the write-lock stall vector.
362            debug_assert!(
363                parsed.len() <= MAX_JWKS_KEYS,
364                "outer keys.len() cap should have already refused this document"
365            );
366            if parsed.len() > MAX_JWKS_KEYS {
367                return Err(IdentityError::Jwks(format!(
368                    "JWKS declares more than {MAX_JWKS_KEYS} Ed25519 OKP keys; refusing to install"
369                )));
370            }
371        }
372        if parsed.is_empty() {
373            // An explicitly-EMPTY `keys` array is authoritative
374            // revocation: the key owner withdrew every signing key
375            // (compromise response). The old blanket error here made
376            // revocation impossible — every refresh failed and the
377            // cached keys verified compromised-key tokens forever.
378            // Retiring the JWKS-tracked keys fails closed: with no
379            // usable keys, every token is refused until the endpoint
380            // publishes keys again. A NON-empty document with zero
381            // usable Ed25519 OKP entries stays an error (that shape is
382            // a misconfigured endpoint — RSA-only, malformed — not a
383            // deliberate withdrawal), so a broken rotation keeps the
384            // last-known-good keys rather than dropping enforcement
385            // into refuse-everything.
386            if keys.is_empty() {
387                let mut loaded = self.keys.write();
388                let mut prior = self.jwks_kids.write();
389                for kid in prior.drain() {
390                    loaded.remove(&kid);
391                }
392                return Ok(0);
393            }
394            return Err(IdentityError::Jwks("no Ed25519 OKP keys found".to_owned()));
395        }
396        let mut loaded = self.keys.write();
397        let mut prior = self.jwks_kids.write();
398        // A manually-added key must not be silently converted to a
399        // JWKS-tracked entry: without this refusal, the next JWKS refresh
400        // that no longer carries the colliding kid would retire (delete)
401        // an operator-configured key. Report the conflict — the operator
402        // can rename either side.
403        for (kid, _) in &parsed {
404            if loaded.contains_key(kid) && !prior.contains(kid) {
405                return Err(IdentityError::Jwks(format!(
406                    "JWKS kid {kid:?} conflicts with a manually-registered key; rename one"
407                )));
408            }
409        }
410        for kid in prior.drain() {
411            loaded.remove(&kid);
412        }
413        for (kid, material) in &parsed {
414            loaded.insert(kid.clone(), material.clone());
415            prior.insert(kid.clone());
416        }
417        Ok(parsed.len())
418    }

435    /// Validate a token and its full delegation chain.
436    pub fn validate(&self, token: &str) -> Result<ValidatedIdentity, IdentityError> {
437        let leaf = self.validate_single(token)?;
438        let mut depth = 0usize;
439        let mut child = leaf.clone();
440        let mut parent_token = leaf.parent_token.clone();
441        while let Some(pt) = parent_token {
442            depth += 1;
443            if depth > self.max_chain_depth {
444                return Err(IdentityError::ChainTooDeep(self.max_chain_depth));
445            }
446            let parent = self.validate_single(&pt)?;
447            // Scope inheritance: child ⊆ parent, with the SAME wildcard
448            // semantics the harness's runtime authorization uses
449            // (`av-harness::pipeline::scope_allows`). Do not use
450            // exact-string equality here: it rejected legitimately-delegated
451            // narrower scopes when the parent held a wildcard — e.g.,
452            // parent `["tool:*"]`, child `["tool:db_write"]` was refused
453            // as `ScopeEscalation("tool:db_write")` even though the
454            // runtime gate would authorize it, so the delegation
455            // machinery was strictly stricter than the authorization
456            // machinery. That asymmetry made wildcard-parent tokens
457            // effectively unusable for narrowing delegation, the
458            // canonical use case for scope subsetting.
459            if let Some(escalated) = child
460                .scopes
461                .iter()
462                .find(|scope| !scope_covered_by(scope, &parent.scopes))
463            {
464                return Err(IdentityError::ScopeEscalation {
465                    scope: escalated.clone(),
466                });
467            }
468            // Child must not outlive parent.
469            if child.exp > parent.exp {
470                return Err(IdentityError::ExpEscalation {
471                    child: child.exp,
472                    parent: parent.exp,
473                });
474            }
475            // Child must not be backdated before its parent. A
476            // hostile HMAC-shared-secret holder could otherwise mint
477            // a child with `iat` up to MAX_TTL_SECS BEFORE the parent
478            // that "delegated" it — asserting authorization causality
479            // that never happened. Consumers that treat `claims.iat`
480            // as "when this identity became authorized" (audit
481            // pipelines, revocation-cache pruning that keys eviction
482            // horizons on `iat + MAX_TTL_SECS`) are then lied to.
483            if child.iat < parent.iat {
484                return Err(IdentityError::IatEscalation {
485                    child: child.iat,
486                    parent: parent.iat,
487                });
488            }
489            // Same posture for `nbf`: a child cannot become usable
490            // before the parent that granted it did. Compare
491            // EFFECTIVE start times uniformly — fall back to `iat`
492            // on either side that omits `nbf`. An earlier version
493            // of this block was guarded by
494            // `if let Some(child_nbf) = child.nbf`, so a child that
495            // OMITTED `nbf` sailed past the check even when its raw
496            // `iat` predated the parent's explicit `nbf`: e.g.
497            // `parent{iat=t0, nbf=t0+15}` +
498            // `child{iat=t0+10, nbf=None}` was accepted, though the
499            // child asserts authority 5 seconds before the parent
500            // grant became active. The `IatEscalation` check above
501            // fires only for `child.iat < parent.iat` and does not
502            // cover this gap. Consumers treating `claims.iat` (or
503            // `nbf`) as "when this identity became authorized" saw
504            // an inverted causal ordering.
505            let parent_start = parent.nbf.unwrap_or(parent.iat);
506            let child_start = child.nbf.unwrap_or(child.iat);
507            if child_start < parent_start {
508                return Err(IdentityError::NbfEscalation {
509                    child: child_start,
510                    parent: parent_start,
511                });
512            }
513            parent_token = parent.parent_token.clone();
514            child = parent;
515        }
516        let now_s = av_core::time::now_ms() / av_core::units::MS_PER_SEC;
517        Ok(ValidatedIdentity {
518            ttl_remaining_s: leaf.exp.saturating_sub(now_s),
519            chain_depth: depth,
520            claims: leaf,
521        })
522    }

524    /// Validate one JWT, in order: pre-auth 8 KiB size cap, header sanity,
525    /// kid lookup, alg/key-type match, signature,
526    /// exp/aud/sub/iss required (+ `nbf` when present), `exp > iat`
527    /// consistency, future-iat, TTL cap, field presence,
528    /// bidi/zero-width spoofing guard, issuer allowlist (when configured).
529    fn validate_single(&self, token: &str) -> Result<NhiClaims, IdentityError> {
530        // Reject oversized tokens up front so an unauthenticated caller
531        // cannot amplify their pre-auth memory footprint through
532        // `jsonwebtoken::decode_header`, which base64-decodes the
533        // header segment before signature verification. RFC-realistic
534        // NHI JWTs are at most a few KiB; 8 KiB is a comfortable
535        // ceiling that blocks the amplification while accepting real
536        // tokens with generous claim sets.
537        const MAX_JWT_BYTES: usize = 8 * 1024;
538        if token.len() > MAX_JWT_BYTES {
539            return Err(IdentityError::Malformed(format!(
540                "token is {} bytes, exceeds pre-auth cap of {MAX_JWT_BYTES}",
541                token.len()
542            )));
543        }
544        let header =
545            jsonwebtoken::decode_header(token).map_err(|e| IdentityError::Malformed(e.to_string()))?;
546        let kid = header.kid.ok_or(IdentityError::MissingKid)?;
547        let keys = self.keys.read();
548        let key = keys
549            .get(&kid)
550            .ok_or_else(|| IdentityError::UnknownKid(kid.clone()))?;
551
552        // Algorithm-confusion defense: the key's type dictates the only
553        // acceptable alg; the token's stated alg must equal it exactly.
554        let (expected_alg, decoding_key) = match key {
555            KeyMaterial::Ed25519Pem(pem) => (
556                Algorithm::EdDSA,
557                DecodingKey::from_ed_pem(pem.as_bytes())
558                    .map_err(|e| IdentityError::Malformed(format!("bad key for kid {kid}: {e}")))?,
559            ),
560            KeyMaterial::Ed25519Jwk(x) => (
561                Algorithm::EdDSA,
562                DecodingKey::from_ed_components(x)
563                    .map_err(|e| IdentityError::Malformed(format!("bad JWK for kid {kid}: {e}")))?,
564            ),
565            KeyMaterial::HmacSecret(secret) => (Algorithm::HS256, DecodingKey::from_secret(secret)),
566        };
567        if header.alg != expected_alg {
568            return Err(IdentityError::AlgorithmRejected {
569                alg: format!("{:?}", header.alg),
570                kid,
571            });
572        }
573
574        let mut validation = Validation::new(expected_alg);
575        validation.set_audience(std::slice::from_ref(&self.audience));
576        validation.set_required_spec_claims(&["exp", "aud", "sub", "iss"]);
577        validation.leeway = self.leeway_secs;
578        validation.validate_nbf = true;
579
580        let data = jsonwebtoken::decode::<NhiClaims>(token, &decoding_key, &validation).map_err(|e| {
581            // Round-107: on expiry, re-decode with the SAME key and
582            // audience but exp validation off — signature integrity
583            // still enforced — purely to recover exp for the error.
584            // The token is still rejected either way; this only
585            // upgrades diagnosability (skewed-host vs stale-token).
586            if matches!(e.kind(), jsonwebtoken::errors::ErrorKind::ExpiredSignature) {
587                let mut relaxed = validation.clone();
588                relaxed.validate_exp = false;
589                if let Ok(peek) = jsonwebtoken::decode::<NhiClaims>(token, &decoding_key, &relaxed) {
590                    let now_s = av_core::time::now_ms() / av_core::units::MS_PER_SEC;
591                    return IdentityError::Expired {
592                        exp: peek.claims.exp,
593                        now: now_s,
594                    };
595                }
596            }
597            IdentityError::Verification(e.to_string())
598        })?;
599        let claims = data.claims;
600
601        if claims.exp <= claims.iat {
602            return Err(IdentityError::BadTimestamps {
603                iat: claims.iat,
604                exp: claims.exp,
605            });
606        }
607        let now_s = av_core::time::now_ms() / av_core::units::MS_PER_SEC;
608        if claims.iat > now_s.saturating_add(self.leeway_secs) {
609            return Err(IdentityError::FutureIat {
610                iat: claims.iat,
611                now: now_s,
612            });
613        }
614        let ttl = claims.exp - claims.iat;
615        if ttl > MAX_TTL_SECS {
616            return Err(IdentityError::TtlTooLong(ttl));
617        }
618        if claims.instance_uid.is_empty() {
619            return Err(IdentityError::EmptyField("instance_uid"));
620        }
621        if claims.charter.is_empty() {
622            return Err(IdentityError::EmptyField("charter"));
623        }
624        if claims.version.is_empty() {
625            return Err(IdentityError::EmptyField("version"));
626        }
627        // docs/reference/LIMITS.md documents a 256-code-point charter cap
628        // ("longer are refused with 400"); the same reasoning applies to
629        // EVERY identity string that flows into logs, receipts, and
630        // event chains — an unbounded (up to the ~7 KiB per-claim JWT
631        // budget) attacker-chosen field is log-spam surface and — since
632        // instance_uid/version bind into every SIGNED receipt — bloats
633        // the JCS-canonicalized signing input on every request. Cap all
634        // identity strings at the same limit for one consistent rule.
635        // Threat model: an HMAC-shared-secret deployment where multiple
636        // principals hold the identity signing key. Any of them can
637        // construct a valid JWT with hostile-length claims.
638        const MAX_IDENTITY_STRING_CHARS: usize = 256;
639        // instance_uid has a TIGHTER cap than the other identity
640        // strings: it flows verbatim into `ai_agent.instance_uid` on
641        // every OCSF event and signed receipt, and the shipped schemas
642        // (ocsf-agent-event, receipt-v2) pin it at maxLength 128 —
643        // matching `av_core::InstanceUid::parse`. A 129–256-char uid
644        // admitted here would build events/receipts that Rust-side
645        // validation accepts but every external schema verifier (and
646        // the console ingest) refuses — the split-verdict class.
647        const MAX_INSTANCE_UID_CHARS: usize = 128;
648        if claims.instance_uid.chars().count() > MAX_INSTANCE_UID_CHARS {
649            return Err(IdentityError::FieldTooLong {
650                field: "instance_uid",
651                max: MAX_INSTANCE_UID_CHARS,
652            });
653        }
654        for (name, value) in [
655            ("instance_uid", claims.instance_uid.as_str()),
656            ("charter", claims.charter.as_str()),
657            ("version", claims.version.as_str()),
658            ("sub", claims.sub.as_str()),
659            ("iss", claims.iss.as_str()),
660            ("jti", claims.jti.as_str()),
661        ] {
662            if value.chars().count() > MAX_IDENTITY_STRING_CHARS {
663                return Err(IdentityError::FieldTooLong {
664                    field: name,
665                    max: MAX_IDENTITY_STRING_CHARS,
666                });
667            }
668        }
669        // Bound the scopes array too: an unbounded list (or an
670        // individually oversized scope) is the same log-spam / receipt-
671        // bloat vector as the strings above, and the delegation-chain
672        // subset check runs a per-element comparison so a 10000-entry
673        // scopes[] amplifies delegation-verification cost per request.
674        const MAX_SCOPES: usize = 64;
675        if claims.scopes.len() > MAX_SCOPES {
676            return Err(IdentityError::FieldTooLong {
677                field: "scopes",
678                max: MAX_SCOPES,
679            });
680        }
681        for scope in &claims.scopes {
682            if scope.chars().count() > MAX_IDENTITY_STRING_CHARS {
683                return Err(IdentityError::FieldTooLong {
684                    field: "scopes[]",
685                    max: MAX_IDENTITY_STRING_CHARS,
686                });
687            }
688        }
689        // Trojan-Source guard: any bidi override or zero-width character in
690        // a rendered identity field would spoof how it looks in operator
691        // logs, receipts, and event chains while remaining part of the raw
692        // bytes on the wire. Scopes must be guarded too — they are the
693        // most audit-prominent identity strings and were the only one
694        // omitted from the original list, an inconsistency that let a
695        // scope like `payout\u{202E}elbast` render as visually-corrupt
696        // junk in operator logs while still surviving the length cap and
697        // the scope-subset check on raw bytes.
698        for (name, value) in [
699            ("instance_uid", claims.instance_uid.as_str()),
700            ("charter", claims.charter.as_str()),
701            ("version", claims.version.as_str()),
702            ("sub", claims.sub.as_str()),
703            ("iss", claims.iss.as_str()),
704            ("jti", claims.jti.as_str()),
705        ] {
706            if av_core::text::contains_bidi_or_zero_width(value) {
707                return Err(IdentityError::SpoofingCharacter(name));
708            }
709        }
710        for scope in &claims.scopes {
711            if av_core::text::contains_bidi_or_zero_width(scope) {
712                return Err(IdentityError::SpoofingCharacter("scopes[]"));
713            }
714        }
715        if let Some(allowed) = &self.allowed_issuers {
716            if !allowed.contains(&claims.iss) {
717                return Err(IdentityError::BadIssuer(claims.iss));
718            }
719        }
720        Ok(claims)
721    }
722}

724/// Return true if `scope` is authorized by any of the `parent_scopes`,
725/// using the SAME wildcard semantics
726/// (`av-harness::pipeline::scope_allows`) applied at runtime
727/// authorization. Duplicating (rather than re-exporting) the check
728/// keeps `av-identity` free of an `av-harness` dependency; if these
729/// two ever diverge, the delegation gate becomes strictly stricter or
730/// looser than the runtime gate — exact-string matching here once
731/// rejected legitimately-delegated narrower scopes under a wildcard
732/// parent, which is the class of bug such divergence produces.
733fn scope_covered_by(scope: &str, parent_scopes: &[String]) -> bool {
734    parent_scopes.iter().any(|parent| {
735        parent == "*"
736            || parent == scope
737            || parent
738                .strip_suffix(":*")
739                .is_some_and(|prefix| scope.starts_with(&format!("{prefix}:")))
740    })
741}

744mod key_material_tests {
745    #![allow(clippy::unwrap_used)]
746
747    use super::KeyMaterial;
748
749    /// Round-19 mutation finding: the hand-written Debug impl exists
750    /// solely so key material can never leak into logs/error chains
751    /// ("never print secrets"), but nothing pinned it — replacing the
752    /// whole impl with a no-op survived the suite, and so would a
753    /// future `#[derive(Debug)]` refactor that prints the secret bytes
754    /// verbatim into any `{:?}` log site. Pin the redaction contract.
755    #[test]
756    fn debug_never_prints_key_material() {
757        const SECRET_PEM: &str = "-----BEGIN PUBLIC KEY-----abc123secretbytes";
758        const SECRET_HMAC: &[u8] = b"hmac-shared-secret-0123456789abcdef";
759        let cases = [
760            (
761                format!("{:?}", KeyMaterial::Ed25519Pem(SECRET_PEM.to_owned())),
762                "Ed25519Pem",
763            ),
764            (
765                format!("{:?}", KeyMaterial::Ed25519Jwk(SECRET_PEM.to_owned())),
766                "Ed25519Jwk",
767            ),
768            (
769                format!("{:?}", KeyMaterial::HmacSecret(SECRET_HMAC.to_vec())),
770                "HmacSecret",
771            ),
772        ];
773        for (rendered, variant) in cases {
774            assert!(
775                rendered.contains(variant),
776                "Debug must still name the variant for diagnosability: {rendered}"
777            );
778            assert!(
779                !rendered.contains("secret")
780                    && !rendered.contains("abc123")
781                    && !rendered.contains("0123456789"),
782                "Debug leaked key material: {rendered}"
783            );
784        }
785    }
