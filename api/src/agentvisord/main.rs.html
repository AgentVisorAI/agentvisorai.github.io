19#[tokio::main]
20async fn main() {
21    // Fail-closed CLI contract: agentvisord historically ignored ALL
22    // arguments, so `agentvisord --config /etc/prod.toml` silently booted
23    // from the env/search-path config instead (possibly permissive
24    // defaults), and `--help` started a server. For a security proxy,
25    // unrecognized arguments must refuse to start. CLI/usage output is
26    // deliberately plain text: it is for the human at the terminal and
27    // happens before the structured logger exists.
28    let config_override = match parse_cli_args(std::env::args().skip(1)) {
29        Ok(CliAction::Run(config_override)) => config_override,
30        Ok(CliAction::Help) => {
31            println!("{USAGE}");
32            return;
33        }
34        Ok(CliAction::Version) => {
35            println!("agentvisord {}", env!("CARGO_PKG_VERSION"));
36            return;
37        }
38        Err(error) => {
39            eprintln!("{error}\n\n{USAGE}");
40            std::process::exit(2);
41        }
42    };
43
44    if let Err(error) = run(config_override).await {
45        // The daemon logs structured JSON, but a fatal
46        // startup/shutdown error used to return through anyhow and
47        // print as a bare `Error:` line plus a backtrace — unparseable
48        // to the log pipeline at exactly the moment it matters most.
49        // Route the fatal reason through the structured logger when it
50        // is up; fall back to stderr only when tracing itself never
51        // initialized (the one failure that cannot be self-reported).
52        if tracing::dispatcher::has_been_set() {
53            tracing::error!(error = format!("{error:#}"), "agentvisord exiting on fatal error");
54        } else {
55            eprintln!("agentvisord exiting on fatal error: {error:#}");
56        }
57        std::process::exit(1);
58    }
59}

61async fn run(config_override: Option<PathBuf>) -> Result<()> {
62    #[cfg(feature = "otel")]
63    let telemetry_provider = init_tracing()?;
64    #[cfg(not(feature = "otel"))]
65    init_tracing()?;
66
67    let (config, config_source) =
68        av_harness::config::load_config_with_override(config_override).map_err(anyhow::Error::msg)?;
69    // Refuse the whole configuration up front, with the
70    // complete list, when it selects backends this binary was compiled
71    // without. The individual build_* sites keep their own bails as
72    // defense in depth, but they fail one at a time and only when
73    // reached — an operator fixing kafka would then discover onnx.
74    let unsupported = config.unsupported_backend_requirements();
75    if !unsupported.is_empty() {
76        anyhow::bail!(
77            "this build cannot run the resolved configuration: {}",
78            unsupported.join("; ")
79        );
80    }
81    let manifest = load_manifest(&config)?;
82    let bridge = build_bridge(&config, &manifest)?;
83
84    let sandbox = load_sandbox(&config)?;
85    let store = build_store(&config)?;
86    let embedder = build_embedder(&config)?;
87    let vector_sink = build_vector_sink(&config, embedder.dim()).await?;
88    // Build the metrics registry BEFORE `build_identity` so the JWKS
89    // refresh loop can bump `av_jwks_refresh_errors_total` on the same
90    // registry that `AppState` will hand to `/metrics`. Otherwise the
91    // JWKS counters would live on a phantom registry no scraper sees,
92    // and a silently-stale key set would
93    // remain unalertable.
94    let metrics = Arc::new(av_core::metrics::Registry::new());
95    let (identity, jwks_refresh) = build_identity(&config, Arc::clone(&metrics)).await?;
96    let signer_path = std::env::var_os("AV_SIGNING_SEED_FILE")
97        .map(PathBuf::from)
98        .unwrap_or_else(|| PathBuf::from("config/signing.seed"));
99    let (raw_signer, signer_newly_generated) = load_or_create_signer(&signer_path)?;
100    // Capture the identifying material BEFORE moving into an `Arc<dyn Signer>`
101    // so the startup banner can name the exact trust anchor the process
102    // will sign under. Omitting this line is how a silent-new-anchor failure
103    // (missing/emptied seed file → freshly generated seed → every receipt
104    // signed by an untrusted key) can go undetected: without the id in the
105    // log, only an external verifier days later notices, at which point the
106    // audit chain for that window is unrecoverable.
107    let signer_key_id = raw_signer.key_id().to_owned();
108    let signer_public_key_hex = hex::encode(raw_signer.public_key_bytes());
109    // Log the trust anchor UNCONDITIONALLY. The startup banner below is
110    // info-level on the default target, so `RUST_LOG=warn` (the common
111    // production setting — and what our own compose/K8s guidance quiets
112    // logs with) silently drops the only steady-state record of which
113    // key this process signs under. The `trust_anchor` target is pinned
114    // to `info` inside init_tracing regardless of RUST_LOG, so this
115    // line survives `warn`, `error`, and even `off`.
116    tracing::info!(
117        target: "trust_anchor",
118        signer_key_id = %signer_key_id,
119        signer_public_key_hex = %signer_public_key_hex,
120        signer_seed_path = %signer_path.display(),
121        freshly_generated = signer_newly_generated,
122        "receipt-signing trust anchor for this process"
123    );
124    if signer_newly_generated {
125        // A fresh anchor is only correct at genuine first boot. Every other
126        // occurrence (Secret failed to mount, emptyDir vanished, seed file
127        // deleted) is a compliance incident: audit consumers that trusted
128        // the previous key will reject every receipt issued from here on.
129        // Emit at WARN so the log pipeline surfaces it without needing a
130        // Prometheus gauge (the metrics registry has no gauge type for
131        // this yet). Uses the always-on `trust_anchor` target so even a
132        // `RUST_LOG=error` deployment records the compliance incident.
133        tracing::warn!(
134            target: "trust_anchor",
135            signer_key_id = %signer_key_id,
136            signer_public_key_hex = %signer_public_key_hex,
137            signer_seed_path = %signer_path.display(),
138            "signing seed was freshly generated at startup — an operator that had trusted a previous key must re-pin this one before any receipt from this instance verifies"
139        );
140    }
141    let signer: Arc<dyn Signer> = Arc::new(raw_signer);
142    bridge
143        .set_control_key(av_harness::control_key_from_signer(signer.as_ref()))
144        .context("configure Bridge control authentication")?;
145    let state = AppState::new_with_backends_and_metrics(
146        config.clone(),
147        store,
148        Arc::new(sandbox),
149        bridge,
150        identity,
151        signer,
152        embedder,
153        vector_sink,
154        Arc::clone(&metrics),
155    )
156    .map_err(anyhow::Error::new)?;
157    // Two daemons on one spool silently split the audit
158    // trail (interleaved journals, racing reconcilers, torn ATIF
159    // artifacts). The README documents single-instance; nothing
160    // enforced it. Hold an exclusive advisory lock on a well-known
161    // spool file for the whole process lifetime — the OS releases it
162    // on ANY exit including SIGKILL, so there is no stale-lock
163    // recovery to get wrong. Everything below (orphaned-temp sweep,
164    // spool recovery, the reconciler) assumes it is the only writer.
165    let _spool_lock = acquire_spool_lock(std::path::Path::new(&config.atif_spool_dir))?;
166    // Boot-time only (no concurrent writer exists yet): sweep temp
167    // files orphaned by a SIGKILL between `create_new` and `rename` —
168    // the RAII unlink cannot run across a crash, so crash loops
169    // accumulate them linearly forever otherwise.
170    match av_core::fsutil::sweep_orphaned_tmp(std::path::Path::new(&config.atif_spool_dir)) {
171        Ok(0) => {}
172        Ok(removed) => tracing::info!(removed, "removed crash-orphaned temp files from the spool"),
173        Err(error) => tracing::warn!(%error, "orphaned-temp sweep failed; stale .tmp files may remain"),
174    }
175    // Fail-closed boot probe: an unwritable or full spool volume used
176    // to boot silently — /health answered 200 while every chat 503'd
177    // and nothing was capturable. Refuse to start instead: the spool
178    // is the audit store; a server that cannot record must not serve.
179    {
180        let probe = std::path::Path::new(&config.atif_spool_dir)
181            .join(format!(".writability-probe-{}.tmp", av_core::new_event_uid()));
182        av_core::fsutil::write_atomic(&probe, b"probe")
183            .and_then(|()| std::fs::remove_file(&probe))
184            .with_context(|| {
185                format!(
186                    "spool directory {:?} is not writable; refusing to serve without a working audit store",
187                    config.atif_spool_dir
188                )
189            })?;
190    }
191    state
192        .finalizer
193        .recover_spooled_sessions(&state.sessions, &config.breaker)
194        .await
195        .context("recover ATIF spool")?;
196    state
197        .finalizer
198        .retry_marked_promotions(&state.sessions)
199        .await
200        .context("retry durable promotions")?;
201    let bridge_maintenance_shutdown = Arc::new(tokio::sync::Notify::new());
202    let bridge_maintenance = spawn_bridge_maintenance(
203        Arc::clone(&state.bridge),
204        Arc::clone(&state.metrics),
205        Arc::clone(&bridge_maintenance_shutdown),
206    );
207    let reconciler_shutdown = Arc::new(tokio::sync::Notify::new());
208    let reconciler = spawn_reconciler(
209        Arc::clone(&state.sessions),
210        state.finalizer.clone(),
211        config.session_idle_close_s,
212        config.reconcile_tick_s,
213        config.breaker.clone(),
214        Arc::clone(&state.metrics),
215        Arc::clone(&reconciler_shutdown),
216    );
217    // Retention: prune sealed ATIF trajectories + sidecars older than
218    // `atif_retention_days` on an hourly cadence. Only sealed pairs are
219    // touched; unpaired remnants stay for the reconciler quarantine
220    // sweep. `None` (the ship default) preserves the historical
221    // "manage-with-external-cron" behaviour.
222    //
223    // The tick body dispatches to `spawn_blocking` inside
224    // `prune_sealed_atif`, and `JoinHandle::abort()` cancels only the
225    // outer async task — a blocking `remove_file` sequence in flight
226    // would keep running against the spool concurrently with process
227    // exit and could leave orphan `.close-complete` markers on
228    // abandonment. Mirror the bridge-maintenance discipline with a
229    // dedicated Notify so shutdown returns the loop between ticks and
230    // the outer `.await` below actually waits for the blocking work
231    // to finish. Same rationale as `bridge_maintenance_shutdown`.
232    let retention_shutdown = Arc::new(tokio::sync::Notify::new());
233    let retention = config.atif_retention_days.map(|days| {
234        let finalizer = state.finalizer.clone();
235        let metrics = Arc::clone(&state.metrics);
236        let shutdown = Arc::clone(&retention_shutdown);
237        let pruned_total = metrics.counter(
238            "av_atif_retention_pruned_total",
239            "Sealed ATIF+sidecar pairs deleted by the retention sweep",
240        );
241        tokio::spawn(async move {
242            let interval = std::time::Duration::from_secs(60 * 60);
243            let max_age = std::time::Duration::from_secs(u64::from(days) * 24 * 60 * 60);
244            let mut ticker = tokio::time::interval(interval);
245            ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Delay);
246            loop {
247                // Race the shutdown notify against the ticker so the
248                // loop can exit between ticks and never launch a fresh
249                // `spawn_blocking` if shutdown fired during the tick's
250                // own await. Mirrors `spawn_bridge_maintenance`.
251                tokio::select! {
252                    biased;
253                    () = shutdown.notified() => return,
254                    _ = ticker.tick() => {}
255                }
256                // Supervise the tick body: a panic here (allocator OOM,
257                // a Display panic on a non-UTF8 error chain, a
258                // parking_lot poison-on-unwind) would otherwise silently
259                // terminate the retention sweep under Tokio's default
260                // UnhandledPanic::Ignore. The 1 h cadence means one
261                // panic invisibly loses one hour of retention — a
262                // persistent panic condition drops retention entirely
263                // until the daemon restarts, and the spool grows
264                // unbounded until it fills the disk. Mirrors the
265                // bridge-maintenance and JWKS-refresh discipline.
266                let pruned_total = pruned_total.clone();
267                let metrics_tick = Arc::clone(&metrics);
268                let finalizer_tick = finalizer.clone();
269                let outcome = std::panic::AssertUnwindSafe(async move {
270                    match finalizer_tick.prune_sealed_atif(max_age).await {
271                        Ok(0) => {}
272                        Ok(n) => {
273                            pruned_total.add(n as u64);
274                            tracing::info!(
275                                pruned = n,
276                                retention_days = days,
277                                "ATIF retention sweep removed sealed evidence pairs"
278                            );
279                        }
280                        Err(error) => {
281                            metrics_tick
282                                .counter(
283                                    "av_atif_retention_errors_total",
284                                    av_harness::pipeline::supervision_help::ATIF_RETENTION_ERRORS,
285                                )
286                                .inc();
287                            tracing::warn!(
288                                %error,
289                                retention_days = days,
290                                "ATIF retention sweep failed; will retry next hour"
291                            );
292                        }
293                    }
294                })
295                .catch_unwind()
296                .await;
297                if let Err(panic) = outcome {
298                    let msg = panic
299                        .downcast_ref::<&'static str>()
300                        .copied()
301                        .or_else(|| panic.downcast_ref::<String>().map(String::as_str))
302                        .unwrap_or("panic payload was not a string");
303                    metrics
304                        .counter(
305                            "av_atif_retention_panics_total",
306                            av_harness::pipeline::supervision_help::ATIF_RETENTION_PANICS,
307                        )
308                        .inc();
309                    tracing::error!(
310                        panic = %msg,
311                        "ATIF retention sweep tick panicked; continuing"
312                    );
313                }
314            }
315        })
316    });
317    let tcp_nodelay_failures = state.metrics.counter(
318        "av_tcp_nodelay_failures_total",
319        "TCP_NODELAY setsockopt failed on an accepted connection; the \
320         suboptimal-latency connection still serves. Rate-limited to a \
321         single tracing warn per process to avoid a per-accept log storm \
322         under half-open flood / SYN flood.",
323    );
324    let listener = tokio::net::TcpListener::bind(&config.listen)
325        .await
326        .with_context(|| format!("bind {}", config.listen))?
327        // Set TCP_NODELAY on every accepted socket. Nagle + delayed-
328        // ACK together produce ~40 ms of avoidable per-frame stall
329        // in the SSE relay: each streamed frame is 50-200 bytes
330        // (one token or a small delta), so the second small frame
331        // sits in the sender's kernel queue until the first is
332        // ACK'd. Compounded across a 500-token stream that's tens
333        // of seconds of inter-token latency the operator would
334        // otherwise trace to "OpenAI feels slower behind the
335        // proxy". The outbound-side reqwest client already sets
336        // `.tcp_keepalive(30 s)` on the upstream direction
337        // (`pipeline.rs`); this closes the reverse-facing hole.
338        //
339        // `tap_io` is axum 0.8's idiomatic hook for per-connection
340        // socket-option tuning without swapping listener types.
341        // Failure to set the option (embedded system without full
342        // socket-option support, half-open flood socket teardown
343        // between accept and setsockopt, ECONNRESET race) logs and
344        // falls through — a suboptimal-latency connection is still
345        // a working one.
346        //
347        // Wrap the warn in a `std::sync::Once`: a SYN-flood or
348        // Slowloris-class probe against an unauthenticated endpoint
349        // could otherwise fire this warn at accept-rate (~20 k/s on
350        // a moderately-sized Linux node), saturating the log
351        // pipeline and amplifying the very DoS the SSE latency
352        // regression is a distant second to. Every occurrence is
353        // still counted via `av_tcp_nodelay_failures_total` so
354        // operators keep visibility of persistent failures without
355        // the log storm. Same dampener discipline as R33's
356        // identity_rejection_window sliding cap.
357        .tap_io(move |tcp_stream| {
358            if let Err(error) = tcp_stream.set_nodelay(true) {
359                static WARNED: std::sync::Once = std::sync::Once::new();
360                WARNED.call_once(|| {
361                    tracing::warn!(
362                        %error,
363                        "failed to set TCP_NODELAY on incoming connection; SSE inter-token \
364                         latency may regress by ~40 ms per frame. Subsequent failures logged \
365                         only via av_tcp_nodelay_failures_total to avoid a per-accept log storm."
366                    );
367                });
368                tcp_nodelay_failures.inc();
369            }
370        });
371    if let Some(segment) = config.duplicated_chat_path_segment() {
372        tracing::warn!(
373            upstream_url = %av_core::url_redact::redact_userinfo(&config.upstream_url),
374            upstream_chat_path = %config.upstream_chat_path,
375            "upstream_url already ends with \"/{segment}\" and upstream_chat_path repeats it; \
376             the joined URL will contain \"/{segment}/{segment}/\" — most providers expect the \
377             base URL without the \"/{segment}\" suffix"
378        );
379    }
380    tracing::info!(
381        listen = %config.listen,
382        config = %config_source,
383        upstream = %av_core::url_redact::redact_userinfo(
384            &format!("{}{}", config.upstream_url.trim_end_matches('/'), config.upstream_chat_path)
385        ),
386        upstream_auth = %av_harness::pipeline::describe_upstream_auth(&config),
387        bridge = %config.bridge_backend,
388        state = %config.state_backend,
389        // Budget counters on an expiring backend
390        // (Redis: 24 h) silently reset for sessions active past the
391        // TTL; in-memory never expires. Surface the value so an
392        // operator diagnosing "my week-long agent's budget reset"
393        // can see the window without reading redis_store.rs.
394        state_counter_ttl_s = ?state.store.counter_ttl_secs(),
395        identity = if config.require_identity { "required" } else { "optional" },
396        // Always surface the exact receipt-signing trust anchor. This is the
397        // simplest observable signal that a Secret mount, an emptyDir volume,
398        // or a hand-managed seed file survived the boot — an operator or an
399        // audit-log consumer can compare it against the previously-published
400        // public key without waiting for the first receipt to fail
401        // downstream.
402        signer_key_id = %signer_key_id,
403        signer_public_key_hex = %signer_public_key_hex,
404        "AgentVisor AI started"
405    );
406    // Publish the signing-key fingerprint as a labelled
407    // gauge whose value is always 1 (Prometheus's convention for
408    // `*_info` metrics). Downstream verifiers can alert on a
409    // key-id change between scrapes even when nothing signed a
410    // receipt in the window — the log-only signal above misses
411    // that if operators only look at metrics dashboards. The
412    // gauge is per-key-id (not just the fingerprint prefix) so a
413    // key rotation shows up as a distinct series rather than a
414    // silent value flip.
415    state
416        .metrics
417        .gauge(
418            &format!(
419                "av_signing_key_info{{key_id=\"{signer_key_id}\",public_key_hex=\"{signer_public_key_hex}\"}}"
420            ),
421            "Signing-key fingerprint — 1 while the current process holds this key",
422        )
423        .set(1);
424    let (shutdown_started_tx, shutdown_started_rx) = tokio::sync::oneshot::channel();
425    // Register the drain-timeout counter up front so alerts can wire
426    // onto it at boot. This surfaces a real hazard: axum's
427    // per-connection tasks are detached `tokio::spawn`s, so dropping
428    // the `Serve` future does NOT cancel in-flight streams. If a
429    // long-lived streaming client outlives the graceful drain budget,
430    // the timeout fires, `state.worker.wait_idle()` also times out,
431    // and any late-arriving job into `finalize_sessions` races the
432    // stream's own drop-time finalizer. A non-zero counter is a
433    // hard-page: it means shutdown ordering is unsafe and every
434    // affected session needs receipt-verify on restart.
435    let drain_timeouts = metrics.counter(
436        "av_http_shutdown_drain_timeouts_total",
437        "HTTP graceful-drain phase exceeded budget; per-connection tasks may still be live",
438    );
439    let draining_flag = Arc::clone(&state.draining);
440    let ready_drain_window = std::time::Duration::from_secs(config.shutdown_ready_drain_s);
441    let server = std::future::IntoFuture::into_future(
442        axum::serve(listener, build_router(state.clone())).with_graceful_shutdown(async move {
443            shutdown_signal().await;
444            // Flip the draining flag FIRST so `/readyz` reports 503 on
445            // every connection accepted from here on. NOTE: axum stops
446            // accepting the moment this future completes, so without
447            // the pre-drain window below a fresh readiness probe sees
448            // connection-refused (also a probe failure, but an LB that
449            // distinguishes "degraded" from "gone" gets no 503, and
450            // in-flight LB routing has zero grace). Every deployment
451            // target uses `shutdown_ready_drain_s` for this window:
452            // the shipped k8s manifest sets it to 5 because the
453            // distroless runtime base has no shell for a preStop
454            // `sleep` hook to run in; docker-compose, systemd, and
455            // bare-VM LBs have no preStop equivalent at all.
456            draining_flag.store(true, std::sync::atomic::Ordering::SeqCst);
457            if ready_drain_window > std::time::Duration::ZERO {
458                tracing::info!(
459                    window_s = ready_drain_window.as_secs(),
460                    "readiness-controlled pre-drain: /readyz now 503, still accepting"
461                );
462                tokio::time::sleep(ready_drain_window).await;
463            }
464            let _ = shutdown_started_tx.send(());
465        }),
466    );
467    tokio::pin!(server);
468    let result = tokio::select! {
469        result = &mut server => result.context("serve AgentVisor AI"),
470        _ = shutdown_started_rx => {
471            match tokio::time::timeout(config.effective_drain_timeout(), &mut server).await {
472                Ok(result) => result.context("serve AgentVisor AI"),
473                Err(_) => {
474                    drain_timeouts.inc();
475                    // Explicitly drop the pinned server future here to
476                    // stop the accepting `TcpListener` from taking new
477                    // connections while the later shutdown phases run.
478                    // Detached per-connection tasks that Axum handed
479                    // off via `tokio::spawn` still leak — a full fix
480                    // requires a broadcast CancellationToken threaded
481                    // through the streaming handlers, which is out of
482                    // scope here. Until then, the counter above makes
483                    // the failure mode observable.
484                    Err(anyhow::anyhow!(
485                        "timed out draining HTTP connections during shutdown"
486                    ))
487                }
488            }
489        }
490    };
491    // R69 F1: Signal reconciler to stop instead of aborting.
492    // JoinHandle::abort() cancels only the outer async task; any
493    // spawn_blocking closure inside the tick body — write_atomic
494    // markers, remove_file cleanup, publish_idempotent for
495    // SESSION_CLOSE, and up to 64 close_session calls per tick —
496    // keeps running orphaned. The reconciler ticks 14 400× more
497    // often than the retention loop below and does far more spool-
498    // mutating work per tick; without Notify, orphan reconciler
499    // closures race `finalize_sessions` (declared below) for the
500    // same session journal / bridge segment on process exit. Same
501    // rationale as retention and bridge_maintenance below.
502    reconciler_shutdown.notify_one();
503    // Signal retention to stop instead of aborting.
504    // JoinHandle::abort() cancels only the outer async task, but the
505    // retention tick body dispatches to spawn_blocking (each tick calls
506    // Finalizer::prune_sealed_atif which is a spawn_blocking wrapper).
507    // An abandoned blocking closure mid-remove_file sequence could leave
508    // an orphan .close-complete marker on the spool. Notify makes the
509    // loop return between ticks so the shutdown `.await` below actually
510    // waits for the blocking work to finish — same rationale as the
511    // bridge-maintenance loop below.
512    retention_shutdown.notify_one();
513    // Signal maintenance to stop instead of aborting.
514    // JoinHandle::abort() only cancels the outer async task; a
515    // spawn_blocking closure that's already running keeps rewriting
516    // Bridge segments to completion and races the process exit.
517    // Notify makes the loop return between ticks so the shutdown
518    // .await below actually waits for the blocking work to finish.
519    bridge_maintenance_shutdown.notify_one();
520    // Abort the JWKS refresh task on shutdown so the
521    // infinite `loop { interval.tick() ... }` cannot outlive the
522    // harness's outbound HTTP hygiene. Previously the JoinHandle was
523    // dropped at spawn time, letting the refresher fire another
524    // request during the shutdown window and race the finalizer for
525    // IdentityValidator key state.
526    if let Some(handle) = jwks_refresh.as_ref() {
527        handle.abort();
528    }
529    // Silence the expected cancellation; log everything else (panic).
530    // The reconciler now uses cooperative shutdown (R69 F1) so a
531    // clean stop returns `Ok(())`, not `Err(is_cancelled)`. Retain
532    // the is_cancelled branch in case of any pathological path (e.g.
533    // a future refactor that aborts on hard timeout).
534    if let Err(error) = reconciler.await {
535        if !error.is_cancelled() {
536            tracing::warn!(%error, "reconciler task exited with an error before shutdown");
537        }
538    }
539    if let Err(error) = bridge_maintenance.await {
540        if !error.is_cancelled() {
541            tracing::warn!(%error, "bridge maintenance task exited with an error before shutdown abort");
542        }
543    }
544    if let Some(handle) = jwks_refresh {
545        if let Err(error) = handle.await {
546            if !error.is_cancelled() {
547                tracing::warn!(%error, "JWKS refresh task exited with an error before shutdown abort");
548            }
549        }
550    }
551    if let Some(handle) = retention {
552        if let Err(error) = handle.await {
553            if !error.is_cancelled() {
554                tracing::warn!(%error, "ATIF retention task exited with an error before shutdown abort");
555            }
556        }
557    }
558    #[cfg(feature = "otel")]
559    let flush_telemetry = move || {
560        if let Some(provider) = telemetry_provider {
561            provider
562                .shutdown_with_timeout(std::time::Duration::from_secs(
563                    av_harness::config::OTEL_FLUSH_SECS,
564                ))
565                .map_err(|error| anyhow::anyhow!("flush OpenTelemetry: {error}"))?;
566        }
567        Ok(())
568    };
569    #[cfg(not(feature = "otel"))]
570    let flush_telemetry = || Ok(());
571    let open_sessions = state.sessions.open_sessions();
572    let shutdown_finalizer = state.finalizer.clone();
573    let finalize_sessions = async move {
574        let mut failures = Vec::new();
575        // Bound each per-session close so a stuck session (a leaked
576        // SessionLease from an axum-cancelled handler, a worker permit
577        // dropped without decrementing, an upstream half-closed after
578        // TCP_KEEPALIVE-less client disconnect) does not starve the
579        // remaining sessions' close budget. `close_session_locked`
580        // internally calls `wait_for_streams` / `wait_for_worker_jobs`
581        // which are unbounded loops on internal counters.
582        //
583        // The per-session deadline MUST be small relative to the outer
584        // `WORKER_FINALIZE_PHASE_SECS` (30 s) budget — otherwise ONE
585        // stuck session eats the whole outer window and every REMAINING
586        // session is still skipped, which is the pathology this fix
587        // exists to close. Set it to 3 s: healthy close_session
588        // completes in < 100 ms so 3 s is 30× headroom for the healthy
589        // path, and up to 10 stuck sessions can fire their per-session
590        // deadline before the outer timer expires — giving the 11th
591        // (and later) healthy sessions a chance to close cleanly.
592        //
593        // Contrast with the reconciler's `IDLE_CLOSE_DEADLINE = 90 s`
594        // (R26): that path is NOT wrapped by an outer timeout so a
595        // large per-session bound is fine there.
596        //
597        // On per-session timeout, bump a pre-registered metric so
598        // operators tuning the deadline (or diagnosing why shutdown
599        // leaves sessions for restart-time recovery) can see the class
600        // separately from `av_http_shutdown_drain_timeouts_total`
601        // (which fires on OUTER drain timeout, a distinct condition).
602        const PER_SESSION_CLOSE_DEADLINE: std::time::Duration = std::time::Duration::from_secs(3);
603        let session_close_timeouts = shutdown_finalizer.metrics().counter(
604            "av_shutdown_session_close_timeouts_total",
605            "Per-session close hit the shutdown-time per-session deadline (3 s) and \
606             was deferred to restart-time spool recovery. A sustained rate > 0 on \
607             every rollout indicates a class of sessions that regularly hang their \
608             close (leaked leases, dropped worker permits, unresponsive bridge \
609             publish) — the coincident session id in the shutdown warn log is the \
610             correlation key. Distinct from `av_http_shutdown_drain_timeouts_total` \
611             which fires on the OUTER phase timeout.",
612        );
613        for session in open_sessions {
614            let session_id = session.id.clone();
615            let outcome = tokio::time::timeout(
616                PER_SESSION_CLOSE_DEADLINE,
617                shutdown_finalizer.close_session(session, av_events::StopReason::SessionClosed),
618            )
619            .await;
620            match outcome {
621                Ok(Ok(_)) => {}
622                Ok(Err(error)) => failures.push(format!("session {session_id}: {error}")),
623                Err(_elapsed) => {
624                    session_close_timeouts.inc();
625                    failures.push(format!(
626                        "session {session_id}: per-session close deadline ({}s) exceeded — deferred to restart-time recovery",
627                        PER_SESSION_CLOSE_DEADLINE.as_secs()
628                    ));
629                }
630            }
631        }
632        // R70 Track B misdiagnosis correction (R71 review): the
633        // previous shape appended `post_finalize_worker.wait_idle()`
634        // here on the claim that `close_session` internally submits
635        // new WORKER jobs (SESSION_CLOSE bridge publish). It does
636        // NOT — `close_session_locked` (`reconciler.rs:961`) awaits
637        // `emit_bridge_event → resolve_lifecycle_ack →
638        // spawn_blocking(bridge.publish_idempotent)` INLINE with no
639        // WorkerHandle::submit. `pending` (the counter `wait_idle`
640        // polls) is only incremented by `routes.rs`/`pipeline.rs`
641        // submissions — none from the finalizer path. The added
642        // `wait_idle` observed `pending == 0` on the first read and
643        // returned immediately: defensive dead-code. Removed to avoid
644        // future maintainers reasoning from a false comment. If a
645        // real "close job races exit" hazard emerges, it lives in the
646        // `spawn_blocking` layer (finalizer bridge publishes) and the
647        // fix is a JoinSet-style tracker, not `WorkerHandle::wait_idle`.
648        if failures.is_empty() {
649            Ok(())
650        } else {
651            Err(anyhow::anyhow!(failures.join("; ")))
652        }
653    };
654    // Compose the worker drain with a pre-step that first drains
655    // in-flight `mcp_call_inner` detached spawns (routes.rs). Axum's
656    // graceful drain only awaits outer handlers; a client-disconnected
657    // mcp_call whose spawned body is mid-execution outlives the drain.
658    // If we skipped straight to `wait_idle`, that body could reach
659    // `worker.try_submit` AFTER `wait_idle` returned, then race
660    // `finalize_sessions` and trip `av_shutdown_session_close_
661    // timeouts_total` on an otherwise-recoverable session.
662    //
663    // The 5 s inner budget is a small fraction of
664    // `WORKER_FINALIZE_PHASE_SECS = 30 s`, leaving 25 s for
665    // `wait_idle` to cover any worker jobs the drained bodies just
666    // submitted. On timeout the `av_shutdown_mcp_drain_timeouts_total`
667    // counter fires (pre-registered in `pipeline.rs`) so an operator
668    // sees the drain miss on the coincident rollout, but shutdown
669    // still proceeds (the affected session's post-drain worker
670    // submission will trip the per-session close deadline and be
671    // recovered on next boot).
672    let mcp_drain_budget = std::time::Duration::from_secs(5);
673    let mcp_inflight = Arc::clone(&state.mcp_inflight);
674    let mcp_metrics = Arc::clone(&state.metrics);
675    let worker_handle = state.worker.clone();
676    let worker_drain = async move {
677        if tokio::time::timeout(mcp_drain_budget, mcp_inflight.wait_drained())
678            .await
679            .is_err()
680        {
681            mcp_metrics
682                .counter(
683                    "av_shutdown_mcp_drain_timeouts_total",
684                    "Shutdown MCP-inflight drain hit its 5 s deadline before every detached \
685                     mcp_call_inner spawn completed",
686                )
687                .inc();
688            tracing::warn!(
689                inflight = mcp_inflight.count(),
690                "timed out draining in-flight MCP tool calls (5 s); shutdown proceeding"
691            );
692        }
693        worker_handle.wait_idle().await;
694    };
695    finish_shutdown(
696        result,
697        std::time::Duration::from_secs(av_harness::config::WORKER_FINALIZE_PHASE_SECS),
698        worker_drain,
699        finalize_sessions,
700        flush_telemetry,
701    )
702    .await
703}

722/// Exclusive advisory lock proving this daemon is the
723/// spool's only writer (see the call site in `run`). The returned
724/// handle must stay alive for the process lifetime; dropping it — or
725/// any process exit, including SIGKILL — releases the lock.
726fn acquire_spool_lock(spool_dir: &Path) -> Result<std::fs::File> {
727    // Use `create_dir_all_synced` (not bare `create_dir_all`) so the
728    // spool ROOT is materialised at 0o700 on Unix. Every subsequent
729    // path (`append_journal`, `write_atomic`, `claim_sync`) short-
730    // circuits its own mkdir with an `is_dir()` fast path, so this
731    // is the ONLY site that ever creates the root — leaving it at
732    // the ambient umask (0o755 under the default 0022) would leak
733    // enumeration of the deterministic `sha256(session-id)[..32]`
734    // filename stems to any co-tenant with `execute` on the parent,
735    // even though the individual spool files are now 0o600.
736    av_core::fsutil::create_dir_all_synced(spool_dir)
737        .with_context(|| format!("create spool directory {}", spool_dir.display()))?;
738    let path = spool_dir.join(".agentvisord.lock");
739    let file = std::fs::OpenOptions::new()
740        .create(true)
741        .truncate(false)
742        .write(true)
743        .open(&path)
744        .with_context(|| format!("open spool lock file {}", path.display()))?;
745    match file.try_lock() {
746        Ok(()) => Ok(file),
747        Err(std::fs::TryLockError::WouldBlock) => anyhow::bail!(
748            "another agentvisord instance already holds the spool lock at {}; \
749             two daemons sharing one spool would silently split the audit trail \
750             (interleaved journals, racing reconcilers, torn artifacts). Run one \
751             instance per spool directory",
752            path.display()
753        ),
754        Err(std::fs::TryLockError::Error(error)) => {
755            Err(anyhow::Error::new(error).context(format!("acquire exclusive spool lock {}", path.display())))
756        }
757    }
758}

760/// Fail-closed argument parsing: anything unrecognized refuses startup
761/// rather than being silently ignored (a mistyped flag must never let a
762/// security proxy boot with a different config than the operator intended).
763fn parse_cli_args(args: impl Iterator<Item = String>) -> Result<CliAction, String> {
764    let mut config: Option<PathBuf> = None;
765    let mut args = args.peekable();
766    while let Some(arg) = args.next() {
767        match arg.as_str() {
768            "-h" | "--help" => return Ok(CliAction::Help),
769            "-V" | "--version" => return Ok(CliAction::Version),
770            "--config" => {
771                let value = args
772                    .next()
773                    .ok_or_else(|| "--config requires a path argument".to_owned())?;
774                if config.replace(PathBuf::from(value)).is_some() {
775                    return Err("--config may only be given once".to_owned());
776                }
777            }
778            other => {
779                if let Some(value) = other.strip_prefix("--config=") {
780                    if value.is_empty() {
781                        return Err("--config requires a non-empty path".to_owned());
782                    }
783                    if config.replace(PathBuf::from(value)).is_some() {
784                        return Err("--config may only be given once".to_owned());
785                    }
786                } else {
787                    return Err(format!("unrecognized argument {other:?}"));
788                }
789            }
790        }
791    }
792    Ok(CliAction::Run(config))
793}

795async fn finish_shutdown<F, G, T>(
796    server_result: Result<()>,
797    worker_timeout: std::time::Duration,
798    worker_drain: F,
799    finalize_sessions: G,
800    flush_telemetry: T,
801) -> Result<()>
802where
803    F: std::future::Future<Output = ()>,
804    G: std::future::Future<Output = Result<()>>,
805    T: FnOnce() -> Result<()>,
806{
807    let mut failures = Vec::new();
808    if let Err(error) = server_result {
809        failures.push(format!("server: {error:#}"));
810    }
811    if tokio::time::timeout(worker_timeout, worker_drain).await.is_err() {
812        failures.push("timed out draining audit worker during shutdown".to_owned());
813    }
814    match tokio::time::timeout(worker_timeout, finalize_sessions).await {
815        Ok(Ok(())) => {}
816        Ok(Err(error)) => failures.push(format!("session finalization: {error:#}")),
817        Err(_) => failures.push("timed out finalizing sessions during shutdown".to_owned()),
818    }
819    if let Err(error) = flush_telemetry() {
820        failures.push(format!("telemetry: {error:#}"));
821    }
822    if failures.is_empty() {
823        Ok(())
824    } else {
825        Err(anyhow::anyhow!("shutdown failures: {}", failures.join("; ")))
826    }
827}

829/// Always-on filter directive for the `trust_anchor` target: the
830/// receipt-signing key identity must reach the logs regardless of
831/// RUST_LOG (a `warn`/`error` deployment otherwise never records which
832/// key it signs under — the silent-new-anchor failure mode). A
833/// target-specific directive outranks any global level directive, so

835fn trust_anchor_directive() -> tracing_subscriber::filter::Directive {

837    // is unreachable but avoids a panic path.
838    "trust_anchor=info"
839        .parse()
840        .unwrap_or_else(|_| tracing_subscriber::filter::LevelFilter::INFO.into())
841}

921fn load_sandbox(config: &HarnessConfig) -> Result<Sandbox> {
922    let mut schemas = std::collections::HashMap::new();
923    if let Some(directory) = config.tool_schema_dir.as_deref() {
924        // Zero-config: the *default* schema directory may simply not exist
925        // (fresh install, empty working dir). That is fine — the sandbox
926        // stays fail-closed and rejects tool calls until schemas are added.
927        // An explicitly configured directory must exist: a typo silently
928        // disabling every schema would be a policy bypass.
929        let is_default_dir = config.uses_default_tool_schema_dir();
930        match std::fs::read_dir(directory) {
931            Err(error) if error.kind() == std::io::ErrorKind::NotFound && is_default_dir => {
932                tracing::warn!(
933                    directory,
934                    "default tool schema directory not found; tool calls will be rejected until schemas are added"
935                );
936            }
937            Err(error) => {
938                return Err(error).with_context(|| format!("read tool schema directory {directory}"));
939            }
940            Ok(entries) => {
941                for entry in entries {
942                    let path = entry?.path();
943                    if path.extension().and_then(std::ffi::OsStr::to_str) != Some("json") {
944                        continue;
945                    }
946                    let tool = path
947                        .file_stem()
948                        .and_then(std::ffi::OsStr::to_str)
949                        .filter(|name| !name.is_empty())
950                        .ok_or_else(|| anyhow::anyhow!("invalid tool schema filename {}", path.display()))?;
951                    let schema: serde_json::Value = serde_json::from_slice(
952                        // Same MAX_CONTROL_BYTES cap and rationale as the
953                        // bridge's `load_validators`: bound
954                        // the allocation before the JSON parser can reject.
955                        &av_core::fsutil::read_capped(&path, av_core::fsutil::MAX_CONTROL_BYTES)
956                            .with_context(|| format!("read tool schema {}", path.display()))?,
957                    )
958                    .with_context(|| format!("parse tool schema {}", path.display()))?;
959                    schemas.insert(tool.to_owned(), schema);
960                }
961            }
962        }
963    }
964    if config.require_tool_schema && schemas.is_empty() {
965        // Only a hard error when tool forwarding is actually in use:
966        // without a tool upstream no /v1/mcp call can be forwarded anyway,
967        // and the sandbox rejects everything unmatched (fail-closed).
968        if config.tool_upstream_url.is_some() {
969            anyhow::bail!(
970                "require_tool_schema=true and tool_upstream_url is set, but no tool schemas were loaded from {:?}",
971                config.tool_schema_dir
972            );
973        }
974        tracing::warn!("no tool schemas loaded; every tool call will be rejected (fail-closed)");
975    }
976    let mut policies: Vec<Box<dyn PolicyEngine>> = Vec::new();
977    for path in &config.wasm_policy_paths {
978        // Zero-config: fall back to the embedded copy of the default
979        // payload-limit policy when the default path is absent on disk.
980        // Explicitly configured paths must exist (typo = hard error).
981        let is_default_policy = HarnessConfig::is_default_policy_path(path);
982        // Bounded read (same discipline as every other boot-time file):
983        // real policies are tiny (the builtin is a few KiB of WAT), so
984        // 16 MiB is generous for any legitimate compiled module while
985        // bounding the allocation before wasmtime can reject the bytes.
986        const MAX_WASM_POLICY_BYTES: u64 = 16 * 1024 * 1024;
987        let bytes = match av_core::fsutil::read_capped(std::path::Path::new(path), MAX_WASM_POLICY_BYTES) {
988            Ok(bytes) => bytes,
989            Err(error) if error.kind() == std::io::ErrorKind::NotFound && is_default_policy => {
990                tracing::info!(path, "policy file not found; using embedded built-in copy");
991                BUILTIN_POLICY_WAT.as_bytes().to_vec()
992            }
993            Err(error) => return Err(error).with_context(|| format!("read WASM policy {path}")),
994        };
995        // The default payload-limit policy
996        // denies bodies above 4 MiB. If the operator raised
997        // `max_request_bytes` past that, chat requests the HTTP body
998        // limit admits get 403 PolicyBlocked — misattributed as a
999        // policy violation and invisible from the config file. Warn
1000        // loudly at boot so the mismatch is discoverable.
1001        const DEFAULT_POLICY_LIMIT_BYTES: usize = 4 * 1024 * 1024;
1002        if is_default_policy && config.max_request_bytes > DEFAULT_POLICY_LIMIT_BYTES {
1003            tracing::warn!(
1004                max_request_bytes = config.max_request_bytes,
1005                policy_limit = DEFAULT_POLICY_LIMIT_BYTES,
1006                "max_request_bytes exceeds the default payload-limit policy's 4 MiB threshold; \
1007                 chat requests between the two sizes will be refused as policy-blocked — raise \
1008                 the constant in the payload_limit.wat policy (or remove it from \
1009                 wasm_policy_paths) to match"
1010            );
1011        }
1012        policies.push(Box::new(
1013            WasmPolicy::from_bytes(path, &bytes).map_err(anyhow::Error::msg)?,
1014        ));
1015    }
1016    // R112 F2: refuse to boot when `wasm_policy_paths = []` is
1017    // explicit AND no tool-schema fail-closed guard is on. Prior
1018    // shape silently accepted an empty policy chain: the
1019    // Sandbox::check_with_principal for-loop over policies became
1020    // a no-op → Gate 3 (policy check) disabled → combined with
1021    // the default require_tool_schema=false → Gate 2 (tool schema
1022    // check) also disabled → the daemon accepts arbitrary tool
1023    // calls under a receipt that still claims policiesEnforced,
1024    // silently defeating the compliance attestation. Compare
1025    // require_tool_schema at line 964 which correctly warns on
1026    // fail-closed; this branch needs a symmetric guard. Warn
1027    // loudly on empty; hard-fail if BOTH gates are off (the
1028    // catastrophic case).
1029    if policies.is_empty() {
1030        if !config.require_tool_schema {
1031            anyhow::bail!(
1032                "wasm_policy_paths is empty AND require_tool_schema is false; \
1033                 refusing to boot with both Gate 3 (policy) and Gate 2 (schema) \
1034                 disabled — receipts would attest 'policiesEnforced' while the \
1035                 daemon accepted arbitrary tool calls. Set at least one \
1036                 wasm_policy_path or set require_tool_schema=true."
1037            );
1038        }
1039        tracing::warn!(
1040            "wasm_policy_paths is empty; policy gate disabled — every tool \
1041             call bypasses Gate 3. Only require_tool_schema fail-closed \
1042             (Gate 2) remains active"
1043        );
1044    }
1045    Sandbox::new(
1046        SandboxConfig {
1047            schemas,

1105/// Load the Bridge manifest, falling back to the embedded built-in when
1106/// the *default* path is absent (zero-config). An explicitly configured
1107/// path that is missing stays a hard error.
1108fn load_manifest(config: &HarnessConfig) -> Result<BridgeManifest> {
1109    // Capped read: `avctl manifest-validate` already refuses manifests
1110    // above MAX_CONTROL_BYTES, and `BridgeManifest::from_yaml` enforces
1111    // its own 256 KiB cap — an uncapped read here only buys the daemon
1112    // an unbounded allocation before that parse-side cap can reject.
1113    let manifest = match av_core::fsutil::read_capped_string(
1114        std::path::Path::new(&config.bridge_manifest_path),
1115        av_core::fsutil::MAX_CONTROL_BYTES,
1116    ) {
1117        Ok(text) => BridgeManifest::from_yaml(&text).map_err(anyhow::Error::new)?,
1118        Err(error) if error.kind() == std::io::ErrorKind::NotFound && config.uses_default_manifest_path() => {
1119            tracing::info!(
1120                path = %config.bridge_manifest_path,
1121                "Bridge manifest not found; using embedded built-in manifest"
1122            );
1123            BridgeManifest::from_yaml(BUILTIN_MANIFEST_YAML).map_err(anyhow::Error::new)?
1124        }
1125        Err(error) => {
1126            return Err(error)
1127                .with_context(|| format!("read Bridge manifest {}", config.bridge_manifest_path));
1128        }
1129    };
1130    // A topic with no cold_uri DROPS records past
1131    // hot_hours — the audit stream half of the system of record
1132    // silently self-deletes (default 720 h = 30 days) while the ATIF
1133    // spool half retains forever. Surface the divergence at boot so
1134    // an operator who relies on broker replay for compliance knows
1135    // the clock is ticking.
1136    for topic in &manifest.topics {
1137        if topic.retention.cold_uri.is_none() {
1138            tracing::warn!(
1139                topic = %topic.name,
1140                hot_hours = topic.retention.hot_hours,
1141                "topic has no cold_uri: records older than hot_hours are DELETED, not archived — configure retention.cold_uri for a durable audit stream"
1142            );
1143        }
1144    }
1145    Ok(manifest)
1146}

1148fn spawn_bridge_maintenance(
1149    bridge: Arc<dyn EventBus>,
1150    metrics: Arc<av_core::metrics::Registry>,
1151    shutdown: Arc<tokio::sync::Notify>,
1152) -> tokio::task::JoinHandle<()> {
1153    tokio::spawn(async move {
1154        let mut interval = tokio::time::interval(std::time::Duration::from_secs(3600));
1155        // Skip missed ticks under transient overload instead of the
1156        // default catch-up burst. Bridge maintenance is fine to run
1157        // once per hour even if the previous run took 30 minutes.
1158        interval.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Delay);
1159        loop {
1160            // Previously `bridge_maintenance.abort()`
1161            // was used at shutdown, but JoinHandle::abort() cancels
1162            // only the outer async task — the `spawn_blocking`
1163            // closure below cannot be cancelled, so the OS thread
1164            // keeps rewriting bridge segments concurrently with
1165            // `flush_telemetry` and process exit. Race the shutdown
1166            // notify against the tick so the loop can exit cleanly
1167            // between ticks, and also skip a fresh spawn_blocking
1168            // if shutdown fired during the tick's own await.
1169            tokio::select! {
1170                biased;
1171                () = shutdown.notified() => return,
1172                _ = interval.tick() => {}
1173            }
1174            // Supervise the tick body: a panic inside the async wrapper
1175            // (e.g., an allocator failure in tracing::warn's Display
1176            // impl) would otherwise silently kill maintenance, and the
1177            // 1 h cadence hides the outage until Bridge hot retention
1178            // grows unbounded and fills the disk.
1179            let outcome = std::panic::AssertUnwindSafe(async {
1180                let maintenance_bridge = Arc::clone(&bridge);
1181                let result = tokio::task::spawn_blocking(move || {
1182                    maintenance_bridge.maintenance(av_core::time::now_ms())
1183                })
1184                .await;
1185                match result {
1186                    Ok(Ok(actions)) => metrics
1187                        .counter(
1188                            "av_bridge_maintenance_actions_total",
1189                            "Bridge retention expirations and cold-export retries",
1190                        )
1191                        .add(actions),
1192                    Ok(Err(error)) => {
1193                        // Previously only tracing::warn.
1194                        // A silent 1-hour cadence made this class of
1195                        // failure invisible to alerts — Bridge hot
1196                        // retention could grow unbounded until disk
1197                        // fills. Bump a counter alongside the log.
1198                        metrics
1199                            .counter(
1200                                "av_bridge_maintenance_errors_total",
1201                                av_harness::pipeline::supervision_help::BRIDGE_MAINTENANCE_ERRORS,
1202                            )
1203                            .inc();
1204                        tracing::warn!(%error, "Bridge maintenance failed");
1205                    }
1206                    Err(error) => {
1207                        metrics
1208                            .counter(
1209                                "av_bridge_maintenance_join_errors_total",
1210                                av_harness::pipeline::supervision_help::BRIDGE_MAINTENANCE_JOIN_ERRORS,
1211                            )
1212                            .inc();
1213                        tracing::warn!(%error, "Bridge maintenance task failed");
1214                    }
1215                }
1216            })
1217            .catch_unwind()
1218            .await;
1219            if let Err(panic) = outcome {
1220                let msg = panic
1221                    .downcast_ref::<&'static str>()
1222                    .copied()
1223                    .or_else(|| panic.downcast_ref::<String>().map(String::as_str))
1224                    .unwrap_or("panic payload was not a string");
1225                metrics
1226                    .counter(
1227                        "av_bridge_maintenance_panics_total",
1228                        av_harness::pipeline::supervision_help::BRIDGE_MAINTENANCE_PANICS,
1229                    )
1230                    .inc();
1231                tracing::error!(panic = %msg, "Bridge maintenance tick panicked; continuing");
1232            }
1233        }
1234    })
1235}

1237async fn build_identity(
1238    config: &HarnessConfig,
1239    metrics: Arc<av_core::metrics::Registry>,
1240) -> Result<(
1241    Option<Arc<IdentityValidator>>,
1242    Option<tokio::task::JoinHandle<()>>,
1243)> {
1244    let has_jwks = config
1245        .identity_jwks_url
1246        .as_deref()
1247        .is_some_and(|url| !url.is_empty());
1248    let has_hmac = config
1249        .identity_hmac_secret_file
1250        .as_deref()
1251        .is_some_and(|path| !path.is_empty());
1252    if !has_jwks && !has_hmac {
1253        return Ok((None, None));
1254    }
1255
1256    let mut validator = IdentityValidator::new(&config.audience);
1257    if !config.identity_allowed_issuers.is_empty() {
1258        validator.allow_issuers(config.identity_allowed_issuers.clone());
1259    }
1260    // Empty string means "unset" — the same posture `has_hmac` above and
1261    // the JWKS branch below already apply. Without the filter, a config
1262    // carrying a valid `identity_jwks_url` plus `identity_hmac_secret_file
1263    // = ""` passed the `has_hmac`-or-`has_jwks` gate and then died here
1264    // trying to stat "".
1265    if let Some(path) = config
1266        .identity_hmac_secret_file
1267        .as_deref()
1268        .filter(|path| !path.is_empty())
1269    {
1270        require_owner_only_mode(Path::new(path))?;
1271        let secret = std::fs::read(path).with_context(|| format!("read identity HMAC secret {path}"))?;
1272        if secret.is_empty() {
1273            anyhow::bail!("identity HMAC secret file {path} is empty");
1274        }
1275        validator
1276            .add_key(&config.identity_hmac_kid, KeyMaterial::HmacSecret(secret))
1277            .context("install identity HMAC key")?;
1278    }
1279
1280    let validator = Arc::new(validator);
1281    let mut refresh_handle: Option<tokio::task::JoinHandle<()>> = None;
1282    // Empty string means "unset" — same posture as `has_jwks` above.
1283    // Without the filter, `identity_jwks_url = ""` built a client and
1284    // fetched "" at boot, killing startup despite the HMAC path being
1285    // fully configured.
1286    if let Some(url) = config.identity_jwks_url.as_deref().filter(|url| !url.is_empty()) {
1287        // Disable redirects: an IdP that returns 302 to an internal URL would
1288        // let a compromised (or misconfigured) JWKS host pivot the harness
1289        // into an SSRF probe against private services.
1290        //
1291        // Total-request timeout of 10 s guards against slowloris IdPs
1292        // that accept the TCP handshake but never send response bytes
1293        // (or send them one byte per minute). Without it, a single
1294        // stuck fetch pins the refresh loop's `tick()` forever and
1295        // subsequent scheduled refreshes never fire.
1296        // Result: revoked keys stay honored until the next process
1297        // restart, with no scheduled recovery.
1298        let client = reqwest::Client::builder()
1299            .connect_timeout(av_harness::pipeline::HTTP_CONNECT_TIMEOUT)
1300            .timeout(std::time::Duration::from_secs(10))
1301            .redirect(reqwest::redirect::Policy::none())
1302            .build()
1303            .context("build JWKS client")?;
1304        refresh_jwks(&client, url, validator.as_ref()).await?;
1305        let url = url.to_owned();
1306        let validator = Arc::clone(&validator);
1307        let refresh_s = config.identity_jwks_refresh_s;
1308        // Register both counters up-front so their TYPE lines are
1309        // present in `/metrics` even before the first failure — ops
1310        // alerts can wire onto them at boot.
1311        let refresh_errors = metrics.counter(
1312            "av_jwks_refresh_errors_total",
1313            "JWKS refresh HTTP/parse/network failures (per attempt)",
1314        );
1315        let refresh_panics = metrics.counter(
1316            "av_jwks_refresh_panics_total",
1317            "JWKS refresh task panicked; loop supervised via catch_unwind",
1318        );
1319        refresh_handle = Some(tokio::spawn(async move {
1320            // Wrap the whole loop in AssertUnwindSafe so an unexpected
1321            // panic (reqwest bug, malformed JWKS, allocator failure)
1322            // does not silently kill the refresher and leave the harness
1323            // running with a stale key set. If the loop *does* panic, we
1324            // log + count + rebuild the loop.
1325            let mut interval = tokio::time::interval(std::time::Duration::from_secs(refresh_s));
1326            // Skip missed ticks instead of the default catch-up burst
1327            // — if a JWKS fetch takes longer than one interval (rare
1328            // but possible under IdP overload) we should not
1329            // immediately re-fire; another key rotation is minutes
1330            // away.
1331            interval.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Delay);
1332            interval.tick().await;
1333            loop {
1334                interval.tick().await;
1335                // Wrap the WHOLE tick body — refresh + result handling
1336                // + tracing + counter — in catch_unwind, mirroring the
1337                // bridge maintenance pattern. Previously only the
1338                // refresh_jwks call itself was supervised, so a panic
1339                // in the match arm (e.g., allocator failure inside
1340                // tracing::warn's Display, or a refresh_panics.inc()
1341                // fault after the registry Arc has been dropped)
1342                // would silently kill the spawned task and leave keys
1343                // frozen until process restart — with no counter to
1344                // page on.
1345                let refresh_errors = Arc::clone(&refresh_errors);
1346                let refresh_panics_for_error = Arc::clone(&refresh_panics);
1347                let client = &client;
1348                let url = &url;
1349                let validator = validator.as_ref();
1350                let tick = std::panic::AssertUnwindSafe(async move {
1351                    match refresh_jwks(client, url, validator).await {
1352                        Ok(_) => {}
1353                        Err(error) => {
1354                            refresh_errors.inc();
1355                            // Do NOT `%error` here. The
1356                            // anyhow chain in `refresh_jwks` used to
1357                            // embed the JWKS URL in every `.with_context`
1358                            // and every `anyhow::bail!` message; anyhow's
1359                            // Display walks the whole chain and prints
1360                            // it. The corp IdP hostname is enterprise-
1361                            // topology-sensitive (reveals SSO vendor,
1362                            // tenant, region) and does not belong in
1363                            // downstream OTLP sinks. `refresh_jwks` now
1364                            // returns URL-free error text; log the
1365                            // stable classifier only.
1366                            tracing::warn!(
1367                                category = %classify_jwks_error(&error),
1368                                "JWKS refresh failed; retaining previously loaded keys"
1369                            );
1370                        }
1371                    }
1372                });
1373                if let Err(panic) = tick.catch_unwind().await {
1374                    refresh_panics_for_error.inc();
1375                    let msg = panic
1376                        .downcast_ref::<&'static str>()
1377                        .copied()
1378                        .or_else(|| panic.downcast_ref::<String>().map(String::as_str))
1379                        .unwrap_or("panic payload was not a string");
1380                    tracing::error!(
1381                        panic = %msg,
1382                        "JWKS refresh task panicked; retaining previously loaded keys and continuing"
1383                    );
1384                }
1385            }
1386        }));
1387    }
1388    if validator.key_count() == 0 {
1389        anyhow::bail!("identity enforcement configured without any verification keys");
1390    }
1391    Ok((Some(validator), refresh_handle))
1392}

1394async fn refresh_jwks(client: &reqwest::Client, url: &str, validator: &IdentityValidator) -> Result<usize> {
1395    // A typical JWKS is < 10 KB (5–20 keys). Even the largest enterprise
1396    // IdPs stay under a few hundred KB. Cap at 4 MiB so a compromised or
1397    // misconfigured JWKS host cannot return a multi-GB payload and OOM
1398    // the harness before the 10 s request timeout fires: at a fast link
1399    // (1 Gbit/s) 1 second is enough to deliver >100 MB, so the timeout
1400    // alone is not a sufficient defense.
1401    //
1402    // Every error return must be URL-free.
1403    // An earlier fix removed the URL from `.with_context()` prepends,
1404    // but `.context("fetch JWKS")` still wraps a `reqwest::Error`
1405    // whose Display embeds the URL (`error sending request for url
1406    // (...)`) — anyhow's Display walks the whole chain and prints
1407    // the inner reqwest error verbatim. Convert the reqwest error
1408    // to a URL-free anyhow::Error using `reqwest::Error::without_url`
1409    // before the context wrap so no downstream `%error` on the boot
1410    // path or future logger can leak the IdP hostname. Keep the
1411    // stripped reqwest error as the anyhow *source* (not a
1412    // stringified message) so `classify_jwks_error`'s downcast can
1413    // still label timeout/connect/body/status failures.
1414    const MAX_JWKS_BYTES: usize = 4 * 1024 * 1024;
1415    let response = client
1416        .get(url)
1417        .send()
1418        .await
1419        .map_err(|error| anyhow::Error::new(error.without_url()).context("fetch JWKS"))?
1420        .error_for_status()
1421        .map_err(|error| {
1422            let status = error
1423                .status()
1424                .map(|s| s.as_u16().to_string())
1425                .unwrap_or_else(|| "unknown".into());
1426            anyhow::Error::new(error.without_url()).context(format!("JWKS endpoint returned status {status}"))
1427        })?;
1428    // Fast reject: if Content-Length is present and already exceeds the
1429    // cap, refuse without allocating anything for the body.
1430    if let Some(len) = response.content_length() {
1431        if len > MAX_JWKS_BYTES as u64 {
1432            anyhow::bail!("JWKS declared Content-Length {len} bytes; cap is {MAX_JWKS_BYTES}");
1433        }
1434    }
1435    use futures::StreamExt as _;
1436    let mut stream = response.bytes_stream();
1437    let mut body = Vec::new();
1438    while let Some(chunk) = stream.next().await {
1439        let chunk =
1440            chunk.map_err(|error| anyhow::Error::new(error.without_url()).context("read JWKS chunk"))?;
1441        let next = body
1442            .len()
1443            .checked_add(chunk.len())
1444            .ok_or_else(|| anyhow::anyhow!("JWKS body size overflowed usize"))?;
1445        if next > MAX_JWKS_BYTES {
1446            anyhow::bail!("JWKS exceeded {MAX_JWKS_BYTES} bytes (received at least {next})");
1447        }
1448        body.extend_from_slice(&chunk);
1449    }
1450    let document: serde_json::Value = serde_json::from_slice(&body).context("parse JWKS JSON")?;
1451    validator.add_jwks(&document).map_err(anyhow::Error::new)
1452}

1454/// Stable string classifier for JWKS refresh failures
1455/// suitable for structured logging without echoing the URL.
1456fn classify_jwks_error(error: &anyhow::Error) -> &'static str {
1457    if let Some(reqwest_err) = error.downcast_ref::<reqwest::Error>() {
1458        if reqwest_err.is_timeout() {
1459            "timeout"
1460        } else if reqwest_err.is_connect() {
1461            "connect"
1462        } else if reqwest_err.is_body() {
1463            "body"
1464        } else if reqwest_err.status().is_some() {
1465            "status"
1466        } else {
1467            "request"
1468        }
1469    } else if error.to_string().contains("exceeded") {
1470        "oversize"
1471    } else if error.to_string().contains("parse JWKS JSON") {
1472        "parse"
1473    } else {
1474        "other"
1475    }
1476}

1478fn build_bridge(config: &HarnessConfig, manifest: &BridgeManifest) -> Result<Arc<dyn EventBus>> {
1479    // `config.bridge()` is the same single parse site `validate()`
1480    // delegates to, so the factory can never disagree with pre-flight:
1481    // an unknown selector or a missing companion was already refused
1482    // at load, and the Kafka/Nats variants carry their endpoint by
1483    // construction.
1484    match config.bridge().map_err(anyhow::Error::msg)? {
1485        BridgeBackend::Embedded => {
1486            // An endpoint without the
1487            // backend that consumes it is a silent misconfiguration —
1488            // e.g. `docker run -e AV_BRIDGE_ENDPOINT=…` against the
1489            // shipped embedded-backend config does nothing. Warn so
1490            // 12-factor operators see why their override is inert.
1491            if config.bridge_endpoint.is_some() {
1492                tracing::warn!(
1493                    "bridge_endpoint (or AV_BRIDGE_ENDPOINT) is set but bridge_backend=\"embedded\" \
1494                     ignores it; set bridge_backend=\"kafka\" or \"nats\" to use the endpoint"
1495                );
1496            }
1497            let path = PathBuf::from(&config.bridge_data_dir);
1498            // Boot-time only, and only in the DAEMON (the single
1499            // writer): sweep temps orphaned by a crash between
1500            // `create_new` and `rename`. This must NOT live inside
1501            // `EmbeddedBroker::open` — `open()` itself is also
1502            // daemon-only now (`avctl event-tail` reads via the
1503            // mutation-free `fetch_read_only`), but keeping the sweep
1504            // here preserves the boot-once discipline.
1505            match av_core::fsutil::sweep_orphaned_tmp(&path) {
1506                Ok(0) => {}
1507                Ok(removed) => {
1508                    tracing::info!(
1509                        removed,
1510                        "removed crash-orphaned temp files from the bridge data dir"
1511                    );
1512                }
1513                Err(error) => {
1514                    tracing::warn!(%error, "bridge orphaned-temp sweep failed; stale .tmp files may remain");
1515                }
1516            }
1517            let bridge = if path.join("manifest.yaml").exists() {
1518                EmbeddedBroker::open(&path)
1519            } else {
1520                EmbeddedBroker::provision(&path, manifest)
1521            }
1522            .context("initialize embedded Bridge")?;
1523            Ok(Arc::new(bridge))
1524        }
1525        BridgeBackend::Kafka { endpoint } => {
1526            #[cfg(feature = "kafka")]
1527            {
1528                let bus = tokio::task::block_in_place(|| {
1529                    av_bridge::kafka_bus::KafkaBus::provision(&endpoint, manifest)
1530                })
1531                .context("initialize Kafka/Redpanda Bridge")?;
1532                Ok(Arc::new(bus))
1533            }
1534            #[cfg(not(feature = "kafka"))]
1535            {
1536                let _ = endpoint;
1537                anyhow::bail!(
1538                    "kafka backend requested but this agentvisord binary was built without the `kafka` feature (rebuild av-harness with --features kafka or full)"
1539                )
1540            }
1541        }
1542        BridgeBackend::Nats { endpoint } => {
1543            #[cfg(feature = "nats")]
1544            {
1545                let bus = tokio::task::block_in_place(|| {
1546                    av_bridge::nats_bus::NatsBus::provision(&endpoint, manifest)
1547                })
1548                .context("initialize NATS JetStream Bridge")?;
1549                Ok(Arc::new(bus))
1550            }
1551            #[cfg(not(feature = "nats"))]
1552            {
1553                let _ = endpoint;
1554                anyhow::bail!(
1555                    "nats backend requested but this agentvisord binary was built without the `nats` feature (rebuild av-harness with --features nats or full)"
1556                )
1557            }
1558        }
1559    }
1560}

1562fn build_store(config: &HarnessConfig) -> Result<Arc<dyn StateStore>> {
1563    match config.state().map_err(anyhow::Error::msg)? {
1564        StateBackend::Memory => {
1565            // Warn on an inert endpoint —
1566            // budget/velocity counters stay process-local (neither
1567            // shared nor durable), exactly what setting a Redis
1568            // endpoint was meant to fix.
1569            if config.state_endpoint.is_some() {
1570                tracing::warn!(
1571                    "state_endpoint (or AV_STATE_ENDPOINT) is set but state_backend=\"memory\" \
1572                     ignores it; set state_backend=\"redis\" to use the endpoint"
1573                );
1574            }
1575            Ok(Arc::new(InMemoryStore::new()))
1576        }
1577        StateBackend::Redis { endpoint } => {
1578            #[cfg(feature = "redis")]

1656fn load_or_create_signer(path: &Path) -> Result<(Ed25519Signer, bool)> {
1657    match read_signer(path) {
1658        Ok(signer) => return Ok((signer, false)),
1659        Err(error)
1660            if error
1661                .downcast_ref::<std::io::Error>()
1662                .is_some_and(|error| error.kind() == std::io::ErrorKind::NotFound) => {}
1663        Err(error) => return Err(error),
1664    }
1665    let signer = Ed25519Signer::generate();
1666    // `signer.seed()` returns
1667    // `Zeroizing<[u8; 32]>` directly, so no temp slot lingers on
1668    // the caller's stack. The hex encoding is separately wrapped
1669    // in Zeroizing so its heap buffer is zeroed on drop too.
1670    let seed = signer.seed();
1671    // `&*seed` (not `*seed`) is required to avoid
1672    // copying the seed bytes onto a fresh un-zeroized temp slot
1673    // for hex::encode. clippy's `needless_borrows_for_generic_args`
1674    // lint is a false positive here — it would suggest `*seed`,
1675    // moving 32 bytes out of the Zeroizing wrapper.
1676    #[allow(clippy::needless_borrows_for_generic_args)]
1677    let encoded_seed = zeroize::Zeroizing::new(hex::encode(&*seed));
1678    if install_seed_exclusive(path, &encoded_seed)? {
1679        Ok((signer, true))
1680    } else {
1681        // A concurrent process installed a seed while we generated ours;
1682        // pick up its key — but this is NOT a fresh-anchor event from our
1683        // point of view, so mark newly_generated = false.
1684        let existing = read_signer(path).context("load signing seed installed by another process")?;
1685        Ok((existing, false))
1686    }
1687}

1689/// Reject a file whose mode is readable by group or other on Unix, and
1690/// refuse to follow a symbolic link at `path` — a pre-planted symlink to
1691/// an attacker-owned 0o600 hex file would otherwise fool the mode check
1692/// (`std::fs::metadata` follows symlinks and returns the *target*'s mode)
1693/// and let `read_to_string` load an attacker-chosen seed. Windows does
1694/// not model POSIX modes so this is a no-op there — deployments on
1695/// Windows must rely on ACLs set by the operator.
1696fn require_owner_only_mode(path: &Path) -> Result<()> {
1697    #[cfg(unix)]
1698    {
1699        use std::os::unix::fs::MetadataExt as _;
1700        // `symlink_metadata` does NOT traverse a symbolic link, so the
1701        // file-type check below is applied to the link itself, not to
1702        // whatever the link happens to point at.
1703        let metadata = std::fs::symlink_metadata(path)
1704            .with_context(|| format!("stat secret file {}", path.display()))?;
1705        let file_type = metadata.file_type();
1706        if file_type.is_symlink() {
1707            anyhow::bail!(
1708                "secret file {} is a symbolic link; refusing to follow (remove the link and store the seed inline at this path)",
1709                path.display(),
1710            );
1711        }
1712        if !file_type.is_file() {
1713            anyhow::bail!(
1714                "secret file {} is not a regular file (type: {file_type:?})",
1715                path.display(),
1716            );
1717        }
1718        // Bottom 9 bits are rwxrwxrwx; any group/other bit set (not
1719        // just read — mode must be exactly owner-only) is refused.
1720        let mode = metadata.mode() & 0o777;
1721        if mode & 0o077 != 0 {
1722            anyhow::bail!(
1723                "secret file {} has mode 0o{mode:03o}; must be 0o600 (chmod 600 {})",
1724                path.display(),
1725                path.display(),
1726            );
1727        }
1728    }
1729    #[cfg(not(unix))]
1730    {
1731        let _ = path;
1732    }
1733    Ok(())
1734}

1736fn read_signer(path: &Path) -> Result<Ed25519Signer> {
1737    require_owner_only_mode(path)?;
1738    // Consistent posture with the CLI's
1739    // read_capped_str — the harness should not use uncapped
1740    // read_to_string on a security-sensitive file just because
1741    // require_owner_only_mode already refused the group/other-
1742    // readable case. A hex-encoded 32-byte seed is 65 bytes with
1743    // newline; MAX_CONTROL_BYTES (1 MiB) is a generous ceiling.
1744    // Wrap the seed intermediates in `Zeroizing<...>`
1745    // so their memory is zeroed on drop rather than leaking a
1746    // recoverable copy in freed heap / stack slots (visible to a
1747    // core dump, minidump upload, or kdump crashkernel image).
1748    // ed25519-dalek 2.2+ with the `zeroize` feature also zeroes
1749    // SigningKey's internal buffer on drop.
1750    use zeroize::Zeroizing;
1751    let encoded = Zeroizing::new(
1752        av_core::fsutil::read_capped_string(path, av_core::fsutil::MAX_CONTROL_BYTES)
1753            .with_context(|| format!("read signing seed {}", path.display()))?,
1754    );
1755    let bytes = Zeroizing::new(hex::decode(encoded.trim()).context("decode signing seed as hex")?);
1756    // Copy directly from the Zeroizing<Vec<u8>> slice
1757    // into a fresh Zeroizing<[u8; 32]>. Historically we did
1758    // `(*bytes).clone().try_into()` which materialized a bare
1759    // `Vec<u8>` intermediate and (on `try_into` failure) an
1760    // `Err(Vec<u8>)` — both unzeroed. `<[u8;32]>::try_from(&[u8])`
1761    // takes only a slice reference and copies bytes straight into
1762    // the caller-supplied slot, so the intermediate is scoped to
1763    // the Zeroizing wrapper for its entire lifetime.
1764    let seed: Zeroizing<[u8; 32]> = Zeroizing::new(
1765        <[u8; 32]>::try_from(bytes.as_slice())
1766            .map_err(|_| anyhow::anyhow!("signing seed must contain exactly 32 bytes"))?,
1767    );
1768    // Refuse known-weak Ed25519 seeds. An all-zero seed
1769    // produces a valid keypair with a globally-known public key — any
1770    // attacker who knows we accepted `[0; 32]` can forge receipts
1771    // that verify against our key id. Same for the all-`0xFF` seed
1772    // (both are the "textbook wrong" values operators sometimes
1773    // paste in when troubleshooting). Fail fast at startup so the
1774    // hazard cannot ship silently.
1775    if *seed == [0u8; 32] {
1776        anyhow::bail!(
1777            "signing seed at {} is all zeros; refusing (this is a known-weak seed with a globally predictable public key)",
1778            path.display()
1779        );
1780    }
1781    if *seed == [0xFFu8; 32] {
1782        anyhow::bail!(
1783            "signing seed at {} is all 0xFF bytes; refusing (this is a known-weak seed with a globally predictable public key)",
1784            path.display()
1785        );
1786    }
1787    Ok(Ed25519Signer::from_seed(&seed))
1788}

1790fn install_seed_exclusive(path: &Path, encoded: &str) -> Result<bool> {
1791    use std::io::Write as _;
1792    let parent = path
1793        .parent()
1794        .filter(|parent| !parent.as_os_str().is_empty())
1795        .unwrap_or_else(|| Path::new("."));
1796    // A bare `create_dir_all` fsyncs no dirent. If `parent` was newly
1797    // created (first-boot install), the ancestor dirents stay
1798    // volatile: the later `sync_directory(parent)` fsyncs the
1799    // CONTENTS of `parent`, not the `parent` dirent inside the
1800    // grandparent. A power loss immediately after install could drop
1801    // the whole `keys/` directory even though the seed file itself
1802    // was fsynced — the next boot would generate a fresh key with a
1803    // different public identity and every already-issued receipt
1804    // would fail signature verification. Route the mkdir through
1805    // `create_dir_all_synced` (which fsyncs every newly-created
1806    // ancestor and sets mode 0o700 atomically at mkdir on Unix,
1807    // closing the shared-tenant enumeration window until the
1808    // seed-file mode gets applied) — same posture as
1809    // `av_core::fsutil::write_atomic`.
1810    av_core::fsutil::create_dir_all_synced(parent)
1811        .with_context(|| format!("create signing seed directory {}", parent.display()))?;
1812    let temporary = parent.join(format!(".signing-seed-{}.tmp", av_core::new_event_uid()));
1813    // Previously an early `?` return from write_all or
1814    // sync_all would orphan the temp file. The two `Err(...)` match
1815    // arms of `hard_link` explicitly removed it, but the earlier
1816    // failure paths did not. Use the same TempPathGuard RAII
1817    // that `write_atomic` uses so every failure path
1818    // unlinks — this is a startup-only code path, but leaving an
1819    // orphan means every subsequent boot leaves another zero-byte
1820    // `.signing-seed-*.tmp` alongside the real seed, and a nervous
1821    // operator debugging a "signing seed exists twice" symptom is
1822    // easily led astray.
1823    let mut guard = av_core::fsutil::TempPathGuard::new(temporary.clone());
1824    let mut options = std::fs::OpenOptions::new();
1825    options.write(true).create_new(true);
1826    #[cfg(unix)]
1827    {
1828        use std::os::unix::fs::OpenOptionsExt as _;
1829        options.mode(0o600);
1830    }
1831    let mut file = options
1832        .open(&temporary)
1833        .with_context(|| format!("create signing seed temporary file {}", temporary.display()))?;
1834    file.write_all(encoded.as_bytes())
1835        .with_context(|| format!("write signing seed temporary file {}", temporary.display()))?;
1836    file.sync_all()
1837        .with_context(|| format!("sync signing seed temporary file {}", temporary.display()))?;
1838    match std::fs::hard_link(&temporary, path) {
1839        Ok(()) => {
1840            // The seed IS installed at `path` at this
1841            // point (hard_link committed). Degrade the remaining
1842            // best-effort ops (tmp unlink, parent fsync) to warn
1843            // rather than returning Err — otherwise a spurious EIO on
1844            // sync_directory made the harness fail startup even
1845            // though the seed was correctly installed, wasting one
1846            // boot cycle to a misleading error. On next boot,
1847            // hard_link → AlreadyExists → Ok(false) and the caller
1848            // reads back the seed — self-corrects, but the noisy
1849            // failure is now avoided at source.
1850            match std::fs::remove_file(&temporary) {
1851                // Only a successful unlink may disarm the guard — an
1852                // unconditional disarm made "guard drop will retry" a
1853                // lie and orphaned a mode-0600 copy of the seed.
1854                Ok(()) => guard.disarm(),
1855                Err(error) => {
1856                    tracing::warn!(
1857                        path = %av_core::fsutil::basename(&temporary),
1858                        %error,
1859                        "signing seed installed, but removing tmp file failed; guard drop will retry"
1860                    );
1861                }
1862            }
1863            if let Err(error) = av_core::fsutil::sync_directory(parent) {
1864                tracing::warn!(
1865                    dir = %av_core::fsutil::basename(parent),
1866                    %error,
1867                    "signing seed installed, but parent directory fsync failed; dirent may not survive an immediate power loss"
1868                );
1869            }
1870            Ok(true)
1871        }
1872        Err(error) if error.kind() == std::io::ErrorKind::AlreadyExists => {
1873            // Guard drop below unlinks the tmp — no explicit remove
1874            // needed. Same for the generic Err arm.
1875            Ok(false)
1876        }
1877        Err(error) => Err(error).with_context(|| format!("install signing seed {}", path.display())),
1878    }
1879}

1881async fn shutdown_signal() {
1882    #[cfg(unix)]
1883    {
1884        // Install SIGTERM first: docker stop / kubectl delete pod send
1885        // SIGTERM (not SIGINT) and expect the process to shut down
1886        // cleanly within its grace period. If we cannot install a
1887        // SIGTERM handler, no orchestrator signal will trigger the
1888        // finalizer — the container silently ignores shutdown until
1889        // it gets SIGKILL, dropping in-flight receipts.
1890        //
1891        // This is a fatal environment misconfiguration (typically
1892        // over-restrictive seccomp), not something to paper over with
1893        // SIGINT-only fallback. Panic so the runtime exits with a
1894        // non-zero status the orchestrator can log.
1895        let mut terminate = match tokio::signal::unix::signal(tokio::signal::unix::SignalKind::terminate()) {
1896            Ok(signal) => signal,
1897            Err(error) => {
1898                tracing::error!(
1899                    %error,
1900                    "failed to install SIGTERM handler — orchestrator shutdowns will not be observed. \
1901                     This is a fatal environment misconfiguration (over-restrictive seccomp, missing \
1902                     signal syscalls). Aborting so the container is not silently unresponsive to \
1903                     `docker stop` / `kubectl delete pod`."
1904                );
1905                std::process::exit(2);
1906            }
1907        };
1908        // Install SIGHUP as an explicitly-ignored signal. The default
1909        // action for SIGHUP on Unix is `Term` — the process gets
1910        // killed immediately, no finalize_sessions, no worker drain,
1911        // no receipt persistence. Real-world triggers that would
1912        // otherwise crash-terminate the daemon: `systemctl reload
1913        // agentvisord` (systemd's default `ExecReload` sends SIGHUP),
1914        // a lost controlling terminal in an interactive `docker exec`
1915        // session, or an operator's `pkill -HUP` muscle memory from
1916        // other daemons that reload on SIGHUP. The daemon does NOT
1917        // support config-reload on SIGHUP (see OPERATIONS.md); this
1918        // handler drains the signal into a no-op so that stance is
1919        // enforced by construction. Installation failure is
1920        // non-fatal (unlike SIGTERM) — the worst case is a returned
1921        // to pre-fix behavior on this ONE signal, and we log a warn.
1922        if let Ok(mut hup) = tokio::signal::unix::signal(tokio::signal::unix::SignalKind::hangup()) {
1923            tokio::spawn(async move {
1924                while hup.recv().await.is_some() {
1925                    tracing::info!(
1926                        "SIGHUP received; ignored (this daemon does not reload config on SIGHUP — see OPERATIONS.md 'Shutdown' section)"
1927                    );
1928                }
1929            });
1930        } else {
1931            tracing::warn!(
1932                "failed to install SIGHUP handler; SIGHUP will kill the daemon per Unix default action \
1933                 (Term). Investigate seccomp / signal syscall restrictions if this appears."
1934            );
1935        }
1936        // R75-F1 (landed R76): Notify-based sequential-consume
1937        // shutdown-signal design with a 100ms coalescing window.
1938        //
1939        // Prior R74 shape had the outer `select!` consume the
1940        // first signal, then a drain loop popped any queued
1941        // permits BEFORE moving the receivers into a force-exit
1942        // spawn. Two failure modes remained: (a) a genuinely-new
1943        // signal arriving during the drain window was silently
1944        // swallowed; (b) the drain vs "concurrent SIGINT+SIGTERM"
1945        // trade-off had no distinction between "operator intent:
1946        // one signal, OS delivered a second concurrently" and
1947        // "operator intent: two signals, force-exit".
1948        //
1949        // Fix: ONE spawn owns both signal receivers and consumes
1950        // them sequentially. On the first signal, it notifies
1951        // the outer via `first_signal_ready` (which returns from
1952        // `wait_for_shutdown` so `finish_shutdown` can begin
1953        // graceful drain). It then sleeps 100 ms and drains any
1954        // additional permits queued during the coalescing window
1955        // — these are treated as PART of the first-signal event
1956        // (concurrent SIGINT+SIGTERM burst, muscle-memory
1957        // Ctrl-C-Ctrl-C within OS scheduler jitter). Only a
1958        // signal arriving AFTER the 100 ms window trips
1959        // exit(130) — matching operator intent for
1960        // "double Ctrl-C to force-exit" (typically 200-500 ms
1961        // between presses).
1962        //
1963        // Concurrent SIGINT+SIGTERM within 100 ms → drained →
1964        // no spurious exit; the graceful drain runs uninterrupted.
1965        // Genuine second signal at t>100ms → captured →
1966        // exit(130) as documented.
1967        let first_signal_ready = Arc::new(tokio::sync::Notify::new());
1968        let notify_outer = Arc::clone(&first_signal_ready);
1969        let mut interrupt = tokio::signal::unix::signal(tokio::signal::unix::SignalKind::interrupt()).ok();
1970        tokio::spawn(async move {
1971            // Consume the FIRST signal.
1972            tokio::select! {
1973                _ = async {
1974                    if let Some(ref mut sig) = interrupt {
1975                        sig.recv().await;
1976                    } else {
1977                        // No SIGINT receiver — mirror the prior shape's
1978                        // `tokio::signal::ctrl_c()` fallback. If ctrl_c()
1979                        // itself errors, log and pend so terminate can
1980                        // still fire.
1981                        if let Err(error) = tokio::signal::ctrl_c().await {
1982                            tracing::error!(%error, "failed to listen for Ctrl-C");
1983                            std::future::pending::<()>().await;
1984                        }
1985                    }
1986                } => {}
1987                _ = terminate.recv() => {}
1988            }
1989            // Wake the outer (main-thread) shutdown flow.
1990            notify_outer.notify_one();
1991            // 100 ms coalescing window: signals arriving in this
1992            // window are treated as part of the same operator
1993            // intent as the first signal (concurrent SIGINT+SIGTERM
1994            // burst, or OS scheduler jitter reordering a single
1995            // logical signal).
1996            tokio::time::sleep(std::time::Duration::from_millis(100)).await;
1997            loop {
1998                let mut drained_any = false;
1999                if let Some(ref mut sig) = interrupt {
2000                    if let Ok(Some(())) =
2001                        tokio::time::timeout(std::time::Duration::from_millis(0), sig.recv()).await
2002                    {
2003                        drained_any = true;
2004                    }
2005                }
2006                if let Ok(Some(())) =
2007                    tokio::time::timeout(std::time::Duration::from_millis(0), terminate.recv()).await
2008                {
2009                    drained_any = true;
2010                }
2011                if !drained_any {
2012                    break;
2013                }
2014            }
2015            // Any signal from HERE on is a genuine second-signal
2016            // event, indicating operator intent to force-exit.
2017            tokio::select! {
2018                _ = async {
2019                    if let Some(ref mut sig) = interrupt {
2020                        sig.recv().await;
2021                    } else {
2022                        std::future::pending::<()>().await;
2023                    }
2024                } => {}
2025                _ = terminate.recv() => {}
2026            }
2027            tracing::warn!(
2028                "second shutdown signal received during graceful shutdown; forcing exit(130) \
2029                 — any in-flight receipt or ATIF write will be picked up by recovery on restart"
2030            );
2031            std::process::exit(130);
2032        });
2033        first_signal_ready.notified().await;
2034    }
2035    #[cfg(not(unix))]
2036    if let Err(error) = tokio::signal::ctrl_c().await {
2037        tracing::error!(%error, "failed to listen for Ctrl-C");
2038    }
2039}
