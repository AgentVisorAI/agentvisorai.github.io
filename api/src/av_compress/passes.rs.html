6/// Tuning knobs (config-file surface; defaults follow the brief).
7#[derive(Debug, Clone)]
8pub struct CompressionConfig {
9    /// Never touch the last N messages (short-term memory preservation).
10    pub keep_tail: usize,
11    /// Tool outputs older than the tail are stubbed when longer than this
12    /// many approximate tokens.
13    pub tool_output_stub_threshold: u64,
14    /// Also collapse exact-duplicate user/assistant messages
15    /// (tool messages are never collapsed).
16    pub collapse_duplicates: bool,
17    /// Normalize whitespace inside JSON-looking tool/assistant string content.
18    pub normalize_json: bool,
19    /// Only bother when the payload reaches at least this many approximate
20    /// tokens (compressing tiny payloads wastes hot-path time).
21    pub min_tokens_to_engage: u64,
22    /// Enable audited middle-history stubbing when other passes cannot reach
23    /// the target on very large histories. Only engages when the payload
24    /// reaches at least 50 000 approximate tokens (hardcoded floor; trumps
25    /// `min_tokens_to_engage` for this pass).
26    pub summarize_middle: bool,
27    /// Minimum target reduction ratio times 1000.
28    pub target_reduction_millis: u64,
29}
30
31impl Default for CompressionConfig {
32    fn default() -> Self {
33        Self {
34            keep_tail: 8,
35            tool_output_stub_threshold: 128,
36            collapse_duplicates: true,
37            normalize_json: true,
38            min_tokens_to_engage: 512,
39            summarize_middle: true,
40            target_reduction_millis: 300,
41        }
42    }
43}

45/// Result of a compression run.
46#[derive(Debug, Clone)]
47pub struct CompressionOutcome {
48    /// The (possibly) compressed payload.
49    pub payload: Value,
50    /// Approximate tokens before.
51    pub tokens_before: u64,
52    /// Approximate tokens after.
53    pub tokens_after: u64,
54    /// Whether any pass modified the payload.
55    pub changed: bool,
56}
57
58impl CompressionOutcome {
59    /// Pruned token count.
60    pub fn pruned_tokens(&self) -> u64 {
61        self.tokens_before.saturating_sub(self.tokens_after)
62    }
63
64    /// Reduction ratio ×1000 (350 = 35.0 %), integer for JCS-exact metrics.
65    pub fn pruning_ratio_millis(&self) -> u64 {
66        if self.tokens_before == 0 {
67            return 0;
68        }
69        self.pruned_tokens().saturating_mul(1000) / self.tokens_before
70    }
71}

73/// Compress a chat payload. Non-chat or unparsable payloads are returned
74/// unchanged (`changed = false`) — this function must never fail the hot path.
75pub fn compress(payload: &Value, cfg: &CompressionConfig) -> CompressionOutcome {
76    let tokens_before = approx_tokens(&payload.to_string());
77    let Some(messages) = payload.get("messages").and_then(Value::as_array) else {
78        return CompressionOutcome {
79            payload: payload.clone(),
80            tokens_before,
81            tokens_after: tokens_before,
82            changed: false,
83        };
84    };
85    if tokens_before < cfg.min_tokens_to_engage || messages.is_empty() {
86        return CompressionOutcome {
87            payload: payload.clone(),
88            tokens_before,
89            tokens_after: tokens_before,
90            changed: false,
91        };
92    }
93
94    let mut out = messages.clone();
95    let tail_start = out.len().saturating_sub(cfg.keep_tail);
96
97    let mut changed = false;
98    // Normalization must run BEFORE the duplicate-collapse passes:
99    // it rewrites assistant/tool content (JSON minification), so two
100    // semantically identical messages that differ only in formatting
101    // become byte-identical only AFTER this pass. With dedupe first,
102    // run 1 kept both, normalized them equal, and run 2 then collapsed
103    // one — violating invariant #4, `compress(compress(x)) ==
104    // compress(x)` (lib.rs). Normalize-first is stable: run 2's
105    // normalize is a no-op on already-minified content, so every later
106    // pass sees the same input it saw in run 1's output.
107    if cfg.normalize_json {
108        changed |= normalize_json_content(&mut out, tail_start);
109    }
110    changed |= collapse_duplicate_system(&mut out, tail_start);
111    if cfg.collapse_duplicates {
112        changed |= collapse_duplicate_messages(&mut out, tail_start);
113    }
114    changed |= stub_stale_tool_outputs(&mut out, tail_start, cfg.tool_output_stub_threshold);
115    if cfg.summarize_middle && tokens_before >= 50_000 {
116        changed |= stub_middle_to_target(
117            payload,
118            &mut out,
119            tail_start,
120            tokens_before,
121            cfg.target_reduction_millis,
122        );
123    }
124
125    if !changed {
126        return CompressionOutcome {
127            payload: payload.clone(),
128            tokens_before,
129            tokens_after: tokens_before,
130            changed: false,
131        };
132    }
133    let mut new_payload = payload.clone();
134    if let Some(obj) = new_payload.as_object_mut() {
135        obj.insert("messages".to_owned(), Value::Array(out));
136    }
137    let tokens_after = approx_tokens(&new_payload.to_string());
138    // Guard: a pass must never grow the payload; if it somehow did, keep the original.
139    if tokens_after > tokens_before {
140        return CompressionOutcome {
141            payload: payload.clone(),
142            tokens_before,
143            tokens_after: tokens_before,
144            changed: false,
145        };
146    }
147    CompressionOutcome {
148        payload: new_payload,
149        tokens_before,
150        tokens_after,
151        changed: true,
152    }
153}

155fn stub_middle_to_target(
156    payload: &Value,
157    messages: &mut [Value],
158    tail_start: usize,
159    tokens_before: u64,
160    target_reduction_millis: u64,
161) -> bool {
162    // Bound the idempotence scan to the middle range only. A legitimate tail
163    // message that happens to quote the marker (a follow-up assistant reply
164    // summarizing a prior compression, or user-controlled content) would
165    // otherwise permanently disable this pass on that conversation.
166    //
167    // TODO(compression-marker): the marker itself is still an
168    // unauthenticated literal substring, so a hostile MIDDLE-range
169    // message can still spoof it and skip stubbing of surrounding
170    // messages. Fixing that requires switching to a keyed marker
171    // (HMAC over prior content) or an out-of-band per-payload
172    // flag — both are larger refactors.
173    //
174    // Tracked as a known limitation (see the
175    // compression-marker entry in SECURITY-AUDIT.md). Attack requires
176    // an already-compromised prior turn (either an operator who
177    // pasted attacker content verbatim, or a system prompt
178    // hardening bypass) — real-world exploitability is bounded to
179    // "attacker who already has message-content control can force
180    // compression pass to skip." A keyed-marker
181    // scheme remains future work, currently unscheduled.
182    let scan_end = tail_start.min(messages.len());
183    // The prior check refused to run a
184    // second pass if ANY pre-tail message content contained the
185    // literal substring "reason: middle history]" — but a
186    // user/assistant message that legitimately quotes that phrase
187    // (or a compromised prior turn that includes it as free text)
188    // could silently disable compression. Narrow the check to
189    // exactly what the marker produces: content must START with
190    // "[pruned:" AND contain the marker tail. A user quoting the
191    // phrase in free text no longer matches (their content will
192    // start with their own words, not "[pruned:"). This is not
193    // the full keyed-marker fix (a known limitation) —
194    // but the spoofing surface shrinks from "any message contains
195    // the substring" to "any message perfectly mimics the marker
196    // prefix + tail." Realistic user text does not shape like a
197    // machine-emitted `[pruned: N tokens, sha256:HEX, reason: middle
198    // history]`.
199    if messages.get(..scan_end).into_iter().flatten().any(|message| {
200        msg_content_str(message).is_some_and(|content| {
201            content.starts_with("[pruned:") && content.contains("reason: middle history]")
202        })
203    }) {
204        return false;
205    }
206    let target_tokens =
207        tokens_before.saturating_mul(1000u64.saturating_sub(target_reduction_millis.min(1000))) / 1000;
208    let mut changed = false;
209    // The prior loop called
210    // `payload_tokens_with_messages(payload, messages)` on every
211    // iteration — each call clones the entire payload AND the
212    // entire messages Vec, then serializes to string. For a 4 MiB
213    // payload with N messages that's O(N²) work AND O(N) transient
214    // allocations per iteration. Instead track a running
215    // `current_tokens` counter that is decremented by the message's
216    // pre-stub token count and incremented by the stub's on each
217    // successful substitution. `payload_tokens_with_messages` is
218    // still called once at loop entry to seed the counter (the
219    // envelope's token count includes non-messages fields like
220    // `model` and `stream`, which the incremental delta preserves).
221    let mut current_tokens = payload_tokens_with_messages(payload, messages);
222    for index in 0..tail_start {
223        if current_tokens <= target_tokens {
224            break;
225        }
226        let Some(message) = messages.get_mut(index) else {
227            continue;
228        };
229        let role = msg_role(message);
230        if role == "system" || role == "tool" || message.get("tool_calls").is_some() {
231            continue;
232        }
233        let Some(content) = msg_content_str(message) else {
234            continue;
235        };
236        if content.starts_with("[pruned:") {
237            continue;
238        }
239        let tokens = approx_tokens(content);
240        if tokens < 32 {
241            continue;
242        }
243        // The running counter was seeded from the
244        // SERIALIZED payload (JSON-escaped: `\n` → two chars, `"` →
245        // `\"`), but decrements used raw-content tokens. Raw ≤ escaped
246        // always, so the loop over-stated remaining tokens and kept
247        // stubbing past the target — destroying messages that should
248        // have survived on code/JSON-heavy content. Measure both the
249        // removed content and the added stub in serialized units.
250        let serialized_tokens =
251            approx_tokens(&serde_json::to_string(&Value::String(content.to_owned())).unwrap_or_default());
252        let digest = av_core::digest::sha256_hex(content.as_bytes());
253        let stub_content = format!("[pruned: {tokens} tokens, sha256:{digest}, reason: middle history]");
254        let stub_serialized_tokens =
255            approx_tokens(&serde_json::to_string(&Value::String(stub_content.clone())).unwrap_or_default());
256        if let Some(object) = message.as_object_mut() {
257            object.insert("content".to_owned(), Value::String(stub_content));
258            // Update the running counter: original message content
259            // contributed `serialized_tokens`, new stub contributes
260            // `stub_serialized_tokens`. Everything else in the payload
261            // envelope is unchanged.
262            current_tokens = current_tokens
263                .saturating_sub(serialized_tokens)
264                .saturating_add(stub_serialized_tokens);
265            changed = true;
266        }
267    }
268    changed
269}

271fn payload_tokens_with_messages(payload: &Value, messages: &[Value]) -> u64 {
272    let mut value = payload.clone();
273    if let Some(object) = value.as_object_mut() {
274        object.insert("messages".to_owned(), Value::Array(messages.to_vec()));
275    }
276    approx_tokens(&value.to_string())

287/// Keep the first system message; collapse later *identical* system messages
288/// (duplicate re-injection is a common agent-framework bug).
289fn collapse_duplicate_system(messages: &mut [Value], tail_start: usize) -> bool {
290    let Some(first_system) = messages.iter().position(|m| msg_role(m) == "system") else {
291        return false;
292    };
293    let reference = messages.get(first_system).cloned();
294    let Some(reference) = reference else { return false };
295    let mut changed = false;
296    for (i, m) in messages.iter_mut().enumerate() {
297        if i <= first_system || i >= tail_start {
298            continue;
299        }
300        if msg_role(m) == "system" && *m == reference {
301            let tokens = approx_tokens(&reference.to_string());
302            let stub = audit_stub("duplicate system message", tokens, &reference);
303            // Stub only when it shrinks the
304            // message — a short duplicate would otherwise GROW the
305            // payload and trip the outer revert-guard into cancelling
306            // the whole compression outcome.
307            if approx_tokens(&stub) >= tokens {
308                continue;
309            }
310            *m = json!({
311                "role": "system",
312                "content": stub,
313            });
314            changed = true;
315        }
316    }
317    changed
318}

320/// Collapse exact-duplicate user/assistant messages outside the tail
321/// (identical retries / framework echoes).
322///
323/// A `HashSet` keyed by content hash replaced
324/// the O(n²) `Vec::contains`, hardened twofold:
325///
326/// 1. **Eligibility**: only messages whose ONLY fields are `role` and
327///    `content` participate. The previous `(role, content)` key
328///    collapsed messages that differed in `name` (multi-participant),
329///    `refusal`, legacy `function_call`, or `audio` — permanently
330///    destroying the distinguishing field (e.g. WHICH named user said
331///    "approve"), violating the crate's "never semantically
332///    destructive" invariant.
333/// 2. **Verification**: the 64-bit hash is a fast filter only; a hash
334///    hit is confirmed with full `Value` equality against the first
335///    occurrence before stubbing. `DefaultHasher` is unkeyed, so a
336///    collision is offline-searchable by anyone who controls message
337///    content — treating a collision as equality silently destroyed a
338///    non-duplicate.
339///
340/// Also applies the stub-size floor: a duplicate is
341/// only stubbed when the stub is actually smaller, so collapsing many
342/// short duplicates can no longer GROW the payload and trip the outer
343/// revert-guard into cancelling all compression for the conversation.
344fn collapse_duplicate_messages(messages: &mut [Value], tail_start: usize) -> bool {
345    use std::collections::HashMap;
346    use std::hash::{DefaultHasher, Hash, Hasher};
347    // hash → index of first occurrence, for equality confirmation.
348    let mut seen: HashMap<u64, usize> = HashMap::new();
349    let mut changed = false;
350    for i in 0..tail_start.min(messages.len()) {
351        let Some(m) = messages.get(i) else { continue };
352        let role = msg_role(m).to_owned();
353        if role != "user" && role != "assistant" {
354            continue;
355        }
356        // Only collapse messages carrying NOTHING beyond role+content —
357        // any extra field (name, refusal, function_call, audio, …) is
358        // semantically distinguishing and must survive verbatim.
359        let only_role_and_content = m
360            .as_object()
361            .is_some_and(|obj| obj.len() == 2 && obj.contains_key("role") && obj.contains_key("content"));
362        if !only_role_and_content {
363            continue;
364        }
365        let Some(content) = msg_content_str(m) else {
366            continue;
367        };
368        // Never re-collapse audit stubs (stub-of-stub would break idempotence).
369        if content.starts_with("[pruned:") {
370            continue;
371        }
372        let mut hasher = DefaultHasher::new();
373        role.hash(&mut hasher);
374        content.hash(&mut hasher);
375        let key = hasher.finish();
376        match seen.get(&key) {
377            None => {
378                seen.insert(key, i);
379            }
380            Some(&first_index) => {
381                // Confirm with real equality — a hash collision must
382                // never destroy a non-duplicate.
383                let is_true_duplicate = messages
384                    .get(first_index)
385                    .is_some_and(|first| messages.get(i).is_some_and(|current| first == current));
386                if !is_true_duplicate {
387                    continue;
388                }
389                let Some(m) = messages.get_mut(i) else { continue };
390                let tokens = approx_tokens(content_of(m).unwrap_or_default());
391                let stub = audit_stub("duplicate message", tokens, m);
392                // Stub-size floor: replacing a 1-token "ok" with a
393                // ~30-token stub grows the payload.
394                if approx_tokens(&stub) >= tokens {
395                    continue;
396                }
397                *m = json!({ "role": role, "content": stub });
398                changed = true;
399            }
400        }
401    }
402    changed
403}

410/// Replace large, stale tool outputs with audit stubs (the brief's "stale tool
411/// responses"). The `tool_call_id` linkage field is preserved so the
412/// conversation graph stays intact.
413fn stub_stale_tool_outputs(messages: &mut [Value], tail_start: usize, threshold: u64) -> bool {
414    let mut changed = false;
415    for (i, m) in messages.iter_mut().enumerate() {
416        if i >= tail_start {
417            break;
418        }
419        if msg_role(m) != "tool" {
420            continue;
421        }
422        let Some(content) = msg_content_str(m) else {
423            continue;
424        };
425        if content.starts_with("[pruned:") {
426            continue; // already stubbed — idempotence
427        }
428        let tokens = approx_tokens(content);
429        if tokens <= threshold {
430            continue;
431        }
432        let digest = av_core::digest::sha256_hex(content.as_bytes());
433        let stub = format!("[pruned: {tokens} tokens, sha256:{digest}]");
434        if let Some(obj) = m.as_object_mut() {
435            obj.insert("content".to_owned(), Value::String(stub));
436            changed = true;
437        }
438    }
439    changed
440}

442/// Minify JSON-formatted string content (strip redundant JSON formatting per
443/// the brief) outside the tail.
444fn normalize_json_content(messages: &mut [Value], tail_start: usize) -> bool {
445    let mut changed = false;
446    for (i, m) in messages.iter_mut().enumerate() {
447        if i >= tail_start {
448            break;
449        }
450        let role = msg_role(m);
451        if role != "tool" && role != "assistant" {
452            continue;
453        }
454        let Some(content) = msg_content_str(m) else {
455            continue;
456        };
457        let trimmed = content.trim_start();
458        if !(trimmed.starts_with('{') || trimmed.starts_with('[')) {
459            continue;
460        }
461        let Ok(parsed) = serde_json::from_str::<Value>(content) else {
462            continue;
463        };
464        let Ok(minified) = serde_json::to_string(&parsed) else {
465            continue;
466        };
467        // The round-trip through `Value` is
468        // lossy for (a) integers outside i64/u64 range — parsed as
469        // f64 and re-serialized with precision loss (128-bit IDs,
470        // large monetary integers common in tool outputs) — and (b)
471        // duplicate keys, collapsed last-wins. Both silently corrupt
472        // tool data the model then reasons over, violating the
473        // "never semantically destructive" invariant. Detect by
474        // re-parsing the minified form and requiring Value-equality
475        // AND matching numeric-literal token streams; skip the pass
476        // for content that does not round-trip exactly.
477        let round_trips = serde_json::from_str::<Value>(&minified)
478            .map(|reparsed| reparsed == parsed)
479            .unwrap_or(false)
480            && numeric_literals(content) == numeric_literals(&minified)
481            && !has_duplicate_keys(content);
482        if !round_trips {
483            continue;
484        }
485        if minified.len() < content.len() {
486            if let Some(obj) = m.as_object_mut() {
487                obj.insert("content".to_owned(), Value::String(minified));
488                changed = true;
489            }
490        }
491    }
492    changed
493}

495fn audit_stub(reason: &str, tokens: u64, original: &Value) -> String {
496    let digest = av_core::digest::sha256_hex(original.to_string().as_bytes());
497    format!("[pruned: {tokens} tokens ({reason}), sha256:{digest}]")
498}

500/// Minification-proof helpers: a single-pass JSON lexer extracting the
501/// ordered numeric-literal tokens and the object-key count from RAW
502/// text (no Value round-trip). Used to prove a minification did not
503/// mutate numeric representations (bigint precision loss) or collapse
504/// duplicate keys.
505fn numeric_literals(raw: &str) -> Vec<String> {
506    lex_json(raw).0
507}
508
509fn has_duplicate_keys(raw: &str) -> bool {
510    // If minification would drop keys, the raw text's key count exceeds
511    // the reparse's. Compare against the canonical no-dup serialization.
512    let Ok(parsed) = serde_json::from_str::<Value>(raw) else {
513        return true; // unparseable — treat conservatively
514    };
515    let Ok(minified) = serde_json::to_string(&parsed) else {
516        return true;
517    };
518    lex_json(raw).1 != lex_json(&minified).1
519}
520
521/// Returns (ordered numeric literals, object-key count).
522fn lex_json(raw: &str) -> (Vec<String>, usize) {
523    let mut numbers = Vec::new();
524    let mut key_count = 0usize;
525    let bytes = raw.as_bytes();
526    let mut i = 0;
527    while let Some(&byte) = bytes.get(i) {
528        match byte {
529            b'"' => {
530                // Skip string, handling escapes; then check for a
531                // following ':' (object key).
532                i += 1;
533                while let Some(&inner) = bytes.get(i) {
534                    match inner {
535                        b'\\' => i += 2,
536                        b'"' => {
537                            i += 1;
538                            break;
539                        }
540                        _ => i += 1,
541                    }
542                }
543                let mut j = i;
544                while bytes.get(j).is_some_and(u8::is_ascii_whitespace) {
545                    j += 1;
546                }
547                if bytes.get(j) == Some(&b':') {
548                    key_count += 1;
549                }
550            }
551            b'-' | b'0'..=b'9' => {
552                let start = i;
553                i += 1;
554                while bytes
555                    .get(i)
556                    .is_some_and(|b| matches!(b, b'0'..=b'9' | b'.' | b'e' | b'E' | b'+' | b'-'))
557                {
558                    i += 1;
559                }
560                if let Some(token) = raw.get(start..i) {
561                    numbers.push(token.to_owned());
562                }
563            }
564            _ => i += 1,
565        }
566    }
567    (numbers, key_count)
568}

625    /// The duplicate-key minification-skip guard is the semantic-

722    /// Mutation-run hardening (round 7): the escape-handling arms of
723    /// `lex_json` had surviving mutants — no test lexed content whose
724    /// strings carry `\"` escapes. The guard must count keys and
725    /// numeric literals correctly through escaped quotes, or dup-key
726    /// content could minify (semantic destruction) and bigint tokens
727    /// could silently mutate.
728    #[test]
729    fn lex_json_handles_escaped_quotes() {
730        // The escaped quote must not terminate the string: the `:9`
731        // inside is string content, not a numeric literal, and the
732        // whole thing is ONE key.
733        let (numbers, keys) = lex_json(r#"{"a\":9":1}"#);
734        assert_eq!(numbers, vec!["1".to_owned()]);
735        assert_eq!(keys, 1);
736        // Structurally rich exact-tuple pin: multiple keys, an escaped
737        // quote inside a key, numbers in values and arrays, and a
738        // digit+colon INSIDE string content — any index-arithmetic slip
739        // in the lexer shifts this tuple.
740        let (numbers, keys) = lex_json(r#"{"x":12,"y\":":3.5e2,"n:1":"t","z":[7,-8]}"#);
741        assert_eq!(
742            numbers,
743            vec![
744                "12".to_owned(),
745                "3.5e2".to_owned(),
746                "7".to_owned(),
747                "-8".to_owned()
748            ]
749        );
750        assert_eq!(keys, 4);
751        // Escape NOT adjacent to a quote (`\n` inside a key whose
752        // content carries `:7`): index-arithmetic slips in the escape
753        // arm re-lex this into phantom numeric literals.
754        let (numbers, keys) = lex_json(r#"{"a\nb:7":1}"#);
755        assert_eq!(numbers, vec!["1".to_owned()]);
756        assert_eq!(keys, 1);
757        // Whitespace between the closing quote and the colon exercises
758        // the lookahead scan; a slip there miscounts keys and turns
759        // every spaced object into a phantom duplicate-key verdict.
760        assert!(!has_duplicate_keys(r#"{"k" : 1, "j" : 2}"#));
761        let (_, keys) = lex_json(r#"{"k" : 1, "j" : 2}"#);
762        assert_eq!(keys, 2);
763        // Duplicate keys that contain escaped quotes must still be
764        // detected (and therefore left unminified end-to-end).
765        assert!(has_duplicate_keys(r#"{"k\"":1, "k\"":2}"#));
766        assert!(!has_duplicate_keys(r#"{"k\"":1, "j\"":2}"#));
767        let dup = r#"{"k\"": 1,  "k\"": 2,  "pad": "padding padding padding"}"#;
768        let mut messages = vec![
769            json!({"role": "assistant", "content": dup}),
770            json!({"role": "user", "content": "tail"}),
