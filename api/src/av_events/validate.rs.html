2//! against) the shipped JSON Schema.

6/// Upper bound (inclusive) for the `time` field on emitted events,
7/// in milliseconds since the Unix epoch.
8///
9/// `253_402_300_799_999` = `9999-12-31T23:59:59.999Z`. The shipped
10/// schema pins `time_iso` to a 4-digit year via its `pattern`, so
11/// any `time` above this bound would render a 6-digit year the
12/// schema rejects. Historically the schema advertised
13/// `2^53` (`JCS_SAFE_MAX`) as the upper bound while the Rust
14/// validator applied this tighter year-10000 cap first — a
15/// split-brain the R51/R52 schema-drift pass closed by aligning the
16/// schema to the same bound. Exposed as a `pub const` so tests can
17/// pin the schema-vs-code equality (see `schema_conformance.rs`)
18/// and so future changes to one side without the other fail CI.
19pub const SCHEMA_TIME_MAX_MS: u64 = 253_402_300_799_999;
20
21/// A structural validation failure.
22#[derive(Debug, thiserror::Error, PartialEq, Eq)]
23#[non_exhaustive]
24pub enum ValidationError {
25    /// A required string field was empty.
26    #[error("required field `{0}` is empty")]
27    EmptyField(&'static str),
28    /// `unmapped` carries inbound-tolerance fields
29    /// that re-serialize to the top level, but the shipped schema
30    /// declares `additionalProperties: false` — such an event passes the
31    /// Rust validator yet fails the broker's schema gate at publish
32    /// (which fail-closes the session). The tolerance is INBOUND-ONLY;
33    /// republishing an event that carries unmapped fields is refused
34    /// here so both validators agree.
35    #[error("event carries {0} unmapped field(s); inbound tolerance is not publishable")]
36    UnmappedNotPublishable(usize),
37    /// `type_uid` doesn't match `class_uid * 100 + activity_id`.
38    #[error("type_uid {type_uid} != class_uid {class_uid} * 100 + activity_id {activity_id}")]
39    TypeUidMismatch {
40        /// Stated type uid.
41        type_uid: u64,
42        /// Stated class uid.
43        class_uid: u32,
44        /// Stated activity id.
45        activity_id: u8,
46    },
47    /// `class_uid` doesn't match the class enum.
48    #[error("class_uid {0} does not match class_name")]
49    ClassUidMismatch(u32),
50    /// Event is not in the OCSF Application Activity category.
51    #[error("category_uid {0} is not Application Activity (6)")]
52    BadCategory(u8),
53    /// Charter is not represented as an OCSF Regular File.
54    #[error("ai_agent.charter.type_id {0} is not Regular File (1)")]
55    BadCharterType(u8),
56    /// Severity outside 1–6.
57    #[error("severity_id {0} outside 1..=6")]
58    BadSeverity(u8),
59    /// Status outside 0–2.
60    #[error("status_id {0} outside 0..=2")]
61    BadStatus(u8),
62    /// Activity_id must be < 100 or the
63    /// `class_uid × 100 + activity_id` bijection collapses:
64    /// `class_uid=9901, activity_id=100` produces the same
65    /// `type_uid = 990200` as `class_uid=9902, activity_id=0`,
66    /// so downstream SIEM pipelines routing/aggregating by
67    /// type_uid mis-classify. Not currently reachable via the
68    /// builder (defaults 1), but `OcsfEvent`'s fields are all pub
69    /// and the struct is not `#[non_exhaustive]`, so callers can
70    /// construct or mutate events directly, bypassing the builder;
71    /// enforce the bijection
72    /// invariant at validate time.
73    #[error("activity_id {0} outside 0..=99 (would collide type_uid namespaces)")]
74    BadActivityId(u8),
75    /// stop_reason caption present without id (or vice versa).
76    #[error("stop_reason and stop_reason_id must be present together")]
77    StopReasonPairMismatch,
78    /// Stop_reason_id maps to a KNOWN `StopReason` variant,
79    /// but the accompanying caption does not match that variant's
80    /// canonical caption. The two fields split downstream analytics
81    /// (SIEM pipelines that group by id disagree with dashboards that
82    /// group by caption) — reject the disagreement so both fields
83    /// mean the same thing.
84    #[error("stop_reason_id {id} maps to {expected_caption:?} but stop_reason is {actual_caption:?}")]
85    StopReasonCaptionMismatch {
86        /// Stated stop reason id.
87        id: u8,
88        /// Canonical caption for that id.
89        expected_caption: &'static str,
90        /// Actual caption that was emitted with the id.
91        actual_caption: String,
92    },
93    /// A numeric field exceeds `av_core::error::JCS_SAFE_MAX`
94    /// (2^53) on the deserialize path. `OcsfEventBuilder::build`
95    /// already refuses values above the JCS-safe integer range, but
96    /// wire deserialization (`serde_json::from_slice::<OcsfEvent>`)
97    /// bypassed the guard — an issuer sending `prompt_tokens = 2^53+1`
98    /// would round-trip through JS-based auditors (JSON.parse) as a
99    /// silently-truncated value, losing 1 or more low bits and
100    /// breaking any receipt hash computed by JS consumers.
101    #[error("field {field} = {value} exceeds JCS-safe integer range 2^53")]
102    JcsUnsafeInteger {
103        /// Field name.
104        field: &'static str,
105        /// Offending value.
106        value: u64,
107    },
108    /// A numeric field is out of its
109    /// schema-declared range. Currently only `pruning_ratio_millis`
110    /// (permille ratio, capped at 1000) enforces this, but the
111    /// variant is variant-generic so future range-bound fields can
112    /// reuse it. Parallel to `JcsUnsafeInteger` — build() and
113    /// `validate_event` (the deserialize backstop) both enforce the
114    /// bound, closing the emitter/validator drift the wire-side path
115    /// used to have.
116    #[error("field {field} = {value} exceeds schema max {max}")]
117    OutOfSchemaRange {
118        /// Field name.
119        field: &'static str,
120        /// Offending value.
121        value: u64,
122        /// Declared inclusive max.
123        max: u64,
124    },
125    /// Timestamp is zero.
126    #[error("time is zero")]
127    ZeroTime,
128    /// Schema parity: `stop_reason_id` outside the schema enum
129    /// {0..=4, 90..=94, 99}.
130    #[error("stop_reason_id {0} is not a known reason id")]
131    UnknownStopReasonId(u8),
132    /// Schema parity: `metadata.version` must be the const "1.10.0".
133    #[error("metadata.version {0:?} is not the pinned OCSF version")]
134    BadOcsfVersion(String),
135    /// Schema parity: fingerprint constants (SHA-256/JCS/64-hex value).
136    #[error("{0} fingerprint violates the SHA-256/JCS/hex-64 shape")]
137    BadFingerprint(&'static str),
138    /// Schema parity: `payload` maxProperties 512.
139    #[error("payload carries {0} properties, above the schema max of 512")]
140    PayloadTooManyProperties(usize),
141    /// Schema parity: `time_iso`'s pattern requires a 4-digit year, so
142    /// epoch-ms beyond 9999-12-31 renders a caption the schema rejects.
143    #[error("time {0} is beyond 9999-12-31 (time_iso year would exceed 4 digits)")]
144    TimeBeyondSchemaRange(u64),
145    /// `time_iso` does not match the canonical ISO-8601 rendering of
146    /// `time`. The two fields are dual representations of one instant;
147    /// a mismatch means a signed/chained event names different times to
148    /// human auditors (caption) and machine tooling (epoch-ms). Same
149    /// class as the receipt's `issued_at`/`issued_at_iso` cross-check.
150    #[error("time_iso {time_iso:?} does not match time ({time}); expected {expected_iso:?}")]
151    TimeIsoMismatch {
152        /// Epoch-milliseconds timestamp.
153        time: u64,
154        /// Caption that was actually emitted.
155        time_iso: String,
156        /// Canonical rendering of `time`.
157        expected_iso: String,
158    },
159}

161/// Validate structural invariants of an event. Returns *all* violations
162/// (collect-all-errors, matching the Harbor validator philosophy).
163pub fn validate_event(ev: &OcsfEvent) -> Result<(), Vec<ValidationError>> {
164    let mut errors = Vec::new();
165    // See UnmappedNotPublishable.
166    if !ev.unmapped.is_empty() {
167        errors.push(ValidationError::UnmappedNotPublishable(ev.unmapped.len()));
168    }
169    if ev.metadata.uid.is_empty() {
170        errors.push(ValidationError::EmptyField("metadata.uid"));
171    }
172    if ev.metadata.version.is_empty() {
173        errors.push(ValidationError::EmptyField("metadata.version"));
174    }
175    if ev.metadata.product.name.is_empty() {
176        errors.push(ValidationError::EmptyField("metadata.product.name"));
177    }
178    if ev.metadata.product.vendor_name.is_empty() {
179        errors.push(ValidationError::EmptyField("metadata.product.vendor_name"));
180    }
181    if ev.metadata.product.version.is_empty() {
182        errors.push(ValidationError::EmptyField("metadata.product.version"));
183    }
184    if ev.session_uid.is_empty() {
185        errors.push(ValidationError::EmptyField("session_uid"));
186    }
187    if ev.ai_agent.version.is_empty() {
188        errors.push(ValidationError::EmptyField("ai_agent.version"));
189    }
190    if ev.ai_agent.charter.name.is_empty() {
191        errors.push(ValidationError::EmptyField("ai_agent.charter.name"));
192    }
193    if ev.ai_agent.charter.type_id != 1 {
194        errors.push(ValidationError::BadCharterType(ev.ai_agent.charter.type_id));
195    }
196    if ev.ai_agent.instance_uid.is_empty() {
197        errors.push(ValidationError::EmptyField("ai_agent.instance_uid"));
198    }
199    if ev.class_uid != ev.class_name.class_uid() {
200        errors.push(ValidationError::ClassUidMismatch(ev.class_uid));
201    }
202    if ev.category_uid != crate::model::CATEGORY_UID {
203        errors.push(ValidationError::BadCategory(ev.category_uid));
204    }
205    let expected_type = u64::from(ev.class_name.class_uid()) * 100 + u64::from(ev.activity_id);
206    if ev.type_uid != expected_type {
207        errors.push(ValidationError::TypeUidMismatch {
208            type_uid: ev.type_uid,
209            class_uid: ev.class_uid,
210            activity_id: ev.activity_id,
211        });
212    }
213    // The `class_uid × 100 + activity_id` invariant
214    // above is only a bijection when activity_id < 100. Enforce
215    // that at validate time so an adversarial event asserting
216    // `class_uid=9901, activity_id=100, type_uid=990200` (which
217    // is self-consistent per the mismatch check but collides
218    // with `class_uid=9902, activity_id=0`) cannot slip through
219    // and mis-route downstream SIEM aggregation by type_uid.
220    if ev.activity_id >= 100 {
221        errors.push(ValidationError::BadActivityId(ev.activity_id));
222    }
223    if !(1..=6).contains(&ev.severity_id) {
224        errors.push(ValidationError::BadSeverity(ev.severity_id));
225    }
226    if ev.status_id > 2 {
227        errors.push(ValidationError::BadStatus(ev.status_id));
228    }
229    if ev.stop_reason_id.is_some() != ev.stop_reason.is_some() {
230        errors.push(ValidationError::StopReasonPairMismatch);
231    }
232    // Schema parity (validator/schema agreement is this module's stated
233    // purpose — see the module doc): the shipped schema pins
234    // `stop_reason_id` to the enum {0..=4, 90..=94, 99}; an id like 50
235    // previously passed here yet failed the broker's schema gate.
236    if let Some(id) = ev.stop_reason_id {
237        if !matches!(id, 0..=4 | 90..=94 | 99) {
238            errors.push(ValidationError::UnknownStopReasonId(id));
239        }
240    }
241    // Schema parity: `metadata.version` is `const "1.10.0"`.
242    if ev.metadata.version != crate::model::OCSF_VERSION {
243        errors.push(ValidationError::BadOcsfVersion(ev.metadata.version.clone()));
244    }
245    // Schema parity: fingerprint constants (algorithm 3/"SHA-256",
246    // serialization 2/"JCS", 64-hex value) — previously entirely
247    // unvalidated on the wire path.
248    for (field, fingerprint) in [
249        ("inventory", ev.inventory.as_ref()),
250        ("prev_inventory", ev.prev_inventory.as_ref()),
251    ] {
252        let Some(fp) = fingerprint else { continue };
253        let hex64 = fp.value.len() == 64
254            && fp
255                .value
256                .bytes()
257                .all(|b| b.is_ascii_hexdigit() && !b.is_ascii_uppercase());
258        if fp.algorithm_id != 3
259            || fp.algorithm != "SHA-256"
260            || fp.serialization_id != 2
261            || fp.serialization != "JCS"
262            || !hex64
263        {
264            errors.push(ValidationError::BadFingerprint(field));
265        }
266    }
267    // Schema parity: `payload` maxProperties 512.
268    if let Some(map) = ev.payload.as_object() {
269        if map.len() > 512 {
270            errors.push(ValidationError::PayloadTooManyProperties(map.len()));
271        }
272    }
273    // Match the shipped JSON schema's
274    // `"stop_reason": {"type": "string", "minLength": 1}`. Some
275    // OpenAI-compatible shims emit `finish_reason: ""` mid-stream; the
276    // parse-site filter in routes.rs::parse_provider_chunk drops that,
277    // but this defense-in-depth check ensures any other emitter path
278    // (`stop_reason_native`, direct builder use) also cannot produce
279    // an event that would pass the Rust validator here yet fail the
280    // schema validator at publish — which permanently fail-closes the
281    // session AFTER the bytes are relayed.
282    if ev.stop_reason.as_deref() == Some("") {
283        errors.push(ValidationError::EmptyField("stop_reason"));
284    }
285    // The initial
286    // fix required `StopReason::from_id(id).caption() == stop_reason`
287    // whenever id was known. That's WRONG — the field docstring at
288    // `model.rs::OcsfEvent::stop_reason` and the builder API
289    // `OcsfEventBuilder::stop_reason_native` both document
290    // `stop_reason` as "the provider's native value when captured".
291    // So `id=1, caption="stop"` (OpenAI native), `id=1,
292    // caption="end_turn"` (Anthropic native), `id=99, caption="Custom
293    // Free Text"` (Other) are ALL legitimate emissions from the
294    // official builder. The over-strict check would have rejected the
295    // vast majority of production events.
296    //
297    // Narrow to the specific cross-wiring case that started this
298    // finding: the caption is ITSELF a canonical caption for a
299    // DIFFERENT known variant. `id=93 (PolicyBlocked)` +
300    // `caption="Loop Detected"` (canonical caption of variant 91) is
301    // an unambiguous mistake — the two fields disagree on which
302    // enforcement fired. Provider-native captions like `"stop"` or
303    // `"tool_calls"` are not canonical captions of any variant, so
304    // they pass. The harness's `map_finish_reason` in routes.rs uses the
305    // lowercase provider tokens; none of those coincide with a
306    // canonical caption (which are Title-cased: `"Stop"`, `"Tool
307    // Use"`, `"Length"`, `"Content Filter"`, …).
308    if let (Some(id), Some(caption)) = (ev.stop_reason_id, ev.stop_reason.as_deref()) {
309        let expected = crate::StopReason::from_id(id).caption();
310        let is_known_id = matches!(id, 0..=4 | 90..=94 | 99);
311        let caption_is_canonical_of_other_variant = is_known_id
312            && expected != caption
313            && [
314                crate::StopReason::Unknown,
315                crate::StopReason::Stop,
316                crate::StopReason::MaxTokens,
317                crate::StopReason::ToolUse,
318                crate::StopReason::SessionClosed,
319                crate::StopReason::ContentFilter,
320                crate::StopReason::LoopDetected,
321                crate::StopReason::BudgetExceeded,
322                crate::StopReason::PolicyBlocked,
323                crate::StopReason::IdentityRejected,
324                crate::StopReason::Other,
325            ]
326            .iter()
327            .any(|other| other.caption() == caption);
328        if caption_is_canonical_of_other_variant {
329            errors.push(ValidationError::StopReasonCaptionMismatch {
330                id,
331                expected_caption: expected,
332                actual_caption: caption.to_owned(),
333            });
334        }
335    }
336    if ev.time == 0 {
337        errors.push(ValidationError::ZeroTime);
338    }
339    // Schema parity: the `time_iso` pattern pins a 4-digit year. A
340    // `time` near 2^53 is JCS-safe and consistent with its own caption
341    // (the cross-check below), but renders a 6-digit year the schema
342    // pattern rejects. `SCHEMA_TIME_MAX_MS` = 9999-12-31T23:59:59.999Z.
343    if ev.time > SCHEMA_TIME_MAX_MS {
344        errors.push(ValidationError::TimeBeyondSchemaRange(ev.time));
345    }
346    // `time_iso` is the second representation of the event's timestamp;
347    // the schema pins its exact format and `OcsfEventBuilder::build`
348    // derives it from `time`. But the wire/deserialize path never
349    // checked the two agree, so a signed/chained event could carry a
350    // `time` and a `time_iso` naming different instants (or a garbage
351    // caption) and still validate — auditors reading the caption and
352    // tooling reading the epoch field would extract contradictory facts
353    // from the same event. Same dual-representation class as the
354    // receipt's issued_at/issued_at_iso cross-check; strict equality
355    // against the canonical rendering enforces format and consistency
356    // in one step.
357    {
358        let expected_iso = av_core::time::iso8601_ms(ev.time);
359        if ev.time_iso != expected_iso {
360            errors.push(ValidationError::TimeIsoMismatch {
361                time: ev.time,
362                time_iso: ev.time_iso.clone(),
363                expected_iso,
364            });
365        }
366    }
367    // JCS-safe integer guard on deserialize path.
368    // `OcsfEventBuilder::build` already enforces this for its own
369    // outputs, but a wire event bypasses build(). Values above
370    // `av_core::error::JCS_SAFE_MAX` (2^53) round-trip through
371    // JS-based JSON parsers as silently-truncated values,
372    // breaking any receipt hash a JS auditor computes over the
373    // event. Cover the fields build() covers: `time`, `seq`, and
374    // every `EventMetrics` counter.
375    for (field, value) in [("time", ev.time), ("metadata.sequence", ev.metadata.sequence)] {
376        if av_core::error::check_jcs_safe(value).is_err() {
377            errors.push(ValidationError::JcsUnsafeInteger { field, value });
378        }
379    }
380    // `ai_agent.ttl_remaining_s` is a schema-typed
381    // u64 with no maximum in the shipped JSON schema. Any value above
382    // 2^53 silently truncates in JS-based auditors and hard-fails
383    // `av_receipts::canonicalize`. `OcsfEventBuilder::build`'s
384    // check_jcs_safe loop already covers the fields listed above; add
385    // ttl_remaining_s here so the wire/deserialize path also refuses.
386    // (`OcsfEvent.ai_agent.ttl_remaining_s` was omitted from the
387    // original JCS-safety sweep.)
388    if let Some(ttl) = ev.ai_agent.ttl_remaining_s {
389        if av_core::error::check_jcs_safe(ttl).is_err() {
390            errors.push(ValidationError::JcsUnsafeInteger {
391                field: "ai_agent.ttl_remaining_s",
392                value: ttl,
393            });
394        }
395    }
396    if let Some(m) = &ev.metrics {
397        for (field, value) in [
398            ("metrics.prompt_tokens", m.prompt_tokens),
399            ("metrics.completion_tokens", m.completion_tokens),
400            ("metrics.cached_tokens", m.cached_tokens),
401            ("metrics.pruned_tokens", m.pruned_tokens),
402            ("metrics.pruning_ratio_millis", m.pruning_ratio_millis),
403        ] {
404            if let Some(v) = value {
405                if av_core::error::check_jcs_safe(v).is_err() {
406                    errors.push(ValidationError::JcsUnsafeInteger { field, value: v });
407                }
408            }
409        }
410        // Mirror the `> 1000` guard from
411        // `OcsfEventBuilder::build`. The schema declares
412        // `pruning_ratio_millis` in `[0, 1000]`; this validator was
413        // introduced as the wire-side backstop for
414        // build's checks, and build later gained the range check too. Adding it here closes the emitter/validator drift
415        // — an event with `pruning_ratio_millis = 5000` now fails
416        // both paths, not just build.
417        if let Some(v) = m.pruning_ratio_millis {
418            if v > 1000 {
419                errors.push(ValidationError::OutOfSchemaRange {
420                    field: "metrics.pruning_ratio_millis",
421                    value: v,
422                    max: 1000,
423                });
424            }
425        }
426    }
427    if errors.is_empty() {
428        Ok(())
429    } else {
430        Err(errors)
431    }
432}

469    /// Pin the fingerprint schema-parity block (previously entirely
470    /// unexercised — every mutation of it survived): a canonical
471    /// SHA-256/JCS fingerprint passes, and each individual violation
472    /// (wrong algorithm id/caption, wrong serialization id/caption,
473    /// non-64-hex value, uppercase hex) is refused on both the
474    /// `inventory` and `prev_inventory` fields.
475    #[test]
476    fn fingerprint_constants_are_enforced_per_field() {
477        use crate::model::Fingerprint;
478        let good = || Fingerprint::sha256_jcs("ab".repeat(32));
479        let mut ev = valid_event();
480        ev.inventory = Some(good());
481        ev.prev_inventory = Some(good());
482        assert!(validate_event(&ev).is_ok(), "canonical fingerprints must pass");
483
484        type Violation = Box<dyn Fn(&mut Fingerprint)>;
485        let violations: Vec<Violation> = vec![
486            Box::new(|f| f.algorithm_id = 2),
487            Box::new(|f| f.algorithm = "SHA-512".into()),
488            Box::new(|f| f.serialization_id = 1),
489            Box::new(|f| f.serialization = "CBOR".into()),
490            Box::new(|f| f.value = "ab".repeat(31)),
491            Box::new(|f| f.value = "zz".repeat(32)),
492            Box::new(|f| f.value = "AB".repeat(32)),
493        ];
494        for (index, violate) in violations.iter().enumerate() {
495            let mut ev = valid_event();
496            let mut fp = good();
497            violate(&mut fp);
498            ev.inventory = Some(fp);
499            let errors = validate_event(&ev).unwrap_err();
500            assert!(
501                errors
502                    .iter()
503                    .any(|e| matches!(e, ValidationError::BadFingerprint("inventory"))),
504                "violation #{index} on inventory must be refused: {errors:?}"
505            );
506            let mut ev = valid_event();
507            let mut fp = good();
508            violate(&mut fp);
509            ev.prev_inventory = Some(fp);
510            let errors = validate_event(&ev).unwrap_err();
511            assert!(
512                errors
513                    .iter()
514                    .any(|e| matches!(e, ValidationError::BadFingerprint("prev_inventory"))),
515                "violation #{index} on prev_inventory must be refused: {errors:?}"
516            );
517        }
518    }

520    /// Pin the payload maxProperties and time-range schema-parity caps
521    /// at their exact boundaries.
522    #[test]
523    fn payload_property_and_time_range_boundaries() {
524        let mut ev = valid_event();
525        let mut payload = serde_json::Map::new();
526        for index in 0..512 {
527            payload.insert(format!("k{index}"), serde_json::Value::Null);
528        }
529        ev.payload = serde_json::Value::Object(payload.clone());
530        assert!(validate_event(&ev).is_ok(), "512 properties is the inclusive max");
531        payload.insert("k512".into(), serde_json::Value::Null);
532        ev.payload = serde_json::Value::Object(payload);
533        assert!(
534            validate_event(&ev)
535                .unwrap_err()
536                .iter()
537                .any(|e| matches!(e, ValidationError::PayloadTooManyProperties(513))),
538            "513 properties must be refused"
539        );
540
541        let mut ev = valid_event();
542        ev.time = 253_402_300_799_999; // 9999-12-31T23:59:59.999Z
543        ev.time_iso = av_core::time::iso8601_ms(ev.time);

670    /// Regression guard: the deserialize path bypasses
671    /// `OcsfEventBuilder::build`'s JCS-safe integer check, so a
672    /// wire event with a token counter above 2^53 would round-trip
673    /// through JS-based JSON parsers as silently-truncated,
674    /// breaking any receipt hash JS consumers compute over the
675    /// event. `validate_event` must flag it on every JCS-safe
676    /// carrier field: `time`, `metadata.sequence`, and every
677    /// `EventMetrics.*_tokens`.
678    #[test]
679    fn jcs_unsafe_integer_flagged_on_every_deserialize_carrier() {
680        let unsafe_value = av_core::error::JCS_SAFE_MAX + 1;
681
682        // time
683        let mut ev = valid_event();
684        ev.time = unsafe_value;
685        let errs = validate_event(&ev).unwrap_err();
686        assert!(
687            errs.iter()
688                .any(|e| matches!(e, ValidationError::JcsUnsafeInteger { field: "time", .. })),
689            "time above JCS_SAFE_MAX must be flagged, got {errs:?}"
690        );
691
692        // metadata.sequence
693        let mut ev = valid_event();
694        ev.metadata.sequence = unsafe_value;
695        let errs = validate_event(&ev).unwrap_err();
696        assert!(
697            errs.iter().any(|e| matches!(
698                e,
699                ValidationError::JcsUnsafeInteger {
700                    field: "metadata.sequence",
701                    ..
702                }
703            )),
704            "sequence above JCS_SAFE_MAX must be flagged, got {errs:?}"
705        );
706
707        // EventMetrics counters
708        for field in [
709            "metrics.prompt_tokens",
710            "metrics.completion_tokens",
711            "metrics.cached_tokens",
712            "metrics.pruned_tokens",
713            "metrics.pruning_ratio_millis",
714        ] {
715            let mut ev = valid_event();
716            let mut metrics = crate::model::EventMetrics::default();
717            match field {
718                "metrics.prompt_tokens" => metrics.prompt_tokens = Some(unsafe_value),
719                "metrics.completion_tokens" => metrics.completion_tokens = Some(unsafe_value),
720                "metrics.cached_tokens" => metrics.cached_tokens = Some(unsafe_value),
721                "metrics.pruned_tokens" => metrics.pruned_tokens = Some(unsafe_value),
722                "metrics.pruning_ratio_millis" => metrics.pruning_ratio_millis = Some(unsafe_value),
723                _ => unreachable!(),
724            }
725            ev.metrics = Some(metrics);
726            let errs = validate_event(&ev).unwrap_err();
727            assert!(
728                errs.iter().any(|e| matches!(
729                    e,
730                    ValidationError::JcsUnsafeInteger { field: f, .. } if *f == field
731                )),
732                "{field} above JCS_SAFE_MAX must be flagged, got {errs:?}"
733            );
734        }
735    }

782    /// downstream SIEM pipelines routing by type_uid mis-classify.
783    /// Not currently reachable via the builder (defaults 1), but the
784    /// all-pub, non-`#[non_exhaustive]` struct is mutable directly
785    /// (as this test does below), bypassing the builder.
786    #[test]
787    fn activity_id_100_is_rejected_even_when_type_uid_matches() {
788        let mut ev = valid_event();
789        ev.activity_id = 100;
790        // Keep type_uid self-consistent so the mismatch check
791        // doesn't fire — this test isolates the bijection guard.
792        ev.type_uid = u64::from(ev.class_name.class_uid()) * 100 + 100;
793        let errs = validate_event(&ev).unwrap_err();
794        assert!(
795            errs.contains(&ValidationError::BadActivityId(100)),
796            "activity_id=100 must be flagged even when type_uid matches; got {errs:?}"

798        // Boundary: 99 must NOT be flagged.
799        ev.activity_id = 99;
800        ev.type_uid = u64::from(ev.class_name.class_uid()) * 100 + 99;
801        let errs = validate_event(&ev).map(|_| Vec::new()).unwrap_or_else(|e| e);
802        assert!(
803            !errs
804                .iter()
805                .any(|e| matches!(e, ValidationError::BadActivityId(_))),
806            "activity_id=99 must be valid; got {errs:?}"
807        );
