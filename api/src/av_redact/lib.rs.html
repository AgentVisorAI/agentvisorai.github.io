1//! Real-time sensitive-data redaction engine for audit event payloads.
2//!
3//! Scans JSON values (or raw JSON bytes) for sensitive data — API keys,
4//! email addresses, credit card numbers, Social Security numbers, public
5//! IPv4 addresses — and replaces matching spans with a configurable
6//! placeholder (default `[REDACTED]`). JSON pointer paths can be used to
7//! unconditionally redact specific subtrees.
8//!
9//! All regex compilation happens once at engine construction. The resulting
10//! [`RedactionEngine`] is `Send + Sync` and safe for use across async
11//! worker tasks.
12
13pub mod engine;
14
15pub use engine::{RedactError, RedactionConfig, RedactionEngine};

445    fn idempotency_across_all_patterns() {
446        let engine = default_engine();
447        let fixtures = vec![
448            json!("Bearer eyJhbGciOiJIUzI1NiJ9.eyJzdWIiOiIxMjM0NTY3ODkwIn0.abc123"),
449            json!("sk-abc123def456ghi789jk"),
450            json!("user@example.com"),
451            json!("4111111111111111"),
452            json!("123-45-6789"),
453            json!("8.8.8.8"),
454            json!({"password": "secret", "email": "a@b.com"}),
455            json!(["Bearer token123456789012", "10.0.0.1"]),
456        ];
457        for original in fixtures {
458            let mut first = original.clone();
459            engine.redact_value(&mut first);
460            let mut second = first.clone();
461            engine.redact_value(&mut second);
462            assert_eq!(first, second, "idempotency failed for {original}");
463        }
464    }

681    fn mixed_payload_redaction() {
682        let engine = default_engine();
683        let mut val = json!({
684            "event": "login",
685            "user_email": "admin@corp.io",
686            "source_ip": "203.0.113.5",
687            "internal_ip": "10.1.2.3",
688            "metadata": {
689                "auth_header": "Bearer eyJhbGciOiJIUzI1NiJ9.test1234",
690                "timestamp": "2024-01-15T10:30:00Z",
691                "count": 42
692            },
693            "tags": ["production", "us-east"]
694        });
695        engine.redact_value(&mut val);
696
697        // Email redacted.
698        let email = val["user_email"].as_str().unwrap();
699        assert!(email.contains("[REDACTED]"), "email not redacted: {email}");
700
701        // Public IP redacted.
702        let src_ip = val["source_ip"].as_str().unwrap();
703        assert!(src_ip.contains("[REDACTED]"), "public IP not redacted: {src_ip}");
704
705        // Private IP preserved.
706        assert_eq!(val["internal_ip"], json!("10.1.2.3"));
707
708        // Bearer token redacted.
709        let auth = val["metadata"]["auth_header"].as_str().unwrap();
710        assert!(auth.contains("[REDACTED]"), "bearer not redacted: {auth}");
711
712        // Timestamp preserved.
713        assert_eq!(val["metadata"]["timestamp"], json!("2024-01-15T10:30:00Z"));
714
715        // Number preserved.
716        assert_eq!(val["metadata"]["count"], json!(42));
717
718        // Array strings preserved.
719        assert_eq!(val["tags"][0], json!("production"));
720    }
