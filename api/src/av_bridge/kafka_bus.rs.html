1//! Kafka-wire connector (feature `kafka`), targeting Redpanda as the reference
2//! self-hosted broker (brief Module F). The event path uses rskafka; a statically
3//! linked librdkafka admin client provisions and verifies topic retention.
4//!
5//! TLS/SASL comes from the environment so secured endpoints need no new
6//! constructor surface: `AV_KAFKA_CA_FILE` pins a root CA (PEM) and enables
7//! TLS on both the admin and event paths; `AV_KAFKA_SASL_USERNAME` /
8//! `AV_KAFKA_SASL_PASSWORD` enable SASL with the mechanism from
9//! `AV_KAFKA_SASL_MECHANISM` (`SCRAM-SHA-256` by default — Redpanda's
10//! native credential store — or `SCRAM-SHA-512` / `PLAIN`). Credentials
11//! are only accepted together with TLS: PLAIN would ship the password in
12//! the clear, and even SCRAM without TLS is exposed to MITM relay. Use
13//! hostname (not bare-IP) broker endpoints with TLS: certificate
14//! verification runs against the dialed name, and IP-SAN support varies
15//! across rustls versions.
16

29/// Broker security material resolved from the environment (module docs).
30struct KafkaSecurity {
31    ca_file: Option<std::path::PathBuf>,
32    credentials: Option<(String, String)>,
33    mechanism: SaslMechanism,
34}
35
36#[derive(Clone, Copy, PartialEq, Eq)]
37enum SaslMechanism {
38    Plain,
39    ScramSha256,
40    ScramSha512,
41}
42
43impl SaslMechanism {
44    fn parse(value: &str) -> Result<Self, BusError> {
45        match value {
46            "PLAIN" => Ok(Self::Plain),
47            "SCRAM-SHA-256" => Ok(Self::ScramSha256),
48            "SCRAM-SHA-512" => Ok(Self::ScramSha512),
49            other => Err(BusError::Backend(format!(
50                "AV_KAFKA_SASL_MECHANISM {other:?} is not supported \
51                 (use PLAIN, SCRAM-SHA-256, or SCRAM-SHA-512)"
52            ))),
53        }
54    }
55
56    /// librdkafka's `sasl.mechanism` spelling.
57    fn librdkafka_name(self) -> &'static str {
58        match self {
59            Self::Plain => "PLAIN",
60            Self::ScramSha256 => "SCRAM-SHA-256",
61            Self::ScramSha512 => "SCRAM-SHA-512",
62        }
63    }
64}

66impl KafkaSecurity {
67    fn from_env() -> Result<Self, BusError> {
68        Self::from_lookup(|name| std::env::var(name).ok())
69    }
70
71    /// Testable core of [`Self::from_env`] (same pattern as the harness
72    /// config's `apply_env_overrides_from`): `get` returns the value of a
73    /// named environment variable, or `None` when unset.
74    fn from_lookup(get: impl Fn(&str) -> Option<String>) -> Result<Self, BusError> {
75        let ca_file = get("AV_KAFKA_CA_FILE").map(std::path::PathBuf::from);
76        let credentials = match (get("AV_KAFKA_SASL_USERNAME"), get("AV_KAFKA_SASL_PASSWORD")) {
77            (Some(username), Some(password)) => Some((username, password)),
78            (None, None) => None,
79            _ => {
80                return Err(BusError::Backend(
81                    "AV_KAFKA_SASL_USERNAME and AV_KAFKA_SASL_PASSWORD must be set together".to_owned(),
82                ))
83            }
84        };
85        let mechanism = match get("AV_KAFKA_SASL_MECHANISM") {
86            Some(value) => SaslMechanism::parse(&value)?,
87            None => SaslMechanism::ScramSha256,
88        };
89        if credentials.is_some() && ca_file.is_none() {
90            return Err(BusError::Backend(
91                "Kafka SASL credentials require AV_KAFKA_CA_FILE: PLAIN ships the password in \
92                 the clear and even SCRAM without TLS is exposed to MITM relay"
93                    .to_owned(),
94            ));
95        }
96        Ok(Self {
97            ca_file,
98            credentials,
99            mechanism,
100        })
101    }
102
103    /// librdkafka admin-client settings mirroring the rskafka event path.
104    fn apply_admin(&self, config: &mut ClientConfig) {
105        let protocol = match (&self.ca_file, &self.credentials) {
106            (Some(_), Some(_)) => "sasl_ssl",
107            (Some(_), None) => "ssl",
108            (None, _) => return,
109        };
110        config.set("security.protocol", protocol);
111        if let Some(ca) = &self.ca_file {
112            config.set("ssl.ca.location", ca.to_string_lossy().as_ref());
113        }
114        if let Some((username, password)) = &self.credentials {
115            config.set("sasl.mechanism", self.mechanism.librdkafka_name());
116            config.set("sasl.username", username);
117            config.set("sasl.password", password);
118        }
119    }
120
121    /// Root-pinned rustls config for the rskafka event path.
122    fn rskafka_tls(&self) -> Result<Option<Arc<rustls_tls::ClientConfig>>, BusError> {
123        let Some(ca) = &self.ca_file else {
124            return Ok(None);
125        };
126        // Same provider discipline as NatsBus::provision: rustls 0.23
127        // resolves its process CryptoProvider lazily and panics when both
128        // `ring` and `aws-lc-rs` are compiled in with none installed.
129        let _ = rustls_tls::crypto::ring::default_provider().install_default();
130        let pem = std::fs::read(ca)
131            .map_err(|error| BusError::Backend(format!("Kafka CA file {}: {error}", ca.display())))?;
132        use rustls_pki_types::pem::PemObject as _;
133        let mut roots = rustls_tls::RootCertStore::empty();
134        let mut certs = 0usize;
135        for cert in rustls_pki_types::CertificateDer::pem_slice_iter(&pem) {
136            let cert = cert
137                .map_err(|error| BusError::Backend(format!("Kafka CA file {}: {error:?}", ca.display())))?;
138            roots
139                .add(cert)
140                .map_err(|error| BusError::Backend(format!("Kafka CA file {}: {error}", ca.display())))?;
141            certs = certs.saturating_add(1);
142        }
143        if certs == 0 {
144            return Err(BusError::Backend(format!(
145                "Kafka CA file {} contains no PEM certificates",
146                ca.display()
147            )));
148        }
149        let config = rustls_tls::ClientConfig::builder()
150            .with_root_certificates(roots)
151            .with_no_client_auth();
152        Ok(Some(Arc::new(config)))
153    }
154
155    fn rskafka_sasl(&self) -> Option<SaslConfig> {
156        self.credentials.as_ref().map(|(username, password)| {
157            let credentials = rskafka::client::Credentials::new(username.clone(), password.clone());
158            match self.mechanism {
159                SaslMechanism::Plain => SaslConfig::Plain(credentials),
160                SaslMechanism::ScramSha256 => SaslConfig::ScramSha256(credentials),
161                SaslMechanism::ScramSha512 => SaslConfig::ScramSha512(credentials),
162            }
163        })
164    }
165}

167/// Kafka/Redpanda bus.
168pub struct KafkaBus {
169    cold_archive: Option<crate::cold_store::ColdArchive>,
170    executor: crate::bus::ConnectorExecutor,
171    /// Per-(topic, partition) clients built once at provision. rskafka's
172    /// `Client::partition_client` performs metadata discovery and broker
173    /// connection setup; constructing one per publish flooded the broker
174    /// under 10k-connection load until audit publishes stalled admission
175    /// past the upstream timeout (observed as 502s in the 10k SLA gate).
176    /// The `Client` itself is not retained: partition clients hold their
177    /// own broker references, and every post-provision operation goes
178    /// through this cache.
179    partition_clients: HashMap<(String, u32), Arc<rskafka::client::partition::PartitionClient>>,
180    topics: HashMap<String, u32>,
181    validators: HashMap<String, jsonschema::Validator>,
182}
183
184impl KafkaBus {
185    /// Connect to `broker` (host:port) and provision topics per the manifest.
186    pub fn provision(broker: &str, manifest: &BridgeManifest) -> Result<Self, BusError> {
187        manifest
188            .validate()
189            .map_err(|e| BusError::Backend(e.to_string()))?;
190        let validators = crate::manifest::compile_topic_validators(manifest)?;
191        let cold_archive = crate::cold_store::ColdArchive::from_manifest(manifest)?;
192        let executor = crate::bus::ConnectorExecutor::new("agentvisor-ai-kafka")?;
193        let security = KafkaSecurity::from_env()?;
194        let mut admin_config = ClientConfig::new();
195        admin_config
196            .set("bootstrap.servers", broker)
197            .set("socket.timeout.ms", "10000");
198        security.apply_admin(&mut admin_config);
199        let admin: AdminClient<DefaultClientContext> = admin_config
200            .create()
201            .map_err(|error| BusError::Backend(error.to_string()))?;
202        let admin = Arc::new(admin);
203        for topic in &manifest.topics {
204            let admin = Arc::clone(&admin);
205            let name = topic.name.clone();
206            let partitions = topic.partitions;
207            let replication_factor = manifest.replication_factor;
208            let retention_ms = u64::from(topic.retention.hot_hours)
209                .checked_mul(av_core::units::MS_PER_HOUR)
210                .ok_or_else(|| BusError::Backend("Kafka retention overflow".to_owned()))?
211                .to_string();
212            executor
213                .run(move || provision_topic(admin, name, partitions, replication_factor, retention_ms))?
214                .map_err(BusError::Backend)?;
215        }
216        drop(admin);
217        // `broker` is a bootstrap list (`host:port[,host:port]`, the same
218        // format rdkafka's `bootstrap.servers` takes above). rskafka wants
219        // one address per element — passing the joined string as a single
220        // entry made every multi-broker list fail to connect.
221        let brokers: Vec<String> = broker
222            .split(',')
223            .map(str::trim)
224            .filter(|entry| !entry.is_empty())
225            .map(str::to_owned)
226            .collect();
227        if brokers.is_empty() {
228            return Err(BusError::Backend("Kafka bootstrap list is empty".to_owned()));
229        }
230        let tls_config = security.rskafka_tls()?;
231        let sasl_config = security.rskafka_sasl();
232        let client = executor
233            .run(move || async move {
234                let mut builder = ClientBuilder::new(brokers);
235                if let Some(tls) = tls_config {
236                    builder = builder.tls_config(tls);
237                }
238                if let Some(sasl) = sasl_config {
239                    builder = builder.sasl_config(sasl);
240                }
241                builder.build().await
242            })?
243            .map_err(|e| BusError::Backend(e.to_string()))?;
244        let client = Arc::new(client);
245        let mut topics = HashMap::new();
246        for t in &manifest.topics {
247            topics.insert(t.name.clone(), t.partitions);
248        }
249        let metadata_client = Arc::clone(&client);
250        let metadata = executor
251            .run(move || async move { metadata_client.list_topics().await })?
252            .map_err(|error| BusError::Backend(error.to_string()))?;
253        for expected in &manifest.topics {
254            let actual = metadata
255                .iter()
256                .find(|topic| topic.name == expected.name)
257                .ok_or_else(|| BusError::Backend(format!("Kafka topic {:?} is missing", expected.name)))?;
258            if actual.partitions.len() != expected.partitions as usize {
259                return Err(BusError::Backend(format!(
260                    "Kafka topic {:?} has {} partitions, manifest requires {}",
261                    expected.name,
262                    actual.partitions.len(),
263                    expected.partitions
264                )));
265            }
266        }
267        let mut partition_clients = HashMap::new();
268        for t in &manifest.topics {
269            for p in 0..t.partitions {
270                let pc_client = Arc::clone(&client);
271                let name = t.name.clone();
272                let pc = executor
273                    .run(move || async move {
274                        pc_client
275                            .partition_client(name, p as i32, UnknownTopicHandling::Error)
276                            .await
277                    })?
278                    .map_err(|error| BusError::Backend(error.to_string()))?;
279                partition_clients.insert((t.name.clone(), p), Arc::new(pc));
280            }
281        }
282        Ok(Self {
283            cold_archive,
284            executor,
285            partition_clients,
286            topics,
287            validators,
288        })
289    }

291    /// Cached per-partition client (built at provision; see struct docs).
292    fn partition_client(
293        &self,
294        topic: &str,
295        partition: u32,
296    ) -> Result<Arc<rskafka::client::partition::PartitionClient>, BusError> {
297        self.partition_clients
298            .get(&(topic.to_owned(), partition))
299            .cloned()
300            .ok_or_else(|| BusError::UnknownTopic(topic.to_owned()))
301    }

303    fn publish_with_uid(
304        &self,
305        topic: &str,
306        key: &str,
307        value: &serde_json::Value,
308        event_uid: Option<&str>,
309    ) -> Result<PublishAck, BusError> {
310        crate::manifest::validate_topic_event(&self.validators, topic, value)?;
311        let partitions = *self
312            .topics
313            .get(topic)
314            .ok_or_else(|| BusError::UnknownTopic(topic.to_owned()))?;
315        let partition = partition_for(key, partitions);
316        let event_uid = event_uid.map_or_else(av_core::new_event_uid, str::to_owned);
317        let stored_at = av_core::time::now_ms();
318        let record = StoredEvent {
319            partition,
320            offset: 0,
321            key: key.to_owned(),
322            value: value.clone(),
323            stored_at,
324        };
325        if let Some(archive) = &self.cold_archive {
326            let already_staged = archive.stage(topic, &record, &event_uid)?;
327            if already_staged {
328                // A prior attempt for this UID staged the intent and may
329                // have produced to the broker before its commit step
330                // failed (ENOSPC persisting the resolved offset, MAC
331                // refusal, crash). rskafka `produce` has no native dedup,
332                // so blindly re-producing would append a second record
333                // for the same logical event — a duplicate in the audit
334                // stream. Consult the partition first, exactly like the
335                // maintenance retry path does.
336                if let Some(existing) = self.find_event_by_uid(topic, key, &event_uid)? {
337                    // The event is durably on the broker; resolving the
338                    // cold intent is best-effort here. On failure the
339                    // durable intent stays queued for the maintenance
340                    // pass — do not fail a publish that succeeded.
341                    if let Err(error) = archive.commit(topic, &event_uid, existing.offset) {
342                        tracing::warn!(
343                            %error,
344                            topic,
345                            event_uid = %event_uid,
346                            "cold intent commit failed after idempotent dedup hit; left for maintenance retry"
347                        );
348                    }
349                    return Ok(existing);
350                }
351            }
352        }
353        let ack = self.publish_broker_only(topic, key, value, stored_at, &event_uid)?;
354        if let Some(archive) = &self.cold_archive {
355            archive.commit(topic, &event_uid, ack.offset)?;
356        }
357        Ok(ack)
358    }

360    fn publish_broker_only(
361        &self,
362        topic: &str,
363        key: &str,
364        value: &serde_json::Value,
365        stored_at: u64,
366        event_uid: &str,
367    ) -> Result<PublishAck, BusError> {
368        crate::manifest::validate_topic_event(&self.validators, topic, value)?;
369        let partitions = *self
370            .topics
371            .get(topic)
372            .ok_or_else(|| BusError::UnknownTopic(topic.to_owned()))?;
373        let partition = partition_for(key, partitions);
374        let record = StoredEvent {
375            partition,
376            offset: 0,
377            key: key.to_owned(),
378            value: value.clone(),
379            stored_at,
380        };
381        let payload = serde_json::to_vec(&record)?;
382        let pc = self.partition_client(topic, partition)?;
383        let key_bytes = event_uid.as_bytes().to_vec();
384        let mut headers = std::collections::BTreeMap::new();
385        headers.insert("agentvisor-event-uid".to_owned(), event_uid.as_bytes().to_vec());
386        let offset = self
387            .executor
388            .run(move || async move {
389                let offsets = pc
390                    .produce(
391                        vec![Record {
392                            key: Some(key_bytes),
393                            value: Some(payload),
394                            headers,
395                            timestamp: chrono_now(),
396                        }],
397                        Compression::NoCompression,
398                    )
399                    .await?;
400                // One record in => exactly one offset out. An empty response
401                // is a broker anomaly; fabricating offset 0 here would persist
402                // a legitimate-looking ack that dedupe/recovery then trusts.
403                Ok::<_, rskafka::client::error::Error>(offsets.first().copied())
404            })?
405            .map_err(|e| BusError::Backend(e.to_string()))?;
406        let offset = offset.ok_or_else(|| {
407            BusError::Backend("Kafka produce succeeded but returned no offset for the record".to_owned())
408        })?;
409        let offset = u64::try_from(offset)
410            .map_err(|_| BusError::Backend(format!("Kafka returned negative offset {offset}")))?;
411        Ok(PublishAck {
412            topic: topic.to_owned(),
413            partition,
414            offset,
415        })
416    }
417}

419async fn provision_topic(
420    admin: Arc<AdminClient<DefaultClientContext>>,
421    name: String,
422    partitions: u32,
423    replication_factor: u32,
424    retention_ms: String,
425) -> Result<(), String> {
426    let partitions = i32::try_from(partitions)
427        .map_err(|_| format!("partition count for Kafka topic {name:?} exceeds i32"))?;
428    let replication_factor = i32::try_from(replication_factor)
429        .map_err(|_| format!("replication factor for Kafka topic {name:?} exceeds i32"))?;
430    let topic = NewTopic::new(&name, partitions, TopicReplication::Fixed(replication_factor))
431        .set("retention.ms", &retention_ms);
432    let results = admin
433        .create_topics(
434            [&topic],
435            &AdminOptions::new().operation_timeout(Some(std::time::Duration::from_secs(5))),
436        )
437        .await
438        .map_err(|error| error.to_string())?;
439    match results.into_iter().next() {
440        Some(Ok(_)) | Some(Err((_, RDKafkaErrorCode::TopicAlreadyExists))) => {}
441        Some(Err((_, error))) => return Err(format!("create Kafka topic {name:?}: {error}")),
442        None => return Err(format!("create Kafka topic {name:?} returned no result")),
443    }
444
445    let resource = ResourceSpecifier::Topic(&name);
446    let results = admin
447        .describe_configs([&resource], &AdminOptions::new())
448        .await
449        .map_err(|error| error.to_string())?;
450    let configuration = results
451        .into_iter()
452        .next()
453        .ok_or_else(|| format!("describe Kafka topic {name:?} returned no result"))?
454        .map_err(|error| format!("describe Kafka topic {name:?}: {error}"))?;
455    let actual = configuration
456        .get("retention.ms")
457        .and_then(|entry| entry.value.as_deref());
458    if actual != Some(retention_ms.as_str()) {
459        return Err(format!(
460            "Kafka topic {name:?} retention.ms is {actual:?}, manifest requires {retention_ms:?}"
461        ));
462    }
463    Ok(())
464}

466impl EventBus for KafkaBus {
467    fn set_control_key(&self, key: [u8; 32]) -> Result<(), BusError> {

475        let event_uid = value
476            .get("metadata")
477            .and_then(|metadata| metadata.get("uid"))
478            .and_then(serde_json::Value::as_str);
479        self.publish_with_uid(topic, key, value, event_uid)
480    }

482    fn publish_idempotent(
483        &self,
484        topic: &str,
485        key: &str,
486        value: &serde_json::Value,

491
492    fn find_event_by_uid(
493        &self,
494        topic: &str,
495        key: &str,
496        event_uid: &str,
497    ) -> Result<Option<PublishAck>, BusError> {
498        let partitions = *self
499            .topics
500            .get(topic)
501            .ok_or_else(|| BusError::UnknownTopic(topic.to_owned()))?;
502        let partition = partition_for(key, partitions);
503        let partition_client = self.partition_client(topic, partition)?;
504        let event_uid = event_uid.to_owned();
505        // One page per `executor.run` call: the executor imposes a hard
506        // 10 s timeout per operation, so running the whole earliest→latest
507        // scan inside a single call made lookups on partitions larger than
508        // one timeout's worth of fetches time out on every attempt, forever
509        // (each retry restarted from earliest under the same cap). Paging
510        // gives each bounded step its own budget — the same shape as the
511        // NATS implementation and the default trait implementation's
512        // per-`fetch` paging.
513        enum Page {
514            Found(i64),
515            Advanced(i64),
516            Empty,
517        }
518        let (mut offset, latest) = {
519            let client = Arc::clone(&partition_client);
520            self.executor
521                .run(move || async move {
522                    let earliest = client
523                        .get_offset(OffsetAt::Earliest)
524                        .await
525                        .map_err(|error| error.to_string())?;
526                    let latest = client
527                        .get_offset(OffsetAt::Latest)
528                        .await
529                        .map_err(|error| error.to_string())?;
530                    Ok::<_, String>((earliest, latest))
531                })?
532                .map_err(BusError::Backend)?
533        };
534        let mut empty_fetches = 0u32;
535        let found = loop {
536            if offset >= latest {
537                break None;
538            }
539            let client = Arc::clone(&partition_client);
540            let page_uid = event_uid.clone();
541            let page = self
542                .executor
543                .run(move || async move {
544                    let (records, _) = client
545                        .fetch_records(offset, 1..(16 * 1024 * 1024), 500)
546                        .await
547                        .map_err(|error| error.to_string())?;
548                    if records.is_empty() {
549                        return Ok::<_, String>(Page::Empty);
550                    }
551                    for record in &records {
552                        // `publish_with_uid` stamps the (possibly freshly
553                        // generated) uid into the `agentvisor-event-uid`
554                        // header even when the payload lacks
555                        // `metadata.uid`. Crash recovery for such an
556                        // event previously scanned only the payload
557                        // field, could never find the record it had in
558                        // fact produced, and re-published a duplicate.
559                        // Consult the header first, then the payload.
560                        if record
561                            .record
562                            .headers
563                            .get("agentvisor-event-uid")
564                            .is_some_and(|uid| uid.as_slice() == page_uid.as_bytes())
565                        {
566                            return Ok(Page::Found(record.offset));
567                        }
568                        if let Some(payload) = record.record.value.as_deref() {
569                            let stored: StoredEvent =
570                                serde_json::from_slice(payload).map_err(|error| error.to_string())?;
571                            if stored
572                                .value
573                                .get("metadata")
574                                .and_then(|metadata| metadata.get("uid"))
575                                .and_then(serde_json::Value::as_str)
576                                == Some(page_uid.as_str())
577                            {
578                                return Ok(Page::Found(record.offset));
579                            }
580                        }
581                    }
582                    let next = records
583                        .last()
584                        .and_then(|record| record.offset.checked_add(1))
585                        .ok_or_else(|| "Kafka event lookup offset overflow".to_owned())?;
586                    Ok(Page::Advanced(next))
587                })?
588                .map_err(BusError::Backend)?;
589            match page {
590                Page::Found(found_offset) => break Some(found_offset),
591                Page::Advanced(next) => {
592                    empty_fetches = 0;
593                    offset = next;
594                }
595                Page::Empty => {
596                    // `offset < latest` proves the broker still holds
597                    // records we have not seen; an empty response here is
598                    // a timing artifact (the 500 ms max_wait expired on a
599                    // loaded/rebalancing broker), NOT proof of absence.
600                    // Returning Ok(None) would let maintenance re-produce
601                    // an already-committed event — a duplicate in the
602                    // audit stream, the exact outcome this lookup exists
603                    // to prevent. Retry bounded, then fail the lookup so
604                    // the caller retries the whole pass later (the cold
605                    // intent is durable).
606                    empty_fetches += 1;
607                    if empty_fetches >= 8 {
608                        return Err(BusError::Backend(format!(
609                            "Kafka event lookup stalled: empty fetch at offset {offset} below latest {latest}"
610                        )));
611                    }
612                }
613            }
614        };
615        found
616            .map(|offset| {
617                u64::try_from(offset)
618                    .map(|offset| PublishAck {
619                        topic: topic.to_owned(),
620                        partition,
621                        offset,
622                    })
623                    .map_err(|_| BusError::Backend(format!("Kafka returned negative offset {offset}")))
624            })
625            .transpose()
626    }
627
628    fn fetch(
629        &self,
630        topic: &str,
631        partition: u32,
632        offset: u64,
633        max: usize,
634    ) -> Result<Vec<StoredEvent>, BusError> {
635        if !self.topics.contains_key(topic) {
636            return Err(BusError::UnknownTopic(topic.to_owned()));
637        }
638        let pc = self.partition_client(topic, partition)?;
639        // Reject offsets above
640        // `i64::MAX` explicitly rather than let `as i64` sign-wrap
641        // into a negative Kafka offset. The Kafka wire protocol
642        // treats -1 as "latest" and -2 as "earliest" (its own
643        // sentinel semantics), so a wrap could silently redirect
644        // the fetch to the log tail instead of the requested
645        // offset — a hard-to-diagnose replay divergence. In
646        // practice `u64` offsets that survive our provisioning
647        // stay well below 2^63, but crash recovery from a
648        // manifest whose offset field was tampered on disk could
649        // still supply a value in `(i64::MAX, u64::MAX]`.
650        let signed_offset = i64::try_from(offset).map_err(|_| {
651            BusError::Backend(format!(
652                "kafka fetch offset {offset} exceeds i64::MAX and would wrap to a negative \
653                 sentinel; refusing the fetch"
654            ))
655        })?;
656        // Contract: "read up to `max` events". The loop below checks the
657        // cap only AFTER pushing, so `max == 0` returned one event —
658        // diverging from the embedded broker, which returns an empty vec.
659        if max == 0 {
660            return Ok(Vec::new());
661        }
662        self.executor
663            .run(move || async move {
664                let (records, _high_watermark) = pc
665                    .fetch_records(signed_offset, 1..(16 * 1024 * 1024), 500)
666                    .await
667                    .map_err(|e| e.to_string())?;
668                let mut out = Vec::new();
669                for r in records {
670                    // Surface the error instead of silently
671                    // dropping a corrupt record. Parity with NatsBus and
672                    // EmbeddedBroker. An auditor or reconciler that sees a
673                    // shorter list than expected — with no error and no
674                    // offset gap — is an evidence gap. Because reconcilers
675                    // advance offset by `events.last().offset + 1`, a
676                    // silently-skipped corrupt record at offset N causes
677                    // the caller to bypass it entirely; keep the record on
678                    // the partition as forensic evidence.
679                    let value = r
680                        .record
681                        .value
682                        .ok_or_else(|| format!("fetch: null record value at offset {}", r.offset))?;
683                    let mut ev: StoredEvent = serde_json::from_slice(&value)
684                        .map_err(|e| format!("fetch decode at offset {}: {e}", r.offset))?;
685                    // Kafka offsets are non-negative by contract; a broken
686                    // or hostile broker returning a negative i64 must fail
687                    // the fetch instead of sign-wrapping into a huge u64
688                    // that corrupts the caller's offset ledger.
689                    ev.offset = u64::try_from(r.offset)
690                        .map_err(|_| format!("fetch: negative broker offset {}", r.offset))?;
691                    out.push(ev);
692                    if out.len() >= max {
693                        break;
694                    }
695                }
696                Ok::<_, String>(out)
697            })?
698            .map_err(BusError::Backend)
699    }
700
701    fn partitions(&self, topic: &str) -> Result<u32, BusError> {
702        self.topics
703            .get(topic)
704            .copied()
705            .ok_or_else(|| BusError::UnknownTopic(topic.to_owned()))
706    }
707
708    fn topics(&self) -> Vec<String> {
709        let mut t: Vec<String> = self.topics.keys().cloned().collect();
710        t.sort();
711        t
712    }
713
714    fn maintenance(&self, _now_ms: u64) -> Result<u64, BusError> {
715        self.cold_archive.as_ref().map_or(Ok(0), |archive| {
716            archive.retry_pending_with(|pending| {
717                // A crash/timeout between a successful produce and
718                // `commit()` leaves the intent offset-None while the event
719                // IS already on the partition; blindly re-producing here
720                // would append a duplicate to the audit stream. Consult the
721                // partition first and only publish when the UID is
722                // genuinely absent.
723                if let Some(ack) = self.find_event_by_uid(&pending.topic, &pending.key, &pending.event_uid)? {
724                    return Ok(ack);
725                }
726                self.publish_broker_only(
727                    &pending.topic,
728                    &pending.key,
729                    &pending.value,
730                    pending.stored_at,
731                    &pending.event_uid,
732                )
733            })
734        })
735    }
