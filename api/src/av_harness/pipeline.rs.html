165/// Pre-resolved metric handles for the request hot path so the
166/// per-stage histogram observation doesn't take the registry mutex.
167/// Same series names as the lazy path — the label suffix matches
168/// `av_stage_duration_seconds{stage="…"}` exactly.
169///
170/// Ordering of the `[Stage; 5]` arrays mirrors the boot registration
171/// in `new_with_backends_and_metrics`: identity, quota, sanitize,
172/// compression, dispatch. `Stage::index()` returns the offset so
173/// `observe_stage` is O(1) with no string alloc on the hot path.

180    /// a false config default.
181    pub(crate) strict_stage_budget: bool,
182    /// Pre-resolved `av_requests_total{route,status_class}` counters,
183    /// indexed by `(Route::index(), StatusClass::index())`. Populated

186    /// per-request `metrics.counter(&format!("..."))` path did 4
187    /// mutex ops on the shared `Registry` per HTTP request plus

191    /// Pre-resolved `av_request_duration_seconds{route}` histograms,
192    /// indexed by `Route::index()`. Same rationale as
193    /// `request_counters` above; the histogram fetch used to hit

196    pub(crate) request_duration_histograms: [Arc<av_core::metrics::Histogram>; 10],
197    /// Pre-resolved `av_upstream_latency_seconds` histogram — hit

218}

490impl HotMetrics {
491    fn new(metrics: &Registry, config_strict_stage_budget: bool) -> Self {
492        // Registrations here mirror the boot loop below. Using the
493        // same `metrics.histogram_with_bounds` / `metrics.counter`
494        // getters means these entries are shared with the lazy path
495        // — a caller that still uses `metrics.counter(name).inc()`
496        // hits the exact same Arc<Counter> we cache here.
497        let stage_histograms: [Arc<av_core::metrics::Histogram>; 5] = Stage::ORDER.map(|stage| {
498            metrics.histogram_with_bounds(
499                &format!("av_stage_duration_seconds{{stage=\"{}\"}}", stage.label()),
500                "Harness stage latency",
501                av_core::metrics::WIDE_LATENCY_BOUNDS_US,
502            )
503        });
504        let stage_strict_budget_counters: [Arc<av_core::metrics::Counter>; 5] = Stage::ORDER.map(|stage| {
505            metrics.counter(
506                &format!("av_strict_budget_breaches_total{{stage=\"{}\"}}", stage.label()),
507                "Middleware stages that exceeded the strict per-stage budget",
508            )
509        });
510        // Pre-resolve the request-metrics middleware's counters +
511        // histograms as a [route][status_class] array so the hot
512        // path is O(1) with zero string alloc and zero Registry
513        // mutex ops. Registration order matches the enum ORDER
514        // constants, matched at read time via `Route::index()` and
515        // `StatusClass::index()`.
516        let request_counters: [[Arc<av_core::metrics::Counter>; 5]; 10] = Route::ORDER.map(|route| {
517            StatusClass::ORDER.map(|status| {
518                metrics.counter(
519                    &format!(
520                        "av_requests_total{{route=\"{}\",status_class=\"{}\"}}",
521                        route.label(),
522                        status.label()
523                    ),
524                    "HTTP requests by route and status class",
525                )
526            })
527        });
528        let request_duration_histograms: [Arc<av_core::metrics::Histogram>; 10] = Route::ORDER.map(|route| {
529            metrics.histogram_with_bounds(
530                &format!("av_request_duration_seconds{{route=\"{}\"}}", route.label()),
531                "End-to-end HTTP request latency by route",
532                av_core::metrics::WIDE_LATENCY_BOUNDS_US,
533            )
534        });
535        // Pre-resolve upstream latency + error counters — hit on
536        // every chat forward. Same rationale as request_counters.
537        let upstream_latency_histogram = metrics.histogram_with_bounds(
538            "av_upstream_latency_seconds",
539            "Time to upstream response headers",
540            av_core::metrics::WIDE_LATENCY_BOUNDS_US,
541        );
542        let upstream_errors_counters: [Arc<av_core::metrics::Counter>; 4] =
543            UpstreamErrorKind::ORDER.map(|kind| {
544                metrics.counter(
545                    &format!("av_upstream_errors_total{{kind=\"{}\"}}", kind.label()),
546                    "Upstream failures by kind",
547                )
548            });
549        // Pre-resolve dashboard endpoint counters + histograms. Same
550        // rationale as request_counters; lower volume but hits the
551        // same shared Registry mutex.
552        let dashboard_request_counters: [[Arc<av_core::metrics::Counter>; 2]; 3] = DashboardEndpoint::ORDER
553            .map(|endpoint| {
554                DashboardStatus::ORDER.map(|status| {
555                    metrics.counter(
556                        &format!(
557                            "av_dashboard_requests_total{{endpoint=\"{}\",status=\"{}\"}}",
558                            endpoint.label(),
559                            status.label()
560                        ),
561                        "Dashboard endpoint requests, labeled by status",
562                    )
563                })
564            });
565        let dashboard_request_duration_histograms: [Arc<av_core::metrics::Histogram>; 3] =
566            DashboardEndpoint::ORDER.map(|endpoint| {
567                metrics.histogram(
568                    &format!(
569                        "av_dashboard_request_duration_seconds{{endpoint=\"{}\"}}",
570                        endpoint.label()
571                    ),
572                    "Dashboard endpoint latency",
573                )
574            });
575        Self {
576            stage_histograms,
577            stage_strict_budget_counters,
578            strict_stage_budget: config_strict_stage_budget || truthy_env("AV_STRICT_BUDGET"),
579            request_counters,
580            request_duration_histograms,
581            upstream_latency_histogram,
582            upstream_errors_counters,
583            dashboard_request_counters,
584            dashboard_request_duration_histograms,
585        }
586    }

606/// Resolve the configured upstream credential into a ready header pair.
607///
608/// Key material is accepted only from an environment variable or an
609/// owner-only file — never from TOML or argv — and the resulting header
610/// value is marked sensitive so `Debug` output redacts it.
611pub(crate) fn resolve_upstream_auth(
612    config: &HarnessConfig,
613) -> Result<Option<(HeaderName, HeaderValue)>, PipelineError> {
614    let key = match read_secret(
615        config.upstream_api_key_env.as_deref(),
616        config.upstream_api_key_file.as_deref(),
617        "upstream API key",
618    )? {
619        Some(key) => key,
620        None => return Ok(None),
621    };
622    let name = HeaderName::try_from(config.upstream_auth_header.as_str()).map_err(|_| {
623        PipelineError::upstream(format!(
624            "upstream_auth_header {:?} is not a valid header name",
625            config.upstream_auth_header
626        ))
627    })?;
628    let rendered = if config.upstream_auth_scheme.is_empty() {
629        key
630    } else {
631        format!("{} {key}", config.upstream_auth_scheme)
632    };
633    let mut value = HeaderValue::from_str(&rendered).map_err(|_| {
634        PipelineError::upstream(
635            "upstream API key contains bytes that cannot appear in an HTTP header".to_owned(),
636        )
637    })?;
638    value.set_sensitive(true);
639    Ok(Some((name, value)))
640}

661/// Read a secret from an env var or an owner-only file, trimming
662/// surrounding whitespace. A configured-but-missing source is a loud startup error:
663/// silently proxying unauthenticated would produce baffling upstream 401s.
664fn read_secret(
665    env_name: Option<&str>,
666    file_path: Option<&str>,
667    what: &str,
668) -> Result<Option<String>, PipelineError> {
669    read_secret_from(|name| std::env::var(name).ok(), env_name, file_path, what)
670}

672/// Testable core of [`read_secret`] with an injected environment.
673fn read_secret_from(
674    get_env: impl Fn(&str) -> Option<String>,
675    env_name: Option<&str>,
676    file_path: Option<&str>,
677    what: &str,
678) -> Result<Option<String>, PipelineError> {
679    if let Some(name) = env_name {
680        let value = get_env(name).ok_or_else(|| {
681            PipelineError::upstream(format!(
682                "{what}: environment variable {name} is not set (export it or update the config)"
683            ))
684        })?;
685        let value = value.trim().to_owned();
686        if value.is_empty() {
687            return Err(PipelineError::upstream(format!(
688                "{what}: environment variable {name} is set but empty"
689            )));
690        }
691        return Ok(Some(value));
692    }
693    if let Some(path) = file_path {
694        require_owner_only_secret(std::path::Path::new(path))
695            .map_err(|error| PipelineError::upstream(format!("{what}: {error}")))?;
696        let value = std::fs::read_to_string(path).map_err(|error| PipelineError::Upstream {
697            context: format!("{what}: read {path}: {error}"),
698            source: Some(Box::new(error)),
699        })?;
700        let value = value.trim().to_owned();
701        if value.is_empty() {
702            return Err(PipelineError::upstream(format!("{what}: file {path} is empty")));
703        }
704        return Ok(Some(value));
705    }
706    Ok(None)
707}

709/// Same posture as the signing-seed loader: refuse symlinks (a pre-planted
710/// link would fool the mode check, CWE-59) and group/other-readable modes
711/// on Unix. Windows deployments rely on operator-set ACLs.
712fn require_owner_only_secret(path: &std::path::Path) -> Result<(), String> {
713    #[cfg(unix)]
714    {
715        use std::os::unix::fs::MetadataExt as _;
716        let metadata = std::fs::symlink_metadata(path)
717            .map_err(|error| format!("stat secret file {}: {error}", path.display()))?;
718        let file_type = metadata.file_type();
719        if file_type.is_symlink() {
720            return Err(format!(
721                "secret file {} is a symbolic link; refusing to follow",
722                path.display()
723            ));
724        }
725        if !file_type.is_file() {
726            return Err(format!("secret file {} is not a regular file", path.display()));
727        }
728        let mode = metadata.mode() & 0o777;
729        if mode & 0o077 != 0 {
730            return Err(format!(
731                "secret file {} has mode 0o{mode:03o}; must be owner-only (chmod 600 {})",
732                path.display(),
733                path.display()
734            ));
735        }
736    }
737    #[cfg(not(unix))]
738    {
739        let _ = path;
740    }
741    Ok(())
742}

799/// RAII owner of the journalled response attempt and (once created) the
800/// durable in-flight response marker, from request admission until a
801/// downstream owner takes over (`AbortFinalizingStream`, a terminal
802/// failure job, or the worker's capture publish).
803///
804/// Axum drops the request future wholesale when the client disconnects.
805/// Every *explicit* failure path already retires the marker and closes
806/// the journalled attempt — but a cancelled future runs none of them:
807/// the journal kept a dangling non-terminal attempt (crash recovery then
808/// quarantines the whole session, see `track_response_attempt`) and the
809/// marker stranded on disk (once the session evicts, every reconciler
810/// tick re-quarantines its id forever). Drop submits the same terminal
811/// failure job the explicit paths use.
812pub(crate) struct ResponseCaptureGuard {
813    worker: crate::worker::WorkerHandle,
814    session: Arc<Session>,
815    identity: AgentIdentity,
816    marker: Option<String>,
817    attempt_id: String,
818    armed: bool,
819}
820
821impl ResponseCaptureGuard {
822    fn new(
823        worker: crate::worker::WorkerHandle,
824        session: Arc<Session>,
825        identity: AgentIdentity,
826        attempt_id: String,
827    ) -> Self {
828        Self {
829            worker,
830            session,
831            identity,
832            marker: None,
833            attempt_id,
834            armed: true,
835        }
836    }
837
838    /// Record the durable in-flight marker so a later drop retires it.
839    pub(crate) fn set_marker(&mut self, marker: String) {
840        self.marker = Some(marker);
841    }
842
843    pub(crate) fn attempt_id(&self) -> &str {
844        &self.attempt_id
845    }
846
847    /// Take ownership of the marker and attempt id, disarming the guard.
848    /// The caller now owns terminal-record submission.
849    pub(crate) fn disarm(mut self) -> (Option<String>, String) {
850        self.armed = false;
851        (self.marker.take(), std::mem::take(&mut self.attempt_id))
852    }
853
854    /// Disarm in place, returning any marker: for callers that submit the
855    /// terminal record themselves while retaining the guard (abandon paths).
856    fn defuse(&mut self) -> Option<String> {
857        self.armed = false;
858        self.marker.take()
859    }
860}

862impl Drop for ResponseCaptureGuard {
863    fn drop(&mut self) {
864        if !self.armed {
865            return;
866        }
867        let job = refused_response_failure_job(
868            Arc::clone(&self.session),
869            self.identity.clone(),
870            "request_cancelled_before_response_capture".to_owned(),
871            self.marker.take(),
872            std::mem::take(&mut self.attempt_id),
873        );
874        // Same fallback as every explicit refusal path: if the queue is
875        // full, fail the session closed rather than dropping the capture
876        // silently.
877        if self.worker.try_submit(job).is_err() {
878            self.session.mark_capture_failed();
879        }
880    }
881}

883/// RAII owner of the admission-time token debit (session AND principal
884/// ledgers), from the post-compression charge in `prepare_chat` until
885/// the request is handed to the upstream HTTP client.
886///
887/// Every *explicit* pre-dispatch abort refunds via `abandon_prepared` —
888/// but axum drops the request future wholesale on client disconnect,
889/// running none of them. Without this guard, a cancellation between
890/// admission and `send()` (the `spawn_blocking` join in
891/// `prepare_chat_nonblocking`, or the awaited marker write in
892/// `forward_chat`) leaked the debit on both ledgers; the principal
893/// ledger persists across sessions and never heals, so a client that
894/// disconnected mid-admission drained its principal budget with zero
895/// LLM work.
896///
897/// [`Self::take_for_dispatch`] disarms the guard at the last
898/// pre-dispatch point: from there the request may reach the provider,
899/// and the debit conservatively stands (mirroring `forward_chat`'s
900/// `is_connect`-only refund on send errors).
901pub(crate) struct AdmissionDebit {
902    store: Arc<dyn StateStore>,
903    session_id: String,
904    budget: av_state::BudgetSpec,
905    principal: Option<(String, av_state::BudgetSpec)>,
906    tokens: u64,
907    /// Metrics registry for the Drop-path panic supervisor.
908    ///
909    /// If the refund closure panics inside the spawn_blocking pool (a
910    /// backend allocator failure, a parking_lot mutex panic during
911    /// unwind, a HashMap grow OOM), tokio's default
912    /// `UnhandledPanic::Ignore` swallows it silently. The debit then
913    /// stays permanent on BOTH the session ledger and — the material
914    /// consequence — the principal ledger, which persists across
915    /// sessions and never heals. Surface the panic via a bumped
916    /// counter and a tracing::error so `rate(av_admission_refund_panics_total)`
917    /// catches it.
918    metrics: Arc<Registry>,
919}
920
921impl AdmissionDebit {
922    /// Disarm and hand the debited amount to the dispatch path; the
923    /// debit stands from here (reconciled against provider usage, or
924    /// refunded by the explicit `is_connect` arm).
925    fn take_for_dispatch(&mut self) -> u64 {
926        std::mem::take(&mut self.tokens)
927    }
928
929    /// Refund both ledgers now, in the order they were spent. Zeroed so
930    /// a later drop (or a hypothetical double abandon) cannot
931    /// double-refund.
932    fn refund_now(&mut self) {
933        let tokens = std::mem::take(&mut self.tokens);
934        if tokens == 0 {
935            return;
936        }
937        ActionBudget::new(self.store.as_ref(), &self.session_id, &self.budget).refund_tokens(tokens);
938        if let Some((principal_id, spec)) = self.principal.as_ref() {
939            ActionBudget::for_principal(self.store.as_ref(), principal_id, spec).refund_tokens(tokens);
940        }
941    }
942}
943
944impl Drop for AdmissionDebit {
945    fn drop(&mut self) {
946        if self.tokens == 0 {
947            return;
948        }
949        // A refund is a synchronous store call (a network round-trip on
950        // the redis backend). A cancelled request future drops on an
951        // async worker thread, where that round-trip would stall every
952        // other stream polled there — move it to the blocking pool when
953        // a runtime is available (drops on the `spawn_blocking` path of
954        // `prepare_chat_nonblocking` already run on blocking threads,
955        // and unit tests may have no runtime at all: refund inline).
956        let tokens = std::mem::take(&mut self.tokens);
957        let store = Arc::clone(&self.store);
958        let session_id = std::mem::take(&mut self.session_id);
959        let budget = std::mem::take(&mut self.budget);
960        let principal = self.principal.take();
961        let metrics = Arc::clone(&self.metrics);
962        // `catch_unwind` around the refund body so a panic inside
963        // spawn_blocking (backend allocator OOM, parking_lot mutex
964        // panic during unwind, a HashMap grow at capacity) doesn't
965        // get swallowed by tokio's default `UnhandledPanic::Ignore`.
966        // The `store: Arc<dyn StateStore>` and `metrics: Arc<Registry>`
967        // captures are already `UnwindSafe` (Arc is UnwindSafe when
968        // its inner T is `RefUnwindSafe`, and the store trait's
969        // implementors are RefUnwindSafe by convention here).
970        // `session_id: String`, `budget: BudgetSpec`, and
971        // `principal: Option<(String, BudgetSpec)>` are all
972        // Send + UnwindSafe. Wrap the whole thing in
973        // `AssertUnwindSafe` to bypass the `RefUnwindSafe` check on
974        // the trait object rather than plumb the bound through the
975        // whole StateStore hierarchy — the invariant we care about
976        // (no shared mutable state leaked from the refund) is
977        // upheld by construction.
978        let sess_for_log = session_id.clone();
979        let refund = move || {
980            let outcome = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
981                ActionBudget::new(store.as_ref(), &session_id, &budget).refund_tokens(tokens);
982                if let Some((principal_id, spec)) = principal.as_ref() {
983                    ActionBudget::for_principal(store.as_ref(), principal_id, spec).refund_tokens(tokens);
984                }
985            }));
986            if let Err(panic) = outcome {
987                let msg = panic
988                    .downcast_ref::<&'static str>()
989                    .copied()
990                    .or_else(|| panic.downcast_ref::<String>().map(String::as_str))
991                    .unwrap_or("panic payload was not a string");
992                metrics
993                    .counter(
994                        "av_admission_refund_panics_total",
995                        "Admission-debit refund closure panicked in Drop; the session \
996                         and principal ledgers retain the debit until operator repair",
997                    )
998                    .inc();
999                tracing::error!(
1000                    session = %sess_for_log,
1001                    tokens,
1002                    panic = %msg,
1003                    "admission-debit refund PANICKED (caught); \
1004                     principal ledger drain persists until operator repair"
1005                );
1006            }
1007        };
1008        match tokio::runtime::Handle::try_current() {
1009            Ok(handle) => {
1010                handle.spawn_blocking(refund);
1011            }
1012            Err(_) => refund(),
1013        }
1014    }
1015}

1070/// Hot-path failures, each carrying an HTTP status mapping.

1073/// `#[source]` instead of a pre-rendered `String`; `context` keeps the
1074/// exact text the old `String` payloads rendered, so Display output
1075/// (embedded verbatim in the OpenAI-shaped HTTP error bodies) is
1076/// unchanged. `Unauthorized` and `Abort` stay plain strings: their
1077/// messages are deliberately classified policy verdicts (the raw
1078/// identity error is logged separately and must not travel with the
1079/// client-facing value).

1083    /// A request header was malformed or inconsistent.
1084    #[error("bad request: {context}")]

1086        /// Human-readable description.
1087        context: String,
1088        /// Underlying typed error, when one exists.
1089        #[source]
1090        source: Option<ErrorSource>,
1091    },
1092    /// Identity was missing or invalid.
1093    #[error("unauthorized: {0}")]
1094    Unauthorized(String),
1095    /// A stateful quota or loop breaker blocked execution.
1096    #[error("request blocked: {context}")]
1097    Blocked {
1098        /// Human-readable description.
1099        context: String,
1100        /// Underlying typed error, when one exists.
1101        #[source]
1102        source: Option<ErrorSource>,
1103    },
1104    /// The upstream provider request failed.
1105    #[error("upstream request failed: {context}")]
1106    Upstream {
1107        /// Human-readable description.
1108        context: String,
1109        /// Underlying typed error, when one exists.
1110        #[source]
1111        source: Option<ErrorSource>,
1112    },
1113    /// The upstream provider accepted the connection but did not
1114    /// finish responding within `upstream_read_timeout_s`. Distinct
1115    /// from [`Self::Upstream`] so the response layer can map to
1116    /// `504 Gateway Timeout` (RFC 7231 §6.6.5) rather than
1117    /// `502 Bad Gateway`, letting operator dashboards distinguish
1118    /// "provider slow" (504 — check provider health) from
1119    /// "provider unreachable" (502 — check DNS / network / config).
1120    /// SDK retry behaviour is unchanged (both retryable).
1121    #[error("upstream request timed out: {context}")]
1122    UpstreamTimeout {
1123        /// Human-readable description.
1124        context: String,
1125        /// Underlying typed error, when one exists.
1126        #[source]
1127        source: Option<ErrorSource>,
1128    },
1129    /// The breaker requested immediate connection closure.
1130    #[error("connection aborted: {0}")]
1131    Abort(String),
1132    /// Required audit capture infrastructure is unavailable.
1133    #[error("audit capture unavailable: {context}")]
1134    Unavailable {
1135        /// Human-readable description.
1136        context: String,
1137        /// Underlying typed error, when one exists.
1138        #[source]
1139        source: Option<ErrorSource>,
1140    },
1141}

1233    /// HTTP status code corresponding to this failure.
1234    ///
1235    /// * `Blocked` / `Abort` are permanent policy verdicts for the current
1236    ///   session: 403 / 409 stop mainstream SDK auto-retry loops that
1237    ///   would otherwise interpret 429 as transient rate limiting and
1238    ///   burn budget re-hitting the same breaker.
1239    /// * `Unavailable` is transient: 503 is paired with `Retry-After` at
1240    ///   the response layer so intermediaries and clients back off in
1241    ///   bounded fashion (RFC 7231 §7.1.3).
1242    /// * `Unauthorized` is paired with `WWW-Authenticate` at the
1243    ///   response layer (RFC 7235 §3.1).
1244    pub fn status(&self) -> axum::http::StatusCode {
1245        match self {
1246            Self::BadRequest { .. } => axum::http::StatusCode::BAD_REQUEST,
1247            Self::Unauthorized(_) => axum::http::StatusCode::UNAUTHORIZED,
1248            Self::Blocked { .. } => axum::http::StatusCode::FORBIDDEN,
1249            Self::Upstream { .. } => axum::http::StatusCode::BAD_GATEWAY,
1250            Self::UpstreamTimeout { .. } => axum::http::StatusCode::GATEWAY_TIMEOUT,
1251            Self::Abort(_) => axum::http::StatusCode::CONFLICT,
1252            Self::Unavailable { .. } => axum::http::StatusCode::SERVICE_UNAVAILABLE,
1253        }
1254    }

1729            .user_agent(concat!("AgentVisor AI/", env!("CARGO_PKG_VERSION")))
1730            // TCP keepalive so pooled connections behind NAT/L4 LBs
1731            // with short idle windows (AWS NLB 350 s, GCP TCP 600 s,
1732            // stateful FWs often 60-120 s) do not turn every first
1733            // request per pool cycle into a full TCP+TLS re-handshake
1734            // that manifests as a spurious 502/`connection reset`.
1735            .tcp_keepalive(std::time::Duration::from_secs(30))
1736            // Cap pool idle at 60 s: shorter than any realistic NAT
1737            // window, longer than any burst-of-requests batch. Bounds
1738            // the pool memory footprint and forces frequent-enough
1739            // TLS refresh under a rolling cert rotation.
1740            .pool_idle_timeout(std::time::Duration::from_secs(60));

1742        // without decompression features, so the proxy cannot decode any
1743        // content-coding an origin might apply. RFC 9110 §12.5.3 says
1744        // an ABSENT Accept-Encoding permits any coding, so a CDN in
1745        // front of the upstream may legally gzip the response — which
1746        // the chat path then 502's on and the tool path (with the
1747        // decompression guard applied above) would also refuse. Advertise
1748        // `Accept-Encoding: identity` so the routine outcome is
1749        // "identity as requested" and the refusal guards are pure
1750        // defense-in-depth.

1753            reqwest::header::ACCEPT_ENCODING,

1767        // than a resource-starving hang.

1769        let read_timeout_s = config
1770            .upstream_read_timeout_s
1771            .unwrap_or(DEFAULT_UPSTREAM_READ_TIMEOUT_S);
1772        client_builder = client_builder.read_timeout(std::time::Duration::from_secs(read_timeout_s));
1773        if config.upstream_http2_prior_knowledge {
1774            client_builder = client_builder.http2_prior_knowledge();
1775        }
1776        let client = client_builder.build().map_err(PipelineError::upstream_source)?;
1777        let upstream_auth = resolve_upstream_auth(&config)?;
1778        let tool_auth = resolve_tool_auth(&config)?;
1779        let hot_metrics = Arc::new(HotMetrics::new(&metrics, config.strict_stage_budget));
1780        // S3: config validation already refused unsupported providers;
1781        // this defense-in-depth error covers direct constructors that
1782        // bypassed validate() (test builders).
1783        let provider_adapter = crate::provider::adapter_for(&config.provider).ok_or_else(|| {
1784            PipelineError::bad_request(format!("unsupported provider {:?}", config.provider))
1785        })?;
1786        tracing::info!(provider = provider_adapter.name(), "provider adapter selected");
1787        let mcp_admission = Arc::new(tokio::sync::Semaphore::new(config.mcp_concurrency));
1788        // R72 review F1 + F2 (landed R74): eager registration
1789        // stores the returned `Arc<Counter>` handle so the
1790        // refusal arm calls `.inc()` directly, bypassing the
1791        // registry mutex on the hot path. Prior R72 shape
1792        // eagerly registered but discarded the handle,
1793        // requiring a re-lookup per refusal.
1794        let mcp_admission_refusals_counter = metrics.counter(
1795            "av_mcp_admission_refusals_total",
1796            "MCP admission was refused because `mcp_concurrency` was already at capacity — \
1797             each admitted call can buffer up to 16 MiB in `read_limited_tool_response`, \
1798             so the cap bounds worst-case MCP resident memory. Sustained > 0 indicates \
1799             tool-upstream slowdown, a burst-traffic event, or an undersized \
1800             `mcp_concurrency`.",
1801        );
1802        Ok(Self {
1803            config,
1804            store,
1805            sandbox,
1806            bridge,
1807            sessions,
1808            worker,
1809            identity,
1810            hot_metrics,
1811            metrics,
1812            client,
1813            upstream_auth,
1814            tool_auth,
1815            tool_audit_gates: Arc::default(),
1816            tool_audits_emitted: Arc::default(),
1817            provider_adapter,
1818            finalizer,
1819            draining: Arc::new(std::sync::atomic::AtomicBool::new(false)),
1820            identity_rejection_window: Arc::new(parking_lot::Mutex::new((Instant::now(), 0))),
1821            journal_key,
1822            mcp_inflight: Arc::new(crate::inflight::InflightTracker::new()),

1831    /// `identity_hint`, when `Some`, is trusted as the resolved identity for
1832    /// this request — the internal `resolve_identity` + audit-enqueue step
1833    /// is skipped. Used by `chat_completions` (R72 F3 pre-admission gate)
1834    /// which already resolved the identity to fail-fast on unauthenticated
1835    /// probes before the body parses. Without this hint, the R72 pre-check
1836    /// path would `enqueue_transient_failure` on a JWKS-rotation race
1837    /// (pre-check success + re-check failure between the two calls),
1838    /// double-counting `av_identity_rejections_total` and burning two
1839    /// of the sampled full-audit slots per rejected request (R73 review
1840    /// F3). `None` preserves the historical behaviour for callers that
1841    /// do NOT pre-resolve.

1852        // accumulates ONE response message per step, so `n > 1` deltas
1853        // from different choices would interleave into a text no model
1854        // produced — and the ATIF step, receipt, and loop-detector
1855        // embedding would attest it (the tool-call accumulator is keyed
1856        // by choice index for exactly this reason). Refuse up front,
1857        // before any budget debit. Fail closed on ANY present `n` that
1858        // is not exactly the integer 1: upstream OpenAI-compatible
1859        // servers lax-coerce `2.0` / `"2"` to a real multi-choice
1860        // request, so an as_u64-only check was bypassable by float or
1861        // string encodings.

1865            Some(_) => {
1866                return Err(PipelineError::bad_request(

1868                ));

1928            // closing window. `get_or_open` recycles completed-close

1930            // complete is mid-close: an SDK auto-retry (e.g. after a
1931            // read-timeout disconnect triggered the background close)
1932            // used to hit a terminal 400 "already closed" — misleading
1933            // (the client closed nothing) and never retried by SDKs,

1935            // against the recycled id. 503 + Retry-After lets standard

1938            // never-complete — permanently (never recycled, never
1939            // evicted). They keep the terminal 400 (permanent
1940            // refusals route to 400 with no Retry-After);

2040                identity.clone(),

2053                payload,
2054                tokens_before: raw_prompt_tokens,

2104                    identity.clone(),

2131                    error.to_string(),

2139                principal_billing_id.as_deref(),

2149                identity.clone(),

2161            Err(error) => {

2165                    .refund_tokens(billed_tokens);

2167                    self.config.principal_budget.as_ref(),
2168                    principal_billing_id.as_deref(),
2169                ) {
2170                    ActionBudget::for_principal(self.store.as_ref(), principal_id, spec)
2171                        .refund_tokens(billed_tokens);

2174                    Arc::clone(&session),
2175                    identity.clone(),
2176                    StopReason::Other,
2177                    error.to_string(),

2185            session: Arc::clone(&session),

2190                "tokens_before": compression.tokens_before,

2194            analyze_loop,
2195            status: StatusId::Success,

2212            }),

2222        // `forward_chat` sets one).
2223        let capture_guard = ResponseCaptureGuard::new(
2224            self.worker.clone(),
2225            Arc::clone(&session),
2226            identity.clone(),
2227            response_attempt_id,
2228        );

2231        let admission_debit = AdmissionDebit {
2232            store: Arc::clone(&self.store),
2233            session_id: session.id.clone(),

2236                .config
2237                .principal_budget
2238                .as_ref()
2239                .map(|spec| (principal_id_for_budget(&identity), spec.clone())),

2244            session,

2250            response_permit: Some(response_permit),

2257    /// Run synchronous local gates without waiting for off-path journal,
2258    /// embedding, or broker work.
2259    ///
2260    /// The inline/blocking decision keys off PAYLOAD
2261    /// SIZE, not off whether a token budget happens to be configured.
2262    /// The cost of `prepare_chat` (parse-adjacent token scanning,
2263    /// sanitize, compression) is a function of body size — measured
2264    /// 1.9 ms p50 of uninterruptible CPU at a 128 KB body, which
2265    /// stalls every in-flight SSE stream's poll on that tokio worker.
2266    /// Previously a configured `budget.max_tokens` (the one thing that
2267    /// makes quotas work) was what moved the work to `spawn_blocking`,
2268    /// which was backwards. Small bodies stay inline (spawn_blocking
2269    /// costs ~10-20 µs of handoff — 2× the total gate cost at 75 B).
2270    ///
2271    /// The `budget.max_tokens.is_none()` clause below is load-bearing,
2272    /// not a leftover: with a budget configured, `prepare_chat` calls
2273    /// `ActionBudget::try_tokens` → `StateStore::try_spend`, which on
2274    /// the redis backend is a synchronous network round-trip — that
2275    /// must never run inline on a tokio worker, whatever the body size.
2276    /// The same applies to `[principal_budget]`: it drives its own
2277    /// `for_principal(...).try_tokens` store round-trip in
2278    /// `prepare_chat` even when the session `max_tokens` is unset, so
2279    /// it must equally force the blocking pool.
2280    /// Async wrapper over [`Self::prepare_chat`] that spawn-blocks on
2281    /// bodies large enough to stall the reactor. See `prepare_chat`
2282    /// for the `identity_hint` contract.
2283    pub async fn prepare_chat_nonblocking(
2284        &self,
2285        headers: &HeaderMap,
2286        payload: Value,
2287        body_bytes: usize,
2288        identity_hint: Option<AgentIdentity>,
2289    ) -> Result<PreparedRequest, PipelineError> {
2290        /// Above this the CPU-bound gates run on the blocking pool.
2291        /// 16 KiB ≈ 250 µs of gate work on the review's 2-vCPU
2292        /// reference box — comfortably under a poll-stall budget.
2293        const INLINE_BODY_LIMIT: usize = 16 * 1024;
2294        if body_bytes <= INLINE_BODY_LIMIT
2295            && self.config.budget.max_tokens.is_none()
2296            && self.config.principal_budget.is_none()
2297        {
2298            return self.prepare_chat(headers, payload, identity_hint);
2299        }
2300        let state = self.clone();
2301        let headers = headers.clone();
2302        tokio::task::spawn_blocking(move || state.prepare_chat(&headers, payload, identity_hint))
2303            .await
2304            .map_err(PipelineError::unavailable_source)?
2305    }

2307    /// Prepare a request and wait until its audit record is durably captured
2308    /// before the provider can observe it.
2309    pub async fn prepare_chat_durable(
2310        &self,
2311        headers: &HeaderMap,
2312        payload: Value,
2313    ) -> Result<PreparedRequest, PipelineError> {
2314        if let Ok(id) = session_id(headers) {
2315            if let Some(session) = self.sessions.get(&id) {
2316                session.wait_for_worker_jobs().await;
2317                if session.capture_failed() {
2318                    return Err(PipelineError::unavailable(
2319                        "session audit capture is incomplete".to_owned(),
2320                    ));
2321                }
2322            }
2323        }
2324        let state = self.clone();
2325        let headers = headers.clone();
2326        let mut prepared = tokio::task::spawn_blocking(move || state.prepare_chat(&headers, payload, None))
2327            .await
2328            .map_err(PipelineError::unavailable_source)??;
2329        prepared.session.wait_for_worker_jobs().await;
2330        if prepared.session.capture_failed() {
2331            let error = PipelineError::unavailable(
2332                "request audit capture failed before provider dispatch".to_owned(),
2333            );
2334            self.abandon_prepared(&mut prepared, StopReason::Other, &error.to_string());
2335            return Err(error);
2336        }
2337        if prepared.session.loop_state.state() == BreakerState::Open {
2338            let action = prepared.session.loop_state.action();
2339            // Latch the terminal enforcement actions only: Inject's
2340            // refusal here is genuinely transient (the client retry
2341            // re-enters prepare_chat, which injects the corrective
2342            // message and proceeds), so it must not make the id
2343            // terminal. Without this latch, a breaker trip delivered
2344            // from THIS site (opened during the audit wait) left the
2345            // id recyclable with a fresh breaker.
2346            if matches!(action, BreakerAction::Abort | BreakerAction::Reject) {
2347                prepared.session.latch_enforcement(StopReason::LoopDetected);
2348            }
2349            let error = match action {
2350                BreakerAction::Abort => {
2351                    PipelineError::Abort("semantic loop circuit breaker opened during audit".to_owned())
2352                }
2353                _ => PipelineError::blocked(
2354                    "semantic loop circuit breaker opened during audit; retry required".to_owned(),
2355                ),
2356            };
2357            self.abandon_prepared(&mut prepared, StopReason::LoopDetected, &error.to_string());
2358            return Err(error);
2359        }
2360        Ok(prepared)
2361    }

2363    /// Close out a prepared request that will never reach `forward_chat`.
2364    ///
2365    /// `prepare_chat` journals a non-terminal [`crate::worker::ResponseAttempt`]
2366    /// with its admission record; the matching terminal record normally comes
2367    /// from `forward_chat`'s failure path or the response relay. A caller that
2368    /// abandons the request after admission must submit the terminal failure
2369    /// record itself — otherwise the journal ends with a dangling non-terminal
2370    /// attempt and a later crash-recovery scan quarantines the whole session
2371    /// over a request the client already saw fail.
2372    fn abandon_prepared(&self, prepared: &mut PreparedRequest, stop_reason: StopReason, reason: &str) {
2373        // Every abandon site runs strictly before upstream dispatch
2374        // (durable-capture failure, breaker-open, marker-write failure),
2375        // so the admission token debit provably bought no LLM work —
2376        // refund it exactly once on BOTH ledgers (the guard zeroes
2377        // itself, so a hypothetical double abandon — or the guard's own
2378        // Drop — cannot double-refund).
2379        prepared.admission_debit.refund_now();
2380        let Some(permit) = prepared.response_permit.take() else {
2381            // Defensive: no permit means the guard's Drop is the only
2382            // resolver left. Leave it armed so the terminal record lands
2383            // through the plain worker queue on drop.
2384            return;
2385        };
2386        // Take the marker so the terminal job below retires it too.
2387        let response_marker = prepared.capture_guard.defuse();
2388        let mut job = self.failure_job(
2389            Arc::clone(&prepared.session),
2390            prepared.identity.clone(),
2391            stop_reason,
2392            reason.to_owned(),
2393        );
2394        job.response_marker = response_marker;
2395        job.response_attempt = Some(crate::worker::ResponseAttempt {
2396            id: prepared.capture_guard.attempt_id().to_owned(),
2397            terminal: true,
2398        });
2399        // Every sibling refusal path (`forward_chat` Err, both
2400        // `chat_completions` arms, the guard's own Drop) fails the
2401        // session closed on submit failure so the dangling attempt
2402        // cannot silently strand. Mirror that here: without it,
2403        // defusing the guard first would lose the safety net this whole
2404        // guard machinery exists to provide.
2405        if permit.submit(&self.worker, job).is_err() {
2406            prepared.session.mark_capture_failed();
2407        }
2408    }

2410    /// Forward a prepared OpenAI-compatible request to the configured provider.
2411    pub async fn forward_chat(
2412        &self,
2413        mut request: PreparedRequest,
2414    ) -> Result<ForwardedResponse, PipelineError> {
2415        // Digest the request payload so the in-flight marker can be
2416        // matched to the observed response bytes at recovery time.
2417        //
2418        // `serde_json::to_vec` on a `Value` is effectively infallible
2419        // (Value can only carry JSON-serialisable data), but we handle
2420        // the theoretical error path anyway: falling back to
2421        // `sha256(b"")` — the well-known empty digest — would make
2422        // every concurrent failed serialisation collide on the same
2423        // request_digest, silently violating the marker's
2424        // one-to-one-with-request invariant. Fall back to a
2425        // session-id-derived digest so a hypothetical failure at
2426        // least keeps distinct sessions distinct.
2427        // Serialize the payload exactly ONCE — the
2428        // bytes feed both the request digest and the upstream body.
2429        // Previously `serde_json::to_vec` here and reqwest's
2430        // `.json(&payload)` one screen apart produced identical bytes
2431        // twice per request.
2432        let payload_bytes = match serde_json::to_vec(&request.payload) {
2433            Ok(bytes) => Some(bytes),
2434            Err(error) => {
2435                tracing::error!(
2436                    %error,
2437                    session = %request.session.id,
2438                    "failed to serialise chat payload for request digest; falling back to session-derived digest"
2439                );
2440                None
2441            }
2442        };
2443        let request_digest = match &payload_bytes {
2444            Some(bytes) => av_core::digest::sha256_hex(bytes),
2445            None => av_core::digest::sha256_hex(request.session.id.as_bytes()),
2446        };
2447        // Cancellation fix: arm the guard with the marker id
2448        // BEFORE the awaited disk write. If the client disconnects
2449        // between here and the write completing, `spawn_blocking` still
2450        // runs the write to durability; the guard's terminal-failure
2451        // job then retires the marker via `clear_response_marker`,
2452        // which now treats an absent file as success — so a never-written
2453        // marker is a clean no-op and a written one is deleted. Under
2454        // the previous ordering, cancellation between the write and
2455        // `set_marker` left the marker stranded, poisoning quarantine
2456        // sets and (across restart) bricking receipts.
2457        let attempt_id = crate::worker::reserve_response_attempt_id();
2458        request.capture_guard.set_marker(attempt_id.clone());
2459        if let Err(error) = crate::worker::write_response_marker(
2460            std::path::Path::new(&self.config.atif_spool_dir),
2461            &self.journal_key,
2462            &request.session.id,
2463            &attempt_id,
2464            request_digest,
2465        )
2466        .await
2467        {
2468            // Fail closed: the marker is what lets a restart-time scan
2469            // (`inflight_response_sessions`) quarantine sessions whose
2470            // provider response may have been observed but never
2471            // captured. Dispatching without it would silently drop that
2472            // crash-durability guarantee, so refuse before upstream I/O
2473            // — terminating the journaled response attempt like every
2474            // other post-admission abort (see `abandon_prepared`).
2475            tracing::warn!(
2476                %error,
2477                session = %request.session.id,
2478                "could not write in-flight response marker; refusing upstream dispatch"
2479            );
2480            let client_error =
2481                PipelineError::unavailable("in-flight response marker could not be persisted".to_owned());
2482            self.abandon_prepared(&mut request, StopReason::Other, &client_error.to_string());
2483            return Err(client_error);
2484        }
2485        let PreparedRequest {
2486            session,
2487            identity,
2488            payload,
2489            lease,
2490            response_permit,
2491            capture_guard,
2492            client_authorization,
2493            upstream_passthrough_headers,
2494            mut admission_debit,
2495            ..
2496        } = request;
2497        let url = format!(
2498            "{}{}",
2499            self.config.upstream_url.trim_end_matches('/'),
2500            self.config.upstream_chat_path
2501        );
2502        // Reuse the digest's serialization for the upstream body; the
2503        // theoretical to_vec failure above falls back to reqwest's
2504        // own `.json()` serialization (same bytes, second attempt).
2505        let mut upstream_request = match payload_bytes {
2506            Some(bytes) => self
2507                .client
2508                .post(url)
2509                .header(reqwest::header::CONTENT_TYPE, "application/json")
2510                .body(bytes),
2511            None => self.client.post(url).json(&payload),
2512        };
2513        // Allowlisted provider-semantic headers (org/project/beta) relay
2514        // verbatim so org-scoped keys and beta-gated features keep
2515        // working through the proxy.
2516        for (name, value) in upstream_passthrough_headers {
2517            upstream_request = upstream_request.header(name, value);
2518        }
2519        if let Some((name, value)) = &self.upstream_auth {
2520            upstream_request = upstream_request.header(name.clone(), value.clone());
2521        } else if let Some(authorization) = client_authorization {
2522            // Passthrough mode: the client's own credential travels to the
2523            // upstream. `validate()` guarantees this is mutually exclusive
2524            // with static keys and with NHI identity enforcement.
2525            upstream_request = upstream_request.header(reqwest::header::AUTHORIZATION, authorization);
2526        }
2527        // Upstream latency (time-to-response-headers)
2528        // and error counters — the two failure-mode-table rows the
2529        // review marked "logs only, no metric".
2530        let upstream_started = Instant::now();
2531        // Last pre-dispatch point: from here the request may reach the
2532        // provider, so the admission debit stands (the `is_connect` arm
2533        // below refunds the one provably-not-sent case). Disarming here
2534        // also ends the cancellation-refund window — a disconnect
2535        // during `send()` keeps the debit, deliberately.
2536        let debited_tokens = admission_debit.take_for_dispatch();
2537        match upstream_request.send().await {
2538            Ok(response) => {
2539                self.hot_metrics
2540                    .upstream_latency_histogram
2541                    .observe_us(elapsed_us(upstream_started));
2542                if response.status().is_server_error() {
2543                    #[allow(clippy::indexing_slicing)]
2544                    {
2545                        self.hot_metrics.upstream_errors_counters[UpstreamErrorKind::Http5xx.index()].inc();
2546                    }
2547                }
2548                Ok(ForwardedResponse {
2549                    response,
2550                    billed_prompt_tokens: debited_tokens,
2551                    lease,
2552                    response_permit,
2553                    capture_guard,
2554                })
2555            }
2556            Err(error) => {
2557                let kind = UpstreamErrorKind::from_reqwest(&error);
2558                #[allow(clippy::indexing_slicing)]
2559                {
2560                    self.hot_metrics.upstream_errors_counters[kind.index()].inc();
2561                }
2562                // `reqwest::Error::Display` embeds the
2563                // request URL (see the redaction rule at
2564                // routes.rs::read_limited_tool_response — the same
2565                // rule applies here). `%error` on `reqwest::Error`
2566                // would render `error sending request for url
2567                // (https://api.openai.com/v1/chat/completions): dns
2568                // error: ...` into every OTLP sink. In multi-tenant
2569                // / air-gapped deployments the upstream URL is not
2570                // something the operator wants leaked to the
2571                // customer's SIEM (it may be an internal LiteLLM
2572                // router, an on-prem Azure resource name, a
2573                // per-tenant model deployment path). Log the
2574                // stable classifier + reqwest's structured
2575                // predicates only.
2576                //
2577                // Also propagate the classifier — NOT
2578                // `error.to_string()` — into the persisted failure
2579                // event so the URL doesn't land in the on-disk
2580                // journal or in the Bridge-published failure
2581                // record either.
2582                let client_reason = classify_upstream_error(&error);
2583                // `is_connect()` proves the request never left this
2584                // process (TCP/TLS establishment failed) — the admission
2585                // token debit bought no LLM work; refund it on BOTH
2586                // ledgers, in the same order they were spent. Any other
2587                // send error (timeout, body, reset mid-response) may have
2588                // reached the provider, so the debit conservatively
2589                // stands. Best-effort; a store blip must never turn a
2590                // 502 into a 5xx cascade.
2591                if error.is_connect() {
2592                    ActionBudget::new(self.store.as_ref(), &session.id, &self.config.budget)
2593                        .refund_tokens(debited_tokens);
2594                    if let Some(spec) = self.config.principal_budget.as_ref() {
2595                        let principal_id = principal_id_for_budget(&identity);
2596                        ActionBudget::for_principal(self.store.as_ref(), &principal_id, spec)
2597                            .refund_tokens(debited_tokens);
2598                    }
2599                }
2600                tracing::warn!(
2601                    session = %session.id,
2602                    category = client_reason,
2603                    error.status = ?error.status(),
2604                    error.is_timeout = error.is_timeout(),
2605                    error.is_connect = error.is_connect(),
2606                    error.is_body = error.is_body(),
2607                    "upstream forwarding failed"
2608                );
2609                // Route timeouts (upstream accepted the connection but
2610                // never finished responding) to 504 Gateway Timeout
2611                // instead of the generic 502 Bad Gateway. Operator
2612                // dashboards keying off HTTP status can now distinguish
2613                // "provider slow" (504) from "provider unreachable"
2614                // (502) without joining against the metric label.
2615                let client_error = if error.is_timeout() {
2616                    PipelineError::upstream_timeout(client_reason.to_owned())
2617                } else {
2618                    PipelineError::upstream(client_reason.to_owned())
2619                };
2620                let persisted_reason = format!("upstream_{client_reason}");
2621                if let Some(permit) = response_permit {
2622                    let (response_marker, response_attempt_id) = capture_guard.disarm();
2623                    let capture_session = Arc::clone(&session);
2624                    let mut job =
2625                        self.failure_job(session, identity, StopReason::Other, persisted_reason.clone());
2626                    job.response_marker = response_marker;
2627                    job.response_attempt = Some(crate::worker::ResponseAttempt {
2628                        id: response_attempt_id,
2629                        terminal: true,
2630                    });
2631                    // This terminal failure capture is the only thing that
2632                    // retires the durable in-flight marker and the
2633                    // journalled response attempt. If the shard is full,
2634                    // fail the session closed instead of dropping the
2635                    // capture silently: otherwise the marker strands, no
2636                    // terminal response event exists, and the session can
2637                    // still close "cleanly" with a receipt over an
2638                    // unresolved response attempt.
2639                    if permit.submit(&self.worker, job).is_err() {
2640                        capture_session.mark_capture_failed();
2641                    }
2642                } else {
2643                    // Defensive (a permit always exists here): keep the
2644                    // guard armed so its drop retires the marker and
2645                    // closes the attempt through the plain worker queue.
2646                    drop(capture_guard);
2647                    self.enqueue_failure(session, identity, StopReason::Other, persisted_reason)?;
2648                }
2649                Err(client_error)
2650            }
2651        }
2652    }

2654    /// Intercept one MCP JSON-RPC tool call, emit its OCSF verdict
2655    /// asynchronously, and return the immediate authorization decision.
2656    pub fn intercept_tool(&self, headers: &HeaderMap, raw: &[u8]) -> Result<ToolVerdict, PipelineError> {
2657        self.intercept_tool_with_session(headers, raw)
2658            .map(|(verdict, _session)| verdict)
2659    }
2660
2661    /// Core of [`Self::intercept_tool`] that also returns the bound session.
2662    ///
2663    /// Callers that must await audit durability need the same session the
2664    /// verdict was recorded under: re-deriving it from headers would mint a
2665    /// fresh random id for a header-less request (see [`session_id`]) and
2666    /// look up a session that does not exist.
2667    fn intercept_tool_with_session(
2668        &self,
2669        headers: &HeaderMap,
2670        raw: &[u8],
2671    ) -> Result<(ToolVerdict, Arc<Session>), PipelineError> {
2672        let session_id = session_id(headers)?;
2673        let workflow = workflow(headers, &self.config.default_workflow)?;
2674        let parsed_call = av_sandbox::parse_tool_call(raw).ok();
2675        let required_scope = parsed_call
2676            .as_ref()
2677            .map(|request| tool_scope(&request.tool))
2678            .unwrap_or_else(|| TOOL_INVOKE_SCOPE.to_owned());
2679        let identity = match self.resolve_identity(headers, Some(&required_scope)) {
2680            Ok(identity) => identity,
2681            Err(error) => {
2682                self.enqueue_transient_failure(&session_id, StopReason::IdentityRejected, error.to_string())?;
2683                return Err(error);
2684            }
2685        };
2686        // Tool interception must NOT resurrect a closed session: a
2687        // tool call extends an in-progress conversation, so silently
2688        // opening a fresh session under the same id would let the
2689        // client accumulate tool calls past a signed receipt boundary
2690        // (chain-of-custody split). Chat requests use `get_or_open`
2691        // (recycles closed ids); the strict `_no_reopen` variant here
2692        // makes the `is_closed()` check below fire.
2693        let session =
2694            self.sessions
2695                .get_or_open_no_reopen(&session_id, workflow, &identity, &self.config.breaker);
2696        let admission = session.admission_guard();
2697        validate_session_binding(&session, workflow, &identity)?;
2698        session.refresh_identity(&identity);
2699        if session.is_closed() {
2700            return Err(PipelineError::bad_request("session is already closed".to_owned()));
2701        }
2702        if session.capture_failed() {
2703            return Err(PipelineError::unavailable(
2704                "session audit capture is incomplete".to_owned(),
2705            ));
2706        }
2707        session.touch();
2708        let worker_permit = self
2709            .worker
2710            .try_reserve(&session_id)
2711            .map_err(PipelineError::unavailable_source)?;
2712        let verdict = match parsed_call.as_ref() {
2713            // `sandbox.check`'s budget gate spends before we return, so we must
2714            // veto workflow-mismatched consequential tools before it runs.
2715            Some(request)
2716                if workflow == Workflow::Unsigned
2717                    && self
2718                        .config
2719                        .consequential_tools
2720                        .iter()
2721                        .any(|required| required == &request.tool) =>
2722            {
2723                let reason = format!("tool {:?} requires a signed workflow", request.tool);
2724                ToolVerdict::Blocked {
2725                    tool: request.tool.clone(),
2726                    stage: "policy",
2727                    reason: reason.clone(),
2728                    response: av_sandbox::rpc::authorization_error(request.id.as_ref(), &reason),
2729                    elapsed_us: 0,
2730                }
2731            }
2732            _ => {
2733                let principal_binding = self
2734                    .config
2735                    .principal_budget
2736                    .as_ref()
2737                    .map(|spec| (principal_id_for_budget(&identity), spec));
2738                let principal_ref = principal_binding.as_ref().map(|(id, spec)| (id.as_str(), *spec));
2739                self.sandbox
2740                    .check_with_principal(self.store.as_ref(), &session_id, principal_ref, raw)
2741            }
2742        };
2743        let (status, payload) = match &verdict {
2744            ToolVerdict::Allowed {
2745                tool,
2746                budget_remaining,
2747                elapsed_us,
2748                ..
2749            } => (
2750                StatusId::Success,
2751                serde_json::json!({
2752                    "tool": tool,
2753                    "allowed": true,
2754                    "budget_remaining": (*budget_remaining != u64::MAX)
2755                        .then_some(*budget_remaining),
2756                    "budget_unlimited": *budget_remaining == u64::MAX,
2757                    "decision_us": elapsed_us,
2758                }),
2759            ),
2760            ToolVerdict::Blocked {
2761                tool,
2762                stage,
2763                reason,
2764                elapsed_us,
2765                ..
2766            } => (
2767                StatusId::Failure,
2768                serde_json::json!({
2769                    "tool": tool,
2770                    "allowed": false,
2771                    "stage": stage,
2772                    "reason": reason,
2773                    "decision_us": elapsed_us,
2774                }),
2775            ),
2776            // `ToolVerdict` is `#[non_exhaustive]` (R56 SemVer
2777            // hardening) — a future variant (e.g. `Warn` for shadow-
2778            // mode, `Deferred` for async policy evaluation) must NOT
2779            // be silently forwarded to the tool server. Fail-closed:
2780            // emit a Blocked-shaped audit payload and log at ERROR
2781            // level so operators discover the unhandled variant
2782            // during rollout. Same defensive posture the sandbox's
2783            // own `is_allowed` uses (whitelist Allowed, treat
2784            // everything else as blocked).
2785            _ => {
2786                tracing::error!(
2787                    "unhandled ToolVerdict variant reached the harness matcher; \
2788                     failing closed. This is a bug — extend the match to handle the new variant."
2789                );
2790                (
2791                    StatusId::Failure,
2792                    serde_json::json!({
2793                        "tool": "<unknown>",
2794                        "allowed": false,
2795                        "stage": "unhandled_verdict",
2796                        "reason": "unhandled ToolVerdict variant; fail-closed",
2797                    }),
2798                )
2799            }
2800        };
2801        let tool_call_id = parsed_call
2802            .as_ref()
2803            .and_then(|request| request.id.as_ref())
2804            .map(|value| value.as_str().map_or_else(|| value.to_string(), str::to_owned))
2805            .unwrap_or_else(av_core::new_event_uid);
2806        let tool_calls = parsed_call.as_ref().map(|request| {
2807            vec![av_atif::ToolCall {
2808                tool_call_id: tool_call_id.clone(),
2809                function_name: request.tool.clone(),
2810                arguments: request.arguments.clone(),
2811                extra: None,
2812            }]
2813        });
2814        let observation = parsed_call.as_ref().map(|_| av_atif::Observation {
2815            results: vec![av_atif::ObservationResult {
2816                source_call_id: Some(tool_call_id),
2817                content: Some(Value::String(payload.to_string())),
2818                subagent_trajectory_ref: None,
2819                extra: None,
2820            }],

3051    pub(crate) fn resolve_identity(
3052        &self,
3053        headers: &HeaderMap,
3054        required_scope: Option<&str>,
3055    ) -> Result<AgentIdentity, PipelineError> {
3056        // Reuse the duplicate-header refusal
3057        // pattern for `Authorization` on the identity hot path.
3058        // Previously `HeaderMap::get(AUTHORIZATION)` returned the
3059        // first value while the `PreparedRequest.client_authorization`
3060        // capture and header-smuggling proxies could observe a
3061        // merged `A, B` form — auth split-brain: the harness
3062        // authenticates as A while log aggregators / WAFs / OTLP
3063        // exporters attribute the request to B. Refuse the multi-
3064        // value case at ingress, symmetric with the X-AV-Session /
3065        // X-AV-Workflow guards.
3066        // RFC 7235 §2.1 declares the auth-scheme token
3067        // case-insensitive. Historically `strip_prefix("Bearer ")`
3068        // silently dropped `bearer eyJ...` / `BEARER eyJ...` /
3069        // `Bearer\teyJ...` to `None`, then the outer match arm
3070        // returned anonymous (when require_identity=false, the
3071        // shipped default) — the caller believed they had
3072        // authenticated while the audit trail attributed the
3073        // request to `anonymous`. That's the exact repudiation
3074        // vector the surrounding refuse-anonymous-with-token
3075        // guard was written to close. Now strip the scheme
3076        // case-insensitively.
3077        let bearer = single_header(headers, "authorization")?

3079            .and_then(strip_bearer_scheme);

3081            (Some(token), Some(validator)) => {

3090                    if let Some(required) = required_scope {
3091                        if !scope_allows(&validated.claims.scopes, required) {
3092                            // RFC 6750 §3.1: a syntactically valid, correctly
3093                            // signed token that lacks the required scope is an
3094                            // AUTHORIZATION failure — 403 insufficient_scope —
3095                            // not a 401. Returning 401 (the old behavior) told
3096                            // SDK token-refreshers to re-authenticate, which
3097                            // cannot help: the grant itself lacks the scope.
3098                            return Err(PipelineError::blocked(format!(

3100                            )));
3101                        }
3102                    }
3103                }

3105            }

3107            // configured. Two legitimate cases and one attack surface:
3108            //  (a) `upstream_authorization_passthrough = true`: the
3109            //      client's bearer is the CREDENTIAL FOR THE UPSTREAM

3111            //      anonymous locally so the request proceeds and the

3114            //  (b) Neither passthrough nor validator: a bearer sent to

3116            //      REJECTED with a diagnostic 401 (see the closing
3117            //      paragraph below and the arm's else branch) rather

3122            //

3124            // and we silently attribute the request as `anonymous`

3126            // e56... by rejecting here. But that broke passthrough
3127            // mode. Restore acceptance when passthrough is on so the
3128            // BYO-key path works, and continue rejecting only when
3129            // NO passthrough is configured and identity was not
3130            // required (i.e. the client presenting a bearer to a

3328/// Strip the RFC 7235 §2.1 `Bearer` scheme from an Authorization header
3329/// value, case-insensitively (per the RFC), and allow one or more
3330/// whitespace characters (`SP` / `HTAB`) between the scheme token and
3331/// the credential (per `token1` production in §2.1).
3332///
3333/// `str::strip_prefix("Bearer ")` used to be the sole
3334/// parser. It missed `bearer`, `BEARER`, and (per the RFC's own
3335/// grammar) `Bearer\t...`, silently dropping to `None` and — when
3336/// `require_identity = false` — letting the request execute as
3337/// anonymous while the caller believed they had authenticated.
3338pub(crate) fn strip_bearer_scheme(value: &str) -> Option<&str> {
3339    let (scheme, rest) = value.split_at_checked(6)?;
3340    if !scheme.eq_ignore_ascii_case("Bearer") {
3341        return None;
3342    }
3343    let rest = rest.trim_start_matches([' ', '\t']);
3344    if rest.len() == value.len() - scheme.len() {
3345        // No whitespace at all after `Bearer` — the credential MUST
3346        // be separated per the ABNF (`token1 = 1*( SP / HTAB )
3347        // token`). Refuse `BearerXYZ` — that's a scheme other than
3348        // Bearer's credential, not a Bearer credential.
3349        return None;
3350    }
3351    if rest.is_empty() {
3352        // Empty credential.
3353        return None;
3354    }
3355    Some(rest)
3356}

3404fn last_message_text(payload: &Value) -> String {
3405    let Some(message) = payload
3406        .get("messages")
3407        .and_then(Value::as_array)
3408        .and_then(|messages| messages.last())
3409    else {
3410        return String::new();
3411    };
3412    // Primary source: string `content`.
3413    if let Some(text) = message.get("content").and_then(Value::as_str) {
3414        if !text.is_empty() {
3415            return text.to_owned();
3416        }
3417    }
3418    // Mirror of the response-side fix: a
3419    // tool-call-only assistant message carries `content: null` and a
3420    // `tool_calls` array. `last_message_text` returning "" here would
3421    // feed a zero-vector embedding into the breaker in worker.rs and
3422    // false-trip legitimate tool-driven agents on the request side.
3423    // Synthesize `tool_name(arguments)` for each tool call, in
3424    // wire-order — same shape as the response-side synthesis in
3425    // routes.rs::AbortFinalizingStream::submit_response_capture.
3426    //
3427    // Beware: `function.arguments` reaches the
3428    // wire in TWO shapes: a JSON-encoded string (OpenAI reference)
3429    // OR a bare JSON object (some clients / most Anthropic-shaped
3430    // gateways). `.as_str()` on the object variant returned None and
3431    // the synthesis collapsed to `tool_name()` — worse than the
3432    // pre-fix behavior, because now two different-argument calls to
3433    // the same tool produced IDENTICAL synthesized text, causing
3434    // false loop trips. Match the tolerant pattern from
3435    // `atif_capture_from_request` in the same file: render the raw
3436    // Value directly when it isn't a stringified JSON. Empty and
3437    // missing collapse to "" and get gated off analyze_loop
3438    // downstream.
3439    if let Some(tool_calls) = message.get("tool_calls").and_then(Value::as_array) {
3440        let synthesized: Vec<String> = tool_calls
3441            .iter()
3442            .filter_map(|call| {
3443                let name = call
3444                    .get("function")
3445                    .and_then(|f| f.get("name"))
3446                    .and_then(Value::as_str)?;
3447                let arguments = call.get("function").and_then(|f| f.get("arguments"));
3448                let args_rendered = match arguments {
3449                    Some(Value::String(s)) => s.clone(),
3450                    Some(other) => serde_json::to_string(other).unwrap_or_default(),
3451                    None => String::new(),
3452                };
3453                Some(format!("{name}({args_rendered})"))
3454            })
3455            .collect();
3456        if !synthesized.is_empty() {
3457            return synthesized.join("\n");
3458        }
3459    }
3460    // Content-parts array (multimodal). Concatenate every text part
3461    // so an assistant reply built from mixed text+image parts still
3462    // feeds a non-zero vector to the breaker rather than "".
3463    if let Some(parts) = message.get("content").and_then(Value::as_array) {
3464        let text: String = parts
3465            .iter()
3466            .filter_map(|part| part.get("text").and_then(Value::as_str))
3467            .collect::<Vec<_>>()
3468            .join("\n");
3469        if !text.is_empty() {
3470            return text;
3471        }
3472    }
3473    String::new()
3474}

3491fn atif_capture_from_request(payload: &Value) -> Result<AtifCapture, PipelineError> {
3492    // Split the two failure classes so support engineers can tell
3493    // "field missing" from "field is the wrong shape" — chasing a
3494    // phantom "no messages" ticket for a caller who typed
3495    // `"messages": {}` used to be the top diagnostic-quality
3496    // complaint. The OpenAI Responses API also uses `input`
3497    // instead of `messages`; the missing-field message now points
3498    // directly at the correct fix.
3499    let messages_value = payload
3500        .get("messages")
3501        .ok_or_else(|| PipelineError::bad_request("chat payload is missing 'messages'".to_owned()))?;
3502    let messages = messages_value.as_array().ok_or_else(|| {
3503        PipelineError::bad_request(format!(
3504            "'messages' must be a JSON array, got {}",
3505            json_type_name(messages_value)
3506        ))
3507    })?;
3508    let message = messages
3509        .last()
3510        .ok_or_else(|| PipelineError::bad_request("chat payload 'messages' is empty".to_owned()))?;
3511    let role = message.get("role").and_then(Value::as_str);
3512    let source = match role {
3513        // "function" is the legacy OpenAI function-calling spelling of
3514        // "tool" (a tool result fed back to the model) — still emitted
3515        // by older LangChain, AutoGen and llama-index versions.
3516        // Refusing it 400'd legitimate agent traffic.
3517        Some("system" | "developer" | "tool" | "function") => av_atif::Source::System,
3518        Some("user") => av_atif::Source::User,
3519        Some("assistant") => av_atif::Source::Agent,
3520        Some(other) => {
3521            return Err(PipelineError::bad_request(format!(
3522                "unsupported chat role {other:?}"
3523            )))
3524        }
3525        None => return Err(PipelineError::bad_request("chat message has no role".to_owned())),
3526    };
3527    let content = message
3528        .get("content")
3529        .filter(|value| value.is_string() || value.is_array())
3530        .cloned()
3531        .unwrap_or_else(|| Value::String(String::new()));
3532    let tool_calls = message.get("tool_calls").and_then(Value::as_array).map(|calls| {
3533        calls
3534            .iter()
3535            .filter_map(|call| {
3536                let function = call.get("function")?;
3537                let arguments = function
3538                    .get("arguments")
3539                    .cloned()
3540                    .unwrap_or(Value::Object(Default::default()));
3541                let arguments = arguments
3542                    .as_str()
3543                    .and_then(|raw| serde_json::from_str(raw).ok())
3544                    .unwrap_or(arguments);
3545                Some(av_atif::ToolCall {
3546                    tool_call_id: call
3547                        .get("id")
3548                        .and_then(Value::as_str)
3549                        .map_or_else(av_core::new_event_uid, str::to_owned),
3550                    function_name: function.get("name")?.as_str()?.to_owned(),
3551                    arguments,
3552                    extra: None,
3553                })
3554            })
3555            .collect()
3556    });
3557    let observation = (role == Some("tool")).then(|| av_atif::Observation {
3558        results: vec![av_atif::ObservationResult {
3559            source_call_id: None,
3560            content: Some(content.clone()),
3561            subagent_trajectory_ref: None,
3562            extra: message
3563                .get("tool_call_id")
3564                .cloned()
3565                .map(|tool_call_id| serde_json::json!({"tool_call_id": tool_call_id})),
3566        }],
3567    });
3568    Ok(AtifCapture {
3569        source,
3570        message: content,
3571        reasoning_content: if source == av_atif::Source::Agent {
3572            message
3573                .get("reasoning_content")
3574                .and_then(Value::as_str)
3575                .map(str::to_owned)
3576        } else {
3577            None
3578        },
3579        model_name: if source == av_atif::Source::Agent {
3580            payload.get("model").and_then(Value::as_str).map(str::to_owned)
3581        } else {
3582            None
3583        },
3584        tool_calls: if source == av_atif::Source::Agent {
3585            tool_calls
3586        } else {
3587            None
3588        },
3589        observation,
3590        llm_call_count: (source == av_atif::Source::Agent).then_some(1),
3591    })
3592}
