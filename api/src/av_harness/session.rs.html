15pub enum Workflow {
16    /// Consequential actions: event chain + receipt at close.

18    /// Exploratory: ATIF capture, receipt only on promotion.

20}

25        match self {

41/// A live session.
42pub struct Session {
43    /// Session id.
44    pub id: String,
45    /// A per-open UUID that disambiguates
46    /// recycled session ids in the off-path vector store. Client
47    /// callers may reuse the same `id` after a finalize; the
48    /// `SessionRegistry::get_or_open` path historically returned a
49    /// fresh `Session` under the same key. A Qdrant vector search
50    /// filtered ONLY on `session_id` would then return the prior
51    /// incarnation's vectors, causing false semantic-loop signals
52    /// from an unrelated past session. Every vector record and
53    /// query is now scoped by
54    /// `session_scope = "{id}#{generation_uid}"` — the UUID is
55    /// fresh for every `Session::new`, so a recycled id lands in
56    /// a distinct scope. The bare `id` remains the primary key for
57    /// on-disk artifacts (spool paths, journal filenames), which
58    /// use crash-safe generation logic of their own.
59    pub session_scope: String,
60    /// Workflow kind.
61    pub workflow: Workflow,
62    /// Agent identity bound at open (from NHI validation).
63    pub identity: AgentIdentity,
64    /// Most recently validated identity, including current TTL.
65    latest_identity: Mutex<AgentIdentity>,
66    /// Monotonic per-session event sequence (authoritative order).
67    seq: AtomicU64,
68    /// Monotonic authenticated-journal record index.
69    journal_index: AtomicU64,
70    /// Loop-breaker state.
71    pub loop_state: av_loopdetect::SessionLoopState,
72    /// Signed workflow: incremental event chain.
73    pub chain: Mutex<EventChain>,
74    /// Unsigned workflow: ATIF steps.
75    pub atif: Mutex<av_atif::TrajectoryBuilder>,
76    /// Aggregates for the receipt.
77    pub totals: Totals,
78    /// Last-activity timestamp (dashboard/display only), epoch ms.
79    /// Wall-clock time subject to VM/NTP jumps; not safe to use for
80    /// idle-eviction decisions — the private `last_activity_instant`
81    /// field carries a monotonic anchor the idle sweeper uses instead.
82    pub last_activity_ms: AtomicU64,
83    /// Monotonic last-activity instant. Used by the idle sweeper so a
84    /// forward wall-clock jump (VM resume, NTP correction) cannot make
85    /// active sessions look premature-idle: `Instant` is bounded by
86    /// real elapsed time on the system's monotonic clock.
87    last_activity_instant: Mutex<Instant>,
88    /// Last normalized provider or enforcement stop reason id.
89    last_stop_reason_id: AtomicU64,
90    /// Sticky enforcement latch (0 = never tripped, else the enforcement
91    /// stop-reason id). Set ONLY at genuine enforcement-refusal sites
92    /// (chat token-budget refusal, mid-stream budget refusal, loop-
93    /// breaker open refusal) — never by the last-write-wins
94    /// `last_stop_reason_id`, which later successful responses
95    /// overwrite and which quota-BACKEND failures and blocked tool
96    /// calls also write, making it an unreliable proxy for "enforcement
97    /// caused this session's end" (false lockouts and losable refusals,
98    /// both verified). Gates same-id recycling after close.
99    enforcement_tripped: AtomicU64,
100    /// Set once closed (idempotent close).
101    pub closed: AtomicU64,
102    /// Serializes close against request admission.
103    admission: RwLock<()>,
104    /// Forwarded chat responses that have not completed or aborted.
105    active_streams: AtomicU64,
106    /// Notification used by finalization to await active response completion.
107    streams_drained: tokio::sync::Notify,
108    /// Issued signed or retroactive receipt.
109    pub receipt: Mutex<Option<av_receipts::Receipt>>,
110    /// Persisted unsigned ATIF artifact.
111    pub atif_path: Mutex<Option<PathBuf>>,
112    /// Set once an unsigned artifact is promoted.
113    promoted: AtomicU64,
114    /// Worker jobs accepted but not yet fully captured.
115    pending_jobs: AtomicU64,
116    /// Notification used by finalization to await a drained worker queue.
117    jobs_drained: tokio::sync::Notify,
118    /// Set when an upstream action could not be captured completely.
119    capture_failed: AtomicU64,
120    /// RAM-cliff guard: count of unsigned ATIF steps
121    /// journaled by the worker. The steps themselves live ONLY in the
122    /// events journal; close rebuilds the trajectory from disk.
123    atif_steps: AtomicU64,
124    /// The lifecycle chain state — the single
125    /// holder of Open/Draining/Sealed/Complete, advanced only via
126    /// CAS `transition` calls that name their legal source states.
127    /// The `closed` claim flag above and the orthogonal properties
128    /// (`capture_failed`, `promoted`, admission) deliberately stay
129    /// outside the chain (see `SessionState` docs).
130    lifecycle: std::sync::atomic::AtomicU8,
131}

133/// The session lifecycle chain, held in
134/// one `AtomicU8` and advanced only through CAS transitions that
135/// name their expected source state — illegal transitions become
136/// impossible rather than untested.
137///
138/// Two flags deliberately stay OUTSIDE the chain (mirroring the S2
139/// plan's `admission_open` exception): `capture_failed` and
140/// `promoted` are orthogonal *properties*, not chain positions — a
141/// capture-failed session still seals and completes (the
142/// capture-failed seal path), and promotion happens to already-
143/// complete sessions. Folding them in would multiply states for
144/// combinations that all remain reachable.
145#[repr(u8)]
146#[derive(Debug, Clone, Copy, PartialEq, Eq)]
147#[non_exhaustive]
148pub enum SessionState {
149    /// Admitting requests.
150    Open = 0,
151    /// `try_close` claimed the close; streams may still be draining.
152    /// A failed finalize returns to `Open` via `reset_close`.
153    Draining = 1,
154    /// The receipt/ATIF artifact is durably committed; the fallible
155    /// close tail (bridge emit + journal removal) is still pending.
156    Sealed = 2,
157    /// The close ran to full completion; the registry may evict.
158    Complete = 3,
159}
160
161impl SessionState {
162    fn decode(raw: u8) -> Self {
163        match raw {
164            1 => Self::Draining,
165            2 => Self::Sealed,
166            3 => Self::Complete,
167            _ => Self::Open,
168        }
169    }
170}

319    /// Attempt to claim the close transition. Only one caller can hold the
320    /// claim at a time; a failed finalize resets it (`reset_close`) so the
321    /// close can be retried.
322    pub fn try_close(&self) -> bool {
323        let claimed = self
324            .closed
325            .compare_exchange(0, 1, Ordering::AcqRel, Ordering::Acquire)
326            .is_ok();
327        if claimed {
328            self.transition(&[SessionState::Open], SessionState::Draining);
329        }
330        claimed
331    }

333    /// The lifecycle chain position.
334    pub fn lifecycle_state(&self) -> SessionState {
335        SessionState::decode(self.lifecycle.load(Ordering::Acquire))
336    }
337
338    /// One CAS transition of the lifecycle state machine (S2 step 3:
339    /// the enum IS the state; the mirror flags are deleted), tolerant
340    /// of two documented behaviors of the close machinery:
341    ///
342    /// * **Idempotent/late re-marks** — a current state at or past
343    ///   `to` in the chain is a no-op. This covers idempotent flag
344    ///   stores AND the `closed` claim flag's post-seal semantics:
345    ///   `reset_close`/`try_close` toggle the claim after an
346    ///   artifact is already committed (failed close-tail retries),
347    ///   where `is_closed()` stays true via `artifact_committed`
348    ///   and the chain must stay Sealed.
349    /// * **Multiple legal sources** — signed recovery seals adopted
350    ///   sessions without a prior claim (Open→Sealed); a live close
351    ///   seals under its claim (Draining→Sealed).
352    ///
353    /// Any other source state is a model violation: the transition
354    /// is REFUSED (the CAS never fires — illegal transitions are
355    /// impossible, not merely untested), and debug builds (the
356    /// entire test suite) additionally panic naming the transition.
357    fn transition(&self, from: &[SessionState], to: SessionState) {
358        let mut current = self.lifecycle.load(Ordering::Acquire);
359        loop {
360            let state = SessionState::decode(current);
361            if from.contains(&state) {
362                match self.lifecycle.compare_exchange_weak(
363                    current,
364                    to as u8,
365                    Ordering::AcqRel,
366                    Ordering::Acquire,
367                ) {
368                    Ok(_) => break,
369                    Err(actual) => current = actual,
370                }
371            } else if state == to || (state as u8) > (to as u8) {
372                break;
373            } else {
374                debug_assert!(
375                    false,
376                    "illegal lifecycle transition {state:?}->{to:?} (expected from {from:?})"
377                );
378                break;
379            }
380        }
381    }

387        self.closed.load(Ordering::Acquire) != 0
388            || matches!(

449    pub(crate) async fn wait_for_streams(&self) {
450        loop {
451            // Subscribe first (via enable), then check the guard; the same Notified
452            // must remain pinned through the await, or notify_waiters() firing in
453            // the interval between drop and re-subscribe is lost forever.
454            let notified = self.streams_drained.notified();
455            let mut notified = std::pin::pin!(notified);
456            notified.as_mut().enable();
457            if self.active_streams.load(Ordering::Acquire) == 0 {
458                return;
459            }
460            notified.await;
461        }
462    }

498    /// Recreate a closed unsigned session from a persisted ATIF trajectory.
499    ///
500    /// `next_seq` is the number of per-session events already published to
501    /// the bridge for this session (steps ↔ journal records are 1:1, so the
502    /// step count is exact). Without restoring it, an adopted session
503    /// reports `peek_seq() == 0` and a post-crash SESSION_CLOSE or
504    /// retroactive-receipt event is minted with `metadata.sequence = 0`,
505    /// colliding with the session's first step event already on the bridge
506    /// (both sibling recovery paths call `restore_next_seq`; this one used
507    /// to be the exception).
508    pub fn recover_unsigned(
509        id: String,
510        identity: AgentIdentity,
511        breaker: av_loopdetect::BreakerConfig,
512        path: PathBuf,
513        metrics: Option<&av_atif::FinalMetrics>,
514        next_seq: u64,
515    ) -> Result<Self, String> {
516        let session = Self::new(id, Workflow::Unsigned, identity, breaker);
517        session.restore_next_seq(next_seq);
518        // A recovered artifact is by definition past close: claim the
519        // close (fresh session — the claim cannot fail) and seal, so
520        // the lifecycle chain walks the same Open→Draining→Sealed
521        // path a live close does.
522        let claimed = session.try_close();
523        debug_assert!(claimed, "recover_unsigned starts from a fresh session");
524        session.mark_artifact_committed();
525        *session.atif_path.lock() = Some(path);
526        if let Some(metrics) = metrics {
527            let prompt_tokens = recovered_counter(metrics.total_prompt_tokens, "prompt tokens")?;
528            let completion_tokens = recovered_counter(metrics.total_completion_tokens, "completion tokens")?;
529            let cached_tokens = recovered_counter(metrics.total_cached_tokens, "cached tokens")?;
530            session
531                .totals
532                .prompt_tokens
533                .store(prompt_tokens, Ordering::Release);
534            session
535                .totals
536                .completion_tokens
537                .store(completion_tokens, Ordering::Release);
538            session
539                .totals
540                .cached_tokens
541                .store(cached_tokens, Ordering::Release);
542            if let Some(extra) = metrics.extra.as_ref() {
543                // Cost has two possible sources on the persisted extra:
544                //   1. `cost_usd_micros` — the current writer's exact
545                //      u64 field.
546                //   2. missing / null — an older writer's format,
547                //      pre-dating the cost_usd_micros key.
548                // Falling straight to `recovered_counter(...)?` on the
549                // missing key returned 0, silently zeroing the session's
550                // recovered cost even when `metrics.total_cost_usd` (the
551                // ATIF-level fallback consumed by the non-extra branch
552                // below) was present. Prefer the exact u64 when it's
553                // there, otherwise use the ATIF-level float — same logic
554                // as the pre-extra branch.
555                let cost_usd_micros = match extra.get("cost_usd_micros").and_then(serde_json::Value::as_u64) {
556                    Some(v) => recovered_counter(Some(v), "cost")?,
557                    None => cost_from_atif_total(metrics.total_cost_usd)?,
558                };
559                session
560                    .totals
561                    .cost_usd_micros
562                    .store(cost_usd_micros, Ordering::Release);
563                let tool_calls = recovered_counter(
564                    extra.get("tool_calls").and_then(serde_json::Value::as_u64),
565                    "tool calls",
566                )?;
567                let tool_allowed = recovered_counter(
568                    extra.get("tool_allowed").and_then(serde_json::Value::as_u64),
569                    "allowed tools",
570                )?;
571                let tool_blocked = recovered_counter(
572                    extra.get("tool_blocked").and_then(serde_json::Value::as_u64),
573                    "blocked tools",
574                )?;
575                if tool_allowed
576                    .checked_add(tool_blocked)
577                    .is_none_or(|classified| classified > tool_calls)
578                {
579                    return Err("recovered tool accounting is inconsistent".to_owned());
580                }
581                session.totals.tool_calls.store(tool_calls, Ordering::Release);
582                session.totals.tool_allowed.store(tool_allowed, Ordering::Release);
583                session.totals.tool_blocked.store(tool_blocked, Ordering::Release);
584                if let Some(id) = extra.get("stop_reason_id").and_then(serde_json::Value::as_u64) {
585                    if id > u64::from(u8::MAX) {
586                        return Err("recovered stop reason exceeds u8".to_owned());
587                    }
588                    session.last_stop_reason_id.store(id, Ordering::Release);
589                }
590                // Latch restoration across restart, from the EXPLICITLY
591                // persisted latch value — never inferred from the stop
592                // reason, which Inject-action breaker trips also record
593                // without latching live (a restart used to upgrade such
594                // deliberately non-terminal trips into permanent id
595                // lockouts). Absent key (pre-latch artifacts,
596                // consolidation-rebuilt artifacts) means no latch.
597                if let Some(latched) = extra
598                    .get("enforcement_latched")
599                    .and_then(serde_json::Value::as_u64)
600                {
601                    if latched > u64::from(u8::MAX) {
602                        return Err("recovered enforcement latch exceeds u8".to_owned());
603                    }
604                    if latched != 0 {

606                    }
607                }
608            } else if let Some(cost) = metrics.total_cost_usd {
609                session
610                    .totals
611                    .cost_usd_micros
612                    .store(cost_from_atif_total(Some(cost))?, Ordering::Release);
613            }
614        }
615        Ok(session)
616    }

781    /// Build the receipt body for this session (signed close or promotion).
782    pub fn receipt_body(
783        &self,
784        subject: av_receipts::ReceiptSubject,
785        stop: StopReason,
786    ) -> av_receipts::ReceiptBody {
787        let recorded_id = self.last_stop_reason_id.load(Ordering::Acquire);
788        let recorded = StopReason::from_id(u8::try_from(recorded_id).unwrap_or(0));
789        // Distinguish "never recorded" (raw id 0) from "recorded an id
790        // this build cannot map" (nonzero → `from_id` says Unknown —
791        // a NEWER node's stop-reason id recovered from its artifact).
792        // Only the former may fall back to the caller's stop: rewriting
793        // a foreign id into e.g. SessionClosed would forge a specific
794        // reason the session never had into the SIGNED receipt. The
795        // honest representation for a foreign id is Unknown.
796        let stop = if recorded_id == 0 { stop } else { recorded };
797        av_receipts::receipt::new_body(
798            self.id.clone(),
799            self.current_identity(),
800            subject,
801            av_receipts::ToolCallSummary {
802                total: self.totals.tool_calls.load(Ordering::Acquire),
803                allowed: self.totals.tool_allowed.load(Ordering::Acquire),
804                blocked: self.totals.tool_blocked.load(Ordering::Acquire),
805            },
806            av_receipts::CostSummary {
807                prompt_tokens: self.totals.prompt_tokens.load(Ordering::Acquire),
808                completion_tokens: self.totals.completion_tokens.load(Ordering::Acquire),
809                cached_tokens: self.totals.cached_tokens.load(Ordering::Acquire),
810                cost_usd_micros: self.totals.cost_usd_micros.load(Ordering::Acquire),
811            },
812            stop,
813        )
814    }

817fn recovered_counter(value: Option<u64>, field: &str) -> Result<u64, String> {
818    let value = value.unwrap_or(0);
819    if value > av_core::error::JCS_SAFE_MAX {
820        return Err(format!("recovered {field} exceeds JCS-safe bounds"));
821    }
822    Ok(value)
823}

825/// Convert an ATIF-level `total_cost_usd` float (dollars) into u64
826/// micros with the same finite/nonnegative/JCS-safe guards the
827/// non-extra branch enforces inline. Returns 0 for a missing value,
828/// mirroring `recovered_counter`. Shared between the two recovery
829/// branches so a cross-version artifact — extra present but without
830/// `cost_usd_micros`, with `total_cost_usd` at the ATIF level — no
831/// longer silently zeroes the recovered cost.
832fn cost_from_atif_total(cost: Option<f64>) -> Result<u64, String> {
833    let Some(cost) = cost else {
834        return Ok(0);
835    };
836    if !cost.is_finite() || cost < 0.0 {
837        return Err("recovered cost is not finite and nonnegative".to_owned());
838    }
839    let micros = (cost * av_core::units::USD_MICROS_PER_DOLLAR as f64).round();
840    if micros > av_core::error::JCS_SAFE_MAX as f64 {
841        return Err("recovered cost exceeds JCS-safe bounds".to_owned());
842    }
843    #[allow(clippy::cast_possible_truncation, clippy::cast_sign_loss)]
844    Ok(micros as u64)
845}

879    /// Get or open a session.
880    ///
881    /// If an entry exists but has completed close (receipt/ATIF durably
882    /// committed, journal removed) and has not yet been reaped by the
883    /// idle sweeper, treat the id as free and open a fresh session —
884    /// otherwise a well-behaved client that reuses a session id after a
885    /// server-side close (retry after a 5xx, network partition, TTL
886    /// refresh) would see `400 session is already closed` for the entire
887    /// eviction window. A session that started close but has not yet
888    /// finished it (`close_complete = 0`) is *not* replaced: reopening
889    /// would race the in-flight close and split the audit trail.
890    ///
891    /// This is the right shape for **chat requests**: the client is
892    /// starting a new turn and it's fine to give them a fresh state
893    /// under the same id. For **tool interception**, use
894    /// [`Self::get_or_open_no_reopen`] — a tool call references an
895    /// in-progress conversation, and silently resurrecting a closed
896    /// session would let the client extend the audit trail past its
897    /// signed receipt.
898    pub fn get_or_open(
899        &self,
900        id: &str,
901        workflow: Workflow,
902        identity: &AgentIdentity,
903        breaker: &av_loopdetect::BreakerConfig,
904    ) -> Arc<Session> {
905        self.get_or_open_inner(
906            id, workflow, identity, breaker, /* reopen_after_close */ true,
907        )
908    }

910    /// Like [`Self::get_or_open`] but hand back the existing session
911    /// **without** recycling completed-close entries — the caller then
912    /// sees `is_closed() == true` and can refuse with `BadRequest`.
913    /// Use this on paths where the caller is trying to extend an
914    /// existing session (tool interception, session-scoped mutations).
915    pub fn get_or_open_no_reopen(
916        &self,
917        id: &str,
918        workflow: Workflow,
919        identity: &AgentIdentity,
920        breaker: &av_loopdetect::BreakerConfig,
921    ) -> Arc<Session> {

923            id, workflow, identity, breaker, /* reopen_after_close */ false,
924        )

927    fn get_or_open_inner(
928        &self,
929        id: &str,
930        workflow: Workflow,
931        identity: &AgentIdentity,
932        breaker: &av_loopdetect::BreakerConfig,
933        reopen_after_close: bool,
934    ) -> Arc<Session> {
935        use dashmap::mapref::entry::Entry;
936        match self.sessions.entry(id.to_owned()) {
937            Entry::Occupied(mut occupied) => {
938                if reopen_after_close && occupied.get().close_complete_flag() {
939                    // Enforcement-triggered closes must NOT silently
940                    // recycle: a session closed because its token budget
941                    // was exhausted (or its loop breaker aborted) would
942                    // otherwise reopen under the SAME id with a fresh
943                    // budget and fresh breaker — verified live as a
944                    // strict 200/403/200/403 alternation that rides the
945                    // per-session cap at ~50% duty forever. Per-session
946                    // budgets cannot stop a client spreading across NEW
947                    // ids (inherent to per-session semantics), but the
948                    // same id must stay terminally refused so each
949                    // incarnation is an explicit, distinct audit entity.
950                    // Keyed on the sticky enforcement latch, not the
951                    // last-write-wins stop reason — later responses
952                    // overwrite that record, and non-enforcement
953                    // failures (quota-backend errors, blocked tool
954                    // calls) also write it, making it an unreliable
955                    // proxy in both directions.
956                    if occupied.get().enforcement_tripped() {
957                        return occupied.get().clone();
958                    }
959                    let fresh = Arc::new(Session::new(
960                        id.to_owned(),
961                        workflow,
962                        identity.clone(),
963                        breaker.clone(),
964                    ));
965                    occupied.insert(Arc::clone(&fresh));
966                    fresh
967                } else {
968                    occupied.get().clone()
969                }
970            }
971            Entry::Vacant(vacant) => {
972                let fresh = Arc::new(Session::new(
973                    id.to_owned(),
974                    workflow,
975                    identity.clone(),
976                    breaker.clone(),
977                ));
978                vacant.insert(Arc::clone(&fresh));
979                fresh
980            }
981        }
982    }

1034    /// Evict signed sessions whose close ran to full completion and that
1035    /// have been idle longer than `idle_s`, returning the evicted sessions.
1036    ///
1037    /// Only signed sessions whose close *fully completed* are eligible: a
1038    /// completed close removed the on-disk journal, so nothing re-inserts
1039    /// them, and a later request or lifecycle call for the id behaves
1040    /// exactly as it would after a process restart. `close_complete` (not
1041    /// `artifact_committed`, which is set before the fallible bridge emits
1042    /// and journal removal) is the gate — a failed or in-flight close must
1043    /// stay resident, or a client reusing the id could open a fresh session
1044    /// whose journal appends collide with the still-on-disk records.
1045    /// Unsigned sessions must stay resident — the recovery scan re-inserts
1046    /// them from their spool artifact on the next tick anyway, and evicting
1047    /// one lets a client reuse its id against the still-present artifact and
1048    /// provenance files, poisoning the new incarnation's close. Capture-failed
1049    /// (quarantined) sessions also stay: they are bounded by real crash
1050    /// events and their in-registry seal is what keeps the fail-closed
1051    /// refusal cheap. Without eviction the registry grows by one entry per
1052    /// client-chosen session id for the process lifetime.
1053    pub fn evict_finalized(&self, idle_s: u64) -> Vec<Arc<Session>> {
1054        // Idle comparison uses the monotonic clock so a
1055        // forward wall-clock jump (VM resume, NTP correction) cannot
1056        // make finalized sessions look premature-idle. Wall-clock
1057        // `last_activity_ms` stays available for dashboard display.
1058        let idle_duration = Duration::from_secs(idle_s);
1059        let mut evicted = Vec::new();
1060        self.sessions.retain(|_, session| {
1061            let evict = session.workflow == Workflow::Signed
1062                && session.close_complete_flag()
1063                && !session.capture_failed()
1064                // Enforcement-latched sessions are retained like
1065                // capture-failed ones: evicting them vacated the id, and
1066                // the next get_or_open built a fresh session with a
1067                // fresh budget/breaker — the same-id refusal silently
1068                // expired after idle_s for signed workflows. Bounded by
1069                // the overflow pass below (unlike capture_failed, the
1070                // latch is client-mintable at line rate — one over-cap
1071                // request per fresh id — so retention needs a cap).
1072                && !session.enforcement_tripped()
1073                && session.active_streams.load(Ordering::Acquire) == 0
1074                && session.pending_jobs.load(Ordering::Acquire) == 0
1075                && session.idle_duration() >= idle_duration;
1076            if evict {
1077                evicted.push(Arc::clone(session));
1078            }
1079            !evict
1080        });
1081        // Overflow pass: cap latched-retained signed sessions. Beyond the
1082        // cap, evict oldest-first — the same-id refusal expires for the
1083        // evicted ids (degraded to the pre-latch idle-bounded behavior),
1084        // which is the accepted cost of bounding client-mintable registry
1085        // growth. Unsigned latched sessions are governed by the existing
1086        // unsigned retention posture and are not evicted here.
1087        const MAX_LATCHED_RETAINED: usize = 4096;
1088        let mut latched: Vec<Arc<Session>> = self
1089            .sessions
1090            .iter()
1091            .filter(|entry| {
1092                entry.workflow == Workflow::Signed
1093                    && entry.close_complete_flag()
1094                    && entry.enforcement_tripped()
1095                    && !entry.capture_failed()
1096                    && entry.active_streams.load(Ordering::Acquire) == 0
1097                    && entry.pending_jobs.load(Ordering::Acquire) == 0
1098            })
1099            .map(|entry| Arc::clone(&entry))
1100            .collect();
1101        if latched.len() > MAX_LATCHED_RETAINED {
1102            latched.sort_by_key(|session| std::cmp::Reverse(session.idle_duration()));
1103            let excess = latched.len() - MAX_LATCHED_RETAINED;
1104            tracing::warn!(
1105                excess,
1106                cap = MAX_LATCHED_RETAINED,
1107                "latched-retained signed sessions exceed the cap; evicting oldest \
1108                 (their same-id enforcement refusal expires)"
1109            );
1110            for session in latched.into_iter().take(excess) {
1111                self.sessions.remove(&session.id);
1112                evicted.push(session);
1113            }
1114        }
1115        // Overflow pass: cap capture-failed (quarantined) session
1116        // retention. Same shape as the latched pass above but for a
1117        // distinct class: capture-failed sessions are supposed to be
1118        // bounded by real crash events, but the paired
1119        // `Finalizer::quarantined_sessions` set is now FIFO-capped at
1120        // MAX_QUARANTINED_SESSIONS. Their in-registry retention needed
1121        // a matching cap to prevent the Session Arcs from growing
1122        // indefinitely across a long-lived process's lifetime. Beyond
1123        // the cap, evict oldest-first — the quarantine artifact on
1124        // disk is what fundamentally gates resurrection, so evicting
1125        // the in-memory Arc is safe (a subsequent recovery pass
1126        // re-inserts if the artifact still lives on disk).
1127        const MAX_CAPTURE_FAILED_RETAINED: usize = 4096;
1128        let mut capture_failed: Vec<Arc<Session>> = self
1129            .sessions
1130            .iter()
1131            .filter(|entry| {
1132                entry.close_complete_flag()
1133                    && entry.capture_failed()
1134                    && entry.active_streams.load(Ordering::Acquire) == 0
1135                    && entry.pending_jobs.load(Ordering::Acquire) == 0
1136            })
1137            .map(|entry| Arc::clone(&entry))
1138            .collect();
1139        if capture_failed.len() > MAX_CAPTURE_FAILED_RETAINED {
1140            capture_failed.sort_by_key(|session| std::cmp::Reverse(session.idle_duration()));
1141            let excess = capture_failed.len() - MAX_CAPTURE_FAILED_RETAINED;
1142            tracing::warn!(
1143                excess,
1144                cap = MAX_CAPTURE_FAILED_RETAINED,
1145                "capture-failed sessions exceed the retention cap; evicting oldest \
1146                 (their same-id fail-closed refusal expires until on-disk quarantine \
1147                 artifact re-inserts on the next recovery tick)"
1148            );
1149            for session in capture_failed.into_iter().take(excess) {
1150                self.sessions.remove(&session.id);
1151                evicted.push(session);
1152            }
1153        }
1154        evicted
1155    }

1157    /// Sessions where `close_session_locked` marked
1158    /// `artifact_committed = 1` but crashed / failed before running
1159    /// the finalization tail (`emit_bridge_event(SESSION_CLOSE)` +
1160    /// `remove_step_journal` + `remove_lifecycle_outbox` +
1161    /// `mark_close_complete`). Without this recovery hook such
1162    /// sessions accumulate in the registry forever: `is_closed()` is
1163    /// true so the idle sweeper skips them; `close_complete = 0` so
1164    /// `evict_finalized` refuses them; recovery scans skip them via
1165    /// the "already in registry" short-circuit. Capture-failed and
1166    /// empty-unsigned quarantines are excluded — they intentionally
1167    /// stay in the registry as evidence of the incident.
1168    ///
1169    /// The empty-unsigned quarantine (the reconciler's

1192        let idle_duration = Duration::from_secs(idle_s);
1193        self.sessions
1194            .iter()
1195            .filter(|e| {
1196                e.idle_duration() >= idle_duration
1197                    && !e.is_closed()
