1//! Provider adapter seam.
2//!
3//! The upstream was treated as OpenAI-shaped throughout the response
4//! path, which made "a second provider" the most painful change for
5//! this design to absorb. This module
6//! introduces the `ProviderAdapter` trait with the OpenAI
7//! adapter as its only implementation — a pure refactor. The SSE
8//! parser keeps its battle-tested implementation in `routes.rs`
9//! verbatim (BOM handling, `[DONE]`, `usage: null`, named-event
10//! refusal, metric-regression rejection); the adapter owns
11//! *selection*, so `AnthropicAdapter` / `GoogleGeminiAdapter`
12//! become additive.
13
14use std::sync::Arc;
15
16/// One upstream provider's wire dialect. Implementations must be
17/// total over arbitrary input (the fuzz target pins this for the
18/// OpenAI adapter) and must never panic on hostile frames.
19pub(crate) trait ProviderAdapter: Send + Sync {
20    /// Stable name, matching the `provider` config value.
21    fn name(&self) -> &'static str;
22    /// Parse one SSE frame (or one buffered non-streaming body) into
23    /// the provider-neutral chunk shape. `Ok(None)` means a
24    /// keepalive/`[DONE]`-style frame carrying nothing attributable.
25    fn parse_sse_chunk(&self, raw: &str) -> Result<Option<crate::routes::ParsedProviderChunk>, String>;
26}
27
28/// OpenAI wire dialect. Also fits vLLM, LiteLLM, Groq, Together,
29/// DeepSeek, OpenRouter, Ollama, LM Studio, llama.cpp, xAI, Mistral
30/// and Azure OpenAI, which emulate it.
31pub(crate) struct OpenAiAdapter;

33impl ProviderAdapter for OpenAiAdapter {
34    fn name(&self) -> &'static str {
35        "openai"
36    }

38    fn parse_sse_chunk(&self, raw: &str) -> Result<Option<crate::routes::ParsedProviderChunk>, String> {

43/// Look up the adapter for a configured `provider` value. `None`
44/// means unsupported — config validation names the supported set so
45/// the daemon refuses at boot, not mid-stream.
46pub(crate) fn adapter_for(provider: &str) -> Option<Arc<dyn ProviderAdapter>> {
47    match provider {
48        "openai" => Some(Arc::new(OpenAiAdapter)),
49        "anthropic" => Some(Arc::new(AnthropicAdapter)),
50        "gemini" => Some(Arc::new(GoogleGeminiAdapter)),
51        _ => None,
52    }
53}

59/// Anthropic Messages API wire dialect (S3 step 2).
60///
61/// Streaming frames are named SSE events (`message_start`,
62/// `content_block_start/delta/stop`, `message_delta`, `message_stop`,
63/// `ping`, `error`); non-streaming bodies are a single
64/// `{"type":"message", "content":[...]}` document. Mapping into the
65/// provider-neutral chunk:
66///
67/// * `text_delta`/`text` blocks → `message`;
68///   `thinking_delta`/`thinking` blocks → `reasoning`.
69/// * `tool_use` blocks / `input_json_delta` → tool-call deltas keyed
70///   by the content-block `index` (Anthropic has no choices array;
71///   `choice_index` is always 0).
72/// * `usage.input_tokens` + `usage.cache_creation_input_tokens` +
73///   `usage.cache_read_input_tokens` → prompt (Anthropic's
74///   `input_tokens` EXCLUDES the cache-accounted input; the checked
75///   sum is the true input volume), `usage.output_tokens` →
76///   completion (Anthropic reports output cumulatively, matching the
77///   `completion_reported` cumulative contract),
78///   `usage.cache_read_input_tokens` → cached.
79/// * `stop_reason` passes through natively — `map_finish_reason`
80///   already folds `end_turn`/`max_tokens`/`tool_use`/
81///   `stop_sequence` into the audit-chain stop taxonomy.
82/// * `event: error` fails the frame (capture must not attribute an
83///   error payload as model output — same posture as the OpenAI
84///   adapter's named-event refusal); `ping`/`message_stop` are
85///   keepalives.
86pub(crate) struct AnthropicAdapter;
87
88/// Anthropic SSE event names whose payloads this adapter captures.
89const ANTHROPIC_CONTENT_EVENTS: &[&str] = &[
90    "message_start",
91    "content_block_start",
92    "content_block_delta",
93    "content_block_stop",
94    "message_delta",
95];

97impl ProviderAdapter for AnthropicAdapter {
98    fn name(&self) -> &'static str {
99        "anthropic"
100    }
101
102    fn parse_sse_chunk(&self, raw: &str) -> Result<Option<crate::routes::ParsedProviderChunk>, String> {
103        // Same SSE §9.2 discipline as the OpenAI parser: strip one
104        // leading BOM, collect `data:` lines (one optional leading
105        // space each), track the frame's `event:` name.
106        let raw = raw.strip_prefix('\u{feff}').unwrap_or(raw);
107        let mut is_sse = false;
108        let mut event_type = String::new();
109        let mut data = Vec::new();
110        for line in raw.split(['\r', '\n']) {
111            if let Some(value) = line.strip_prefix("data:") {
112                is_sse = true;
113                data.push(value.strip_prefix(' ').unwrap_or(value));
114            } else if let Some(value) = line.strip_prefix("event:") {
115                is_sse = true;
116                event_type = value.strip_prefix(' ').unwrap_or(value).to_owned();
117            } else if line == "data"
118                || line.starts_with("id:")
119                || line.starts_with("retry:")
120                || line.starts_with(':')
121            {
122                is_sse = true;
123            }
124        }
125        let candidate = if is_sse {
126            if data.iter().all(|entry| entry.trim().is_empty()) {
127                // Dataless frames — named keepalives included — carry
128                // nothing attributable.
129                return Ok(None);
130            }
131            data.join("\n")
132        } else {
133            raw.trim().to_owned()
134        };
135        if candidate.is_empty() {
136            return Ok(None);
137        }
138        // `[DONE]` is technically OpenAI's terminator, not Anthropic's,
139        // but OpenAI-compat gateways (LiteLLM in `--rewrite openai`
140        // mode, OpenRouter's Anthropic passthrough with
141        // `stream_options`) commonly append it. Treat it as a keepalive
142        // rather than trying to `serde_json::from_str("[DONE]")` and
143        // aborting the whole stream on its last frame. Same handling
144        // is applied by the OpenAI parser directly.
145        if candidate.trim() == "[DONE]" {
146            return Ok(None);
147        }
148        let value: serde_json::Value = serde_json::from_str(&candidate)
149            .map_err(|error| format!("invalid provider JSON frame: {error}"))?;
150        let frame_type = value
151            .get("type")
152            .and_then(serde_json::Value::as_str)
153            .unwrap_or_default()
154            .to_owned();
155        // The SSE event name and the payload's `type` field are
156        // redundant in the Anthropic dialect; require agreement when
157        // both are present so a hostile frame cannot smuggle an error
158        // payload under a content event name (or vice versa).
159        //
160        // Exception: SSE §9.2.6 makes `message` the default event name
161        // — an empty `event:` field, or its absence, MUST behave as if
162        // `event: message` was set. Gateways (LiteLLM, OpenRouter,
163        // some enterprise OpenAI-compat proxies) commonly normalise
164        // Anthropic's named events under this default when relaying
165        // to a client that consumed the OpenAI shape. Treating
166        // `event: message` as SSE-default (equivalent to unset) here
167        // matches the OpenAI parser's ordering at routes.rs:3749, so
168        // an Anthropic wire that a gateway relabels under `message`
169        // does not fail-close where the same content on OpenAI passes.
170        if is_sse && !event_type.is_empty() && event_type != "message" && event_type != frame_type {
171            return Err(format!(
172                "provider SSE event name {event_type:?} does not match payload type {frame_type:?}"
173            ));
174        }
175        match frame_type.as_str() {
176            "ping" | "message_stop" => return Ok(None),
177            "error" => {
178                return Err(format!(
179                    "provider streamed an error event: {}",
180                    value
181                        .pointer("/error/message")
182                        .and_then(serde_json::Value::as_str)
183                        .unwrap_or("(no message)")
184                ));
185            }
186            "message" => {} // non-streaming body
187            other if ANTHROPIC_CONTENT_EVENTS.contains(&other) => {}
188            other => {
189                return Err(format!(
190                    "provider frame carries unsupported Anthropic type {other:?}; refusing to \
191                     attribute it to the audit surface"
192                ));
193            }
194        }
195        parse_anthropic_payload(&frame_type, &value)
196    }
197}

199/// Shared payload → chunk mapping for streaming events and the
200/// non-streaming `message` document.
201#[allow(clippy::too_many_lines)]
202fn parse_anthropic_payload(
203    frame_type: &str,
204    value: &serde_json::Value,
205) -> Result<Option<crate::routes::ParsedProviderChunk>, String> {
206    use crate::routes::{provider_u64, ParsedProviderChunk, ProviderToolCallDelta};
207    use serde_json::Value;
208
209    fn input_is_empty_object(v: &Value) -> bool {
210        v.as_object().is_some_and(serde_json::Map::is_empty)
211    }
212
213    let mut message = String::new();
214    let mut reasoning = String::new();
215    let mut finish_reason = None;
216    let mut tool_call_deltas = Vec::new();
217
218    // Usage may sit at the top level (message_delta, non-streaming) or
219    // under `message` (message_start).
220    let usage = value
221        .get("usage")
222        .or_else(|| value.pointer("/message/usage"))
223        .filter(|usage| !usage.is_null());
224    let mut usage_reported = false;
225    let mut prompt_tokens = None;
226    let mut completion_tokens = None;
227    let mut cached_tokens = None;
228    if let Some(usage) = usage {
229        usage_reported = true;
230        // Anthropic's `input_tokens` EXCLUDES the cache-accounted
231        // input: `cache_creation_input_tokens` and
232        // `cache_read_input_tokens` are siblings, and the true input
233        // volume is the sum of all three. Recording bare input_tokens
234        // understated prompt usage by the entire cached prefix (a
235        // 200k-token cached system prompt billed as ~10 prompt
236        // tokens), understating receipts AND letting sessions sail
237        // past prompt budgets. Checked addition; `provider_u64`
238        // already JCS-bounds each addend and the sum is re-bounded by
239        // the same gate downstream.
240        let input = provider_u64(usage.get("input_tokens"), "input_tokens")?;
241        let cache_creation = provider_u64(
242            usage.get("cache_creation_input_tokens"),
243            "cache_creation_input_tokens",
244        )?;
245        let cache_read = provider_u64(usage.get("cache_read_input_tokens"), "cache_read_input_tokens")?;
246        prompt_tokens = match (input, cache_creation, cache_read) {
247            (None, None, None) => None,
248            (a, b, c) => {
249                let sum = a
250                    .unwrap_or(0)
251                    .saturating_add(b.unwrap_or(0))
252                    .saturating_add(c.unwrap_or(0));
253                if sum > av_core::error::JCS_SAFE_MAX {
254                    return Err("provider input token sum exceeds JCS-safe bounds".to_owned());
255                }
256                Some(sum)
257            }
258        };
259        completion_tokens = provider_u64(usage.get("output_tokens"), "output_tokens")?;
260        cached_tokens = cache_read;
261    }
262    let model_name = value
263        .get("model")
264        .or_else(|| value.pointer("/message/model"))
265        .and_then(Value::as_str)
266        .map(str::to_owned);
267    if let Some(reason) = value
268        .get("stop_reason")
269        .or_else(|| value.pointer("/delta/stop_reason"))
270        .or_else(|| value.pointer("/message/stop_reason"))
271        .and_then(Value::as_str)
272        .filter(|reason| !reason.is_empty())
273    {
274        finish_reason = Some(reason.to_owned());
275    }
276
277    let mut absorb_block = |index: u64, block: &Value, is_streaming_start: bool| -> Result<(), String> {
278        match block.get("type").and_then(Value::as_str).unwrap_or_default() {
279            "text" | "text_delta" => {
280                if let Some(text) = block.get("text").and_then(Value::as_str) {
281                    message.push_str(text);
282                }
283            }
284            "thinking" | "thinking_delta" => {
285                if let Some(text) = block
286                    .get("thinking")
287                    .or_else(|| block.get("text"))
288                    .and_then(Value::as_str)
289                {
290                    reasoning.push_str(text);
291                }
292            }
293            "tool_use" => {
294                // Two distinct scenarios share this arm:
295                //
296                //  1. NON-STREAMING (`message` / `message_start` with
297                //     the content array inline): `input` carries the
298                //     COMPLETE argument object. A zero-arg call arrives
299                //     as `"input":{}` and must attest as `"{}"` — the
300                //     same wire shape OpenAI produces (`"arguments":
301                //     "{}"`) — so the same logical call signs
302                //     identically regardless of provider. Pre-R14 this
303                //     branch fell through to `arguments = ""` and the
304                //     downstream fallback wrote `{"raw":""}` into the
305                //     signed ATIF step.
306                //
307                //  2. STREAMING (`content_block_start`): Anthropic's
308                //     protocol ALWAYS ships `"input":{}` here as a
309                //     PLACEHOLDER, then follows with
310                //     `input_json_delta.partial_json` chunks that
311                //     `absorb_frame` CONCATENATES into `arguments`.
312                //     Emitting `"{}"` here would corrupt every
313                //     non-empty streaming tool_use to
314                //     `{}{"city":"…"}` — invalid JSON, and the
315                //     `.unwrap_or_else(|_| json!({"raw": …}))`
316                //     fallback at routes.rs:4024 would attest the
317                //     mangled string. Emit `""` instead; the truly
318                //     zero-arg streaming case (no delta frames follow)
319                //     is normalised at that same finalize point,
320                //     where empty accumulated args become `"{}"`
321                //     before the parse attempt.
322                let arguments = match block.get("input") {
323                    Some(input)
324                        if !input.is_null() && !(is_streaming_start && input_is_empty_object(input)) =>
325                    {
326                        serde_json::to_string(input)
327                            .map_err(|error| format!("provider tool input is not serializable: {error}"))?
328                    }
329                    _ if is_streaming_start => String::new(),
330                    _ => "{}".to_owned(),
331                };
332                tool_call_deltas.push(ProviderToolCallDelta {
333                    choice_index: 0,
334                    index,
335                    complete: false,
336                    id: block.get("id").and_then(Value::as_str).map(str::to_owned),
337                    name: block.get("name").and_then(Value::as_str).map(str::to_owned),
338                    arguments,
339                });
340            }
341            "input_json_delta" => {
342                tool_call_deltas.push(ProviderToolCallDelta {
343                    choice_index: 0,
344                    index,
345                    complete: false,
346                    id: None,
347                    name: None,
348                    arguments: block
349                        .get("partial_json")
350                        .and_then(Value::as_str)
351                        .unwrap_or_default()
352                        .to_owned(),
353                });
354            }
355            // signature_delta, redacted_thinking, server_tool_use
356            // results, citations: nothing attributable to capture.
357            _ => {}
358        }
359        Ok(())
360    };
361
362    match frame_type {
363        "content_block_start" | "content_block_delta" => {
364            let index = provider_u64(value.get("index"), "content block index")?
365                .ok_or_else(|| "provider content block has no index".to_owned())?;
366            let block = value
367                .get("content_block")
368                .or_else(|| value.get("delta"))
369                .ok_or_else(|| "provider content frame has no block payload".to_owned())?;
370            // Only the START frame carries the tool_use placeholder;
371            // `content_block_delta` frames of type `input_json_delta`
372            // are the concatenated chunks, never a placeholder.
373            let is_streaming_start = frame_type == "content_block_start";
374            absorb_block(index, block, is_streaming_start)?;
375        }
376        "message" | "message_start" => {
377            let content = value.get("content").or_else(|| value.pointer("/message/content"));
378            if let Some(Value::Array(blocks)) = content {
379                for (position, block) in blocks.iter().enumerate() {
380                    let index =
381                        u64::try_from(position).map_err(|_| "provider content index overflow".to_owned())?;
382                    absorb_block(index, block, false)?;
383                }
384            }
385        }
386        // message_delta / content_block_stop carry no content blocks.
387        _ => {}
388    }
389
390    let completion_reported = completion_tokens.is_some();
391    let completion_tokens = completion_tokens.unwrap_or_else(|| {
392        let mut estimated = av_core::tokens::approx_tokens(&message)
393            .saturating_add(av_core::tokens::approx_tokens(&reasoning));
394        for call in &tool_call_deltas {
395            estimated = estimated
396                .saturating_add(call.name.as_deref().map_or(0, av_core::tokens::approx_tokens))
397                .saturating_add(av_core::tokens::approx_tokens(&call.arguments));
398        }
399        estimated
400    });
401    Ok(Some(ParsedProviderChunk {
402        message,
403        reasoning: (!reasoning.is_empty()).then_some(reasoning),
404        model_name,
405        metrics: av_events::EventMetrics {
406            prompt_tokens,
407            completion_tokens: Some(completion_tokens),
408            cached_tokens,
409            pruned_tokens: None,
410            pruning_ratio_millis: None,
411        },
412        usage_reported,
413        completion_reported,
414        finish_reason,
415        // Anthropic does not report request cost on the wire.
416        cost_usd_micros: 0,
417        cost_reported: false,
418        // A recognized Anthropic frame is the dialect's equivalent of
419        // "has a choices array": absorb_frame's empty-200 guard keys
420        // on this.
421        has_choices: true,
422        tool_call_deltas,
423    }))
424}

426/// Google Gemini `generateContent`/`streamGenerateContent` wire
427/// dialect (S3 step 3).
428///
429/// Streaming uses UNNAMED `data:` frames (`?alt=sse`); each frame and
430/// the non-streaming body share one shape:
431/// `candidates[].content.parts[]` for text/`thought`/`functionCall`
432/// parts, `usageMetadata` for accounting, `finishReason` (SCREAMING
433/// case) on the final candidate frame. Mapping into the
434/// provider-neutral chunk:
435///
436/// * text parts → `message`; parts flagged `"thought": true` →
437///   `reasoning`.
438/// * `functionCall` parts → tool-call deltas keyed by
439///   (candidate index, part position); Gemini function calls arrive
440///   complete, with `args` as a JSON object serialized verbatim.
441/// * `promptTokenCount` → prompt, `candidatesTokenCount` +
442///   `thoughtsTokenCount` → completion (cumulative across frames;
443///   thinking tokens are billed output and must count),
444///   `cachedContentTokenCount` → cached.
445/// * `finishReason` folds into the OpenAI-lowercase taxonomy the
446///   audit chain understands: `STOP`→`stop`, `MAX_TOKENS`→
447///   `max_tokens`, safety-class refusals → `content_filter`; other
448///   values pass through lowercased (→ `StopReason::Other`).
449/// * `promptFeedback.blockReason` (request refused before any
450///   candidate) → `content_filter`.
451/// * Named SSE events are refused when they carry data (Gemini never
452///   names its frames; a named frame is not attributable), mirroring
453///   the OpenAI adapter's posture.
454pub(crate) struct GoogleGeminiAdapter;
455
456impl ProviderAdapter for GoogleGeminiAdapter {
457    fn name(&self) -> &'static str {
458        "gemini"
459    }
460
461    fn parse_sse_chunk(&self, raw: &str) -> Result<Option<crate::routes::ParsedProviderChunk>, String> {
462        use crate::routes::{provider_u64, ParsedProviderChunk, ProviderToolCallDelta};
463        use serde_json::Value;
464
465        let raw = raw.strip_prefix('\u{feff}').unwrap_or(raw);
466        let mut is_sse = false;
467        let mut event_type = String::new();
468        let mut data = Vec::new();
469        for line in raw.split(['\r', '\n']) {
470            if let Some(value) = line.strip_prefix("data:") {
471                is_sse = true;
472                data.push(value.strip_prefix(' ').unwrap_or(value));
473            } else if let Some(value) = line.strip_prefix("event:") {
474                is_sse = true;
475                event_type = value.strip_prefix(' ').unwrap_or(value).to_owned();
476            } else if line == "data"
477                || line.starts_with("id:")
478                || line.starts_with("retry:")
479                || line.starts_with(':')
480            {
481                is_sse = true;
482            }
483        }
484        if is_sse && !event_type.is_empty() && event_type != "message" {
485            // Same [DONE]-under-named-event carve-out as the OpenAI
486            // parser (routes.rs:3765) and the Anthropic parser's
487            // ordering (`[DONE]` short-circuits before the event-
488            // type check). `[DONE]` is a sentinel; a named event
489            // carrying only [DONE] lines has no attributable content
490            // and must not fail-close the whole stream. OpenAI-compat
491            // gateways fronting Gemini (per the comment below at
492            // line 472) commonly batch a final `[DONE]` under a
493            // summary event as they finalise their own event stream.
494            if data.iter().all(|entry| {
495                let trimmed = entry.trim();
496                trimmed.is_empty() || trimmed == "[DONE]"
497            }) {
498                return Ok(None);
499            }
500            return Err(format!(
501                "provider SSE frame carries unsupported event type {event_type:?}; the Gemini \
502                 dialect uses unnamed data frames, so a named frame is not attributable"
503            ));
504        }
505        let candidate = if is_sse {
506            if data.iter().all(|entry| entry.trim().is_empty()) {
507                return Ok(None);
508            }
509            data.join("\n")
510        } else {
511            raw.trim().to_owned()
512        };
513        if candidate.is_empty() {
514            return Ok(None);
515        }
516        // `[DONE]` is OpenAI-shaped, not Gemini's native terminator,
517        // but OpenAI-compat gateways in front of Gemini commonly
518        // append it. Treat as keepalive instead of aborting the
519        // stream on its last frame. Same handling as OpenAI and
520        // Anthropic parsers.
521        if candidate.trim() == "[DONE]" {
522            return Ok(None);
523        }
524        let value: Value = serde_json::from_str(&candidate)
525            .map_err(|error| format!("invalid provider JSON frame: {error}"))?;
526
527        // Same top-level error handling as the OpenAI parser: a
528        // Gemini-shaped stream can carry `{"error":{"code":...,
529        // "message":"...","status":"..."}}` mid-stream. Pre-fix this
530        // fell through to `has_choices = false` (no candidates, no
531        // usage, no finishReason) and aborted with the generic "no
532        // choices array" — the provider's real reason (INVALID_ARGUMENT,
533        // RESOURCE_EXHAUSTED, PERMISSION_DENIED) never reached the
534        // receipt.
535        if let Some(err_obj) = value.get("error").filter(|v| !v.is_null()) {
536            let message = err_obj
537                .get("message")
538                .and_then(Value::as_str)
539                .unwrap_or("provider ended stream with an unspecified error");
540            let status = err_obj.get("status").and_then(Value::as_str).unwrap_or("");
541            return Err(if status.is_empty() {
542                format!("provider stream error: {message}")
543            } else {
544                format!("provider stream error [{status}]: {message}")
545            });
546        }
547
548        let mut message = String::new();
549        let mut reasoning = String::new();
550        let mut finish_reason = None;
551        let mut tool_call_deltas = Vec::new();
552
553        let mut usage_reported = false;
554        let mut prompt_tokens = None;
555        let mut completion_tokens = None;
556        let mut cached_tokens = None;
557        if let Some(usage) = value.get("usageMetadata").filter(|usage| !usage.is_null()) {
558            usage_reported = true;
559            prompt_tokens = provider_u64(usage.get("promptTokenCount"), "promptTokenCount")?;
560            // Gemini reports thinking tokens SEPARATELY:
561            // `candidatesTokenCount` covers only the visible candidate
562            // text, `thoughtsTokenCount` the (often much larger)
563            // reasoning budget — both are billed as output. Recording
564            // bare candidatesTokenCount understated completion usage
565            // by the whole thinking spend (a 1000-thought/10-text
566            // response accounted as 10), understating receipts and
567            // completion budgets.
568            let candidates = provider_u64(usage.get("candidatesTokenCount"), "candidatesTokenCount")?;
569            let thoughts = provider_u64(usage.get("thoughtsTokenCount"), "thoughtsTokenCount")?;
570            completion_tokens = match (candidates, thoughts) {
571                (None, None) => None,
572                (a, b) => {
573                    let sum = a.unwrap_or(0).saturating_add(b.unwrap_or(0));
574                    if sum > av_core::error::JCS_SAFE_MAX {
575                        return Err("provider completion token sum exceeds JCS-safe bounds".to_owned());
576                    }
577                    Some(sum)
578                }
579            };
580            cached_tokens = provider_u64(usage.get("cachedContentTokenCount"), "cachedContentTokenCount")?;
581        }
582        let model_name = value
583            .get("modelVersion")
584            .and_then(Value::as_str)
585            .map(str::to_owned);
586        // A safety block before any candidate is a terminal refusal.
587        if value
588            .pointer("/promptFeedback/blockReason")
589            .and_then(Value::as_str)
590            .is_some_and(|reason| !reason.is_empty())
591        {
592            finish_reason = Some("content_filter".to_owned());
593        }
594
595        let has_candidates = value.get("candidates").is_some();
596        let candidates = match value.get("candidates") {
597            Some(Value::Array(candidates)) => Some(candidates),
598            Some(_) => return Err("provider candidates is not an array".to_owned()),
599            None => None,
600        };
601        if let Some(candidates) = candidates {
602            for (candidate_position, entry) in candidates.iter().enumerate() {
603                let choice_index = entry
604                    .get("index")
605                    .and_then(Value::as_u64)
606                    .or_else(|| u64::try_from(candidate_position).ok())
607                    .unwrap_or(u64::MAX);
608                if let Some(reason) = entry
609                    .get("finishReason")
610                    .and_then(Value::as_str)
611                    .filter(|reason| !reason.is_empty())
612                {
613                    finish_reason = Some(map_gemini_finish_reason(reason));
614                }
615                let parts = match entry.pointer("/content/parts") {
616                    Some(Value::Array(parts)) => parts.as_slice(),
617                    Some(Value::Null) | None => &[],
618                    Some(_) => return Err("provider content parts is not an array".to_owned()),
619                };
620                for (position, part) in parts.iter().enumerate() {
621                    if let Some(text) = part.get("text").and_then(Value::as_str) {
622                        if part.get("thought").and_then(Value::as_bool).unwrap_or(false) {
623                            reasoning.push_str(text);
624                        } else {
625                            message.push_str(text);
626                        }
627                    }
628                    if let Some(call) = part.get("functionCall").filter(|call| !call.is_null()) {
629                        // Emit `"{}"` when args are absent/null, mirroring the
630                        // Anthropic tool_use fix above and OpenAI's
631                        // `"arguments":"{}"` for a zero-arg tool call.
632                        // Falling through to `String::new()` here made the
633                        // ATIF step attest `{"raw":""}` on Gemini while the
634                        // OpenAI equivalent for the same logical call
635                        // attested `{}`, breaking the "same signature ⇔
636                        // same bytes" auditor invariant across providers.
637                        let arguments = match call.get("args") {
638                            Some(args) if !args.is_null() => {
639                                serde_json::to_string(args).map_err(|error| {
640                                    format!("provider function args are not serializable: {error}")
641                                })?
642                            }
643                            _ => "{}".to_owned(),
644                        };
645                        tool_call_deltas.push(ProviderToolCallDelta {
646                            choice_index,
647                            // Frame-local part position, NOT stable
648                            // across streamed chunks (Gemini restarts
649                            // `parts` at 0 every frame). `complete`
650                            // tells `absorb_frame` to allocate a fresh
651                            // per-choice slot instead of merging on it.
652                            index: u64::try_from(position)
653                                .map_err(|_| "provider part index overflow".to_owned())?,
654                            complete: true,
655                            id: call.get("id").and_then(Value::as_str).map(str::to_owned),
656                            name: call.get("name").and_then(Value::as_str).map(str::to_owned),
657                            arguments,
658                        });
659                    }
660                }
661            }
662        }
663        let completion_reported = completion_tokens.is_some();
664        let completion_tokens = completion_tokens.unwrap_or_else(|| {
665            let mut estimated = av_core::tokens::approx_tokens(&message)
666                .saturating_add(av_core::tokens::approx_tokens(&reasoning));
667            for call in &tool_call_deltas {
668                estimated = estimated
669                    .saturating_add(call.name.as_deref().map_or(0, av_core::tokens::approx_tokens))
670                    .saturating_add(av_core::tokens::approx_tokens(&call.arguments));
671            }
672            estimated
673        });
674        // Candidates or usage-only frames are the dialect's
675        // equivalent of a choices array for the empty-200 guard. A
676        // prompt-safety block (`promptFeedback.blockReason`) is a
677        // terminal refusal that Gemini may emit with NEITHER
678        // candidates nor usageMetadata — the parser maps it to
679        // `content_filter` above, so it must count as a valid frame
680        // or the guard misclassifies a clean refusal as a capture
681        // failure (502 instead of a content-filter stop).
682        let has_choices = has_candidates || usage_reported || finish_reason.is_some();
683        Ok(Some(ParsedProviderChunk {
684            message,
685            reasoning: (!reasoning.is_empty()).then_some(reasoning),
686            model_name,
687            metrics: av_events::EventMetrics {
688                prompt_tokens,
689                completion_tokens: Some(completion_tokens),
690                cached_tokens,
691                pruned_tokens: None,
692                pruning_ratio_millis: None,
693            },
694            usage_reported,
695            completion_reported,
696            finish_reason,
697            // Gemini does not report request cost on the wire.
698            cost_usd_micros: 0,
699            cost_reported: false,
700            has_choices,
701            tool_call_deltas,
702        }))
703    }
704}

706/// Fold Gemini's SCREAMING_CASE `finishReason` into the lowercase
707/// taxonomy `map_finish_reason` understands. Safety-class refusals
708/// all map to `content_filter`; unknown values pass through
709/// lowercased and land on `StopReason::Other`.
710fn map_gemini_finish_reason(reason: &str) -> String {
711    match reason {
712        "STOP" => "stop".to_owned(),
713        "MAX_TOKENS" => "max_tokens".to_owned(),
714        "SAFETY" | "RECITATION" | "BLOCKLIST" | "PROHIBITED_CONTENT" | "SPII" => "content_filter".to_owned(),
715        // Gemini emits these when the model tried to invoke a tool
716        // but produced invalid JSON args, or invoked a tool that
717        // wasn't declared. Both are semantically tool-shaped stops
718        // (a signal to the auditor: "the model tried to use a
719        // tool"), not generic "Other" — dashboards grouping
720        // tool-call correctness failures need them in the ToolUse
721        // bucket rather than in Other's grab-bag.
722        "MALFORMED_FUNCTION_CALL" | "UNEXPECTED_TOOL_CALL" => "tool_calls".to_owned(),
723        other => other.to_ascii_lowercase(),
724    }
725}

1260    /// R15 (review-of-R14): Anthropic streaming tool_use concatenation
1261    /// regression. `content_block_start` ships `"input":{}` as a
1262    /// placeholder — R14 initially emitted `"{}"` here, which
1263    /// concatenated with subsequent `input_json_delta` chunks into
1264    /// `{}{"city":…}` (invalid JSON). This test walks a real
1265    /// three-frame streaming sequence, concatenates the deltas the
1266    /// way `absorb_frame` does (`push_bounded`), and asserts:
1267    ///   (a) the wire-level placeholder is empty
1268    ///   (b) the accumulated arguments parse as valid JSON
1269    ///   (c) `normalize_tool_call_arguments` renders the final
1270    ///       attested `Value` — proving audit parity end-to-end.
1271    #[test]
1272    fn anthropic_streaming_tool_use_concatenates_to_valid_json_end_to_end() {
1273        let adapter = adapter_for("anthropic").unwrap();
1274        // Frame 1: content_block_start, placeholder input.
1275        let start = adapter
1276            .parse_sse_chunk(concat!(
1277                r#"event: content_block_start"#,
1278                "\n",
1279                r#"data: {"type":"content_block_start","index":1,"content_block":{"type":"tool_use","id":"toolu_1","name":"get_weather","input":{}}}"#,
1280            ))
1281            .unwrap()
1282            .unwrap();
1283        assert_eq!(
1284            start.tool_call_deltas[0].arguments, "",
1285            "placeholder MUST be empty — otherwise concatenation with subsequent \
1286             input_json_delta chunks corrupts to `{{}}{{...}}`"
1287        );
1288
1289        // Frame 2 + 3: input_json_delta chunks that concatenate.
1290        let d1 = adapter
1291            .parse_sse_chunk(
1292                r#"data: {"type":"content_block_delta","index":1,"delta":{"type":"input_json_delta","partial_json":"{\"city\":"}}"#,
1293            )
1294            .unwrap()
1295            .unwrap();
1296        let d2 = adapter
1297            .parse_sse_chunk(
1298                r#"data: {"type":"content_block_delta","index":1,"delta":{"type":"input_json_delta","partial_json":"\"NYC\"}"}}"#,
1299            )
1300            .unwrap()
1301            .unwrap();
1302
1303        // Simulate absorb_frame's per-slot push_bounded concatenation.
1304        let mut accum = String::new();
1305        accum.push_str(&start.tool_call_deltas[0].arguments);
1306        accum.push_str(&d1.tool_call_deltas[0].arguments);
1307        accum.push_str(&d2.tool_call_deltas[0].arguments);
1308        assert_eq!(accum, r#"{"city":"NYC"}"#);
1309        // And finalize — the attested Value.
1310        let attested = crate::routes::normalize_tool_call_arguments(&accum);
1311        assert_eq!(attested, serde_json::json!({"city": "NYC"}));
1312    }
