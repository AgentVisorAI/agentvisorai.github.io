1//! Spool subdirectory names shared across the harness.
2
3/// Directory under the spool root that holds durable lifecycle-outbox files.
4pub(crate) const OUTBOX: &str = "outbox";
5/// Directory under the spool root that holds tool intent/outcome/audited files.
6pub(crate) const TOOL_EXECUTIONS: &str = "tool-executions";
7/// Directory under the spool root that holds authenticated response markers.
8pub(crate) const INFLIGHT_RESPONSES: &str = "inflight-responses";
9
10/// On-disk file suffixes for the tool-execution state machine.
11pub(crate) const TOOL_INTENT_SUFFIX: &str = ".intent.json";
12pub(crate) const TOOL_OUTCOME_SUFFIX: &str = ".outcome.json";
13pub(crate) const TOOL_AUDITED_SUFFIX: &str = ".audited";
14
15#[cfg(test)]
16mod tests {
17    use super::*;
18
19    /// Register item 30 sub-clause: "write a one-page 'spool layout
20    /// and recovery' doc". docs/reference/SPOOL-AND-RECOVERY.md exists
21    /// and its file inventory must actually enumerate every real
22    /// on-disk artifact the harness writes. When a new suffix or
23    /// subdirectory constant is added here, this test forces it into
24    /// the doc — otherwise the layout grows silently and an operator
25    /// inspecting a live spool sees files they can't reason about.
26    ///
27    /// Behaviorally verified: adding a new constant here and NOT
28    /// touching the doc fails this test with a message naming the
29    /// exact missing string.
30    #[test]
31    fn spool_constants_are_documented_in_spool_and_recovery() {
32        let doc = include_str!("../../../docs/reference/SPOOL-AND-RECOVERY.md");
33        for (name, value) in [
34            ("OUTBOX", OUTBOX),
35            ("TOOL_EXECUTIONS", TOOL_EXECUTIONS),
36            ("INFLIGHT_RESPONSES", INFLIGHT_RESPONSES),
37            ("TOOL_INTENT_SUFFIX", TOOL_INTENT_SUFFIX),
38            ("TOOL_OUTCOME_SUFFIX", TOOL_OUTCOME_SUFFIX),
39            ("TOOL_AUDITED_SUFFIX", TOOL_AUDITED_SUFFIX),
40        ] {
41            assert!(
42                doc.contains(value),
43                "spool constant {name} = {value:?} is not documented in \
44                 docs/reference/SPOOL-AND-RECOVERY.md. Every on-disk artifact \
45                 the harness writes must appear in the file-inventory table \
46                 so operators inspecting a live spool can identify it."
47            );
48        }
49    }
50}
