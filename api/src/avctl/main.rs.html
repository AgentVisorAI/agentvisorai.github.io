201#[tokio::main]
202async fn main() -> std::process::ExitCode {
203    // Returning `Result<()>` relied on the default
204    // Termination impl printing anyhow's DEBUG format — which appends
205    // a full captured backtrace whenever RUST_BACKTRACE is exported,
206    // leaking 12 lines of internal frames on every well-formed user
207    // error (bad TOML, missing file). Print the Display chain only.
208    match run(Cli::parse()).await {
209        Ok(()) => std::process::ExitCode::SUCCESS,
210        Err(error) => {
211            eprintln!("Error: {error:#}");
212            std::process::ExitCode::FAILURE
213        }
214    }
215}

217async fn run(cli: Cli) -> Result<()> {
218    let Some(command) = cli.command else {
219        return wizard_then_maybe_start().await;
220    };
221    match command {
222        Command::Setup => wizard_then_maybe_start().await,
223        Command::Start => setup::start().await,
224        Command::Init {
225            preset,
226            output,
227            force,
228            upstream_url,
229            key_env,
230        } => setup::init(
231            preset,
232            &output,
233            force,
234            upstream_url.as_deref(),
235            key_env.as_deref(),
236        ),
237        Command::Doctor { offline } => setup::doctor(offline).await,
238        Command::Health { url } => setup::health(&url).await,
239        Command::Keygen { output } => keygen(&output),
240        Command::Pubkey { seed } => pubkey(&seed),
241        Command::ReceiptLocate { session_id, spool } => receipt_locate(&session_id, &spool),
242        Command::SpoolPrune {
243            spool,
244            retention_days,
245        } => spool_prune(&spool, retention_days),
246        Command::ReceiptVerify { path, public_key_hex } => receipt_verify(&path, &public_key_hex),
247        Command::AtifValidate { path, mode } => atif_validate(&path, mode),
248        Command::ManifestValidate { path } => manifest_validate(&path),
249        Command::BridgeProvision { manifest, data_dir } => bridge_provision(&manifest, &data_dir),
250        Command::EventTail {
251            data_dir,
252            topic,
253            partition,
254            offset,
255            max,
256        } => event_tail(&data_dir, &topic, partition, offset, max),
257        Command::SessionPromote {
258            url,
259            id,
260            bearer_token_file,
261        } => session_promote(&url, &id, bearer_token_file.as_deref()).await,
262        Command::ConfigValidate {
263            path,
264            structural_only,
265        } => config_validate(&path, structural_only),
266        Command::Loadgen {
267            url,
268            connections,
269            workflow,
270            bearer_token_file,
271        } => loadgen(&url, connections, &workflow, bearer_token_file.as_deref()).await,
272    }
273}

296        println!("When you're ready: avctl start");

301fn keygen(path: &Path) -> Result<()> {
302    // `signer.seed()` returns `Zeroizing<[u8; 32]>`
303    // directly, and the hex encoding is separately wrapped so both
304    // stack and heap copies zero on drop rather than lingering in
305    // freed memory recoverable from a core dump.
306    use zeroize::Zeroizing;
307    let signer = Ed25519Signer::generate();
308    let seed = signer.seed();
309    // `&*seed` avoids clippy's needless_borrows lint
310    // suggestion (`*seed`) which would move 32 bytes out of the
311    // Zeroizing wrapper into a fresh un-zeroized temp slot.
312    #[allow(clippy::needless_borrows_for_generic_args)]
313    let encoded = Zeroizing::new(hex::encode(&*seed));
314    if !install_seed_exclusive(path, &encoded)? {
315        anyhow::bail!("refusing to overwrite existing key {}", path.display());
316    }
317    println!(
318        "{}",
319        serde_json::json!({
320            "key_id": signer.key_id(),
321            "public_key_hex": hex::encode(signer.public_key_bytes()),
322            "seed_file": path,
323        })
324    );
325    Ok(())
326}

328/// Load an Ed25519 seed file and print the derived public key + key_id.
329/// This exists so operators can pin a trusted public key for
330/// `avctl receipt-verify` without ever exposing seed material — the
331/// only bytes read from disk are the hex-encoded seed, and only the
332/// derived public key is printed.
333fn pubkey(path: &Path) -> Result<()> {
334    use zeroize::Zeroizing;
335    let encoded = Zeroizing::new(
336        av_core::fsutil::read_capped_string(path, av_core::fsutil::MAX_CONTROL_BYTES)
337            .with_context(|| format!("read signing seed {}", path.display()))?,
338    );
339    let bytes = Zeroizing::new(hex::decode(encoded.trim()).context("decode signing seed as hex")?);
340    let seed: Zeroizing<[u8; 32]> = Zeroizing::new(
341        <[u8; 32]>::try_from(bytes.as_slice())
342            .map_err(|_| anyhow::anyhow!("signing seed must contain exactly 32 bytes"))?,
343    );
344    // Mirror the startup-guard refusal — printing the public key of a
345    // known-weak seed would give an operator false confidence that
346    // their pinned key is unique to this deployment.
347    if *seed == [0u8; 32] || *seed == [0xFFu8; 32] {
348        anyhow::bail!(
349            "signing seed at {} is a known-weak seed with a globally predictable public key; refusing to print",
350            path.display()
351        );
352    }
353    let signer = Ed25519Signer::from_seed(&seed);
354    println!(
355        "{}",
356        serde_json::json!({
357            "key_id": signer.key_id(),
358            "public_key_hex": hex::encode(signer.public_key_bytes()),
359            "seed_file": path,
360        })
361    );
362    Ok(())
363}

365fn install_seed_exclusive(path: &Path, encoded: &str) -> Result<bool> {
366    use std::io::Write as _;
367    let parent = path
368        .parent()
369        .filter(|parent| !parent.as_os_str().is_empty())
370        .unwrap_or_else(|| Path::new("."));
371    // Same rationale as harness main.rs `install_seed_exclusive`:
372    // route through `create_dir_all_synced` so newly-created ancestor
373    // dirents are fsynced (a bare `create_dir_all` leaves them
374    // volatile — power loss after install can drop the whole `keys/`
375    // directory even though the seed itself was fsynced) and so the
376    // leaf dir mode is 0o700 atomically at mkdir on Unix.
377    av_core::fsutil::create_dir_all_synced(parent)
378        .with_context(|| format!("create key directory {}", parent.display()))?;
379    let temporary = parent.join(format!(".avctl-key-{}.tmp", av_core::new_event_uid()));
380    // Parity with setup.rs, harness main.rs, and
381    // cold_store.rs — arm an RAII guard so a transient IO failure
382    // (ENOSPC on write, EIO on sync, EROFS after hard_link, etc.)
383    // cannot leave a `.avctl-key-<uuidv7>.tmp` containing live
384    // Ed25519 seed material on disk. Mode 0600 limits exposure to
385    // the running user, but the tmp survives reboots, gets picked
386    // up by backups, and appears in coredumps that snapshot the
387    // filesystem — none of which the operator will know to purge.
388    let mut guard = av_core::fsutil::TempPathGuard::new(temporary.clone());
389    let mut options = std::fs::OpenOptions::new();
390    options.write(true).create_new(true);
391    #[cfg(unix)]
392    {
393        use std::os::unix::fs::OpenOptionsExt as _;
394        options.mode(0o600);
395    }
396    let mut file = options
397        .open(&temporary)
398        .with_context(|| format!("create key temporary file {}", temporary.display()))?;
399    file.write_all(encoded.as_bytes())
400        .with_context(|| format!("write key temporary file {}", temporary.display()))?;
401    file.sync_all()
402        .with_context(|| format!("sync key temporary file {}", temporary.display()))?;
403    match std::fs::hard_link(&temporary, path) {
404        Ok(()) => {
405            std::fs::remove_file(&temporary)
406                .with_context(|| format!("remove key temporary file {}", temporary.display()))?;
407            // The tmp is gone; disarm the guard so its Drop is a
408            // no-op (unlinking a non-existent path would race with a
409            // fresh keygen that happened to reuse the UUIDv7 tail).
410            guard.disarm();
411            // Downgrade post-hard-link sync_directory
412            // failures to a warn. The seed is already installed at
413            // `path`; returning Err misleads the operator into
414            // thinking keygen failed and can prompt them to delete
415            // the "half-installed" file, which is actually the live
416            // seed. Same discipline as `write_atomic` in
417            // av_core::fsutil.
418            if let Err(error) = av_core::fsutil::sync_directory(parent) {
419                eprintln!(
420                    "warning: post-install directory fsync failed at {}: {error}; the seed is visible but its dirent may not survive an immediate power loss",
421                    parent.display()
422                );
423            }
424            Ok(true)
425        }
426        Err(error) if error.kind() == std::io::ErrorKind::AlreadyExists => {
427            // Guard's Drop unlinks the tmp; no manual removal needed.
428            Ok(false)
429        }
430        Err(error) => Err(error).with_context(|| format!("install key {}", path.display())),
431    }
432}

446/// Neutralise terminal-escape injection when printing
447/// attacker-influenced strings.
448///
449/// Anything reachable through a trusted third party — a signed receipt
450/// whose `receipt_id` was minted by a rotated/compromised signer, an
451/// ATIF file that a peer produced, an HTTP body from an upstream —
452/// can carry ESC / CSI / DEL / C1 bytes that reprogram the operator's
453/// terminal (clear screen, spoof a green "OK" line, rewrite the
454/// window title, or poison paste buffers on vulnerable emulators —
455/// CVE-2003-0063 class). Replace every control byte (C0 except TAB,
456/// DEL, and every C1) with U+FFFD before it flows into
457/// println!/eprintln!. Also replace the Trojan-Source family
458/// (CVE-2021-42574: bidi overrides/isolates, zero-width glyphs, and
459/// the U+2028/U+2029 line separators — av-core's shared dangerous
460/// set): a receipt id like "safe\u{202E}suoicilam" would otherwise
461/// render reversed, letting two visually identical ids differ on
462/// the wire.
463fn sanitize_for_terminal(input: &str) -> String {
464    input
465        .chars()
466        .map(|c| {
467            let cp = c as u32;
468            // Preserve TAB (0x09) as the one C0 whitespace we allow;
469            // callers that need line-based output use format! + '\n'
470            // themselves before this function sees the string.
471            if cp == 0x09 {
472                c
473            } else if cp < 0x20
474                || cp == 0x7f
475                || (0x80..=0x9f).contains(&cp)
476                || av_core::text::is_bidi_or_zero_width(c)
477            {
478                '\u{FFFD}'
479            } else {
480                c
481            }
482        })
483        .collect()
484}

498/// "the filename is `sha256(session_id)[..32]` with no
499/// lookup command" — the offline-verification workflow had no way to
500/// go from a session id to its receipt. Compute the stem and report
501/// every artifact class for it, including `archived-*` prior
502/// incarnations (a recycled session id archives the previous
503/// incarnation's artifact rather than overwriting it).
504fn receipt_locate(session_id: &str, spool: &Path) -> Result<()> {
505    let digest = av_core::digest::sha256_hex(session_id.as_bytes());
506    let stem = digest.get(..32).unwrap_or(&digest);
507    let mut artifacts = serde_json::Map::new();
508    let candidates: [(&str, PathBuf); 5] = [
509        ("receipt", spool.join("receipts").join(format!("{stem}.json"))),
510        ("atif_trajectory", spool.join(format!("{stem}.json"))),
511        ("atif_provenance", spool.join(format!("{stem}.atif-auth"))),
512        ("event_journal", spool.join(format!("{stem}.events.ndjson"))),
513        ("journal_metadata", spool.join(format!("{stem}.session.json"))),
514    ];
515    for (label, path) in &candidates {
516        artifacts.insert(
517            (*label).to_owned(),
518            serde_json::json!({
519                "path": path,
520                "exists": path.exists(),
521            }),
522        );
523    }
524    // Archived prior incarnations: `{stem}.archived-*` in either the
525    // spool root (ATIF collisions) or receipts/ (receipt collisions).
526    let mut archived: Vec<String> = Vec::new();
527    for dir in [spool.to_path_buf(), spool.join("receipts")] {
528        let Ok(entries) = std::fs::read_dir(&dir) else {
529            continue;
530        };
531        for entry in entries.flatten() {
532            let name = entry.file_name();
533            let Some(name) = name.to_str() else { continue };
534            if name.starts_with(stem) && name.contains("archived-") {
535                archived.push(entry.path().display().to_string());
536            }
537        }
538    }
539    archived.sort();
540    println!(
541        "{}",
542        serde_json::json!({
543            "session_id": session_id,
544            "stem": stem,
545            "artifacts": artifacts,
546            "archived_prior_incarnations": archived,
547        })
548    );
549    Ok(())
550}

552/// Manual, offline entry point to the sealed-ATIF retention sweep —
553/// the same `prune_sealed_atif_blocking` the harness's hourly
554/// `atif_retention_days` task runs, callable without a running
555/// harness (external cron, one-off reclaim, decommissioning).
556fn spool_prune(spool: &Path, retention_days: u32) -> Result<()> {
557    let max_age = std::time::Duration::from_secs(u64::from(retention_days) * 24 * 60 * 60);
558    let pruned = av_harness::reconciler::prune_sealed_atif_blocking(spool, max_age)
559        .with_context(|| format!("prune spool {}", spool.display()))?;
560    println!(
561        "{}",
562        serde_json::json!({
563            "spool": spool,
564            "retention_days": retention_days,
565            "pruned_pairs": pruned,
566        })
567    );
568    Ok(())
569}

571fn receipt_verify(path: &Path, public_keys_hex: &[String]) -> Result<()> {
572    // Use the strict deserializer that refuses duplicate
573    // JSON keys at any nesting level. `avctl receipt-verify` is the
574    // primary offline audit tool — a receipt that verified here but
575    // showed different content in `jq` (first-wins) would defeat
576    // the whole audit posture. `Receipt::from_json_slice` closes
577    // that split-brain uniformly.
578    let bytes = read_capped(path, MAX_RECEIPT_BYTES, "receipt")?;
579    let receipt =
580        Receipt::from_json_slice(&bytes).with_context(|| format!("parse receipt {}", path.display()))?;
581    // Accept multiple trusted keys so a key-rotation
582    // window is operable — av_receipts::Keyring was always
583    // multi-key (selected by the receipt's key_id); only this CLI
584    // flag was single-valued.
585    let mut keyring = Keyring::new();
586    for (index, key_hex) in public_keys_hex.iter().enumerate() {
587        // Accept hex (what `avctl pubkey` and the startup banner print)
588        // AND standard base64 (what the receipt's `public_key_b64`
589        // field carries) — auditors were hand-converting
590        // between the two encodings of the same 32 bytes.
591        let decoded = hex::decode(key_hex.trim()).or_else(|_| {
592            use base64::Engine as _;
593            base64::engine::general_purpose::STANDARD
594                .decode(key_hex.trim())
595                .map_err(|_| hex::FromHexError::InvalidStringLength)
596        });
597        let public_key: [u8; 32] = decoded
598            .with_context(|| {
599                format!(
600                    "trusted public key #{} is neither hexadecimal nor base64",
601                    index + 1
602                )
603            })?
604            .try_into()
605            .map_err(|_| {
606                anyhow::anyhow!("trusted public key #{} must contain exactly 32 bytes", index + 1)
607            })?;
608        keyring
609            .add_key_bytes(&public_key)
610            .with_context(|| format!("trusted public key #{} is invalid", index + 1))?;
611    }
612    receipt.verify(&keyring).context("receipt verification failed")?;
613    // Sanitise the receipt_id before printing. A signer
614    // trusted by the operator could otherwise mint a receipt whose
615    // receipt_id contains ESC/CSI bytes, reprogramming the auditor's
616    // terminal after the "verified" line.
617    println!("verified {}", sanitize_for_terminal(&receipt.body.receipt_id));
618    Ok(())
619}

621fn atif_validate(path: &Path, mode: ValidationMode) -> Result<()> {
622    let bytes = read_capped(path, MAX_ATIF_BYTES, "trajectory")?;
623    let mode = match mode {
624        ValidationMode::Strict => av_atif::Mode::Strict,
625        ValidationMode::Compat => av_atif::Mode::Compat,
626    };
627    // `validate_bytes` refuses duplicate keys before
628    // parsing (parallel to `Receipt::from_json_slice`'s
629    // strict scanner) and runs `validate_value` on the untyped
630    // form, which exercises unknown-field checks that the typed
631    // `Trajectory` (no `deny_unknown_fields`) silently drops.
632    let issues = match av_atif::validate_bytes(&bytes, mode) {
633        Ok(issues) => issues,
634        Err(reason) => anyhow::bail!("parse {}: {reason}", path.display()),
635    };
636    if !issues.is_empty() {
637        // The reconciler already
638        // caps its render at first 16 + total; the CLI mirrors
639        // that, and also detects the av_atif truncation
640        // marker (message contains "issue cap") so the summary
641        // says "at least N" rather than reporting the capped
642        // count as if it were exact.
643        const ATIF_HEAD: usize = 16;
644        let truncated = issues.last().is_some_and(|i| i.message.contains("issue cap"));
645        // Number of "real" issues: strip the synthetic marker if
646        // present. The total the user sees is a lower bound when
647        // truncated, exact otherwise.
648        let real_total = if truncated { issues.len() - 1 } else { issues.len() };
649        let shown = issues.iter().take(ATIF_HEAD);
650        for issue in shown {
651            // Sanitise every field of the issue since
652            // both `path` (a JSON pointer built from user JSON keys)
653            // and `message` (may embed a value snippet) come from
654            // attacker-supplied ATIF content.
655            eprintln!(
656                "{}: {}",
657                sanitize_for_terminal(&issue.path),
658                sanitize_for_terminal(&issue.message)
659            );
660        }
661        if real_total > ATIF_HEAD {
662            let suppressed = real_total - ATIF_HEAD;
663            let qualifier = if truncated { "at least " } else { "" };
664            eprintln!("... {qualifier}{suppressed} more issue(s) suppressed (showing first {ATIF_HEAD})");
665        }
666        if truncated {
667            anyhow::bail!("ATIF validation failed with at least {real_total} issue(s) (validator truncated)");
668        }
669        anyhow::bail!("ATIF validation failed with {real_total} issue(s)");
670    }
671    println!("valid {}", path.display());
672    Ok(())
673}

706fn event_tail(data_dir: &Path, topic: &str, partition: u32, offset: u64, max: usize) -> Result<()> {
707    // Cap `--max`. Every neighbouring CLI flag has a
708    // documented cap (loadgen --connections <= 10_000, dashboard
709    // limit <= 500, receipt/atif/config file caps); event-tail did
710    // not, so `avctl event-tail --max 4000000000` allocates a Vec
711    // large enough to OOM the CLI on a 64-bit host. 100_000 is
712    // three orders of magnitude above any operator-friendly page
713    // size — a real drain use case should page with --offset.
714    const MAX_EVENT_TAIL: usize = 100_000;
715    if max > MAX_EVENT_TAIL {
716        anyhow::bail!("--max {max} exceeds the safety cap of {MAX_EVENT_TAIL}; page with --offset instead");
717    }
718    // Read-only by construction: `EmbeddedBroker::open` recovers state
719    // (torn-tail `set_len`, sidecar rewrite, append-handle creation),
720    // which must never run beside a live daemon mid-append — a second
721    // process's "repair" could truncate bytes the daemon then acks as
722    // durable. `fetch_read_only` touches nothing on disk.
723    for event in
724        EmbeddedBroker::fetch_read_only(data_dir, topic, partition, offset, max).context("fetch events")?
725    {
726        println!("{}", serde_json::to_string(&event)?);
727    }
728    Ok(())
729}

731async fn session_promote(base_url: &str, id: &str, token_file: Option<&Path>) -> Result<()> {
732    let url = promote_url(base_url, id)?;
733    let token = bearer_token(token_file)?;
734    // Every other CLI probe bounds its wait (see `doctor` and `probe_endpoint`
735    // at 3 s); without an explicit timeout, a hung harness leaves this call
736    // waiting forever.
737    let client = reqwest::Client::builder()
738        .timeout(std::time::Duration::from_secs(60))
739        .connect_timeout(std::time::Duration::from_secs(5))
740        // Never follow redirects: the request carries bearer_auth, and a
741        // 3xx from an in-path proxy would silently re-send the NHI token
742        // to whatever the Location header names. Matches every other
743        // credential-bearing client in this crate (doctor, health).
744        .redirect(reqwest::redirect::Policy::none())
745        .build()
746        .context("build promotion client")?;
747    let mut request = client.post(url);
748    if let Some(token) = token.as_ref() {
749        request = request.bearer_auth(token.as_str());
750    }
751    let response = request.send().await.context("promote session")?;
752    let status = response.status();
753    // Cap the response body read. reqwest's
754    // `response.text()` buffers the whole body with no built-in
755    // limit; a misbehaving harness (or a stalled proxy) could keep
756    // the CLI holding unbounded RAM waiting for EOF. Stream chunks
757    // and refuse past MAX_CONTROL_BYTES (1 MiB — a promotion
758    // receipt is a few hundred bytes).
759    let body = read_capped_response(response, av_core::fsutil::MAX_CONTROL_BYTES).await?;
760    if !status.is_success() {
761        // Harness error body is attacker-influencable
762        // via header echoing / error messages — sanitise before
763        // it lands on the operator's terminal.
764        anyhow::bail!("promotion failed ({status}): {}", sanitize_for_terminal(&body));
765    }
766    // The promoted receipt is JSON — printing it
767    // through serde_json::to_string is not attacker-controllable at
768    // this layer, but the raw body could carry stray control bytes
769    // if a proxy adulterates the response. Sanitise to be safe.
770    println!("{}", sanitize_for_terminal(&body));
771    Ok(())
772}

774/// Build the promotion URL with proper path-segment encoding. Session
775/// ids accept any printable ASCII (see `SessionId::parse`), including
776/// `/`, `?` and `#`; raw string interpolation let such an id split the
777/// path or start a query string, sending the promotion to the wrong
778/// route. The harness router percent-decodes `{id}` captures, so
779/// encoding here round-trips exactly.
780fn promote_url(base_url: &str, id: &str) -> Result<reqwest::Url> {
781    let mut url =
782        reqwest::Url::parse(base_url).with_context(|| format!("parse harness base URL {base_url:?}"))?;
783    url.path_segments_mut()
784        .map_err(|()| anyhow::anyhow!("harness base URL {base_url:?} cannot carry a path"))?
785        .pop_if_empty()
786        .extend(["v1", "sessions", id, "promote"]);
787    Ok(url)
788}

790/// Capped, streaming replacement for
791/// `response.text().await`. Refuses to buffer more than `max_bytes`
792/// before an EOF and surfaces the cap in the error text.
793async fn read_capped_response(response: reqwest::Response, max_bytes: u64) -> Result<String> {
794    use futures::StreamExt as _;
795    let mut stream = response.bytes_stream();
796    let mut buf: Vec<u8> = Vec::new();
797    let cap = usize::try_from(max_bytes).unwrap_or(usize::MAX);
798    while let Some(chunk) = stream.next().await {
799        let chunk = chunk.context("read response chunk")?;
800        if buf.len().saturating_add(chunk.len()) > cap {
801            anyhow::bail!("response exceeded {max_bytes} bytes; refusing to buffer");
802        }
803        buf.extend_from_slice(&chunk);
804    }
805    String::from_utf8(buf).context("response body was not valid UTF-8")
806}

832        config.config_version, config.listen

837async fn loadgen(
838    base_url: &str,
839    connections: usize,
840    workflow: &str,
841    token_file: Option<&Path>,
842) -> Result<()> {
843    if connections == 0 {
844        anyhow::bail!("connections must be greater than zero");
845    }
846    // Cap the requested concurrency: at multi-hundred-thousand
847    // connections the loadgen exhausts source ports / FDs / RAM on the
848    // operator's own host before any latency numbers land, and the load
849    // *test* becomes the very thing it was supposed to measure. 10k is
850    // the stated SLA gate — the previous 100k cap
851    // encouraged runs that a single host cannot sustain: default Linux
852    // `net.ipv4.ip_local_port_range` (~28k) exhausts before hyper can
853    // dial, and `RLIMIT_NOFILE` (1024–65536) trips producing thousands
854    // of `Address not available` failures that look like the harness
855    // is broken.
856    const MAX_CONNECTIONS: usize = 10_000;
857    if connections > MAX_CONNECTIONS {
858        anyhow::bail!(
859            "connections must be <= {MAX_CONNECTIONS} — larger values \
860             exhaust source ports / FDs before producing usable results"
861        );
862    }
863    if av_harness::session::Workflow::parse(workflow).is_none() {
864        anyhow::bail!("workflow must be signed or unsigned");
865    }
866    // A hung upstream must not turn `loadgen --connections 10000` into an
867    // unbounded resource consumer that never fails — the load *test* would
868    // otherwise mask the failure it is supposed to expose. Per-request
869    // timeout bounds each in-flight probe; connect_timeout catches
870    // upstream unreachable faster than TCP retransmit does. Idle pool
871    // capped at 1024 — more than that is never useful and just wastes
872    // FDs after the burst ends.
873    let client = reqwest::Client::builder()
874        .pool_max_idle_per_host(connections.min(1024))
875        .timeout(std::time::Duration::from_secs(30))
876        .connect_timeout(std::time::Duration::from_secs(5))
877        // No redirects: requests may carry bearer_auth (token leak on a
878        // proxy-injected 3xx), and following one would silently measure
879        // the redirect target instead of the configured endpoint.
880        .redirect(reqwest::redirect::Policy::none())
881        .build()
882        .context("build loadgen client")?;
883    let url = format!("{}/v1/chat/completions", base_url.trim_end_matches('/'));
884    let token = bearer_token(token_file)?.map(Arc::new);
885    // Unique per run: reusing ids like `load-0` across runs would bind to
886    // sessions a previous run left behind — mid-close ones refuse reuse
887    // with 503 "session close is completing" and quarantined ones with a
888    // 400.
889    let run_id = av_core::new_event_uid();
890    let started = Instant::now();
891    let results: Vec<Result<u64, String>> = stream::iter(0..connections)
892        .map(|index| {
893            let client = client.clone();
894            let url = url.clone();
895            let workflow = workflow.to_owned();
896            let token = token.clone();
897            let run_id = run_id.clone();
898            async move {
899                let request_started = Instant::now();
900                let mut request = client
901                    .post(url)
902                    .header("x-av-session", format!("load-{run_id}-{index}"))
903                    .header("x-av-workflow", workflow)
904                    .json(&serde_json::json!({
905                        "model": "loadgen",
906                        "stream": true,
907                        "messages": [{"role": "user", "content": "health check"}],
908                    }));
909                if let Some(token) = token.as_ref() {
910                    request = request.bearer_auth(token.as_str());
911                }
912                let response = request.send().await.map_err(|error| error.to_string())?;
913                let status = response.status();
914                // Cap loadgen response body reads. With
915                // `stream: true`, an SSE response accumulates whole
916                // via `.bytes()` — a stalled server could pin each
917                // of `--connections 10_000` tasks to unbounded RAM,
918                // turning the load test into the fault it was meant
919                // to expose. Refuse past 4 MiB per response.
920                use futures::StreamExt as _;
921                let mut stream = response.bytes_stream();
922                let mut total = 0usize;
923                const LOADGEN_MAX_RESPONSE: usize = 4 * 1024 * 1024;
924                while let Some(chunk) = stream.next().await {
925                    let bytes = chunk.map_err(|error| error.to_string())?;
926                    total = total.saturating_add(bytes.len());
927                    if total > LOADGEN_MAX_RESPONSE {
928                        return Err(format!("response exceeded {LOADGEN_MAX_RESPONSE} bytes"));
929                    }
930                }
931                if !status.is_success() {
932                    return Err(format!("HTTP {status}"));
933                }
934                Ok(u64::try_from(request_started.elapsed().as_micros()).unwrap_or(u64::MAX))
935            }
936        })
937        .buffer_unordered(connections)
938        .collect()
939        .await;
940    let mut latencies = Vec::with_capacity(connections);
941    let mut failures = Vec::new();
942    for result in results {
943        match result {
944            Ok(latency) => latencies.push(latency),
945            Err(error) => failures.push(error),
946        }
947    }
948    latencies.sort_unstable();
949    println!(
950        "connections={} success={} failed={} wall_ms={} p50_us={} p95_us={} p99_us={}",
951        connections,
952        latencies.len(),
953        failures.len(),
954        started.elapsed().as_millis(),
955        percentile(&latencies, 50),
956        percentile(&latencies, 95),
957        percentile(&latencies, 99),
958    );
959    if !failures.is_empty() {
960        anyhow::bail!(
961            "{} load requests failed; first: {}",
962            failures.len(),
963            failures.first().map(String::as_str).unwrap_or("unknown")
964        );
965    }
966    Ok(())
967}

969/// Read the NHI bearer token into a `Zeroizing` buffer. The token is a
970/// credential at least as sensitive as an API key — every allocation
971/// that holds it (the raw read and the trimmed copy) is zero-on-drop so
972/// a post-run core dump or heap scan does not expose it verbatim.
973/// Mirrors the wizard's `ask_secret_line` discipline.
974fn bearer_token(path: Option<&Path>) -> Result<Option<zeroize::Zeroizing<String>>> {
975    let path = path
976        .map(Path::to_path_buf)
977        .or_else(|| std::env::var_os("AV_BEARER_TOKEN_FILE").map(PathBuf::from));
978    path.map(|path| {
979        // Refuse a bearer token file that any other local
980        // user can read. On a shared-tenant host, a 0o644 token file
981        // in ~/.avctl/ leaks the operator's credentials to every
982        // process running as another uid. Mirrors the harness's own
983        // discipline for signing seed / HMAC secret files (see
984        // `require_owner_only_mode`).
985        #[cfg(unix)]
986        {
987            use std::os::unix::fs::MetadataExt as _;
988            let metadata = std::fs::symlink_metadata(&path)
989                .with_context(|| format!("stat bearer token {}", path.display()))?;
990            if metadata.file_type().is_symlink() {
991                anyhow::bail!(
992                    "bearer token file {} is a symbolic link; refuse to follow (planted-symlink hazard)",
993                    path.display()
994                );
995            }
996            let mode = metadata.mode() & 0o777;
997            if mode & 0o077 != 0 {
998                anyhow::bail!(
999                    "bearer token file {} has mode 0o{mode:03o}; must be 0o600 (chmod 600 {})",
1000                    path.display(),
1001                    path.display()
1002                );
1003            }
1004        }
1005        let raw = zeroize::Zeroizing::new(read_capped_str(&path, MAX_CONFIG_BYTES, "bearer token")?);
1006        let token = zeroize::Zeroizing::new(raw.trim().to_owned());
1007        if token.is_empty() {
1008            anyhow::bail!("bearer token file {} is empty", path.display());
1009        }
1010        Ok(token)
1011    })
1012    .transpose()
1013}

1015fn percentile(sorted: &[u64], percentile: usize) -> u64 {
1016    if sorted.is_empty() {
1017        return 0;
1018    }
1019    let index = sorted.len().saturating_mul(percentile).saturating_add(99) / 100;
1020    sorted.get(index.saturating_sub(1)).copied().unwrap_or(0)
1021}

1157    /// `sanitize_for_terminal` neutralises every control
1158    /// byte that could reprogram the operator's terminal. A signer
1159    /// trusted by the operator could otherwise mint a receipt whose
1160    /// receipt_id contains ESC/CSI/DEL/C1 bytes and hijack the
1161    /// output of `avctl receipt-verify`.
1162    #[test]
1163    fn sanitize_for_terminal_neutralises_ansi_escapes() {
1164        // ESC (0x1b) — the C0 prefix for ANSI CSI sequences.
1165        let evil = "OK\u{1b}[2J\u{1b}[Hspoofed";
1166        let cleaned = sanitize_for_terminal(evil);
1167        assert!(!cleaned.contains('\u{1b}'), "ESC survived: {cleaned:?}");
1168        // CSI (0x9b) — the single-byte 8-bit-terminal equivalent.
1169        let evil2 = "line\u{9b}2J";
1170        assert!(!sanitize_for_terminal(evil2).contains('\u{9b}'));
1171        // DEL (0x7f).
1172        assert!(!sanitize_for_terminal("a\u{7f}b").contains('\u{7f}'));
1173        // NUL and other C0 (except TAB which we preserve).
1174        assert!(!sanitize_for_terminal("a\u{00}b").contains('\u{00}'));
1175        assert!(!sanitize_for_terminal("a\nb").contains('\n'));
1176        assert_eq!(sanitize_for_terminal("safe\tstring"), "safe\tstring");
1177        // Regular UTF-8 characters (multi-byte) survive.
1178        assert_eq!(sanitize_for_terminal("héllo"), "héllo");
1179    }
