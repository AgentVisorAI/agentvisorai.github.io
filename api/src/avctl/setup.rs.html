1//! `avctl init` / `avctl doctor` / `avctl health` — first-run setup and
2//! environment diagnosis.
3//!
4//! `init` writes a provider-specific `agentvisor.toml` that the harness
5//! picks up automatically (it is first in the config search path), then
6//! prints copy-paste next steps. `doctor` re-resolves configuration the

138/// Why a candidate upstream URL cannot be accepted, if anything.
139/// (Also guards the TOML interpolation: quote/backslash/control are out.)
140fn upstream_url_error(url: &str) -> Option<&'static str> {
141    if !url.starts_with("http://") && !url.starts_with("https://") {
142        return Some("must start with http:// or https://");
143    }
144    if url
145        .chars()
146        .any(|c| c == '"' || c == '\\' || c.is_control() || c == ' ')
147    {
148        return Some("contains characters not allowed in a URL");
149    }
150    None
151}

153/// Reject values that could escape a TOML string literal.
154fn assert_toml_safe(label: &str, value: &str) -> Result<()> {
155    if value.chars().any(|c| c == '"' || c == '\\' || c.is_control()) {
156        anyhow::bail!("{label} contains characters that cannot be stored in the config file");
157    }
158    Ok(())
159}

161/// Split a pasted upstream URL that already embeds a chat-completions path.
162///
163/// The harness joins `upstream_url` + `upstream_chat_path`, so storing a full
164/// endpoint URL (which "paste your endpoint" naturally invites) would produce
165/// `/v1/chat/completions/v1/chat/completions` and a confusing provider 404.
166/// Returns `(base_url, chat_path_override)`; the override is `None` when the
167/// embedded path is exactly the default that the harness appends anyway.
168fn split_embedded_chat_path(url: String) -> (String, Option<String>) {
169    const DEFAULT_CHAT_PATH: &str = "/v1/chat/completions";
170    let path_start = match url.find("://").and_then(|scheme| {
171        let host = scheme + 3;
172        url[host..].find('/').map(|slash| host + slash)
173    }) {
174        Some(index) => index,
175        None => return (url, None),
176    };
177    // Tolerate the trailing slash a copy-paste commonly carries
178    // (`.../v1/chat/completions/`): without the trim, no split happens,
179    // the full URL becomes `upstream_url`, and the harness appends the
180    // default chat path — producing `.../chat/completions//v1/chat/…`
181    // and a provider 404 on every call from a config that
182    // self-validated cleanly.
183    let path = url[path_start..].trim_end_matches('/');
184    let embeds_chat = path.ends_with("/chat/completions") || path.contains("/chat/completions?");
185    if !embeds_chat {
186        return (url, None);
187    }
188    let base = url[..path_start].to_owned();
189    if path == DEFAULT_CHAT_PATH {
190        return (base, None);
191    }
192    let path = path.to_owned();
193    (base, Some(path))
194}

196/// Render the annotated TOML for a plan.
197fn render_config(plan: &ConfigPlan) -> String {
198    let spec = spec(plan.preset);
199    let chat_path = plan.chat_path.as_deref().or(spec.chat_path);
200    let upstream_url = &plan.upstream_url;
201    let mut out = String::new();
202    out.push_str("# AgentVisor AI harness configuration (generated by `avctl init`).\n");
203    out.push_str("# Every omitted field keeps a safe default; see config/harness.example.toml\n");
204    out.push_str("# in the AgentVisor AI repository for the full annotated reference.\n\n");
205    out.push_str("config_version = 1\n\n");
206    out.push_str("# Address the proxy listens on. Use \"0.0.0.0:8484\" inside containers.\n");
207    out.push_str("listen = \"127.0.0.1:8484\"\n\n");
208    out.push_str("# OpenAI-compatible provider base URL.\n");
209    out.push_str(&format!("upstream_url = \"{upstream_url}\"\n"));
210    if let Some(path) = chat_path {
211        out.push_str("# Provider-specific chat completions path.\n");
212        out.push_str(&format!("upstream_chat_path = \"{path}\"\n"));
213    }
214    match &plan.key {
215        KeySpec::Env(env) => {
216            out.push_str("\n# API key is read from this environment variable at startup.\n");
217            out.push_str("# (Alternative: upstream_api_key_file = \"/path/to/key\" with mode 0600.)\n");
218            out.push_str(&format!("upstream_api_key_env = \"{env}\"\n"));
219        }
220        KeySpec::File(path) => {
221            out.push_str("\n# API key is read from this file at startup. Keep it private:\n");
222            out.push_str("# the server refuses group- or world-readable key files.\n");
223            out.push_str(&format!("upstream_api_key_file = \"{path}\"\n"));
224        }
225        KeySpec::NoKey => {
226            out.push_str("\n# No API key needed for this endpoint. To add one later:\n");
227            out.push_str("#   upstream_api_key_env = \"PROVIDER_API_KEY\"\n");
228        }
229    }
230    if let Some(header) = spec.auth_header {
231        out.push_str(&format!("upstream_auth_header = \"{header}\"\n"));
232    }
233    if let Some(scheme) = spec.auth_scheme {
234        out.push_str("# Empty scheme sends the key raw (no \"Bearer \" prefix).\n");
235        out.push_str(&format!("upstream_auth_scheme = \"{scheme}\"\n"));
236    }
237    out.push_str("\n# OpenAI SDKs require an api_key and always send an Authorization header.\n");
238    out.push_str("# This dev-mode flag accepts and DISCARDS that header (it is never\n");
239    out.push_str("# forwarded upstream; the key above is injected instead), so the\n");
240    out.push_str("# one-line quickstart works with any stock SDK. Remove it when you\n");
241    out.push_str("# turn on require_identity.\n");
242    out.push_str("ignore_client_authorization = true\n\n");
243    out.push_str("# Sign every session close with an Ed25519 receipt (the product's\n");
244    out.push_str("# core promise). \"unsigned\" produces ATIF trajectories without\n");
245    out.push_str("# signatures; clients can override per request via X-AV-Workflow.\n");
246    out.push_str("default_workflow = \"signed\"\n");
247    out.push_str("\n# Signed receipts, OCSF events, and ATIF trajectories are written here.\n");
248    match &plan.data_root {
249        Some(root) => {
250            // These paths mirror the harness `default_spool` / `default_bridge`
251            // shapes (see crates/av-harness/src/config.rs) so that removing a
252            // key from the generated config leaves the runtime pointing at the
253            // same directory the generated config named. Diverging silently
254            // relocates the spool the moment an operator edits the file.
255            out.push_str(&format!("atif_spool_dir = \"{root}/spool/atif\"\n"));
256            out.push_str(&format!("bridge_data_dir = \"{root}/data/bridge\"\n"));
257        }
258        None => {
259            out.push_str("atif_spool_dir = \"spool/atif\"\n");
260            out.push_str("bridge_data_dir = \"data/bridge\"\n");
261        }
262    }
263    // LAST in the template: a TOML table header captures every key
264    // after it, so [budget] must follow all top-level keys.
265    out.push_str("\n# Module-B action budget: without a [budget] block NOTHING binds —\n");
266    out.push_str("# token, payout and per-tool caps are all unlimited, so a runaway\n");
267    out.push_str("# agent (or an SDK retry loop) spends against your provider key\n");
268    out.push_str("# unbounded. These mirror config/harness.example.toml; tune per\n");
269    out.push_str("# deployment. With require_identity, add [principal_budget] with\n");
270    out.push_str("# the same shape so limits survive session-id rotation.\n");
271    out.push_str("[budget]\n");
272    out.push_str("max_tokens = 200000\n");
273    out.push_str("max_payout_usd_micros = 50000000\n");
274    out.push_str("max_total_tool_calls = 100\n");
275    out
276}

278/// `avctl init` — write a provider-specific config and print next steps.
279pub fn init(
280    preset: Preset,
281    output: &Path,
282    force: bool,
283    upstream_url: Option<&str>,
284    key_env: Option<&str>,
285) -> Result<()> {
286    let spec_defaults = spec(preset);
287    let upstream_url = match (upstream_url, spec_defaults.upstream_url) {
288        (Some(url), _) => url.to_owned(),
289        (None, "") => anyhow::bail!("--preset custom requires --upstream-url"),
290        (None, default) => default.to_owned(),
291    };
292    // The values are interpolated into TOML below: reject anything that
293    // could escape the string literal (quote, backslash, control bytes)
294    // instead of silently writing a config that means something else.
295    if let Some(reason) = upstream_url_error(&upstream_url) {
296        anyhow::bail!("--upstream-url {reason}");
297    }
298    let key_env = key_env
299        .map(str::to_owned)
300        .or_else(|| spec_defaults.key_env.map(str::to_owned));
301    if let Some(env) = &key_env {
302        let valid = !env.is_empty()
303            && !env.starts_with(|c: char| c.is_ascii_digit())
304            && env.chars().all(|c| c.is_ascii_alphanumeric() || c == '_');
305        if !valid {
306            anyhow::bail!("--key-env must be a valid environment variable name, got {env:?}");
307        }
308    }
309    let plan = ConfigPlan {
310        preset,
311        upstream_url,
312        chat_path: None,
313        key: match &key_env {
314            Some(env) => KeySpec::Env(env.clone()),
315            None => KeySpec::NoKey,
316        },
317        data_root: None,
318    };
319    // Same footgun as the wizard: a full endpoint URL passed via
320    // --upstream-url would double the chat path once the harness joins them.
321    let (upstream_url, chat_path) = split_embedded_chat_path(plan.upstream_url);
322    let plan = ConfigPlan {
323        upstream_url,
324        chat_path: chat_path.or(plan.chat_path),
325        ..plan
326    };
327    let rendered = render_config(&plan);
328    // Refuse to write anything the harness itself would reject.
329    av_harness::HarnessConfig::from_toml(&rendered)
330        .map_err(|error| anyhow::anyhow!("generated config failed validation (bug): {error}"))?;
331    if output.exists() && !force {
332        anyhow::bail!("{} already exists; pass --force to overwrite", output.display());
333    }
334    // Refuse to follow a pre-planted symlink at
335    // `output`. The wizard path is protected by `write_atomic`'s
336    // rename-over semantics
337    // (see `wizard_replaces_config_symlink_instead_of_following_it`);
338    // `init --force` used to differ, so an attacker who could plant
339    // ~/.agentvisor/agentvisor.toml as a symlink to
340    // ~/.bashrc or /etc/nginx/conf.d/upstream.conf could redirect
341    // the TOML write into the target when the operator ran
342    // `avctl init --force`. Unlink the symlink first so the
343    // subsequent std::fs::write hits a fresh regular file at
344    // `output`.
345    if let Ok(metadata) = std::fs::symlink_metadata(output) {
346        if metadata.file_type().is_symlink() {
347            std::fs::remove_file(output)
348                .with_context(|| format!("remove pre-existing symlink at {}", output.display()))?;
349        }
350    }
351    if let Some(parent) = output.parent().filter(|p| !p.as_os_str().is_empty()) {
352        std::fs::create_dir_all(parent).with_context(|| format!("create directory {}", parent.display()))?;
353    }
354    // `std::fs::write` after the symlink check above was
355    // a TOCTOU — an attacker replanting the symlink in the
356    // check→write window redirected the TOML into the target (and
357    // created it 0666&~umask before the post-hoc chmod).
358    // `write_atomic`'s rename-over semantics replace the link ITSELF
359    // rather than following it, closing the race the same way the
360    // wizard path does.
361    av_core::fsutil::write_atomic(output, rendered.as_bytes())
362        .with_context(|| format!("write {}", output.display()))?;
363    // Match the key-file installer
364    // (`write_private_key_file`) which explicitly opens with
365    // mode(0o600). The rendered config carries
366    // upstream_url / upstream_api_key_env / upstream_auth_header hints
367    // — capability-sensitive metadata that a co-tenant on a shared
368    // workstation could otherwise read via the default umask=022
369    // world-readable bits. Silently no-op on non-Unix (Windows ACLs
370    // are inherited from the parent dir and 0o600 has no analog).
371    #[cfg(unix)]
372    {
373        use std::os::unix::fs::PermissionsExt as _;
374        if let Err(error) = std::fs::set_permissions(output, std::fs::Permissions::from_mode(0o600)) {
375            eprintln!(
376                "warning: failed to chmod 0600 on {}: {error}; contents may be world-readable",
377                output.display(),
378            );
379        }
380    }
381    println!("wrote {}", output.display());
382    for note in spec_defaults.notes {
383        println!("note: {note}");
384    }
385    println!();
386    println!("Next steps:");
387    let mut step = 1;
388    if let Some(env) = &key_env {
389        println!("  {step}. export {env}=<your api key>");
390        step += 1;
391    }
392    if preset == Preset::Azure {
393        println!(
394            "  {step}. edit {} and fill in your resource + deployment",
395            output.display()
396        );
397        step += 1;
398    }
399    println!("  {step}. agentvisord       # starts on 127.0.0.1:8484");
400    step += 1;
401    println!("  {step}. point your OpenAI-compatible client at http://127.0.0.1:8484/v1");
402    println!();
403    println!("Check the environment any time with: avctl doctor");
404    Ok(())
405}

489/// Print a prompt and read one trimmed line; EOF is a hard error so a
490/// closed pipe can never spin the ask-again loops.
491fn ask_line(prompt: &str, input: &mut dyn std::io::BufRead) -> Result<String> {
492    use std::io::Write as _;
493    print!("{prompt}");
494    std::io::stdout().flush().context("flush prompt")?;
495    let mut line = String::new();
496    if input.read_line(&mut line).context("read answer")? == 0 {
497        anyhow::bail!("setup input ended unexpectedly");
498    }
499    Ok(line.trim().to_owned())
500}

502/// Read one line into a `Zeroizing<String>` from the
503/// very first allocation, so the pasted secret + trailing `\n` are
504/// zeroed on drop. `ask_line` above stores the line in a plain
505/// `String` and returns a fresh `.trim().to_owned()` — both the
506/// original buffer and the trimmed copy sit un-zeroed in the
507/// allocator when the piped-stdin (non-TTY) Plain path fed the
508/// wizard a real API key. An earlier fix wrapped only the outer
509/// return; this closes the intermediate copies. The caller trims
510/// via `line.trim()` (a `&str` borrow into the Zeroizing buffer),
511/// so no additional un-zeroed intermediate is created.
512fn ask_secret_line(prompt: &str, input: &mut dyn std::io::BufRead) -> Result<zeroize::Zeroizing<String>> {
513    use std::io::Write as _;
514    print!("{prompt}");
515    std::io::stdout().flush().context("flush prompt")?;
516    let mut line = zeroize::Zeroizing::new(String::new());
517    if input.read_line(&mut line).context("read answer")? == 0 {
518        anyhow::bail!("setup input ended unexpectedly");
519    }
520    Ok(line)
521}

523/// `Zeroizing<String>` end-to-end so
524/// every intermediate copy of the pasted secret zeroes its heap
525/// allocation on drop.
526///
527/// * Hidden (TTY) branch: `rpassword::prompt_password` returns a
528///   plain `String`; wrap it immediately at the boundary so the
529///   raw allocation never outlives this function.
530/// * Plain (non-TTY / piped stdin) branch: use `ask_secret_line`
531///   which allocates the line buffer INSIDE `Zeroizing` from the
532///   start. `ask_line` (used for non-secret prompts) still returns
533///   plain `String` — cheaper, but MUST NOT be reached from
534///   secret-carrying paths. Rejected retries in the caller's loop
535///   then drop-zero the previous attempt's buffer as well.
536///
537/// Residual: `StdinLock`'s internal `BufReader` buffer is outside
538/// our control (std/kernel-owned); we cannot zero that copy. All
539/// intermediates we ourselves allocate are zeroed.
540fn ask_secret(
541    prompt: &str,
542    input: &mut dyn std::io::BufRead,
543    mode: &SecretInput,
544) -> Result<zeroize::Zeroizing<String>> {
545    match mode {
546        SecretInput::Hidden => {
547            let raw =
548                rpassword::prompt_password(format!("{prompt} (typing is hidden): ")).context("read key")?;
549            Ok(zeroize::Zeroizing::new(raw))
550        }
551        SecretInput::Plain => ask_secret_line(&format!("{prompt}: "), input),
552    }
553}

599/// Write `key` to `path` with owner-only permissions, replacing any
600/// previous file (including a symlink, which the server would refuse).
601///
602/// Atomic install (tmp file + rename + parent sync) so at every instant
603/// the final path holds either the old key or the new one — never empty
604/// or truncated. Same posture as the server's signing-seed installer in
605/// av-harness/main.rs.
606fn write_private_key_file(path: &Path, key: &str) -> Result<()> {
607    use std::io::Write as _;
608    let parent = path
609        .parent()
610        .filter(|p| !p.as_os_str().is_empty())
611        .unwrap_or_else(|| Path::new("."));
612    // `create_dir_all_synced` atomically sets mode 0o700 at mkdir on
613    // Unix (via `DirBuilder::mode`), closing the world-listable
614    // window that a bare `create_dir_all` + `set_permissions(0o700)`
615    // opens: between the mkdir at `create_dir_all(0o777 & !umask)`
616    // and the follow-up chmod, a co-tenant with execute on the
617    // grandparent could `ls parent/` and see the deterministic
618    // `.agentvisor-key-<uuidv7>.tmp` name that's about to hold the
619    // private key. The synced helper also fsyncs every newly-created
620    // ancestor dirent — same first-boot power-loss durability guard
621    // as `install_seed_exclusive`. Skip the explicit
622    // `set_permissions` when the dir is fresh (the mkdir already
623    // narrowed); a pre-existing shared parent keeps its prior mode
624    // (rather than this call widening OTHER files' visibility by
625    // ratcheting the shared parent to 0o700).
626    av_core::fsutil::create_dir_all_synced(parent)
627        .with_context(|| format!("create key directory {}", parent.display()))?;
628    let temporary = parent.join(format!(".agentvisor-key-{}.tmp", av_core::new_event_uid()));
629    // Sibling of the harness's `install_seed_exclusive`.
630    // Without the RAII guard, an early `?` return from write_all /
631    // sync_all leaves a zero-byte `.agentvisor-key-*.tmp` behind
632    // that only cargoes up on next boot as a confused operator
633    // symptom ("two key files?"). Use the shared TempPathGuard so
634    // every failure path unlinks.
635    let mut guard = av_core::fsutil::TempPathGuard::new(temporary.clone());
636    let mut options = std::fs::OpenOptions::new();
637    options.write(true).create_new(true);
638    #[cfg(unix)]
639    {
640        use std::os::unix::fs::OpenOptionsExt as _;
641        options.mode(0o600);
642    }
643    let mut file = options
644        .open(&temporary)
645        .with_context(|| format!("create key file {}", temporary.display()))?;
646    file.write_all(key.as_bytes())
647        .with_context(|| format!("write key file {}", temporary.display()))?;
648    file.sync_all()
649        .with_context(|| format!("sync key file {}", temporary.display()))?;
650    // Install with a single atomic rename. The previous shape
651    // (`remove_file(path)` then `hard_link`) destroyed the ONLY copy of
652    // the existing key whenever the hard_link failed — EPERM on a
653    // re-chmoded parent, EMLINK, ENOSPC on the directory update — because
654    // the old key was already unlinked and the tmp was unconditionally
655    // removed afterwards. `rename` replaces the destination atomically
656    // (POSIX), so at every instant `path` holds either the old key or the
657    // new one, and it replaces (never follows) a pre-planted symlink at
658    // `path`. On failure the old key is untouched and the guard's Drop
659    // unlinks the tmp.
660    std::fs::rename(&temporary, path).with_context(|| format!("install key file {}", path.display()))?;
661    guard.disarm();
662    // Mirror of the harness's install_seed_exclusive
663    // and fsutil::write_atomic. Once
664    // rename commits, the key IS at `path` — a subsequent
665    // sync_directory failure means the dirent may not survive an
666    // immediate power loss, but every observer running now sees the
667    // key. Returning Err here made `avctl init` report failure and
668    // the whole install cycle retry needlessly.
669    // Downgrade to warn+Ok so this last-stanza fsync is best-effort.
670    if let Err(error) = av_core::fsutil::sync_directory(parent) {
671        eprintln!(
672            "warning: key installed at {}, but parent directory fsync failed: {error}; \
673             dirent may not survive an immediate power loss",
674            parent.display()
675        );
676    }
677    Ok(())
678}

701/// The guided setup: a few quick prompts (provider, key, start-now),
702/// everything stored
703/// under `home/.agentvisor/` so no environment variables, exports, or
704/// file editing are ever needed. Drive `input` with scripted answers and
705/// `SecretInput::Plain` in tests.
706pub fn wizard(home: &Path, input: &mut dyn std::io::BufRead, secrets: &SecretInput) -> Result<WizardOutcome> {
707    let home = absolute_home(home)?;
708    let home = home.as_path();
709    println!("Welcome to AgentVisor AI! A couple of quick questions and you're done.");
710    println!();
711    println!("Where do your AI models come from?");
712    println!();
713    for (index, (label, _)) in WIZARD_MENU.iter().enumerate() {
714        println!("  {:>2}) {label}", index + 1);
715    }
716    let mut preset = None;
717    for _ in 0..5 {
718        let answer = ask_line("\nType a number and press Enter [1]: ", input)?;
719        if answer.is_empty() {
720            preset = WIZARD_MENU.first().map(|entry| entry.1);
721            break;
722        }
723        if let Ok(choice) = answer.parse::<usize>() {
724            if let Some(entry) = choice.checked_sub(1).and_then(|index| WIZARD_MENU.get(index)) {
725                preset = Some(entry.1);
726                break;
727            }
728        }
729        println!("Please type a number between 1 and {}.", WIZARD_MENU.len());
730    }
731    let Some(preset) = preset else {
732        anyhow::bail!("no valid choice made");
733    };
734
735    // Provider-specific follow-ups.
736    let (upstream_url, chat_path, needs_key) = match preset {
737        Preset::Azure => {
738            println!("\nFrom your Azure OpenAI resource (shown in the Azure portal):");
739            let resource = ask_validated("  Resource name: ", input, azure_name_error)?;
740            let deployment = ask_validated("  Deployment name: ", input, azure_name_error)?;
741            (
742                format!("https://{resource}.openai.azure.com"),
743                Some(format!(
744                    "/openai/deployments/{deployment}/chat/completions?api-version=2024-10-21"
745                )),
746                true,
747            )
748        }
749        Preset::Custom => {
750            let url = ask_validated(
751                "\nFull URL of your OpenAI-compatible endpoint (e.g. http://10.0.0.5:8000): ",
752                input,
753                |value| upstream_url_error(value).map(|reason| format!("That URL {reason}.")),
754            )?;
755            let needs_key = ask_yes_no("Does it need an API key? [y/N]: ", input, false)?;
756            // Pasting the complete endpoint URL must work: split any embedded
757            // chat path so base + upstream_chat_path join back to what was
758            // pasted instead of doubling the path.
759            let (url, chat_path) = split_embedded_chat_path(url);
760            (url, chat_path, needs_key)
761        }
762        other => {
763            let spec = spec(other);
764            (
765                spec.upstream_url.to_owned(),
766                spec.chat_path.map(str::to_owned),
767                spec.key_env.is_some(),
768            )
769        }
770    };
771    if let Some(reason) = upstream_url_error(&upstream_url) {
772        anyhow::bail!("upstream URL {reason}");
773    }
774
775    // Key: pasted once, stored privately — no export step, ever.
776    let root = home.join(".agentvisor");
777    // Same rationale as `write_private_key_file` above: atomic 0o700
778    // at mkdir + fsync of newly-created ancestor dirents. Closes the
779    // world-listable window that a bare `create_dir_all` +
780    // `set_permissions(0o700)` opens, and hardens against first-time
781    // install crash-loss of the `~/.agentvisor/` directory.
782    av_core::fsutil::create_dir_all_synced(&root)
783        .with_context(|| format!("create settings directory {}", root.display()))?;
784    let key_file = if needs_key {
785        let name = friendly_name(preset);
786        // `key` holds the accepted secret wrapped in
787        // Zeroizing so the heap allocation is zeroed on drop. Each
788        // failed retry's `pasted` binding drops-zeros before the
789        // next iteration reaches the reader.
790        let mut key: Option<zeroize::Zeroizing<String>> = None;
791        for _ in 0..5 {
792            let pasted = ask_secret(&format!("\nPaste your {name} API key"), input, secrets)?;
793            // `pasted.trim()` returns a &str borrowed from the
794            // zeroing buffer; we validate against the borrow and
795            // only take ownership of a zeroing copy after all
796            // checks pass, so a rejected paste never spawns a
797            // second un-zeroed String on the heap.
798            let trimmed = pasted.trim();
799            if trimmed.is_empty() {
800                println!("The key looked empty — please paste it again.");
801                continue;
802            }
803            if trimmed.chars().any(char::is_control) {
804                println!("That key contains characters that don't belong — please try again.");
805                continue;
806            }
807            // Real API keys are plain ASCII; anything else (emoji, accents,
808            // smart quotes from a rich-text paste) would only fail later
809            // with a cryptic HTTP-header error, so catch it here.
810            if !trimmed.is_ascii() {
811                println!(
812                    "That key contains non-ASCII characters (accents, emoji, or smart quotes) — API keys never do. Please paste it again."
813                );
814                continue;
815            }
816            if trimmed.len() > 8192 {
817                println!(
818                    "That looks far too long to be an API key — it may be the wrong clipboard content. Please paste just the key."
819                );
820                continue;
821            }
822            // Build the accepted-key buffer directly
823            // from the trimmed slice into a fresh Zeroizing so the
824            // stored copy is minimal and zero-on-drop.
825            key = Some(zeroize::Zeroizing::new(trimmed.to_owned()));
826            break;
827        }
828        let Some(key) = key else {
829            anyhow::bail!("no API key provided");
830        };
831        let path = root.join("keys").join(format!("{}.key", slug(preset)));
832        // Pass by `&str` borrow so the write path
833        // never allocates a plain String copy of the secret.
834        write_private_key_file(&path, key.as_str())?;
835        Some(path)
836    } else {
837        None
838    };
839
840    // Assemble, self-validate, and save the config.
841    let data_root = root.join("data");
842    let key = match &key_file {
843        Some(path) => KeySpec::File(path_for_config("your key file location", path)?),
844        None => KeySpec::NoKey,
845    };
846    let plan = ConfigPlan {
847        preset,
848        upstream_url,
849        chat_path,
850        key,
851        data_root: Some(path_for_config("your home folder", &data_root)?),
852    };
853    let rendered = render_config(&plan);
854    av_harness::HarnessConfig::from_toml(&rendered)
855        .map_err(|error| anyhow::anyhow!("generated config failed validation (bug): {error}"))?;
856    let config_path = av_harness::config::user_config_path_from(home);
857    if config_path.exists() {
858        let backup = config_path.with_extension("toml.bak");
859        // `write_atomic`, NOT `std::fs::copy`: copy creates/truncates
860        // the destination FOLLOWING a symlink, so a pre-planted
861        // `agentvisor.toml.bak -> ~/.bashrc` would be clobbered with
862        // TOML. Every other write in this file already defends against
863        // exactly that (init's unlink + write_atomic, the key
864        // installer's rename); the backup was the one exception.
865        let previous = std::fs::read(&config_path)
866            .with_context(|| format!("read {} for backup", config_path.display()))?;
867        av_core::fsutil::write_atomic(&backup, &previous)
868            .with_context(|| format!("back up {}", config_path.display()))?;
869        // The old `fs::copy` inherited the config's 0600; `write_atomic`
870        // creates default-mode (world-readable under umask 022) files.
871        // The backup carries the same upstream_url / key-env metadata the
872        // 0600 policy on the live config exists to protect.
873        #[cfg(unix)]
874        {
875            use std::os::unix::fs::PermissionsExt as _;
876            if let Err(error) = std::fs::set_permissions(&backup, std::fs::Permissions::from_mode(0o600)) {
877                eprintln!(
878                    "warning: failed to chmod 0600 on {}: {error}; contents may be world-readable",
879                    backup.display(),
880                );
881            }
882        }
883        println!("\n(Your previous settings were saved to {})", backup.display());
884    }
885    // Two concurrent wizards on the same $HOME must not interleave into a
886    // fixed `.toml.tmp` name — `write_atomic` uses a UUID-suffixed tmp file
887    // and rename semantics that replace any pre-existing symlink at the
888    // destination without following it.
889    av_core::fsutil::write_atomic(&config_path, rendered.as_bytes())
890        .with_context(|| format!("install {}", config_path.display()))?;
891    // Chmod 0600 to match the key-file installer
892    // (`write_private_key_file`) and prevent co-tenants on a shared
893    // workstation from reading
894    // upstream_url / upstream_api_key_env metadata via default umask
895    // 022 world-readable bits. `write_atomic` opens with default mode
896    // via `OpenOptions::create_new`; setting the perm post-rename
897    // covers the final path atomically since rename is what makes the
898    // dst visible.
899    #[cfg(unix)]
900    {
901        use std::os::unix::fs::PermissionsExt as _;
902        if let Err(error) = std::fs::set_permissions(&config_path, std::fs::Permissions::from_mode(0o600)) {
903            eprintln!(
904                "warning: failed to chmod 0600 on {}: {error}; contents may be world-readable",
905                config_path.display(),
906            );
907        }
908    }
909
910    println!("\nAll set!");
911    println!();
912    println!("  Settings: {}", config_path.display());
913    // The wizard writes the LOWEST-priority config location. A
914    // cwd-local `agentvisor.toml` / `config/harness.toml` (e.g. from an
915    // earlier `avctl init` in this directory) or an exported
916    // `AV_CONFIG` silently outranks it — the user pastes a fresh key,
917    // sees "All set!", and the daemon then loads the stale file. Detect
918    // the shadowing NOW and say so loudly, instead of leaving the only
919    // hint in `start`'s easily-missed "(settings: ...)" line.
920    if let Ok(av_harness::config::ConfigSource::File(effective)) = av_harness::config::resolve_config_source()
921    {
922        let same = match (
923            std::fs::canonicalize(&effective),
924            std::fs::canonicalize(&config_path),
925        ) {
926            (Ok(a), Ok(b)) => a == b,
927            _ => effective == config_path,
928        };
929        if !same {
930            println!();
931            println!("  WARNING: these settings will NOT be used yet.");
932            println!("  A higher-priority config shadows them: {}", effective.display());
933            println!("  Remove/rename that file (or unset AV_CONFIG) so the settings above take effect.");
934        }
935    }
936    if let Some(path) = &key_file {
937        println!("  API key:  {} (private to your user account)", path.display());
938    }
939    if matches!(preset, Preset::Ollama | Preset::Lmstudio) {
940        println!(
941            "\n  Reminder: make sure {} is running on this computer.",
942            friendly_name(preset)
943        );
944    }
945    // Setup already succeeded and the config is on disk, so a closed
946    // stdin here (e.g. `printf '11\n' | avctl`) must not turn into a
947    // scary error — it just means "don't start now".
948    let start_now = ask_yes_no("\nStart AgentVisor AI now? [Y/n]: ", input, true).unwrap_or(false);
949    Ok(WizardOutcome {
950        config_path,
951        start_now,
952    })
953}

1007/// Print the last `count` log lines, unwrapping the JSON envelope when
1008/// possible so failures read like sentences instead of telemetry.
1009///
1010/// JSON-decoding the tracing envelope RESURRECTS
1011/// escaped control bytes (tracing writes `\u001b`; `serde_json` decodes
1012/// it back to live ESC). An attacker who can influence a logged string
1013/// (any upstream/tool response payload that got captured into an error
1014/// message) could then paint spoofed diagnostics on the operator's
1015/// terminal via CSI sequences. Route both the JSON-decoded message and
1016/// the raw-line fallback through the same terminal sanitizer used
1017/// everywhere else in this crate.
1018fn print_log_tail(path: &Path, count: usize) {
1019    // Bounded read: the log is opened append-only with no rotation, so
1020    // it grows without bound across runs; slurping the whole file (the
1021    // old `read_to_string`) risked a multi-GB allocation at exactly the
1022    // moment the operator is diagnosing a failure. Read only a fixed
1023    // suffix — generous for `count` lines of tracing JSON.
1024    const TAIL_BYTES: u64 = 256 * 1024;
1025    let Ok(mut file) = std::fs::File::open(path) else {
1026        return;
1027    };
1028    let Ok(len) = file.metadata().map(|m| m.len()) else {
1029        return;
1030    };
1031    use std::io::{Read as _, Seek as _};
1032    let start = len.saturating_sub(TAIL_BYTES);
1033    if start > 0 && file.seek(std::io::SeekFrom::Start(start)).is_err() {
1034        return;
1035    }
1036    let mut bytes = Vec::with_capacity(usize::try_from(len - start).unwrap_or(0));
1037    if file.take(TAIL_BYTES).read_to_end(&mut bytes).is_err() {
1038        return;
1039    }
1040    let text = String::from_utf8_lossy(&bytes);
1041    // Drop the first (possibly truncated) line when we started mid-file.
1042    let text = if start > 0 {
1043        text.split_once('\n').map_or("", |(_, rest)| rest)
1044    } else {
1045        &text
1046    };
1047    let mut tail: Vec<&str> = text.lines().rev().take(count).collect();
1048    tail.reverse();
1049    for line in tail {
1050        let friendly = serde_json::from_str::<serde_json::Value>(line)
1051            .ok()
1052            .and_then(|value| {
1053                let level = value.get("level")?.as_str()?.to_owned();
1054                let message = value.get("fields")?.get("message")?.as_str()?.to_owned();
1055                Some(format!("{level:<5} {message}"))
1056            })
1057            .unwrap_or_else(|| line.to_owned());
1058        eprintln!("    {}", crate::sanitize_for_terminal(&friendly));
1059    }
1060}

1070/// `avctl start` — run the server in the foreground with friendly output.
1071/// Logs go to a file so the terminal stays readable; Ctrl-C stops both
1072/// processes gracefully (the server flushes trajectories on shutdown).
1073pub async fn start() -> Result<()> {
1074    // Friendly pre-checks with the exact resolution the server uses.
1075    let not_set_up = matches!(
1076        av_harness::config::resolve_config_source(),
1077        Ok(av_harness::config::ConfigSource::BuiltIn)
1078    ) && std::env::var_os("AV_UPSTREAM_URL").is_none_or(|value| value.is_empty());
1079    if not_set_up {
1080        anyhow::bail!(
1081            "AgentVisor AI is not set up yet.\nRun `avctl` (no arguments) and answer a couple of quick questions, then try again."
1082        );
1083    }
1084    let (config, source) = av_harness::config::load_config().map_err(|error| {
1085        anyhow::anyhow!("configuration problem: {error}\nRun `avctl doctor` for a full diagnosis.")
1086    })?;
1087    // Rewrite the listen address to a
1088    // loopback probe URL. The prior implementation was a pair of
1089    // string replacements (`0.0.0.0`->`127.0.0.1`, `[::]`->`[::1]`)
1090    // which missed three real forms:
1091    //   1. IPv6 link-local with a zone identifier
1092    //      (`[fe80::1%eth0]:8484`) — the raw `%` in the URL fails
1093    //      RFC 3986 parsing and every probe URL-parses to `Err`,
1094    //      so `avctl start` blocks the full 20 s deadline and
1095    //      kills a healthy child.
1096    //   2. Fully expanded unspecified `[0:0:0:0:0:0:0:0]:8484`
1097    //      (Rust `SocketAddr::to_string()` for `::` may emit
1098    //      the shortened form, but operator-typed configs can be
1099    //      expanded).
1100    //   3. Non-`0.0.0.0` / `[::]` listen literals that happen to
1101    //      resolve to an unspecified address.
1102    // Parse via `SocketAddr` so the address family drives the
1103    // loopback rewrite, and drop any zone identifier since a
1104    // loopback connect never needs it.
1105    let base = build_probe_base(&config.listen);
1106
1107    if health_ok(&base).await {
1108        println!("AgentVisor AI is already running at {base}");
1109        // The running daemon loaded its settings at ITS boot — it does
1110        // not reload on config rewrites. Right after the wizard (the
1111        // common path into here) that means the fresh settings are NOT
1112        // in effect until the old process is restarted; without this
1113        // hint the success banner reads as "new settings active".
1114        println!("Note: it keeps the settings it started with. To apply new settings, stop that process and run `avctl start` again.");
1115        print_usage_banner(&base);
1116        return Ok(());
1117    }
1118
1119    let binary = find_server_binary();
1120    let log_path = av_harness::config::user_config_path()
1121        .and_then(|path| Some(path.parent()?.join("agentvisor-ai.log")))
1122        .unwrap_or_else(|| PathBuf::from("agentvisor-ai.log"));
1123    if let Some(parent) = log_path.parent() {
1124        let _ = std::fs::create_dir_all(parent);
1125    }
1126    let log = std::fs::OpenOptions::new()
1127        .create(true)
1128        .append(true)
1129        .open(&log_path)
1130        .with_context(|| format!("open log file {}", log_path.display()))?;
1131    let log_err = log.try_clone().context("clone log handle")?;
1132
1133    println!("Starting AgentVisor AI (settings: {source})...");
1134    let mut command = tokio::process::Command::new(&binary);
1135    command
1136        .stdout(std::process::Stdio::from(log))
1137        .stderr(std::process::Stdio::from(log_err));
1138    // When running from the per-user config, keep every file the server
1139    // creates (signing seed included) inside ~/.agentvisor/ instead of
1140    // scattering a config/ directory into whatever folder we run from.
1141    if std::env::var_os("AV_SIGNING_SEED_FILE").is_none() {
1142        if let av_harness::config::ConfigSource::File(config_path) = &source {
1143            if Some(config_path) == av_harness::config::user_config_path().as_ref() {
1144                if let Some(root) = config_path.parent() {
1145                    command.env("AV_SIGNING_SEED_FILE", root.join("signing.seed"));
1146                }
1147            }
1148        }
1149    }
1150    let mut child = command.spawn().map_err(|error| {
1151        if error.kind() == std::io::ErrorKind::NotFound {
1152            anyhow::anyhow!(
1153                "could not find the AgentVisor AI server program ({}).\n\
1154                 It normally lives next to avctl. To install it:\n\
1155                 cargo install av-harness   # or: cargo install --path crates/av-harness",
1156                binary.display()
1157            )
1158        } else {
1159            anyhow::Error::new(error).context(format!("could not run {}", binary.display()))
1160        }
1161    })?;
1162
1163    // Wait until healthy — or explain why it died. Stop requests during
1164    // startup must reach the child too: without polling the signal
1165    // streams here, a SIGTERM/SIGINT in this window would kill avctl at
1166    // the default disposition and orphan the server mid-boot.
1167    let mut interrupt = std::pin::pin!(tokio::signal::ctrl_c());
1168    let mut terminate = std::pin::pin!(terminate_signal());
1169    let deadline = std::time::Instant::now() + Duration::from_secs(20);
1170    loop {
1171        if let Some(status) = child.try_wait().context("check server status")? {
1172            eprintln!("AgentVisor AI stopped right away ({status}). Last log lines:");
1173            print_log_tail(&log_path, 12);
1174            anyhow::bail!("startup failed — run `avctl doctor` for a diagnosis");
1175        }
1176        if health_ok(&base).await {
1177            break;
1178        }
1179        if std::time::Instant::now() >= deadline {
1180            let _ = child.start_kill();
1181            let _ = child.wait().await;
1182            print_log_tail(&log_path, 12);
1183            anyhow::bail!("the server did not become ready within 20 seconds");
1184        }
1185        tokio::select! {
1186            () = tokio::time::sleep(Duration::from_millis(250)) => {}
1187            _ = &mut interrupt => {
1188                println!("\nStopping AgentVisor AI...");
1189                forward_signal(&child, "-INT");
1190                wait_for_graceful_exit(&mut child).await;
1191                return Ok(());
1192            }
1193            () = &mut terminate => {
1194                println!("\nStopping AgentVisor AI...");
1195                forward_signal(&child, "-TERM");
1196                wait_for_graceful_exit(&mut child).await;
1197                return Ok(());
1198            }
1199        }
1200    }
1201
1202    println!("✓ AgentVisor AI is running at {base}");
1203    print_usage_banner(&base);
1204    println!("  Log file: {}", log_path.display());
1205    println!("  Press Ctrl-C to stop.");
1206
1207    tokio::select! {
1208        status = child.wait() => {
1209            let status = status.context("wait for server")?;
1210            if status.success() {
1211                println!("AgentVisor AI stopped.");
1212            } else {
1213                eprintln!("AgentVisor AI exited unexpectedly ({status}). Last log lines:");
1214                print_log_tail(&log_path, 12);
1215                anyhow::bail!("server exited unexpectedly");
1216            }
1217        }
1218        _ = &mut interrupt => {
1219            println!("\nStopping AgentVisor AI...");
1220            // Terminal Ctrl-C reaches the child through the foreground
1221            // process group, but a direct `kill -INT <avctl>` does not —
1222            // forward to be sure. A duplicate SIGINT is harmless: the
1223            // server's signal stream is already registered, so it can't
1224            // fall back to the abrupt default disposition.
1225            forward_signal(&child, "-INT");
1226            wait_for_graceful_exit(&mut child).await;
1227        }
1228        () = &mut terminate => {
1229            // SIGTERM (Activity Monitor "Quit", plain `kill`, service
1230            // managers) only reaches avctl — without forwarding it the
1231            // server would keep running unsupervised as an orphan.
1232            println!("\nStopping AgentVisor AI...");
1233            forward_signal(&child, "-TERM");
1234            wait_for_graceful_exit(&mut child).await;
1235        }
1236    }
1237    Ok(())
1238}

1240/// Resolves when the process receives SIGTERM; never resolves where
1241/// SIGTERM does not exist (or the handler cannot be installed) so the
1242/// other `select!` arms keep working.
1243#[cfg(unix)]
1244async fn terminate_signal() {
1245    match tokio::signal::unix::signal(tokio::signal::unix::SignalKind::terminate()) {
1246        Ok(mut stream) => {
1247            stream.recv().await;
1248        }
1249        Err(_) => std::future::pending().await,
1250    }
1251}
1252
1253#[cfg(not(unix))]
1254async fn terminate_signal() {
1255    std::future::pending::<()>().await;
1256}

1258/// Forward `signal` (e.g. "-INT") to the server child so it shuts down
1259/// gracefully even when only avctl was signalled. Uses the `kill`
1260/// program because the workspace forbids unsafe code (no raw libc).
1261#[cfg(unix)]
1262fn forward_signal(child: &tokio::process::Child, signal: &str) {
1263    if let Some(pid) = child.id() {
1264        let _ = std::process::Command::new("kill")
1265            .arg(signal)
1266            .arg(pid.to_string())
1267            .status();
1268    }
1269}
1270
1271#[cfg(not(unix))]
1272fn forward_signal(_child: &tokio::process::Child, _signal: &str) {}

1274/// Wait for the server's four-phase graceful shutdown, then force-kill
1275/// only if it truly hangs.
1276///
1277/// The harness's documented shutdown budget is
1278/// up to 95 s (30 s HTTP drain + 30 s worker wait_idle + 30 s
1279/// finalize_sessions + 5 s telemetry flush) — the systemd unit and
1280/// compose files are tuned to 120–125 s for exactly this reason. The
1281/// prior 10 s SIGKILL here dropped receipts and truncated trajectories
1282/// mid-finalize on the desktop path while printing "Your data is
1283/// saved." Print progress so a long flush doesn't look like a hang.
1284async fn wait_for_graceful_exit(child: &mut tokio::process::Child) {
1285    const GRACEFUL_BUDGET_S: u64 = 100;
1286    match tokio::time::timeout(Duration::from_secs(10), child.wait()).await {
1287        Ok(_) => {
1288            println!("Stopped. Your data is saved.");
1289            return;
1290        }
1291        Err(_) => {
1292            println!("Waiting for the server to flush sessions and receipts (up to {GRACEFUL_BUDGET_S}s)...");
1293        }
1294    }
1295    match tokio::time::timeout(Duration::from_secs(GRACEFUL_BUDGET_S - 10), child.wait()).await {
1296        Ok(_) => println!("Stopped. Your data is saved."),
1297        Err(_) => {
1298            let _ = child.start_kill();
1299            let _ = child.wait().await;
1300            // Do NOT claim data safety: a SIGKILL mid-finalize can
1301            // leave receipts unminted and trajectories truncated —
1302            // recovery will quarantine them on next start.
1303            println!("Stopped (forced). Some sessions may need recovery on next start.");
1304        }
1305    }
1306}

1308/// One diagnostic line: pass/warn/fail plus a short explanation.
1309enum Check {
1310    Pass(String),
1311    Warn(String),
1312    Fail(String),
1313}
1314
1315fn report(checks: &[Check]) -> (usize, usize) {
1316    let mut warnings = 0;
1317    let mut failures = 0;
1318    for check in checks {
1319        match check {
1320            Check::Pass(message) => println!("  ok    {message}"),
1321            Check::Warn(message) => {
1322                warnings += 1;
1323                println!("  warn  {message}");
1324            }
1325            Check::Fail(message) => {
1326                failures += 1;
1327                println!("  FAIL  {message}");
1328            }
1329        }
1330    }
1331    (warnings, failures)
1332}

1334/// `avctl doctor` — resolve config exactly like the server and verify every
1335/// runtime prerequisite. Never prints secret values or URL userinfo —
1336/// key sources by name only, endpoints with credentials redacted.
1337pub async fn doctor(offline: bool) -> Result<()> {
1338    let mut checks = Vec::new();
1339
1340    // 1. Config resolution (same search order as the server).
1341    let resolved = av_harness::config::load_config();
1342    let (config, source) = match resolved {
1343        Ok((config, source)) => {
1344            checks.push(Check::Pass(format!("config: {source}")));
1345            (Some(config), Some(source))
1346        }
1347        Err(error) => {
1348            checks.push(Check::Fail(format!("config: {error}")));
1349            (None, None)
1350        }
1351    };
1352
1353    if let Some(config) = &config {
1354        // 2. Common URL footgun: base URL already contains the chat path
1355        // prefix (e.g. ".../v1" + "/v1/chat/completions").
1356        if let Some(segment) = config.duplicated_chat_path_segment() {
1357            checks.push(Check::Warn(format!(
1358                "upstream_url ends with \"/{segment}\" and upstream_chat_path repeats it; \
1359                 the joined URL will contain \"/{segment}/{segment}/\" — drop the suffix from upstream_url"
1360            )));
1361        }
1362
1363        // 3. Upstream auth source (names only, never values).
1364        let auth = av_harness::pipeline::describe_upstream_auth(config);
1365        if let Some(env) = &config.upstream_api_key_env {
1366            match std::env::var(env) {
1367                Ok(value) if !value.trim().is_empty() => {
1368                    checks.push(Check::Pass(format!("upstream auth: {auth} (set)")));
1369                }
1370                _ => checks.push(Check::Fail(format!(
1371                    "upstream auth: {auth} — environment variable {env} is unset or empty"
1372                ))),
1373            }
1374        } else if let Some(file) = &config.upstream_api_key_file {
1375            // Mirror `av_harness::pipeline::require_owner_only_secret`
1376            // exactly — the runtime refuses symlinks, non-regular files,
1377            // group/other-readable modes, and empty content. Doctor used to
1378            // only `metadata(file).ok()`-check existence, so an operator
1379            // whose key file was chmod 644 or a dangling symlink would see
1380            // `avctl doctor` PASS and `avctl start`/harness startup FAIL.
1381            match check_owner_only_secret(std::path::Path::new(file)) {
1382                Ok(()) => checks.push(Check::Pass(format!("upstream auth: {auth}"))),
1383                Err(reason) => checks.push(Check::Fail(format!("upstream auth: {auth} — {reason}"))),
1384            }
1385        } else {
1386            checks.push(Check::Pass(format!("upstream auth: {auth}")));
1387        }
1388
1389        // Parity with the upstream API key
1390        // check above. `tool_upstream_bearer_env/_file` is the MCP
1391        // path's bearer token; the runtime applies the SAME
1392        // `require_owner_only_secret` posture at first tool call,
1393        // but doctor used to check neither.
1394        if let Some(env) = &config.tool_upstream_bearer_env {
1395            match std::env::var(env) {
1396                Ok(value) if !value.trim().is_empty() => {
1397                    checks.push(Check::Pass(format!("tool bearer: env {env} (set)")));
1398                }
1399                _ => checks.push(Check::Fail(format!(
1400                    "tool bearer: environment variable {env} is unset or empty"
1401                ))),
1402            }
1403        } else if let Some(file) = &config.tool_upstream_bearer_file {
1404            match check_owner_only_secret(std::path::Path::new(file)) {
1405                Ok(()) => checks.push(Check::Pass(format!("tool bearer: {file}"))),
1406                Err(reason) => {
1407                    checks.push(Check::Fail(format!("tool bearer: {file} — {reason}")));
1408                }
1409            }
1410        } else if config.tool_upstream_url.is_some() {
1411            checks.push(Check::Warn(
1412                "tool bearer: none configured (MCP calls will proxy unauthenticated)".to_owned(),
1413            ));
1414        }
1415
1416        // 4. Upstream reachability (TCP + HTTP; any HTTP status counts).
1417        if offline {
1418            checks.push(Check::Warn("upstream: skipped (--offline)".to_owned()));
1419        } else {
1420            let url = config.upstream_url.trim_end_matches('/').to_owned();
1421            let client = reqwest::Client::builder()
1422                .redirect(reqwest::redirect::Policy::none())
1423                .timeout(Duration::from_secs(3))
1424                .build()
1425                .context("build probe client")?;
1426            match client.get(&url).send().await {
1427                Ok(response) => checks.push(Check::Pass(format!(
1428                    "upstream: {} reachable (HTTP {})",
1429                    redact_userinfo(&url),
1430                    response.status().as_u16()
1431                ))),
1432                Err(error) => {
1433                    // Redact userinfo here too — this failure
1434                    // line used to leak what the success line redacted.
1435                    // `without_url()` strips reqwest's own embedded copy of
1436                    // the URL from the error Display as well.
1437                    checks.push(Check::Fail(format!(
1438                        "upstream: {} unreachable ({})",
1439                        redact_userinfo(&url),
1440                        error.without_url()
1441                    )));
1442                }
1443            }
1444        }
1445
1446        // 5. Bridge manifest. Same MAX_CONTROL_BYTES cap as
1447        // `manifest-validate` and the daemon's own load, so all three
1448        // reach the same verdict on the same file.
1449        match av_core::fsutil::read_capped_string(
1450            Path::new(&config.bridge_manifest_path),
1451            av_core::fsutil::MAX_CONTROL_BYTES,
1452        ) {
1453            Ok(text) => match av_bridge::BridgeManifest::from_yaml(&text) {
1454                Ok(manifest) => checks.push(Check::Pass(format!(
1455                    "manifest: {} ({} topics)",
1456                    config.bridge_manifest_path,
1457                    manifest.topics.len()
1458                ))),
1459                Err(error) => checks.push(Check::Fail(format!(
1460                    "manifest: {} invalid: {error}",
1461                    config.bridge_manifest_path
1462                ))),
1463            },
1464            Err(_) if config.uses_default_manifest_path() => checks.push(Check::Pass(
1465                "manifest: using embedded built-in (no file on disk)".to_owned(),
1466            )),
1467            Err(error) => checks.push(Check::Fail(format!(
1468                "manifest: {}: {error}",
1469                config.bridge_manifest_path
1470            ))),
1471        }
1472
1473        // 6. WASM policies.
1474        for path in &config.wasm_policy_paths {
1475            if Path::new(path).is_file() {
1476                checks.push(Check::Pass(format!("policy: {path}")));
1477            } else if av_harness::HarnessConfig::is_default_policy_path(path) {
1478                checks.push(Check::Pass(format!("policy: {path} (embedded built-in)")));
1479            } else {
1480                checks.push(Check::Fail(format!("policy: {path} not found")));
1481            }
1482        }
1483
1484        // 7. Tool schemas. The consequence differs by posture:
1485        // require_tool_schema = true (default) rejects schema-less tool
1486        // calls; false lets them through the schema gate unvalidated.
1487        let no_schema_consequence = if config.require_tool_schema {
1488            "tool calls will be rejected"
1489        } else {
1490            "require_tool_schema = false, so tool arguments will skip schema validation (policy and budget gates still apply)"
1491        };
1492        match config.tool_schema_dir.as_deref() {
1493            Some(dir) => match std::fs::read_dir(dir) {
1494                Ok(entries) => {
1495                    let count = entries
1496                        .filter_map(Result::ok)
1497                        .filter(|e| e.path().extension().is_some_and(|x| x == "json"))
1498                        .count();
1499                    if count > 0 {
1500                        checks.push(Check::Pass(format!("tool schemas: {count} in {dir}")));
1501                    } else if config.tool_upstream_url.is_some() {
1502                        checks.push(Check::Fail(format!(
1503                            "tool schemas: none in {dir} but tool_upstream_url is set"
1504                        )));
1505                    } else {
1506                        checks.push(Check::Warn(format!(
1507                            "tool schemas: none in {dir}; {no_schema_consequence}"
1508                        )));
1509                    }
1510                }
1511                Err(_) if config.uses_default_tool_schema_dir() => checks.push(Check::Warn(format!(
1512                    "tool schemas: default directory missing; {no_schema_consequence}"
1513                ))),
1514                Err(error) => checks.push(Check::Fail(format!("tool schemas: {dir}: {error}"))),
1515            },
1516            None => checks.push(Check::Warn(format!(
1517                "tool schemas: no directory configured; {no_schema_consequence}"
1518            ))),
1519        }
1520
1521        // 8. Signing seed (created on first run if absent). Pick
1522        // the same path `avctl start` will actually pass to the
1523        // harness — when the user config is at `user_config_path()`,
1524        // start overrides `AV_SIGNING_SEED_FILE` to
1525        // `<user_config_dir>/signing.seed` (see main.rs
1526        // `avctl_start`, "Starting AgentVisor AI"). Doctor used to
1527        // probe `config/signing.seed` regardless, so a bad seed in
1528        // the user directory would pass doctor and fail startup.
1529        // Additionally, verify the seed content (hex, 32 bytes, not
1530        // known-weak) so a corrupt seed cannot survive doctor —
1531        // `av_harness::main::read_signer` refuses those at startup.
1532        let seed_path = seed_path_for(source.as_ref());
1533        if seed_path.is_file() {
1534            #[cfg(unix)]
1535            {
1536                use std::os::unix::fs::MetadataExt as _;
1537                match std::fs::symlink_metadata(&seed_path) {
1538                    Ok(meta) if meta.file_type().is_symlink() => checks.push(Check::Fail(format!(
1539                        "signing seed: {} is a symbolic link",
1540                        seed_path.display()
1541                    ))),
1542                    Ok(meta) if meta.mode() & 0o077 != 0 => checks.push(Check::Fail(format!(
1543                        "signing seed: {} is group/world accessible (chmod 600 it)",
1544                        seed_path.display()
1545                    ))),
1546                    Ok(_) => match check_signing_seed_content(&seed_path) {
1547                        Ok(()) => checks.push(Check::Pass(format!("signing seed: {}", seed_path.display()))),
1548                        Err(reason) => checks.push(Check::Fail(format!(
1549                            "signing seed: {}: {reason}",
1550                            seed_path.display()
1551                        ))),
1552                    },
1553                    Err(error) => {
1554                        checks.push(Check::Fail(format!(
1555                            "signing seed: {}: {error}",
1556                            seed_path.display()
1557                        )));
1558                    }
1559                }
1560            }
1561            #[cfg(not(unix))]
1562            match check_signing_seed_content(&seed_path) {
1563                Ok(()) => checks.push(Check::Pass(format!("signing seed: {}", seed_path.display()))),
1564                Err(reason) => checks.push(Check::Fail(format!(
1565                    "signing seed: {}: {reason}",
1566                    seed_path.display()
1567                ))),
1568            }
1569        } else {
1570            checks.push(Check::Pass(format!(
1571                "signing seed: {} will be generated on first run",
1572                seed_path.display()
1573            )));
1574        }
1575
1576        // 9. Data directories writable (created on demand by the server).
1577        for (label, dir) in [
1578            ("spool dir", config.atif_spool_dir.as_str()),
1579            ("bridge dir", config.bridge_data_dir.as_str()),
1580        ] {
1581            let path = Path::new(dir);
1582            let probe_parent = if path.exists() {
1583                path.to_path_buf()
1584            } else {
1585                // Find the nearest existing ancestor to test writability.
1586                let mut ancestor = path.to_path_buf();
1587                while !ancestor.exists() {
1588                    match ancestor.parent() {
1589                        Some(parent) if !parent.as_os_str().is_empty() => {
1590                            ancestor = parent.to_path_buf();
1591                        }
1592                        _ => {
1593                            ancestor = PathBuf::from(".");
1594                            break;
1595                        }
1596                    }
1597                }
1598                ancestor
1599            };
1600            // A real create+delete probe: permission bits alone lie about
1601            // ownership, ACLs, and read-only mounts. Write
1602            // actual DATA bytes, not just a dirent — a filesystem
1603            // out of data blocks (the most likely spool outage) can
1604            // still create zero-byte files from reserved inode space,
1605            // so a bare create_new passed while every request 503'd.
1606            let probe = probe_parent.join(format!(".avctl-doctor-{}", std::process::id()));
1607            let write_outcome = std::fs::OpenOptions::new()
1608                .write(true)
1609                .create_new(true)
1610                .open(&probe)
1611                .and_then(|mut file| {
1612                    use std::io::Write as _;
1613                    file.write_all(&[0u8; 4096])?;
1614                    file.sync_all()
1615                });
1616            match write_outcome {
1617                Ok(()) => {
1618                    let _ = std::fs::remove_file(&probe);
1619                    checks.push(Check::Pass(format!("{label}: {dir}")));
1620                }
1621                Err(error) => {
1622                    let _ = std::fs::remove_file(&probe);
1623                    checks.push(Check::Fail(format!(
1624                        "{label}: cannot write 4 KiB in {} ({error})",
1625                        probe_parent.display()
1626                    )));
1627                }
1628            }
1629            // Surface the spool's current footprint so an operator can
1630            // see growth (there is no automatic pruning without
1631            // atif_retention_days).
1632            if *label == *"spool dir" && path.exists() {
1633                let (bytes, files) = std::fs::read_dir(path)
1634                    .map(|entries| {
1635                        entries
1636                            .flatten()
1637                            .filter_map(|entry| entry.metadata().ok())
1638                            .filter(std::fs::Metadata::is_file)
1639                            .fold((0u64, 0u64), |(bytes, files), metadata| {
1640                                (bytes.saturating_add(metadata.len()), files + 1)
1641                            })
1642                    })
1643                    .unwrap_or((0, 0));
1644                checks.push(Check::Pass(format!(
1645                    "spool footprint: {files} files, {} KiB (top level)",
1646                    bytes / 1024
1647                )));
1648            }
1649        }
1650
1651        // 9b. Backends the config selects that THIS
1652        // build cannot run. Previously doctor passed every check while
1653        // the daemon was guaranteed to hard-fail at boot on a
1654        // compiled-out backend.
1655        for problem in config.unsupported_backend_requirements() {
1656            checks.push(Check::Fail(problem));
1657        }
1658
1659        // 9c. The three checkout-path footguns doctor
1660        // used to miss entirely.
1661        let loopback = listen_is_loopback(&config.listen);
1662        if !loopback {
1663            checks.push(Check::Warn(format!(
1664                "listen = {:?} is not loopback: anyone who can reach this address gets a \
1665                 fully-authenticated proxy to your provider account (the key is injected \
1666                 server-side). Keep 127.0.0.1 unless an outer network layer controls access",
1667                config.listen
1668            )));
1669        }
1670        if config.default_workflow == "unsigned" {
1671            checks.push(Check::Warn(
1672                "default_workflow = \"unsigned\": sessions produce ATIF trajectories but NO \
1673                 signed receipts. Set default_workflow = \"signed\" (or send \
1674                 X-AV-Workflow: signed per request) if you need verifiable receipts"
1675                    .to_owned(),
1676            ));
1677        }
1678        // Budget posture. All dimensions are optional, so a config with
1679        // no (or an empty) [budget] block parses cleanly and enforces
1680        // NOTHING — unlimited tokens, payout, and tool calls per
1681        // session. `avctl init` writes binding limits, but hand-written
1682        // and pre-init configs hit this silently; the whole point of
1683        // doctor is to catch exactly this class of quiet misposture.
1684        if config.budget.binds_anything() {
1685            checks.push(Check::Pass("budget: at least one binding limit".to_owned()));
1686        } else {
1687            checks.push(Check::Warn(
1688                "budget: no [budget] limits configured — sessions run with UNLIMITED \
1689                 tokens, payout, and tool calls. Add a [budget] block (see `avctl init` \
1690                 output) with max_tokens / max_payout_usd_micros / max_total_tool_calls"
1691                    .to_owned(),
1692            ));
1693        }
1694        if config
1695            .principal_budget
1696            .as_ref()
1697            .is_some_and(|spec| !spec.binds_anything())
1698        {
1699            checks.push(Check::Warn(
1700                "principal_budget: [principal_budget] block present but no limits set — \
1701                 it enforces nothing"
1702                    .to_owned(),
1703            ));
1704        }
1705        if config.dashboard_enabled && !loopback {
1706            checks.push(Check::Warn(format!(
1707                "dashboard_enabled = true with non-loopback listen {:?}: /dashboard and \
1708                 /api/v1/dashboard/* are UNAUTHENTICATED and expose per-session identity, \
1709                 token and cost metadata to anyone who can reach the port",
1710                config.listen
1711            )));
1712        }
1713
1714        // 10. Backend endpoints (TCP probe only; skipped offline).
1715        // Gate each probe on the BACKEND SELECTOR, not on the endpoint
1716        // being set: the daemon ignores `state_endpoint` under
1717        // `state_backend = "memory"` (and validates that as fine), so
1718        // probing an unused endpoint made doctor FAIL configs the
1719        // daemon runs happily — e.g. a compose template exporting
1720        // AV_STATE_ENDPOINT while the config keeps the memory backend.
1721        if !offline {
1722            for (label, endpoint, used) in [
1723                (
1724                    "state (redis)",
1725                    config.state_endpoint.as_deref(),
1726                    config.state_backend != "memory",
1727                ),
1728                (
1729                    "bridge endpoint",
1730                    config.bridge_endpoint.as_deref(),
1731                    config.bridge_backend != "embedded",
1732                ),
1733                (
1734                    "qdrant",
1735                    config.qdrant_url.as_deref(),
1736                    config.vector_backend == "qdrant",
1737                ),
1738            ] {
1739                let Some(endpoint) = endpoint.filter(|e| !e.is_empty()) else {
1740                    continue;
1741                };
1742                if !used {
1743                    checks.push(Check::Warn(format!(
1744                        "{label}: endpoint configured but the selected backend does not use it \
1745                         (the daemon ignores it); not probed"
1746                    )));
1747                    continue;
1748                }
1749                match probe_endpoint_any(endpoint).await {
1750                    Ok(()) => checks.push(Check::Pass(format!(
1751                        // The success line must
1752                        // redact userinfo too, not just the failure line.
1753                        "{label}: {} reachable",
1754                        redact_userinfo(endpoint)
1755                    ))),
1756                    Err(error) => {
1757                        // DO NOT echo `endpoint` verbatim
1758                        // in the failure line — a redis/qdrant URL may
1759                        // embed userinfo (`redis://user:pass@host/0`)
1760                        // and this text lands on stderr, which CI
1761                        // pipelines routinely capture. Report the
1762                        // label + probe error only.
1763                        checks.push(Check::Fail(format!("{label}: unreachable ({error})")));
1764                    }
1765                }
1766            }
1767        }
1768
1769        // 11. Identity posture.
1770        if config.require_identity {
1771            checks.push(Check::Pass("identity: required (production posture)".to_owned()));
1772        } else {
1773            checks.push(Check::Warn(
1774                "identity: optional (dev mode; set require_identity=true in production)".to_owned(),
1775            ));
1776        }
1777    }
1778
1779    println!("avctl doctor");
1780    let (warnings, failures) = report(&checks);
1781    println!();
1782    if failures > 0 {
1783        anyhow::bail!("{failures} check(s) failed, {warnings} warning(s)");
1784    }
1785    println!("all checks passed ({warnings} warning(s))");
1786    Ok(())
1787}

1799/// TCP-connect to `host:port` extracted from a URL or bare `host:port`.
1800async fn probe_endpoint(endpoint: &str) -> Result<()> {

1802    tokio::time::timeout(Duration::from_secs(3), tokio::net::TcpStream::connect(&target))
1803        .await
1804        .map_err(|_| anyhow::anyhow!("timeout"))?
1805        .map(|_| ())
1806        .map_err(anyhow::Error::from)
1807}

1809/// Probe a config-shaped endpoint. Two forms that
1810/// `probe_endpoint` alone cannot handle without emitting a spurious
1811/// "unreachable" for a perfectly valid config:
1812///
1813/// 1. Redis `unix:` (and `redis+unix:`) — a local UDS socket, not a
1814///    TCP host:port. Config-validate at `HarnessConfig::validate`
1815///    (state_endpoint scheme allowlist) accepts `unix:...` for Redis.
1816///    We probe the socket file's existence and move on.
1817///
1818/// 2. Kafka bootstrap list — a comma-separated `host:port,host:port,...`
1819///    string. The bare list has no scheme, so `probe_target` would
1820///    misparse the first comma as trailing garbage in the port.
1821///    Probing every bootstrap is overkill; the runtime only needs
1822///    ONE reachable broker. We treat the whole list as "reachable"
1823///    if any single member is reachable.
1824async fn probe_endpoint_any(endpoint: &str) -> Result<()> {
1825    if endpoint.starts_with("unix:") || endpoint.starts_with("redis+unix:") {
1826        let path = endpoint
1827            .strip_prefix("redis+unix:")
1828            .or_else(|| endpoint.strip_prefix("unix:"))
1829            .unwrap_or(endpoint)
1830            .trim_start_matches("//");
1831        if path.is_empty() {
1832            anyhow::bail!("unix endpoint has empty path");
1833        }
1834        match std::fs::metadata(path) {
1835            // Existence alone is not reachability: a regular file at
1836            // the socket path passed doctor while the runtime's Redis
1837            // client requires an actual Unix domain socket — a false
1838            // "reachable" verdict on exactly the misconfiguration
1839            // doctor exists to catch.
1840            Ok(metadata) => {
1841                #[cfg(unix)]
1842                {
1843                    use std::os::unix::fs::FileTypeExt as _;
1844                    if !metadata.file_type().is_socket() {
1845                        anyhow::bail!(
1846                            "unix socket {path}: exists but is not a socket (type: {:?})",
1847                            metadata.file_type()
1848                        );
1849                    }
1850                    // A socket FILE with no listener behind it (stale
1851                    // socket left by a dead daemon) is the second false-
1852                    // PASS shape: metadata says "socket", but the
1853                    // runtime's connect will get ECONNREFUSED. Probe
1854                    // with a real connect, same 3 s bound as the TCP
1855                    // probe.
1856                    tokio::time::timeout(Duration::from_secs(3), tokio::net::UnixStream::connect(path))
1857                        .await
1858                        .map_err(|_| anyhow::anyhow!("unix socket {path}: connect timeout"))?
1859                        .map_err(|error| {
1860                            anyhow::anyhow!("unix socket {path}: exists but refuses connections ({error})")
1861                        })?;
1862                }
1863                #[cfg(not(unix))]
1864                let _ = metadata;
1865                return Ok(());
1866            }
1867            Err(error) => anyhow::bail!("unix socket {path}: {error}"),
1868        }
1869    }
1870    if endpoint.contains(',') {
1871        let mut last_err: Option<anyhow::Error> = None;
1872        for member in endpoint.split(',').map(str::trim).filter(|m| !m.is_empty()) {
1873            match probe_endpoint(member).await {
1874                Ok(()) => return Ok(()),
1875                Err(error) => last_err = Some(error),
1876            }
1877        }
1878        return Err(last_err.unwrap_or_else(|| anyhow::anyhow!("no bootstrap members configured")));
1879    }
1880    probe_endpoint(endpoint).await
1881}

1883/// Mirror `av_harness::pipeline::require_owner_only_secret`
1884/// posture so `avctl doctor` cannot falsely PASS a key file the
1885/// harness would refuse: symlinks, non-regular files,
1886/// group/other-readable modes on Unix, and empty content.
1887fn check_owner_only_secret(path: &Path) -> std::result::Result<(), String> {
1888    #[cfg(unix)]
1889    {
1890        use std::os::unix::fs::MetadataExt as _;
1891        let metadata =
1892            std::fs::symlink_metadata(path).map_err(|error| format!("stat {}: {error}", path.display()))?;
1893        let file_type = metadata.file_type();
1894        if file_type.is_symlink() {
1895            return Err(format!(
1896                "{} is a symbolic link; refusing to follow",
1897                path.display()
1898            ));
1899        }
1900        if !file_type.is_file() {
1901            return Err(format!("{} is not a regular file", path.display()));
1902        }
1903        let mode = metadata.mode() & 0o777;
1904        if mode & 0o077 != 0 {
1905            return Err(format!(
1906                "{} has mode 0o{mode:03o}; must be owner-only (chmod 600 {})",
1907                path.display(),
1908                path.display()
1909            ));
1910        }
1911    }
1912    #[cfg(not(unix))]
1913    {
1914        let metadata =
1915            std::fs::metadata(path).map_err(|error| format!("stat {}: {error}", path.display()))?;
1916        if !metadata.is_file() {
1917            return Err(format!("{} is not a regular file", path.display()));
1918        }
1919    }
1920    let contents = std::fs::read(path).map_err(|error| format!("read {}: {error}", path.display()))?;
1921    // Mirror the daemon exactly: auth resolution uses `read_to_string`,
1922    // which hard-errors on invalid UTF-8. Without this check a UTF-16
1923    // key file (PowerShell `Out-File` default) passed doctor (0x00
1924    // bytes are not ASCII whitespace) and then failed in the daemon.
1925    if std::str::from_utf8(&contents).is_err() {
1926        return Err(format!(
1927            "{} is not valid UTF-8 (the daemon reads key files with read_to_string and will \
1928             refuse it); re-save it as plain UTF-8/ASCII",
1929            path.display()
1930        ));
1931    }
1932    if contents.iter().all(u8::is_ascii_whitespace) {
1933        return Err(format!("{} is empty", path.display()));
1934    }
1935    Ok(())
1936}

1962/// Validate signing-seed contents the way
1963/// `av_harness::main::read_signer` will at startup — hex-decoded,
1964/// exactly 32 bytes, and not a known-weak seed (all-zero or all-0xFF).
1965/// A doctor that only checks file mode / symlink status lets a
1966/// truncated or textbook-wrong seed pass, and the operator only
1967/// discovers the problem when `avctl start` fails.
1968fn check_signing_seed_content(path: &Path) -> std::result::Result<(), String> {
1969    let contents =
1970        std::fs::read_to_string(path).map_err(|error| format!("read {}: {error}", path.display()))?;
1971    let decoded = hex::decode(contents.trim()).map_err(|error| format!("decode as hex: {error}"))?;
1972    if decoded.len() != 32 {
1973        return Err(format!("must be exactly 32 bytes, got {}", decoded.len()));
1974    }
1975    if decoded.iter().all(|byte| *byte == 0x00) {
1976        return Err("all-zero seed (known-weak, globally predictable public key)".to_owned());
1977    }
1978    if decoded.iter().all(|byte| *byte == 0xFF) {
1979        return Err("all-0xFF seed (known-weak, globally predictable public key)".to_owned());
1980    }
1981    Ok(())
1982}

1984/// Extract the `host:port` from a URL or bare
1985/// `host:port` string. Split out as a pure helper so tests can lock
1986/// in three formerly-buggy behaviours: scheme-driven default
1987/// ports for HTTPS/redis/nats/rediss, userinfo stripping so
1988/// credentials never leak into the connect target or into error text,
1989/// and IPv6 bracketed literals staying intact.
1990fn probe_target(endpoint: &str) -> Result<String> {
1991    let (scheme, rest) = match endpoint.split_once("://") {
1992        Some((scheme, rest)) => (Some(scheme), rest),
1993        None => (None, endpoint),
1994    };
1995    // Trim path/query/fragment BEFORE stripping userinfo: RFC 3986
1996    // userinfo cannot contain '/', so an '@' after the first '/' is
1997    // path data, not credentials ("http://h/p@th" must probe h, not
1998    // "th"). Mirrors redact_userinfo's authority-first parse.
1999    let hostpart = rest.split(['/', '?', '#']).next().unwrap_or(rest);
2000    // Strip userinfo (`user:pass@`).
2001    let host_and_port = hostpart
2002        .rsplit_once('@')
2003        .map_or(hostpart, |(_userinfo, host)| host);
2004    let (host, port_opt) = if let Some(stripped) = host_and_port.strip_prefix('[') {
2005        match stripped.split_once(']') {
2006            Some((h, tail)) => {
2007                let port = tail.strip_prefix(':').and_then(|p| p.parse::<u16>().ok());
2008                (format!("[{h}]"), port)
2009            }
2010            // An opening bracket with no close is a config typo; falling
2011            // through used to hand "[::1:6379" to TcpStream::connect,
2012            // whose deep parse error misled the operator about the real
2013            // defect (the unbalanced bracket in the endpoint).
2014            None => anyhow::bail!("malformed IPv6 authority (missing ']') in {host_and_port:?}"),
2015        }
2016    } else {
2017        match host_and_port.rsplit_once(':') {
2018            Some((h, p)) if p.parse::<u16>().is_ok() => (h.to_owned(), p.parse::<u16>().ok()),
2019            _ => (host_and_port.to_owned(), None),
2020        }
2021    };
2022    let default_port: u16 = match scheme {
2023        Some("https") | Some("wss") | Some("qdrant+https") => 443,
2024        Some("nats") | Some("tls+nats") | Some("nats+tls") | Some("tls") => 4222,
2025        // `rediss://` is TLS Redis, not HTTPS: redis-rs applies
2026        // `port().unwrap_or(6379)` for BOTH redis:// and rediss://, so
2027        // the daemon connects to :6379 on a portless endpoint. Probing
2028        // :443 (the old mapping) reported FAIL "unreachable" for a
2029        // working production config — or a false PASS if an unrelated
2030        // HTTPS service listened there.
2031        Some("redis") | Some("rediss") => 6379,
2032        _ => 80,
2033    };
2034    let port = port_opt.unwrap_or(default_port);
2035    if host.is_empty() {
2036        anyhow::bail!("no host in endpoint");
2037    }
2038    Ok(format!("{host}:{port}"))
2039}

2041/// "Is this listen address reachable off-host?" — answered by parsing
2042/// the address, not by string prefix: `[::1]:8484` IS loopback (the old
2043/// `starts_with("127.0.0.1")` test warned on it, training operators to
2044/// ignore that warning class and miss the real one on `[::]:8484`).
2045/// The `localhost` prefix heuristic is kept only for the
2046/// unparsable-hostname form.
2047fn listen_is_loopback(listen: &str) -> bool {
2048    listen
2049        .parse::<std::net::SocketAddr>()
2050        .map(|addr| addr.ip().is_loopback())
2051        .unwrap_or_else(|_| listen.starts_with("localhost"))
2052}

2054/// Parse the operator-configured listen address into a
2055/// loopback probe URL. Handles four forms explicitly:
2056///
2057/// * IPv4/IPv6 unspecified addresses map to loopback (matches the
2058///   family: `0.0.0.0` -> `127.0.0.1`, `[::]` / expanded
2059///   `[0:0:0:0:0:0:0:0]` -> `[::1]`).
2060/// * Any other IPv4/IPv6 literal is preserved (an operator binding
2061///   to a specific interface probes that same interface).
2062/// * IPv6 addresses with a zone identifier (`[fe80::1%eth0]`) drop
2063///   the zone id — loopback connects don't need it and the raw `%`
2064///   in an unencoded URL breaks RFC 3986 parsers.
2065/// * Anything unparsable (hostname, `*:port`, mangled config) falls
2066///   back to the original string-replace behaviour so the
2067///   change is strictly additive.
2068fn build_probe_base(listen: &str) -> String {
2069    if let Ok(sa) = listen.parse::<std::net::SocketAddr>() {
2070        let host = if sa.ip().is_unspecified() {
2071            if sa.is_ipv6() {
2072                "[::1]".to_owned()
2073            } else {
2074                "127.0.0.1".to_owned()
2075            }
2076        } else {
2077            match sa.ip() {
2078                std::net::IpAddr::V4(v4) => v4.to_string(),
2079                // Rust's `Ipv6Addr::Display` already emits the
2080                // shortened form and does not include a zone id
2081                // (`sa.ip()` discards any scope id that parsed), so this branch
2082                // gives us a stable bracketed form.
2083                std::net::IpAddr::V6(v6) => format!("[{v6}]"),
2084            }
2085        };
2086        format!("http://{host}:{}", sa.port())
2087    } else {
2088        // Fallback: preserve the string-replace behaviour when parse
2089        // fails (bare hostname, `*:port`, IPv6 with an interface-NAME
2090        // zone id like `%eth0` — numeric ids such as `%1` parse fine
2091        // on stable and take the branch above, where `sa.ip()` drops
2092        // the scope); drop the zone id via a
2093        // second replace so at least the loopback rewrite still
2094        // runs.
2095        let rewritten = listen.replace("0.0.0.0", "127.0.0.1").replace("[::]", "[::1]");
2096        // Strip an IPv6 zone identifier if present: `[fe80::1%eth0]`
2097        // -> `[fe80::1]`. Loopback connects don't need it, and the
2098        // raw `%` breaks URL parsing downstream.
2099        let rewritten = if let (Some(start), Some(pct), Some(end)) =
2100            (rewritten.find('['), rewritten.find('%'), rewritten.find(']'))
2101        {
2102            if start < pct && pct < end {
2103                let mut owned = String::with_capacity(rewritten.len());
2104                owned.push_str(&rewritten[..pct]);
2105                owned.push_str(&rewritten[end..]);
2106                owned
2107            } else {
2108                rewritten
2109            }
2110        } else {
2111            rewritten
2112        };
2113        format!("http://{rewritten}")
2114    }
2115}

2117/// `avctl health` — probe a running harness; exit 0 only on an HTTP 2xx.
2118pub async fn health(base_url: &str) -> Result<()> {
2119    // Accept the base URL (documented), a pasted full /health URL, and
2120    // pasted /livez//readyz probe URLs — container healthchecks probe
2121    // READINESS (a full disk keeps /health and /livez green while every
2122    // request 503s; /readyz's writability probe is the one that flips).
2123    // Passing those paths through verbatim lets compose healthchecks
2124    // say `avctl health --url http://127.0.0.1:8484/readyz`; silently
2125    // appending a second path segment would probe a nonexistent route
2126    // and report a healthy harness as unhealthy.
2127    let base = base_url.trim_end_matches('/');

2718                .unwrap()

2769    fn round_28_probe_target_parses_urls_correctly() {
2770        // F1: scheme-driven defaults, not the old hardcoded :80.
2771        assert_eq!(
2772            probe_target("https://api.example.com").unwrap(),
2773            "api.example.com:443"
2774        );
2775        assert_eq!(

2777            "cache.example.com:6379"
2778        );
2779        assert_eq!(
2780            probe_target("rediss://cache.example.com").unwrap(),
2781            // TLS Redis defaults to 6379 like plain redis:// — redis-rs
2782            // (which the daemon uses) applies port().unwrap_or(6379)
2783            // for both schemes, and doctor must probe what the daemon
2784            // dials.
2785            "cache.example.com:6379"
2786        );
2787        assert_eq!(
2788            probe_target("nats://bus.example.com").unwrap(),
2789            "bus.example.com:4222"
2790        );
2791        // Explicit port always wins over scheme default.
2792        assert_eq!(
2793            probe_target("https://api.example.com:6333").unwrap(),
2794            "api.example.com:6333"
2795        );
2796        assert_eq!(
2797            probe_target("http://api.example.com:8080/path?q=1").unwrap(),
2798            "api.example.com:8080"
2799        );
2800        // Bare host without scheme defaults to :80.
2801        assert_eq!(probe_target("localhost").unwrap(), "localhost:80");
2802        assert_eq!(probe_target("localhost:6379").unwrap(), "localhost:6379");
2803        // F5: userinfo is stripped before the connect target so
2804        // credentials never reach TcpStream::connect or the failure
2805        // path's error text.
2806        let target = probe_target("redis://user:pass@cache.example.com/0").unwrap();
2807        assert_eq!(target, "cache.example.com:6379");
2808        assert!(!target.contains("user"), "userinfo leaked into target: {target}");
2809        assert!(!target.contains("pass"), "userinfo leaked into target: {target}");
2810        // IPv6 literals stay bracketed and preserve default port.
2811        assert_eq!(probe_target("http://[::1]").unwrap(), "[::1]:80");
2812        assert_eq!(
2813            probe_target("https://[2001:db8::1]:8443").unwrap(),
2814            "[2001:db8::1]:8443"
2815        );
2816        // '@' after the authority (in path/query) is not userinfo — the
2817        // authority must be isolated before stripping credentials.
2818        assert_eq!(probe_target("http://h/p@th").unwrap(), "h:80");

2827        assert!(probe_target("redis://[::1").is_err());
