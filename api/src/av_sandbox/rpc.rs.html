1//! JSON-RPC 2.0 / MCP `tools/call` parsing. Total: never panics on arbitrary
2//! bytes (property-tested), returns typed errors for every malformed shape.

4use serde_json::Value;

6/// A parsed MCP tool-call request.
7#[derive(Debug, Clone, PartialEq)]
8pub struct ToolCallRequest {
9    /// JSON-RPC id (absent for notifications).
10    pub id: Option<Value>,
11    /// Tool name (`params.name`).
12    pub tool: String,
13    /// Tool arguments (`params.arguments`, defaults to `{}`).
14    pub arguments: Value,
15}
16
17/// Parse failures.
18#[derive(Debug, thiserror::Error, PartialEq, Eq)]
19#[non_exhaustive]
20pub enum RpcError {
21    /// Not valid JSON at all.
22    #[error("invalid JSON: {0}")]
23    Json(String),
24    /// Valid JSON but not a JSON-RPC 2.0 object.
25    #[error("not a JSON-RPC 2.0 request: {0}")]
26    NotJsonRpc(String),
27    /// A method other than `tools/call` (passthrough, not an error verdict —
28    /// the caller decides what to do with non-tool traffic).
29    #[error("method {0:?} is not tools/call")]
30    NotToolCall(String),
31    /// `tools/call` missing/invalid params.
32    #[error("invalid tools/call params: {0}")]
33    BadParams(String),
34    /// Payload exceeds the configured size bound (DoS guard).
35    #[error("payload of {0} bytes exceeds the {1}-byte bound")]
36    TooLarge(usize, usize),
37}
38
39/// Hard byte bound applied before parsing (attacker-controlled input).
40pub const MAX_PAYLOAD_BYTES: usize = 4 * 1024 * 1024;
41
42/// Depth bound applied post-parse (deeply nested JSON as a DoS vector).
43pub const MAX_JSON_DEPTH: usize = 64;
44
45/// Parse raw bytes as an MCP `tools/call`.
46pub fn parse_tool_call(raw: &[u8]) -> Result<ToolCallRequest, RpcError> {
47    if raw.len() > MAX_PAYLOAD_BYTES {
48        return Err(RpcError::TooLarge(raw.len(), MAX_PAYLOAD_BYTES));
49    }
50    // Duplicate-key rejection: `serde_json` silently keeps the LAST value
51    // on duplicate keys, but the raw body is forwarded unchanged to the
52    // configured tool upstream. If the upstream's JSON decoder is
53    // first-wins (jackson, some Go decoders) or exposes both values, a
54    // request like
55    //   {"params":{"name":"safe_read","name":"db_write","arguments":{}}}
56    // parses here as `tool = "db_write"` (last-wins), passes the
57    // safe_read policy/budget gate, then executes `db_write` upstream —
58    // a permissions-model split with full audit-trail mismatch. Reject
59    // duplicates so gate and upstream see the same request or neither
60    // does. RFC 8259 §4 makes duplicate-name handling "implementation-
61    // defined", so the trust-boundary policy is: refuse ambiguity.
62    reject_duplicate_keys(raw)?;
63    let v: Value = serde_json::from_slice(raw).map_err(|e| RpcError::Json(e.to_string()))?;
64    if depth_of(&v, 0) > MAX_JSON_DEPTH {
65        return Err(RpcError::NotJsonRpc("nesting exceeds depth bound".into()));
66    }
67    let obj = v
68        .as_object()
69        .ok_or_else(|| RpcError::NotJsonRpc("root is not an object".into()))?;
70    match obj.get("jsonrpc").and_then(Value::as_str) {
71        Some("2.0") => {}
72        other => {
73            return Err(RpcError::NotJsonRpc(format!(
74                "jsonrpc field is {other:?}, need \"2.0\""
75            )))
76        }
77    }
78    let method = obj
79        .get("method")
80        .and_then(Value::as_str)
81        .ok_or_else(|| RpcError::NotJsonRpc("method missing or not a string".into()))?;
82    if method != "tools/call" {
83        return Err(RpcError::NotToolCall(method.to_owned()));
84    }
85    let params = obj
86        .get("params")
87        .and_then(Value::as_object)
88        .ok_or_else(|| RpcError::BadParams("params missing or not an object".into()))?;
89    let tool = params
90        .get("name")
91        .and_then(Value::as_str)
92        .filter(|s| !s.is_empty())
93        .ok_or_else(|| RpcError::BadParams("params.name missing or empty".into()))?;
94    if av_core::text::contains_bidi_or_zero_width(tool) {
95        return Err(RpcError::BadParams(
96            "params.name carries a bidi/zero-width spoofing character".into(),
97        ));
98    }
99    // Reject ASCII control characters and inner whitespace: without this a
100    // request for `"db_write\n"` would slip past exact-string matchers for
101    // per-tool schemas, per-tool policy deny-lists, per-tool budgets, and
102    // the harness consequential-tools workflow veto, then be normalized by
103    // most downstream MCP servers to `"db_write"`. Also cover the same
104    // trailing/leading whitespace shape.
105    if tool.chars().any(|c| c.is_control() || c.is_whitespace()) {
106        return Err(RpcError::BadParams(
107            "params.name contains a control character or whitespace".into(),
108        ));
109    }
110    // Refuse any non-ASCII byte in the tool name. Every downstream
111    // comparison — per-tool schema lookup, deny-list matching, per-tool
112    // budget key — is raw byte-equality. A caller supplying
113    // `de\u{0301}lete` (decomposed) or `𝐝𝐛_𝐰𝐫𝐢𝐭𝐞` (mathematical
114    // bold, NFKC-folds to `db_write`) sees a different byte string
115    // than a policy that lists `delete` / `db_write`, but most MCP
116    // servers apply NFC/NFKC before dispatch — so the visually
117    // "different" name resolves to the same tool downstream.
118    // Restricting to ASCII collapses the Unicode-normalization attack
119    // surface without adding a runtime `unicode-normalization`
120    // dependency to the parse gate.
121    if !tool.is_ascii() {
122        return Err(RpcError::BadParams(
123            "params.name must be ASCII: non-ASCII tool names introduce a Unicode-normalization mismatch \
124             between this proxy's exact-byte matching and downstream MCP servers that fold NFC/NFKC"
125                .into(),
126        ));
127    }
128    // Refuse any uppercase byte. Downstream tool-name comparisons are
129    // exact-byte, but most MCP servers apply ASCII case folding. A
130    // policy that denies `db_write` sees `DB_write` slip through the
131    // deny-list byte-eq check while the server executes the same
132    // tool. Requiring the caller to normalize to lowercase up front
133    // eliminates the class.
134    if tool.chars().any(|c| c.is_ascii_uppercase()) {
135        return Err(RpcError::BadParams(
136            "params.name must be lowercase: mixed-case tool names bypass policy deny-lists that use \
137             exact-byte matching while downstream MCP servers apply ASCII case folding"
138                .into(),
139        ));
140    }
141    // JSON-RPC 2.0 §4 requires `id` to be a string, number, or null: reject
142    // objects/arrays/bool up front so downstream correlation tables that key
143    // on id-as-string cannot be confused by structured ids.
144    let id = obj.get("id").cloned();
145    match id.as_ref() {
146        None => {}
147        Some(Value::String(_) | Value::Number(_) | Value::Null) => {}
148        Some(_) => {
149            return Err(RpcError::BadParams(
150                "id must be a string, number, or null per JSON-RPC 2.0 §4".into(),
151            ));
152        }
153    }
154    // Per JSON-RPC 2.0 §4.1 a request with no `id` is a notification: a
155    // notification MUST NOT elicit a response. Refusing them here keeps
156    // attackers from fire-and-forgetting `tools/call` to drain budget
157    // (max_total_tool_calls, payout) without needing to consume responses.
158    if id.is_none() {
159        return Err(RpcError::BadParams(
160            "tools/call requires an id; JSON-RPC notifications are not accepted".into(),
161        ));
162    }
163    let arguments = params
164        .get("arguments")
165        .cloned()
166        .unwrap_or_else(|| Value::Object(Default::default()));
167    if !arguments.is_object() {
168        return Err(RpcError::BadParams("params.arguments must be an object".into()));
169    }
170    Ok(ToolCallRequest {
171        id,
172        tool: tool.to_owned(),
173        arguments,
174    })
175}

177/// Compute the maximum nesting depth of a JSON value. Exposed so
178/// callers outside this crate (chat-completions ingress and any
179/// admin body parser) can apply the same trust-boundary
180/// depth cap without duplicating logic.
181pub fn depth_of(v: &Value, current: usize) -> usize {
182    if current > MAX_JSON_DEPTH {
183        return current; // early-out: no need to recurse further
184    }
185    match v {
186        Value::Array(items) => items
187            .iter()
188            .map(|i| depth_of(i, current + 1))
189            .max()
190            .unwrap_or(current + 1),
191        Value::Object(map) => map
192            .values()
193            .map(|i| depth_of(i, current + 1))
194            .max()
195            .unwrap_or(current + 1),
196        _ => current,
197    }
198}

200/// The general-purpose duplicate-key
201/// rejection primitive, exposed for callers outside this crate (chat
202/// completions ingress, admin body parsers) that want the same
203/// trust-boundary "refuse ambiguous JSON" semantics without pulling in
204/// `RpcError`.
205///
206/// Returns `Ok(())` if `raw` contains no duplicate keys at any nesting
207/// level; otherwise a description of the offending key.
208pub fn refuse_duplicate_json_keys(raw: &[u8]) -> Result<(), String> {
209    reject_duplicate_keys(raw).map_err(|error| error.to_string())
210}

212/// Refuse any JSON object with duplicate keys anywhere in the payload.
213/// `serde_json` silently keeps the last value; downstream MCP servers may
214/// keep the first, expose both, or interpret the collapse differently.
215/// The trust-boundary policy is "refuse ambiguity" — see the call-site
216/// comment in `parse_tool_call`.
217///
218/// Distinguish scanner-created "duplicate key"
219/// errors from underlying `serde_json` parse errors (EOF, unbalanced
220/// braces, invalid escape, recursion limit). The prior blanket
221/// `.map_err(|e| RpcError::Json(format!("duplicate JSON key rejected:
222/// {e}")))` gave EVERY scanner error the same misleading
223/// "duplicate JSON key rejected" prefix, so operator triage on
224/// dup-key alerts fired on any malformed JSON. Mirrors the
225/// sentinel fix in `av_receipts::check_no_duplicate_keys`.
226///
227/// Also refuse trailing garbage AFTER a
228/// valid JSON value. `deserialize_any` returns after the first
229/// complete value, so an input like `{"ok":1}garbage` was silently
230/// accepted. `Deserializer::end()` returns Err if any non-whitespace
231/// bytes remain, closing the class of "smuggled second document"
232/// attacks (JSON body-splitting through a proxy that only inspected
233/// the first value).
234fn reject_duplicate_keys(raw: &[u8]) -> Result<(), RpcError> {
235    use serde::de::{Deserializer, MapAccess, SeqAccess, Visitor};
236    use std::fmt;
237
238    const DUP_KEY_SENTINEL: &str = "__av_sandbox_dup:";
239
240    struct NoDupKeys;
241
242    impl<'de> Visitor<'de> for NoDupKeys {
243        type Value = ();
244
245        fn expecting(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
246            f.write_str("any JSON value without duplicate object keys")
247        }
248
249        fn visit_bool<E>(self, _: bool) -> Result<(), E> {
250            Ok(())
251        }
252        fn visit_i64<E>(self, _: i64) -> Result<(), E> {
253            Ok(())
254        }
255        fn visit_u64<E>(self, _: u64) -> Result<(), E> {
256            Ok(())
257        }
258        fn visit_f64<E>(self, _: f64) -> Result<(), E> {
259            Ok(())
260        }
261        fn visit_str<E>(self, _: &str) -> Result<(), E> {
262            Ok(())
263        }
264        fn visit_string<E>(self, _: String) -> Result<(), E> {
265            Ok(())
266        }
267        fn visit_none<E>(self) -> Result<(), E> {
268            Ok(())
269        }
270        fn visit_unit<E>(self) -> Result<(), E> {
271            Ok(())
272        }
273
274        fn visit_some<D>(self, deserializer: D) -> Result<(), D::Error>
275        where
276            D: Deserializer<'de>,
277        {
278            deserializer.deserialize_any(NoDupKeys)
279        }
280
281        fn visit_seq<A>(self, mut seq: A) -> Result<(), A::Error>
282        where
283            A: SeqAccess<'de>,
284        {
285            while seq.next_element::<NoDupWrap>()?.is_some() {}
286            Ok(())
287        }
288
289        fn visit_map<A>(self, mut map: A) -> Result<(), A::Error>
290        where
291            A: MapAccess<'de>,
292        {
293            let mut seen: std::collections::HashSet<String> = std::collections::HashSet::new();
294            while let Some(key) = map.next_key::<String>()? {
295                if !seen.insert(key.clone()) {
296                    return Err(serde::de::Error::custom(format!(
297                        "{DUP_KEY_SENTINEL}duplicate object key `{}`",
298                        key.escape_debug()
299                    )));
300                }
301                let _: NoDupWrap = map.next_value()?;
302            }
303            Ok(())
304        }
305    }
306
307    // Wrapper so nested structures use the same visitor.
308    struct NoDupWrap;
309    impl<'de> serde::Deserialize<'de> for NoDupWrap {
310        fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
311        where
312            D: Deserializer<'de>,
313        {
314            deserializer.deserialize_any(NoDupKeys)?;
315            Ok(NoDupWrap)
316        }
317    }
318
319    let mut de = serde_json::Deserializer::from_slice(raw);
320    if let Err(error) = de.deserialize_any(NoDupKeys) {
321        let msg = error.to_string();
322        return if let Some(rest) = msg.strip_prefix(DUP_KEY_SENTINEL) {
323            Err(RpcError::Json(format!("duplicate JSON key rejected: {rest}")))
324        } else {
325            // Not a duplicate-key error — the underlying serde_json
326            // error (EOF, unbalanced braces, invalid escape,
327            // recursion limit) surfaces with its own prefix so
328            // triage doesn't misattribute the class.
329            Err(RpcError::Json(msg))
330        };
331    }
332    // Refuse trailing content after the first complete
333    // JSON value. `de.end()` returns Err on any non-whitespace
334    // trailing bytes.
335    de.end()
336        .map_err(|error| RpcError::Json(format!("trailing content after JSON value: {error}")))?;
337    Ok(())
338}

340/// Build the JSON-RPC error response for a blocked call (the "immediate
341/// authorization error back to the agent loop" from the brief).
342pub fn authorization_error(id: Option<&Value>, reason: &str) -> Value {
343    serde_json::json!({
344        "jsonrpc": "2.0",
345        "id": id.cloned().unwrap_or(Value::Null),
346        "error": {
347            "code": -32001,
348            "message": "tool call blocked by AgentVisor AI policy",
349            "data": { "reason": reason }
350        }
351    })
352}

354/// Recover the request `id` for an error response, per JSON-RPC 2.0 §5:
355/// the response `id` MUST echo the request's when it is detectable, and
356/// be `null` only "if there was an error in detecting the id" (parse
357/// error / invalid request). [`RpcError::NotToolCall`] and
358/// [`RpcError::BadParams`] both mean the envelope parsed as a
359/// well-formed JSON-RPC 2.0 object, so the id IS detectable — replying
360/// `null` broke client correlation (a pending `initialize` call hung
361/// until timeout because its error response carried `id: null`).
362/// Parse-class failures (invalid JSON, duplicate keys, oversize,
363/// non-2.0 envelope) keep `null`: the id is absent, ambiguous, or
364/// untrustworthy there.
365#[must_use]
366pub fn detectable_error_id(raw: &[u8], error: &RpcError) -> Option<Value> {
367    if !matches!(error, RpcError::NotToolCall(_) | RpcError::BadParams(_)) {
368        return None;
369    }
370    // Both variants are only produced after the size, duplicate-key and
371    // depth gates passed, so this bounded re-parse cannot do more work
372    // than the original.
373    let value: Value = serde_json::from_slice(raw).ok()?;
374    let id = value.as_object()?.get("id")?;
375    // §4: id must be a string, number, or null — echo nothing else.
376    matches!(id, Value::String(_) | Value::Number(_) | Value::Null).then(|| id.clone())
377}

379/// JSON-RPC 2.0 reserves -32700 (parse
380/// error), -32600 (invalid request), and -32602 (invalid params) for
381/// requests that never reached application logic. Reporting those as
382/// -32001 "blocked by policy" (with HTTP 403) falsely told clients an
383/// authorization decision was made about a request that never parsed.
384pub fn protocol_error(id: Option<&Value>, error: &RpcError) -> Value {
385    let (code, message) = match error {
386        RpcError::Json(_) => (-32700, "parse error"),
387        RpcError::NotJsonRpc(_) | RpcError::TooLarge(..) => (-32600, "invalid request"),
388        RpcError::BadParams(_) => (-32602, "invalid params"),
389        // Passthrough refusal is an application-level policy choice.
390        RpcError::NotToolCall(_) => {
391            return authorization_error(id, &error.to_string());
392        }
393    };
394    serde_json::json!({
395        "jsonrpc": "2.0",
396        "id": id.cloned().unwrap_or(Value::Null),
397        "error": {
398            "code": code,
399            "message": message,
400            "data": { "reason": error.to_string() }
401        }
402    })
403}

464    /// JSON-RPC 2.0 §5: an error response MUST echo the request `id`
465    /// whenever it is detectable, and use `null` only when detection
466    /// failed. Pre-fix every parse-gate refusal replied `id: null`,
467    /// including well-formed envelopes (`initialize` passthrough, bad
468    /// params) whose id was plainly readable — SDK correlation tables
469    /// never matched the response, so the pending call hung to timeout.
470    #[test]
471    fn error_responses_echo_a_detectable_request_id() {
472        // Well-formed envelope, non-tools/call method: id 42 detectable.
473        let raw = br#"{"jsonrpc":"2.0","id":42,"method":"initialize","params":{}}"#;
474        let error = parse_tool_call(raw).unwrap_err();
475        assert!(matches!(error, RpcError::NotToolCall(_)));
476        let id = detectable_error_id(raw, &error);
477        assert_eq!(id, Some(json!(42)));
478        let response = protocol_error(id.as_ref(), &error);
479        assert_eq!(response["id"], json!(42));
480
481        // Well-formed envelope, bad params: string id detectable.
482        let raw = br#"{"jsonrpc":"2.0","id":"req-1","method":"tools/call","params":{}}"#;
483        let error = parse_tool_call(raw).unwrap_err();
484        assert!(matches!(error, RpcError::BadParams(_)));
485        assert_eq!(detectable_error_id(raw, &error), Some(json!("req-1")));
486
487        // Structured (spec-invalid) id: refused as BadParams, and the
488        // echo must NOT propagate the invalid shape — null stands.
489        let raw = br#"{"jsonrpc":"2.0","id":{"nested":1},"method":"tools/call","params":{"name":"t"}}"#;
490        let error = parse_tool_call(raw).unwrap_err();
491        assert!(matches!(error, RpcError::BadParams(_)));
492        assert_eq!(detectable_error_id(raw, &error), None);
493
494        // Parse-class failures: id undetectable per spec — null.
495        let raw = b"not json";
496        let error = parse_tool_call(raw).unwrap_err();
497        assert_eq!(detectable_error_id(raw, &error), None);
498        let response = protocol_error(None, &error);
499        assert_eq!(response["id"], Value::Null);
500    }

775    /// `reject_duplicate_keys`
776    /// used to wrap EVERY scanner error with the "duplicate JSON key
777    /// rejected: ..." prefix — misleading operator triage on
778    /// dup-key alerts because malformed JSON, EOF, and recursion
779    /// limits were all reported as if they were duplicate-key
780    /// rejections. Mirror the sentinel fix from
781    /// `av_receipts`: real duplicate-key errors still get the
782    /// dup-key prefix; parse errors surface their own class.
783    #[test]
784    fn duplicate_key_class_distinguished_from_generic_parse_error() {
785        // Duplicate top-level key → "duplicate JSON key rejected: ..."
786        let real_dup = serde_json::to_vec(&json!({
787            "jsonrpc": "2.0",
788            "id": "dup",
789            "method": "tools/call",
790            "params": {"name": "read", "arguments": {}}
791        }))
792        .unwrap();
793        // Inject a duplicate by hand — serde_json's default parser
794        // would take last-wins.
795        let dup_raw: Vec<u8> = {
796            let mut v = real_dup.clone();
797            // Insert a second `"jsonrpc": "2.0"` right after the
798            // opening brace.
799            let insertion = br#""jsonrpc":"2.0","#;
800            v.splice(1..1, insertion.iter().copied());
801            v
802        };
803        let err = parse_tool_call(&dup_raw).unwrap_err();
804        assert!(
805            matches!(&err, RpcError::Json(msg) if msg.contains("duplicate JSON key rejected")),
806            "real dup-key must carry the dup-key prefix, got {err:?}"
807        );
808
809        // Malformed JSON (unbalanced brace) → NO dup-key prefix.
810        let malformed: &[u8] = b"{\"jsonrpc\":\"2.0\",\"id\":1,";
811        let err = parse_tool_call(malformed).unwrap_err();
812        match &err {
813            RpcError::Json(msg) => assert!(
814                !msg.contains("duplicate JSON key rejected"),
815                "malformed JSON must not be labeled as duplicate-key, got {msg:?}"
816            ),
817            other => panic!("expected RpcError::Json for malformed input, got {other:?}"),
818        }
819
820        // Empty input → NO dup-key prefix.
821        let err = parse_tool_call(b"").unwrap_err();
822        if let RpcError::Json(msg) = &err {
823            assert!(
824                !msg.contains("duplicate JSON key rejected"),
825                "empty input must not be labeled as duplicate-key, got {msg:?}"
826            );
827        }
828    }

871    /// Mutation-run hardening: the depth cap was only proven far
872    /// past the limit, so boundary mutants (`current + 1` -> `current * 1`,
873    /// deleting the Array arm) survived. Pin the exact boundary through the
874    /// public parser for BOTH container kinds. Depth accounting: the
875    /// envelope contributes 3 levels (root -> params -> arguments), and each
876    /// wrapper inside `arguments.k` adds one, so 61 wrappers sit exactly at
877    /// MAX_JSON_DEPTH = 64 and 62 exceed it.
878    #[test]
879    fn json_depth_boundary_is_exact_for_arrays_and_objects() {
880        let build = |inner: String| {
881            format!(
882                r#"{{"jsonrpc":"2.0","id":1,"method":"tools/call","params":{{"name":"t","arguments":{{"k":{inner}}}}}}}"#
883            )
884        };
885        let arrays = |n: usize| format!("{}0{}", "[".repeat(n), "]".repeat(n));
886        let objects = |n: usize| format!("{}0{}", "{\"x\":".repeat(n), "}".repeat(n));
887        for shape in [&arrays as &dyn Fn(usize) -> String, &objects] {
888            let at_limit = build(shape(MAX_JSON_DEPTH - 3));
889            assert!(
890                parse_tool_call(at_limit.as_bytes()).is_ok(),
891                "depth exactly at MAX_JSON_DEPTH must parse"
892            );
893            let past = build(shape(MAX_JSON_DEPTH - 2));
894            let outcome = parse_tool_call(past.as_bytes());
895            assert!(
896                matches!(outcome, Err(RpcError::NotJsonRpc(ref m)) if m.contains("depth")),
897                "one past MAX_JSON_DEPTH must be refused, got {outcome:?}"
898            );
899        }
900        // Empty containers exercise the `unwrap_or(current + 1)` fallback
901        // arms — an empty object/array is still one level deep, so one
902        // sitting past the cap must be refused too.
903        for leaf in ["{}", "[]"] {
904            let wrappers = "{\"x\":".repeat(MAX_JSON_DEPTH - 3);
905            let closers = "}".repeat(MAX_JSON_DEPTH - 3);
906            let past = build(format!("{wrappers}{leaf}{closers}"));
907            let outcome = parse_tool_call(past.as_bytes());
908            assert!(
909                matches!(outcome, Err(RpcError::NotJsonRpc(ref m)) if m.contains("depth")),
910                "empty {leaf} leaf past MAX_JSON_DEPTH must be refused, got {outcome:?}"
911            );
912        }
913    }
