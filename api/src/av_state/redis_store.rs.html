1//! Redis-backed `StateStore` — the distributed budget/state layer for
2//! multi-replica deployments.
3//!
4//! Check-and-spend runs as a server-side Lua script so it is atomic across
5//! distributed clients — same contract as `InMemoryStore::try_spend`.
6//! Contract tests live in `tests/redis_contract.rs`, gated on `AV_REDIS_URL`
7//! (skipped loudly when unset — a skipped gate prints, never silently passes).

12/// TTL (seconds) applied to every counter key this store touches (the
13/// spend/add INCRBY paths and the refund clamp). 24 h: counters are
14/// session-scoped and the TTL is the backstop that keeps abandoned
15/// sessions from leaking Redis memory forever. NOTE the prod/dev
16/// divergence this creates: `InMemoryStore` never expires, so a session
17/// idle longer than this window has its budget counters silently reset
18/// against Redis only. Callers must keep session lifetimes within this
19/// window (or persist budgets elsewhere) — see the `StateStore` trait
20/// docs on counter lifetime.
21const BUDGET_COUNTER_TTL_SECS: u64 = 86_400;
22/// One-shot refund replay markers (see `REFUND_LUA`) only need to
23/// outlive the cluster driver's transparent retry window (seconds);
24/// 5 minutes is generous without accumulating a day of marker keys.
25const REFUND_MARKER_TTL_SECS: u64 = 300;
26
27/// Atomic check-and-spend using subtraction so `current + amount` never rounds.
28static TRY_SPEND_LUA: std::sync::LazyLock<String> = std::sync::LazyLock::new(|| {
29    format!(
30        r"
31for i, key in ipairs(KEYS) do
32    local current = tonumber(redis.call('GET', key) or '0')
33    local amount = tonumber(ARGV[(i - 1) * 2 + 1])
34    local limit = tonumber(ARGV[(i - 1) * 2 + 2])
35    if current > limit or amount > limit - current then
36        -- Refresh TTL on every EXISTING key
37        -- in this invocation before returning the refusal. Without
38        -- this, a session that hit its cap and keeps retrying (every
39        -- attempt refused, commit loop never reached) has its counter
40        -- expire {BUDGET_COUNTER_TTL_SECS}s after the last SUCCESSFUL
41        -- spend — the next attempt then sees current=0 and the cap
42        -- silently RESETS mid-session, doubling the budget. The TTL's
43        -- purpose is reclaiming ABANDONED sessions; an actively
44        -- retrying session is not abandoned.
45        for j, k in ipairs(KEYS) do
46            if redis.call('EXISTS', k) == 1 then
47                redis.call('EXPIRE', k, {BUDGET_COUNTER_TTL_SECS})
48            end
49        end
50        -- Refusal shape: {{-1, refusal_index_1_based}}.
51        -- The leading -1 discriminates from the commit shape
52        -- ({{post_commit_min_remaining, 0}}). See TrySpendOutcome
53        -- in store.rs for rationale.
54        return {{-1, i}}
55    end
56end
57-- R66 F3: compute post-commit min-headroom inside the SAME atomic
58-- EVAL that commits the spends, so `BudgetDecision::Allowed
59-- {{ remaining }}` cannot race with a concurrent
60-- remove_prefix/spend/refund on the same keys. Every key shares
61-- the caller's session hash tag, so the whole script runs on ONE
62-- server-side slot.
63local min_remaining = -1
64for i, key in ipairs(KEYS) do
65    local amount = tonumber(ARGV[(i - 1) * 2 + 1])
66    local limit = tonumber(ARGV[(i - 1) * 2 + 2])
67    local new = redis.call('INCRBY', key, amount)
68    redis.call('EXPIRE', key, {BUDGET_COUNTER_TTL_SECS})
69    local remaining = limit - new
70    if remaining < 0 then
71        remaining = 0
72    end
73    if min_remaining == -1 or remaining < min_remaining then
74        min_remaining = remaining
75    end
76end
77if min_remaining == -1 then
78    -- Empty KEYS: sentinel meaning `u64::MAX` on the client side.
79    min_remaining = -1
80end
81-- Commit shape: {{min_remaining, 0}}. See TRY_SPEND_LUA_REFUSED_SENTINEL
82-- in redis_store.rs.
83return {{min_remaining, 0}}
84"
85    )
86});

88static ADD_LUA: std::sync::LazyLock<String> = std::sync::LazyLock::new(|| {
89    format!(
90        r"
91local current = tonumber(redis.call('GET', KEYS[1]) or '0')
92local amount = tonumber(ARGV[1])
93local limit = tonumber(ARGV[2])
94if current > limit or amount > limit - current then
95    -- Keep an actively-refused counter
96    -- alive — see the twin comment in TRY_SPEND_LUA.
97    if redis.call('EXISTS', KEYS[1]) == 1 then
98        redis.call('EXPIRE', KEYS[1], {BUDGET_COUNTER_TTL_SECS})
99    end
100    return -1
101end
102local result = redis.call('INCRBY', KEYS[1], ARGV[1])
103-- Match TRY_SPEND_LUA's TTL. Without this, any counter touched
104-- only through `add()` (bookkeeping, telemetry, non-budget spending)
105-- persists forever in Redis; over a long-running deployment that
106-- silently leaks memory until Redis OOMs. The two APIs must be
107-- interchangeable from the persistence perspective.
108redis.call('EXPIRE', KEYS[1], {BUDGET_COUNTER_TTL_SECS})
109return result
110"
111    )
112});

114/// Saturating single-key refund. `EXISTS`-gated to never resurrect
115/// a `remove_prefix`-cleared cell; `SET key 0 EX {BUDGET_COUNTER_TTL_SECS}`
116/// on underflow keeps the counter TTL-aligned; `EXPIRE` on the
117/// non-underflow path matches the same discipline. Extracted to a
118/// static (R69) so `every_counter_script_applies_the_shared_ttl`
119/// enrolls it in the TTL-drift regression test (R67 review L3).
120///
121/// KEYS[2] is a one-shot idempotency marker (same hash tag as the
122/// counter — a suffix on KEYS[1] — so cluster mode keeps the script
123/// single-slot). The cluster driver transparently RETRIES commands on
124/// connection loss (redis-rs `DEFAULT_RETRIES`), and a refund replayed
125/// after its first execution committed-but-lost-the-reply would credit
126/// the budget twice — refunding tokens that were only debited once,
127/// i.e. spendable budget minted out of a network flap. `SET NX` on the
128/// marker makes the replay a no-op. Spend-side scripts deliberately
129/// have no marker: their replay double-DEBITS, which fails
130/// conservative (the documented EVAL-uncertainty posture), never
131/// budget-inflating.
132static REFUND_LUA: std::sync::LazyLock<String> = std::sync::LazyLock::new(|| {
133    format!(
134        r"
135if redis.call('SET', KEYS[2], '1', 'NX', 'EX', {REFUND_MARKER_TTL_SECS}) == false then
136    return -1
137end
138if redis.call('EXISTS', KEYS[1]) == 0 then
139    return 0
140end
141local new = redis.call('DECRBY', KEYS[1], ARGV[1])
142if new < 0 then
143    redis.call('SET', KEYS[1], 0, 'EX', {BUDGET_COUNTER_TTL_SECS})
144    return 0
145end
146redis.call('EXPIRE', KEYS[1], {BUDGET_COUNTER_TTL_SECS})
147return new
148"
149    )
150});

152/// Atomic multi-key refund. Per-key semantics match `REFUND_LUA`
153/// exactly (EXISTS gate + DECRBY + clamp-at-zero via SET with
154/// fresh TTL / EXPIRE refresh on non-underflow). Extracted to a
155/// static (R69) so the same TTL-drift test enrolls it.
156///
157/// The trailing key (`KEYS[#KEYS]`) is the replay marker — see
158/// `REFUND_LUA`. The decrement passes `ARGV[i]` to DECRBY as the raw
159/// integer STRING: the previous `tonumber(ARGV[i])` converted through
160/// a Lua double, so an `i64::MAX`-clamped amount rounded to 2^63,
161/// DECRBY rejected it mid-loop, and earlier keys in the same batch
162/// stayed decremented while later ones were skipped — a partial
163/// commit the in-memory backend (which clamps every entry) never
164/// produces. Redis parses the string as an exact i64.
165static REFUND_MANY_LUA: std::sync::LazyLock<String> = std::sync::LazyLock::new(|| {
166    format!(
167        r"
168if redis.call('SET', KEYS[#KEYS], '1', 'NX', 'EX', {REFUND_MARKER_TTL_SECS}) == false then
169    return -1
170end
171for i = 1, #KEYS - 1 do
172    local key = KEYS[i]
173    if redis.call('EXISTS', key) == 1 then
174        local new = redis.call('DECRBY', key, ARGV[i])
175        if new < 0 then
176            redis.call('SET', key, 0, 'EX', {BUDGET_COUNTER_TTL_SECS})
177        else
178            redis.call('EXPIRE', key, {BUDGET_COUNTER_TTL_SECS})
179        end
180    end
181end
182return 0
183"
184    )
185});

187/// Redis-backed store. Connections are pooled internally (r2d2 for both
188/// single-node and cluster; the redis crate implements
189/// `r2d2::ManageConnection` for `ClusterClient` directly).
190///
191/// **EVAL uncertainty on network drop.** `try_spend_many` / `add` invoke
192/// atomic Lua scripts server-side; if the connection breaks between the
193/// server's INCRBY commit and the client's response read, the caller
194/// sees `StateError::Backend` and cannot distinguish "commit succeeded,
195/// response lost" from "commit never ran". Sandbox path treats
196/// backend-error as blocked (fails closed for the request) — but a
197/// subsequent client retry that lands with the same intent then hits a
198/// pre-debited counter and cumulates the two spends into one intended
199/// tool call, overcharging the budget by the retry's amount. The
200/// 24 h counter TTL bounds unbounded growth, and refund-on-loss-race
201/// (`refund_tool_call`) recovers the common cases, but the strictly
202/// idempotent shape would require a client-supplied request nonce
203/// stored briefly in Redis — deferred as a design change.
204pub struct RedisStore {
205    backend: RedisBackend,
206}
207
208enum RedisBackend {
209    Single(r2d2::Pool<RedisConnectionManager>),
210    Cluster(r2d2::Pool<redis::cluster::ClusterClient>),
211}

213struct RedisConnectionManager {
214    client: redis::Client,
215    timeout: std::time::Duration,
216}

218impl r2d2::ManageConnection for RedisConnectionManager {
219    type Connection = redis::Connection;
220    type Error = redis::RedisError;
221
222    fn connect(&self) -> Result<Self::Connection, Self::Error> {
223        let connection = self.client.get_connection_with_timeout(self.timeout)?;
224        connection.set_read_timeout(Some(self.timeout))?;
225        connection.set_write_timeout(Some(self.timeout))?;
226        Ok(connection)
227    }
228
229    fn is_valid(&self, connection: &mut Self::Connection) -> Result<(), Self::Error> {
230        redis::cmd("PING").query(connection)
231    }
232
233    fn has_broken(&self, _connection: &mut Self::Connection) -> bool {
234        false
235    }
236}

238impl RedisStore {
239    /// Connect to `url` (e.g. `redis://127.0.0.1:6379`).
240    /// Comma-separated URLs select Redis Cluster mode.
241    pub fn connect(url: &str) -> Result<Self, StateError> {
242        let nodes: Vec<String> = url
243            .split(',')
244            .map(str::trim)
245            .filter(|node| !node.is_empty())
246            .map(str::to_owned)
247            .collect();
248        if nodes.len() > 1 {
249            // Previously one `ClusterConnection` behind a mutex —
250            // every quota/budget operation across all sessions serialized on
251            // a single socket, which under 10k-connection load stalled
252            // admission long enough to blow upstream timeouts (observed as
253            // 502 "upstream timed out" in the 10k SLA gate). Pool cluster
254            // connections exactly like the single-node path.
255            let client = redis::cluster::ClusterClientBuilder::new(nodes)
256                .connection_timeout(std::time::Duration::from_secs(2))
257                .response_timeout(std::time::Duration::from_secs(2))
258                .build()
259                .map_err(|e| StateError::Backend(e.to_string()))?;
260            let pool = r2d2::Pool::builder()
261                .max_size(32)
262                .connection_timeout(std::time::Duration::from_secs(2))
263                .build(client)
264                .map_err(|e| StateError::Backend(e.to_string()))?;
265            return Ok(Self {
266                backend: RedisBackend::Cluster(pool),
267            });
268        }
269        let client = redis::Client::open(url).map_err(|e| StateError::Backend(e.to_string()))?;
270        let manager = RedisConnectionManager {
271            client,
272            timeout: std::time::Duration::from_secs(2),
273        };
274        let pool = r2d2::Pool::builder()
275            .max_size(32)
276            .connection_timeout(std::time::Duration::from_secs(2))
277            .build(manager)
278            .map_err(|e| StateError::Backend(e.to_string()))?;
279        Ok(Self {
280            backend: RedisBackend::Single(pool),
281        })
282    }
283}

285fn add_on<C: redis::ConnectionLike>(conn: &mut C, key: &str, delta: u64) -> Result<u64, StateError> {
286    if delta > av_core::error::JCS_SAFE_MAX {
287        return Err(StateError::Overflow(key.to_owned()));
288    }
289    let value: i64 = redis::Script::new(&ADD_LUA)
290        .key(key)
291        .arg(delta)
292        .arg(av_core::error::JCS_SAFE_MAX)
293        .invoke(conn)
294        .map_err(|e| StateError::Backend(e.to_string()))?;
295    if value < 0 {
296        return Err(StateError::Overflow(key.to_owned()));
297    }
298    u64::try_from(value).map_err(|_| StateError::Overflow(key.to_owned()))
299}
300
301fn get_on<C: redis::ConnectionLike>(conn: &mut C, key: &str) -> Result<u64, StateError> {
302    let value: Option<i64> = conn.get(key).map_err(|e| StateError::Backend(e.to_string()))?;
303    let value = match value {
304        None => 0,
305        Some(v) if v < 0 => return Err(StateError::Overflow(key.to_owned())),
306        Some(v) => u64::try_from(v).map_err(|_| StateError::Overflow(key.to_owned()))?,
307    };
308    if value > av_core::error::JCS_SAFE_MAX {
309        return Err(StateError::Overflow(key.to_owned()));
310    }
311    Ok(value)
312}

314fn spend_many_on<C: redis::ConnectionLike>(
315    conn: &mut C,
316    spends: &[Spend],
317) -> Result<crate::TrySpendOutcome, StateError> {
318    // Shared duplicate-key guard: the Lua script reads
319    // GET(key) once per iteration in the check phase, so two spends on
320    // the same key each see the pre-commit value — same hazard as the
321    // in-memory backend, one implementation.
322    crate::store::refuse_duplicate_spend_keys(spends)?;
323    for spend in spends {
324        if spend.amount > av_core::error::JCS_SAFE_MAX || spend.limit > av_core::error::JCS_SAFE_MAX {
325            return Err(StateError::Overflow(spend.key.clone()));
326        }
327    }
328    if spends.is_empty() {
329        // Match the sentinel InMemoryStore returns for the empty-
330        // slice case. No EVAL round-trip; nothing to spend.
331        return Ok(crate::TrySpendOutcome::Committed {
332            post_commit_min_remaining: u64::MAX,
333        });
334    }
335    let script = redis::Script::new(&TRY_SPEND_LUA);
336    let mut invocation = script.prepare_invoke();
337    for spend in spends {
338        invocation.key(&spend.key).arg(spend.amount).arg(spend.limit);
339    }
340    // Return shape: [i64, i64]. Two disjoint carriers:
341    //   Commit:  [post_commit_min_remaining, 0]  (min_remaining
342    //            may be -1 to sentinel u64::MAX for empty KEYS;
343    //            spends.is_empty() short-circuits above so we
344    //            should not see -1 here).
345    //   Refusal: [-1, refusal_index_1_based]
346    let outcome: (i64, i64) = invocation
347        .invoke(conn)
348        .map_err(|e| StateError::Backend(e.to_string()))?;
349    match outcome {
350        (-1, refused) => {
351            let refused = usize::try_from(refused - 1)
352                .map_err(|_| StateError::Backend(format!("invalid Lua failure index {refused}")))?;
353            Ok(crate::TrySpendOutcome::Refused { index: refused })
354        }
355        (min_remaining, 0) => {
356            // -1 is the empty-KEYS sentinel (see the Lua); in practice
357            // unreachable here (guarded above) but honour it.
358            let remaining = if min_remaining < 0 {
359                u64::MAX
360            } else {
361                u64::try_from(min_remaining).unwrap_or(0)
362            };
363            Ok(crate::TrySpendOutcome::Committed {
364                post_commit_min_remaining: remaining,
365            })
366        }
367        (a, b) => Err(StateError::Backend(format!(
368            "TRY_SPEND_LUA returned unrecognised shape [{a}, {b}]"
369        ))),
370    }
371}

373impl StateStore for RedisStore {
374    fn counter_ttl_secs(&self) -> Option<u64> {
375        Some(BUDGET_COUNTER_TTL_SECS)
376    }
377
378    fn add(&self, key: &str, delta: u64) -> Result<u64, StateError> {
379        match &self.backend {
380            RedisBackend::Single(pool) => add_on(
381                &mut pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
382                key,
383                delta,
384            ),
385            RedisBackend::Cluster(pool) => add_on(
386                &mut *pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
387                key,
388                delta,
389            ),
390        }
391    }
392
393    fn get(&self, key: &str) -> Result<u64, StateError> {
394        match &self.backend {
395            RedisBackend::Single(pool) => get_on(
396                &mut pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
397                key,
398            ),
399            RedisBackend::Cluster(pool) => get_on(
400                &mut *pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
401                key,
402            ),
403        }
404    }
405
406    // `try_spend` uses the default trait impl (routes through
407    // `try_spend_many` with a 1-element slice). Removed the standalone
408    // wrapper in R68 — the default is exactly the same shape.
409
410    fn try_spend_many(&self, spends: &[Spend]) -> Result<crate::TrySpendOutcome, StateError> {
411        match &self.backend {
412            RedisBackend::Single(pool) => spend_many_on(
413                &mut pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
414                spends,
415            ),
416            RedisBackend::Cluster(pool) => spend_many_on(
417                &mut *pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
418                spends,
419            ),
420        }
421    }
422
423    fn remove(&self, key: &str) {
424        // Mirror the `refund` treatment
425        // — best-effort by contract, but a completely-silent Redis
426        // failure on the cleanup path leaves an operator with no
427        // signal that a session's stale key wasn't removed. The next
428        // recycled-session-id can then inherit the counter. Warn on
429        // both pool-get and DEL failure with structured fields.
430        let outcome: Result<(), redis::RedisError> = match &self.backend {
431            RedisBackend::Single(pool) => match pool.get() {
432                Ok(mut connection) => connection.del(key),
433                Err(error) => Err(redis::RedisError::from((
434                    redis::ErrorKind::IoError,
435                    "connection pool exhausted",
436                    error.to_string(),
437                ))),
438            },
439            RedisBackend::Cluster(pool) => match pool.get() {
440                Ok(mut connection) => connection.del(key),
441                Err(error) => Err(redis::RedisError::from((
442                    redis::ErrorKind::IoError,
443                    "connection pool exhausted",
444                    error.to_string(),
445                ))),
446            },
447        };
448        if let Err(error) = outcome {
449            let error_kind = error.kind();
450            tracing::warn!(
451                target: "av_state::redis",
452                kind = ?error_kind,
453                detail = %error,
454                "Redis remove failed silently; per-key cleanup was not applied — \
455                 a subsequent session recycling this key may inherit its counter \
456                 until the 24 h TTL expires"
457            );
458        }
459    }
460
461    /// Saturating refund via `DECRBY` + a MAX(0) clamp.
462    /// Best-effort — errors are silently swallowed so a Redis blip on
463    /// the compensation path can never turn a lost-race response into
464    /// a 5xx.
465    ///
466    /// NEVER resurrect a key that was already `DEL`'d
467    /// by a concurrent `remove_prefix`. The prior implementation
468    /// called `DECRBY` on the raw key: Redis initialises a missing
469    /// key to 0 first, so `DECRBY` returned `-amount` and the
470    /// `MAX(0)` clamp branch did `SET key 0` (no `EX`), producing
471    /// a permanent TTL-less key. Under the lost-claim-
472    /// plus-idle-close ordering (mcp_call sandbox-gate debit
473    /// races with the reconciler's clear_budget_state), the
474    /// refund path leaked one-to-three TTL-less keys per session
475    /// — attacker-choosable memory growth against the exact class
476    /// the `clear_budget_state` doc in reconciler.rs documents as
477    /// impossible. Fix: gate
478    /// the whole DECRBY on `EXISTS`. If the session was cleared,
479    /// the budget is already gone and there is nothing to
480    /// compensate; the refund is a silent no-op. If the session
481    /// is alive, we DECRBY-clamp AND refresh the counter TTL to
482    /// match `TRY_SPEND_LUA` (a plain DECRBY does not refresh).
483    fn refund(&self, key: &str, amount: u64) {
484        // Redis DECRBY takes i64. Cap at i64::MAX so a caller passing
485        // u64::MAX cannot silently wrap into a negative value.
486        let amount = i64::try_from(amount).unwrap_or(i64::MAX);
487        // Fresh replay marker per logical refund: the same-slot suffix
488        // keeps cluster mode single-slot; a driver-level retry reuses
489        // the SAME marker and no-ops (see REFUND_LUA).
490        let marker = format!("{key}:r:{}", av_core::new_event_uid());
491        // Refund is best-effort by the trait contract, but
492        // its silent-swallow used to be COMPLETELY invisible — no log,
493        // no metric — so an operator seeing budget depletion during a
494        // Redis outage had no signal that compensation had been
495        // failing. Log the failed refund with structured fields so
496        // downstream aggregators (Vector → OTLP → SIEM) can alert on
497        // it. The response path still stays 200 OK so the caller
498        // never learns a compensation failure as a 5xx.
499        let outcome: Result<i64, redis::RedisError> = match &self.backend {
500            RedisBackend::Single(pool) => match pool.get() {
501                Ok(mut connection) => redis::Script::new(&REFUND_LUA)
502                    .key(key)
503                    .key(&marker)
504                    .arg(amount)
505                    .invoke(&mut *connection),
506                Err(error) => Err(redis::RedisError::from((
507                    redis::ErrorKind::IoError,
508                    "connection pool exhausted",
509                    error.to_string(),
510                ))),
511            },
512            RedisBackend::Cluster(pool) => match pool.get() {
513                Ok(mut connection) => redis::Script::new(&REFUND_LUA)
514                    .key(key)
515                    .key(&marker)
516                    .arg(amount)
517                    .invoke(&mut *connection),
518                Err(error) => Err(redis::RedisError::from((
519                    redis::ErrorKind::IoError,
520                    "connection pool exhausted",
521                    error.to_string(),
522                ))),
523            },
524        };
525        if let Err(error) = outcome {
526            let error_kind = error.kind();
527            tracing::warn!(
528                target: "av_state::redis",
529                kind = ?error_kind,
530                detail = %error,
531                "Redis refund failed silently; per-key compensation was not applied — \
532                 budget counter may remain over-charged until the 24 h TTL expires or a \
533                 successful spend/refund lands"
534            );
535        }
536    }

538    /// R66 F2: atomic multi-key refund via a single Lua script.
539    /// All keys share the caller's Redis Cluster hash tag (guaranteed
540    /// by `ActionBudget::session_prefix` which wraps its digest in
541    /// `{...}`), so cross-slot dispatch is impossible and the script
542    /// runs on ONE server-side EVAL. Per-key semantics match
543    /// `refund` exactly: EXISTS gate (never resurrect a
544    /// remove_prefix-cleared cell), DECRBY, clamp-at-zero via SET
545    /// with fresh TTL on underflow, EXPIRE refresh on non-underflow.
546    /// Best-effort like `refund` — Redis errors log-warn but never
547    /// propagate.
548    fn refund_many(&self, refunds: &[crate::Refund]) {
549        let Some(first) = refunds.first() else {
550            return;
551        };
552        // Replay marker key (see REFUND_LUA): derived from the first
553        // refund key so it shares the batch's hash tag (all keys carry
554        // the same `{...}` tag per the invariant above) — single-slot
555        // in cluster mode.
556        let marker = format!("{}:r:{}", first.key, av_core::new_event_uid());
557        let outcome: Result<i64, redis::RedisError> = match &self.backend {
558            RedisBackend::Single(pool) => match pool.get() {
559                Ok(mut connection) => {
560                    let script = redis::Script::new(&REFUND_MANY_LUA);
561                    let mut invocation = script.prepare_invoke();
562                    for r in refunds {
563                        invocation.key(&r.key);
564                        // Cap at i64::MAX so a caller passing u64::MAX
565                        // cannot silently wrap into a negative value.
566                        invocation.arg(i64::try_from(r.amount).unwrap_or(i64::MAX));
567                    }
568                    invocation.key(&marker);
569                    invocation.invoke(&mut *connection)
570                }
571                Err(error) => Err(redis::RedisError::from((
572                    redis::ErrorKind::IoError,
573                    "connection pool exhausted",
574                    error.to_string(),
575                ))),
576            },
577            RedisBackend::Cluster(pool) => match pool.get() {
578                Ok(mut connection) => {
579                    let script = redis::Script::new(&REFUND_MANY_LUA);
580                    let mut invocation = script.prepare_invoke();
581                    for r in refunds {
582                        invocation.key(&r.key);
583                        invocation.arg(i64::try_from(r.amount).unwrap_or(i64::MAX));
584                    }
585                    invocation.key(&marker);
586                    invocation.invoke(&mut *connection)
587                }
588                Err(error) => Err(redis::RedisError::from((
589                    redis::ErrorKind::IoError,
590                    "connection pool exhausted",
591                    error.to_string(),
592                ))),
593            },
594        };
595        if let Err(error) = outcome {
596            let error_kind = error.kind();
597            tracing::warn!(
598                target: "av_state::redis",
599                kind = ?error_kind,
600                detail = %error,
601                refunds = refunds.len(),
602                "Redis refund_many failed silently; per-key compensation was not applied — \
603                 budget counters may remain over-charged until the 24 h TTL expires or a \
604                 successful spend/refund lands"
605            );
606        }
607    }
608
609    /// Whole-session counter cleanup. Previously left as the
610    /// trait's default no-op on the assumption that the 24 h TTL made it
611    /// pure hygiene — but `SessionRegistry::get_or_open` recycles a
612    /// finalized session id into a fresh open session, which then spends
613    /// against the SAME hash-tagged keys. Against `InMemoryStore` (dev,
614    /// CI, and every budget test) the recycled incarnation starts from
615    /// zero; against Redis it silently inherited up to 24 h of the prior
616    /// incarnation's `tokens`/`total_calls`/`tool:*`/`payout` counters —
617    /// the exact cross-backend divergence class the
618    /// `add`/`try_spend_many` alignment closed.
619    ///
620    /// All of a session's keys share one cluster slot by construction
621    /// (`ActionBudget::session_prefix` wraps the digest in a `{hash-tag}`),
622    /// so a SCAN cursor + bounded DEL batches — routed by the prefix
623    /// pattern's hash-tag to the owning master in cluster mode — reach
624    /// every key on both single-node and cluster backends.
625    ///
626    /// SCAN, not KEYS. The first take used
627    /// `redis.call('KEYS', pattern)` inside a Lua script, which is
628    /// O(entire keyspace per node) AND blocks the Redis event loop
629    /// atomically for the whole scan+delete. On a shared Redis with high
630    /// session churn or attacker-chosen ids, every close stalled all
631    /// other Redis clients (including hot-path `try_spend_many` gates)
632    /// for the duration. SCAN is O(matches) amortized and yields between
633    /// batches, so per-session cleanup no longer blocks concurrent
634    /// traffic.
635    ///
636    /// `route_command` with an explicit slot for cluster
637    /// mode. An intermediate revision used
638    /// `Commands::scan_match` on the cluster connection, but bare
639    /// `SCAN` has no key argument — `RoutingInfo::for_routable` returns
640    /// `None`, and `ClusterConnection::request` treats that as an
641    /// immediate `UNROUTABLE_ERROR`. The iterator failed on
642    /// construction, the `Err(_) => return` arm swallowed it, and
643    /// cleanup silently did nothing on every cluster deployment.
644    /// Best-effort like `remove`/`refund`: any Redis blip on the
645    /// cleanup path must not fail the close.
646    fn remove_prefix(&self, prefix: &str) {
647        // `MATCH` glob treats `* ? [ ] \` as metacharacters. The prefix
648        // is `budget:{<32 hex>}:` today (no metacharacters), but escape
649        // defensively so a future prefix shape cannot over-match.
650        let mut pattern = String::with_capacity(prefix.len() + 1);
651        for c in prefix.chars() {
652            if matches!(c, '*' | '?' | '[' | ']' | '\\') {
653                pattern.push('\\');
654            }
655            pattern.push(c);
656        }
657        pattern.push('*');
658        match &self.backend {
659            RedisBackend::Single(pool) => {
660                if let Ok(mut connection) = pool.get() {
661                    scan_and_delete_single(&mut connection, &pattern);
662                }
663            }
664            RedisBackend::Cluster(pool) => {
665                if let Ok(mut connection) = pool.get() {
666                    // Route by the ORIGINAL (unescaped) prefix so the
667                    // hash-tag slot matches the one Redis derives from
668                    // the on-cluster keys. If the future adds a prefix
669                    // shape that contains any glob metachar, computing
670                    // the slot from the ESCAPED pattern would embed
671                    // `\` characters that never appear in the on-cluster
672                    // keys — SCAN would route to a slot no key lives on
673                    // and `remove_prefix` would silently no-op, letting
674                    // budget counters survive up to the 24 h TTL. Today's
675                    // prefix is metachar-free so escaping is a no-op and
676                    // this bug is latent; the fix protects against the
677                    // future shape change.
678                    scan_and_delete_cluster(&mut connection, &pattern, prefix);
679                }
680            }
681        }
682    }
683}

685/// Bounded SCAN+DEL loop against a single-node connection. Each SCAN
686/// call is O(COUNT) at the server; the loop terminates when the cursor
687/// returns to 0. DEL is batched so one round-trip retires up to
688/// `SCAN_COUNT` keys.
689fn scan_and_delete_single(conn: &mut redis::Connection, pattern: &str) {
690    const SCAN_COUNT: usize = 500;
691    let mut cursor: u64 = 0;
692    loop {
693        let scan: Result<(u64, Vec<String>), _> = redis::cmd("SCAN")
694            .arg(cursor)
695            .arg("MATCH")
696            .arg(pattern)
697            .arg("COUNT")
698            .arg(SCAN_COUNT)
699            .query(conn);
700        let (next, batch) = match scan {
701            Ok(pair) => pair,
702            Err(error) => {
703                // SCAN failing means we bail
704                // out mid-cleanup — the remaining keys survive with
705                // their TTLs, and a future session recycling this id
706                // (within 24 h) would inherit the leftover counters
707                // silently. Log with structured fields so an operator
708                // aggregating warns can spot chronic backend outages.
709                let error_kind = error.kind();
710                tracing::warn!(
711                    target: "av_state::redis",
712                    kind = ?error_kind,
713                    detail = %error,
714                    "Redis SCAN failed during remove_prefix; partial cleanup — surviving \
715                     keys will expire at the 24 h TTL or on the next successful \
716                     remove_prefix for the same prefix"
717                );
718                return;
719            }
720        };
721        if !batch.is_empty() {
722            let mut del = redis::cmd("DEL");
723            for key in &batch {
724                del.arg(key);
725            }
726            let outcome: Result<i64, _> = del.query(conn);
727            if let Err(error) = outcome {
728                let error_kind = error.kind();
729                tracing::warn!(
730                    target: "av_state::redis",
731                    kind = ?error_kind,
732                    detail = %error,
733                    batch_len = batch.len(),
734                    "Redis DEL batch failed during remove_prefix; these keys survive \
735                     until the 24 h TTL expires"
736                );
737            }
738        }
739        if next == 0 {
740            return;
741        }
742        cursor = next;
743    }
744}
745
746/// Cluster variant: bare `SCAN` has no key argument so
747/// `redis::cluster_routing::RoutingInfo::for_routable` returns `None`
748/// on it (the sync ClusterConnection then fails with
749/// `UNROUTABLE_ERROR`, silently deleting nothing — the
750/// regression this replaces). All matches share one hash-slot by
751/// construction (`ActionBudget::session_prefix` wraps the digest in
752/// `{hash-tag}`), so compute the slot from the prefix and route both
753/// SCAN and DEL through `route_command` to that specific master.
754///
755/// `pattern` carries the ESCAPED glob (metachars prefixed with `\`)
756/// for the SCAN MATCH argument. `route_prefix` carries the ORIGINAL
757/// unescaped prefix used SOLELY for slot computation — the hash-tag
758/// content Redis extracts from an on-cluster key is the RAW bytes,
759/// so a slot computed from an escaped pattern would miss on any
760/// prefix shape containing a glob metachar. Today's prefix is
761/// metachar-free so escape is a no-op; the separation protects
762/// against a future prefix shape change.
763fn scan_and_delete_cluster(conn: &mut redis::cluster::ClusterConnection, pattern: &str, route_prefix: &str) {
764    use redis::cluster_routing::{Route, RoutingInfo, SingleNodeRoutingInfo, SlotAddr};
765    const SCAN_COUNT: usize = 500;
766    let route_key = route_prefix.trim_end_matches('*');
767    let slot = redis::cluster_routing::get_slot(route_key.as_bytes());
768    let routing = RoutingInfo::SingleNode(SingleNodeRoutingInfo::SpecificNode(Route::new(
769        slot,
770        SlotAddr::Master,
771    )));
772    let mut cursor: u64 = 0;
773    loop {
774        let scan_cmd = {
775            let mut cmd = redis::cmd("SCAN");
776            cmd.arg(cursor)
777                .arg("MATCH")
778                .arg(pattern)
779                .arg("COUNT")
780                .arg(SCAN_COUNT);
781            cmd
782        };
783        let value = match conn.route_command(&scan_cmd, routing.clone()) {
784            Ok(value) => value,
785            Err(error) => {
786                // Parity with
787                // `scan_and_delete_single`. Cluster-mode SCAN failure
788                // was completely silent, so an operator watching
789                // `av_state::redis::warn` for cleanup problems saw
790                // nothing on cluster deployments even under Redis
791                // slowdown.
792                let error_kind = error.kind();
793                tracing::warn!(
794                    target: "av_state::redis",
795                    kind = ?error_kind,
796                    detail = %error,
797                    slot,
798                    "Redis cluster SCAN failed during remove_prefix; partial cleanup — \
799                     surviving keys will expire at the 24 h TTL or on the next successful \
800                     remove_prefix for the same prefix"
801                );
802                return;
803            }
804        };
805        let parsed: Result<(u64, Vec<String>), _> = redis::FromRedisValue::from_redis_value(&value);
806        let (next, batch) = match parsed {
807            Ok(pair) => pair,
808            Err(error) => {
809                let error_kind = error.kind();
810                tracing::warn!(
811                    target: "av_state::redis",
812                    kind = ?error_kind,
813                    detail = %error,
814                    "Redis cluster SCAN returned an unparsable response during remove_prefix; \
815                     aborting further cleanup for this prefix"
816                );
817                return;
818            }
819        };
820        if !batch.is_empty() {
821            let mut del = redis::cmd("DEL");
822            for key in &batch {
823                del.arg(key);
824            }
825            // Multi-key DEL is safe here because every match shares the
826            // slot we already computed above — route it explicitly so
827            // the driver does not need to inspect the keys again.
828            if let Err(error) = conn.route_command(&del, routing.clone()) {
829                let error_kind = error.kind();
830                tracing::warn!(
831                    target: "av_state::redis",
832                    kind = ?error_kind,
833                    detail = %error,
834                    slot,
835                    batch_len = batch.len(),
836                    "Redis cluster DEL batch failed during remove_prefix; these keys survive \
837                     until the 24 h TTL expires"
838                );
839            }
840        }
841        if next == 0 {
842            return;
843        }
844        cursor = next;
845    }
846}

854    /// Every Lua script that mutates a counter must apply the single
855    /// shared TTL constant — a hardcoded-literal drift between the
856    /// spend/add/refund paths would make some counters outlive others
857    /// and silently split the session-expiry policy.
858    ///
859    /// R69: extended to enrol REFUND_LUA and REFUND_MANY_LUA. Both
860    /// were inline `format!` strings pre-R69 (a per-call allocation);
861    /// extraction to statics closed R67 review L3's coverage gap so
862    /// a future edit that drops `EXPIRE` from either would be caught
863    /// here.
864    #[test]
865    fn every_counter_script_applies_the_shared_ttl() {
866        let ttl = BUDGET_COUNTER_TTL_SECS.to_string();
867        for (name, script) in [
868            ("TRY_SPEND_LUA", TRY_SPEND_LUA.as_str()),
869            ("ADD_LUA", ADD_LUA.as_str()),
870            ("REFUND_LUA", REFUND_LUA.as_str()),
871            ("REFUND_MANY_LUA", REFUND_MANY_LUA.as_str()),
872        ] {
873            assert!(
874                script.contains(&format!("EXPIRE', KEYS[1], {ttl}"))
875                    || script.contains(&format!("EXPIRE', key, {ttl}")),
876                "{name} must EXPIRE with BUDGET_COUNTER_TTL_SECS: {script}"
877            );
878            // R111 F1: the R69 review-L3 guard was `!script.contains('{')`
879            // — designed to catch unexpanded `{BUDGET_COUNTER_TTL_SECS}`
880            // format placeholders left over from the format!() template.
881            // But TRY_SPEND_LUA legitimately contains literal `{`
882            // characters: the raw-string template writes `{{-1, i}}`
883            // and `{{min_remaining, 0}}` for Lua-return-table syntax
884            // (double-braces escape in format!()), producing literal
885            // `{...}` in the compiled script. The blanket contains-'{'
886            // scan panicked on every run — the guard has been silently
887            // broken since it landed. Fix: check for the ONLY
888            // placeholder name we actually use (BUDGET_COUNTER_TTL_SECS)
889            // — any surviving occurrence of the identifier means the
890            // format!() expansion didn't happen.
891            assert!(
892                !script.contains("BUDGET_COUNTER_TTL_SECS"),
893                "{name} has an unexpanded BUDGET_COUNTER_TTL_SECS placeholder"
894            );
895        }
896        // Refund replay/precision pins (pass-7 findings):
897        // 1. Both refund scripts open with the SET-NX replay marker —
898        //    the cluster driver transparently retries scripts on
899        //    connection loss, and an unmarked refund replay CREDITS the
900        //    budget twice (spendable budget minted from a network flap).
901        // 2. REFUND_MANY passes ARGV[i] to DECRBY as the raw integer
902        //    string — `tonumber` converted through a Lua double, so an
903        //    i64::MAX amount rounded to 2^63, DECRBY rejected mid-loop,
904        //    and the batch PARTIALLY committed (in-memory clamps every
905        //    entry; the backends must agree).
906        for (name, script) in [
907            ("REFUND_LUA", REFUND_LUA.as_str()),
908            ("REFUND_MANY_LUA", REFUND_MANY_LUA.as_str()),
909        ] {
910            assert!(
911                script.contains("'NX', 'EX'"),
912                "{name} must guard against driver-level replay with a SET-NX marker: {script}"
913            );
914        }
915        assert!(
916            !REFUND_MANY_LUA.contains("tonumber(ARGV"),
917            "REFUND_MANY_LUA must pass ARGV to DECRBY as an exact integer string, \
918             not through a Lua double: {}",
919            REFUND_MANY_LUA.as_str()
920        );
921    }
