1//! Redis-backed `StateStore` — the distributed budget/state layer for
2//! multi-replica deployments.
3//!
4//! Check-and-spend runs as a server-side Lua script so it is atomic across
5//! distributed clients — same contract as `InMemoryStore::try_spend`.
6//! Contract tests live in `tests/redis_contract.rs`, gated on `AV_REDIS_URL`
7//! (skipped loudly when unset — a skipped gate prints, never silently passes).
8
9use crate::store::{Spend, StateError, StateStore};
10use redis::Commands;

12/// TTL (seconds) applied to every counter key this store touches (the
13/// spend/add INCRBY paths and the refund clamp). 24 h: counters are
14/// session-scoped and the TTL is the backstop that keeps abandoned
15/// sessions from leaking Redis memory forever. NOTE the prod/dev
16/// divergence this creates: `InMemoryStore` never expires, so a session
17/// idle longer than this window has its budget counters silently reset
18/// against Redis only. Callers must keep session lifetimes within this
19/// window (or persist budgets elsewhere) — see the `StateStore` trait
20/// docs on counter lifetime.
21const BUDGET_COUNTER_TTL_SECS: u64 = 86_400;
22
23/// Atomic check-and-spend using subtraction so `current + amount` never rounds.
24static TRY_SPEND_LUA: std::sync::LazyLock<String> = std::sync::LazyLock::new(|| {
25    format!(
26        r"
27for i, key in ipairs(KEYS) do
28    local current = tonumber(redis.call('GET', key) or '0')
29    local amount = tonumber(ARGV[(i - 1) * 2 + 1])
30    local limit = tonumber(ARGV[(i - 1) * 2 + 2])
31    if current > limit or amount > limit - current then
32        -- Refresh TTL on every EXISTING key
33        -- in this invocation before returning the refusal. Without
34        -- this, a session that hit its cap and keeps retrying (every
35        -- attempt refused, commit loop never reached) has its counter
36        -- expire {BUDGET_COUNTER_TTL_SECS}s after the last SUCCESSFUL
37        -- spend — the next attempt then sees current=0 and the cap
38        -- silently RESETS mid-session, doubling the budget. The TTL's
39        -- purpose is reclaiming ABANDONED sessions; an actively
40        -- retrying session is not abandoned.
41        for j, k in ipairs(KEYS) do
42            if redis.call('EXISTS', k) == 1 then
43                redis.call('EXPIRE', k, {BUDGET_COUNTER_TTL_SECS})
44            end
45        end
46        -- Refusal shape: {{-1, refusal_index_1_based}}.
47        -- The leading -1 discriminates from the commit shape
48        -- ({{post_commit_min_remaining, 0}}). See TrySpendOutcome
49        -- in store.rs for rationale.
50        return {{-1, i}}
51    end
52end
53-- R66 F3: compute post-commit min-headroom inside the SAME atomic
54-- EVAL that commits the spends, so `BudgetDecision::Allowed
55-- {{ remaining }}` cannot race with a concurrent
56-- remove_prefix/spend/refund on the same keys. Every key shares
57-- the caller's session hash tag, so the whole script runs on ONE
58-- server-side slot.
59local min_remaining = -1
60for i, key in ipairs(KEYS) do
61    local amount = tonumber(ARGV[(i - 1) * 2 + 1])
62    local limit = tonumber(ARGV[(i - 1) * 2 + 2])
63    local new = redis.call('INCRBY', key, amount)
64    redis.call('EXPIRE', key, {BUDGET_COUNTER_TTL_SECS})
65    local remaining = limit - new
66    if remaining < 0 then
67        remaining = 0
68    end
69    if min_remaining == -1 or remaining < min_remaining then
70        min_remaining = remaining
71    end
72end
73if min_remaining == -1 then
74    -- Empty KEYS: sentinel meaning `u64::MAX` on the client side.
75    min_remaining = -1
76end
77-- Commit shape: {{min_remaining, 0}}. See TRY_SPEND_LUA_REFUSED_SENTINEL
78-- in redis_store.rs.
79return {{min_remaining, 0}}
80"
81    )
82});

84static ADD_LUA: std::sync::LazyLock<String> = std::sync::LazyLock::new(|| {
85    format!(
86        r"
87local current = tonumber(redis.call('GET', KEYS[1]) or '0')
88local amount = tonumber(ARGV[1])
89local limit = tonumber(ARGV[2])
90if current > limit or amount > limit - current then
91    -- Keep an actively-refused counter
92    -- alive — see the twin comment in TRY_SPEND_LUA.
93    if redis.call('EXISTS', KEYS[1]) == 1 then
94        redis.call('EXPIRE', KEYS[1], {BUDGET_COUNTER_TTL_SECS})
95    end
96    return -1
97end
98local result = redis.call('INCRBY', KEYS[1], ARGV[1])
99-- Match TRY_SPEND_LUA's TTL. Without this, any counter touched
100-- only through `add()` (bookkeeping, telemetry, non-budget spending)
101-- persists forever in Redis; over a long-running deployment that
102-- silently leaks memory until Redis OOMs. The two APIs must be
103-- interchangeable from the persistence perspective.
104redis.call('EXPIRE', KEYS[1], {BUDGET_COUNTER_TTL_SECS})
105return result
106"
107    )
108});

110/// Saturating single-key refund. `EXISTS`-gated to never resurrect
111/// a `remove_prefix`-cleared cell; `SET key 0 EX {BUDGET_COUNTER_TTL_SECS}`
112/// on underflow keeps the counter TTL-aligned; `EXPIRE` on the
113/// non-underflow path matches the same discipline. Extracted to a
114/// static (R69) so `every_counter_script_applies_the_shared_ttl`
115/// enrolls it in the TTL-drift regression test (R67 review L3).
116static REFUND_LUA: std::sync::LazyLock<String> = std::sync::LazyLock::new(|| {
117    format!(
118        r"
119if redis.call('EXISTS', KEYS[1]) == 0 then
120    return 0
121end
122local new = redis.call('DECRBY', KEYS[1], ARGV[1])
123if new < 0 then
124    redis.call('SET', KEYS[1], 0, 'EX', {BUDGET_COUNTER_TTL_SECS})
125    return 0
126end
127redis.call('EXPIRE', KEYS[1], {BUDGET_COUNTER_TTL_SECS})
128return new
129"
130    )
131});
132
133/// Atomic multi-key refund. Per-key semantics match `REFUND_LUA`
134/// exactly (EXISTS gate + DECRBY + clamp-at-zero via SET with
135/// fresh TTL / EXPIRE refresh on non-underflow). Extracted to a
136/// static (R69) so the same TTL-drift test enrolls it.
137static REFUND_MANY_LUA: std::sync::LazyLock<String> = std::sync::LazyLock::new(|| {
138    format!(
139        r"
140for i, key in ipairs(KEYS) do
141    if redis.call('EXISTS', key) == 1 then
142        local amount = tonumber(ARGV[i])
143        local new = redis.call('DECRBY', key, amount)
144        if new < 0 then
145            redis.call('SET', key, 0, 'EX', {BUDGET_COUNTER_TTL_SECS})
146        else
147            redis.call('EXPIRE', key, {BUDGET_COUNTER_TTL_SECS})
148        end
149    end
150end
151return 0
152"
153    )
154});

156/// Redis-backed store. Connections are pooled internally (r2d2 for both
157/// single-node and cluster; the redis crate implements
158/// `r2d2::ManageConnection` for `ClusterClient` directly).
159///
160/// **EVAL uncertainty on network drop.** `try_spend_many` / `add` invoke
161/// atomic Lua scripts server-side; if the connection breaks between the
162/// server's INCRBY commit and the client's response read, the caller
163/// sees `StateError::Backend` and cannot distinguish "commit succeeded,
164/// response lost" from "commit never ran". Sandbox path treats
165/// backend-error as blocked (fails closed for the request) — but a
166/// subsequent client retry that lands with the same intent then hits a
167/// pre-debited counter and cumulates the two spends into one intended
168/// tool call, overcharging the budget by the retry's amount. The
169/// 24 h counter TTL bounds unbounded growth, and refund-on-loss-race
170/// (`refund_tool_call`) recovers the common cases, but the strictly
171/// idempotent shape would require a client-supplied request nonce
172/// stored briefly in Redis — deferred as a design change.
173pub struct RedisStore {
174    backend: RedisBackend,
175}
176
177enum RedisBackend {
178    Single(r2d2::Pool<RedisConnectionManager>),
179    Cluster(r2d2::Pool<redis::cluster::ClusterClient>),
180}
181
182struct RedisConnectionManager {
183    client: redis::Client,
184    timeout: std::time::Duration,
185}
186
187impl r2d2::ManageConnection for RedisConnectionManager {
188    type Connection = redis::Connection;
189    type Error = redis::RedisError;
190
191    fn connect(&self) -> Result<Self::Connection, Self::Error> {
192        let connection = self.client.get_connection_with_timeout(self.timeout)?;
193        connection.set_read_timeout(Some(self.timeout))?;
194        connection.set_write_timeout(Some(self.timeout))?;
195        Ok(connection)
196    }
197
198    fn is_valid(&self, connection: &mut Self::Connection) -> Result<(), Self::Error> {
199        redis::cmd("PING").query(connection)
200    }
201
202    fn has_broken(&self, _connection: &mut Self::Connection) -> bool {
203        false
204    }
205}

207impl RedisStore {
208    /// Connect to `url` (e.g. `redis://127.0.0.1:6379`).
209    /// Comma-separated URLs select Redis Cluster mode.
210    pub fn connect(url: &str) -> Result<Self, StateError> {
211        let nodes: Vec<String> = url
212            .split(',')
213            .map(str::trim)
214            .filter(|node| !node.is_empty())
215            .map(str::to_owned)
216            .collect();
217        if nodes.len() > 1 {
218            // Previously one `ClusterConnection` behind a mutex —
219            // every quota/budget operation across all sessions serialized on
220            // a single socket, which under 10k-connection load stalled
221            // admission long enough to blow upstream timeouts (observed as
222            // 502 "upstream timed out" in the 10k SLA gate). Pool cluster
223            // connections exactly like the single-node path.
224            let client = redis::cluster::ClusterClientBuilder::new(nodes)
225                .connection_timeout(std::time::Duration::from_secs(2))
226                .response_timeout(std::time::Duration::from_secs(2))
227                .build()
228                .map_err(|e| StateError::Backend(e.to_string()))?;
229            let pool = r2d2::Pool::builder()
230                .max_size(32)
231                .connection_timeout(std::time::Duration::from_secs(2))
232                .build(client)
233                .map_err(|e| StateError::Backend(e.to_string()))?;
234            return Ok(Self {
235                backend: RedisBackend::Cluster(pool),
236            });
237        }
238        let client = redis::Client::open(url).map_err(|e| StateError::Backend(e.to_string()))?;
239        let manager = RedisConnectionManager {
240            client,
241            timeout: std::time::Duration::from_secs(2),
242        };
243        let pool = r2d2::Pool::builder()
244            .max_size(32)
245            .connection_timeout(std::time::Duration::from_secs(2))
246            .build(manager)
247            .map_err(|e| StateError::Backend(e.to_string()))?;
248        Ok(Self {
249            backend: RedisBackend::Single(pool),
250        })
251    }
252}

254fn add_on<C: redis::ConnectionLike>(conn: &mut C, key: &str, delta: u64) -> Result<u64, StateError> {
255    if delta > av_core::error::JCS_SAFE_MAX {
256        return Err(StateError::Overflow(key.to_owned()));
257    }
258    let value: i64 = redis::Script::new(&ADD_LUA)
259        .key(key)
260        .arg(delta)
261        .arg(av_core::error::JCS_SAFE_MAX)
262        .invoke(conn)
263        .map_err(|e| StateError::Backend(e.to_string()))?;
264    if value < 0 {
265        return Err(StateError::Overflow(key.to_owned()));
266    }
267    u64::try_from(value).map_err(|_| StateError::Overflow(key.to_owned()))
268}
269
270fn get_on<C: redis::ConnectionLike>(conn: &mut C, key: &str) -> Result<u64, StateError> {
271    let value: Option<i64> = conn.get(key).map_err(|e| StateError::Backend(e.to_string()))?;
272    let value = match value {
273        None => 0,
274        Some(v) if v < 0 => return Err(StateError::Overflow(key.to_owned())),
275        Some(v) => u64::try_from(v).map_err(|_| StateError::Overflow(key.to_owned()))?,
276    };
277    if value > av_core::error::JCS_SAFE_MAX {
278        return Err(StateError::Overflow(key.to_owned()));
279    }
280    Ok(value)
281}

283fn spend_many_on<C: redis::ConnectionLike>(
284    conn: &mut C,
285    spends: &[Spend],
286) -> Result<crate::TrySpendOutcome, StateError> {
287    // Shared duplicate-key guard: the Lua script reads
288    // GET(key) once per iteration in the check phase, so two spends on
289    // the same key each see the pre-commit value — same hazard as the
290    // in-memory backend, one implementation.
291    crate::store::refuse_duplicate_spend_keys(spends)?;
292    for spend in spends {
293        if spend.amount > av_core::error::JCS_SAFE_MAX || spend.limit > av_core::error::JCS_SAFE_MAX {
294            return Err(StateError::Overflow(spend.key.clone()));
295        }
296    }
297    if spends.is_empty() {
298        // Match the sentinel InMemoryStore returns for the empty-
299        // slice case. No EVAL round-trip; nothing to spend.
300        return Ok(crate::TrySpendOutcome::Committed {
301            post_commit_min_remaining: u64::MAX,
302        });
303    }
304    let script = redis::Script::new(&TRY_SPEND_LUA);
305    let mut invocation = script.prepare_invoke();
306    for spend in spends {
307        invocation.key(&spend.key).arg(spend.amount).arg(spend.limit);
308    }
309    // Return shape: [i64, i64]. Two disjoint carriers:
310    //   Commit:  [post_commit_min_remaining, 0]  (min_remaining
311    //            may be -1 to sentinel u64::MAX for empty KEYS;
312    //            spends.is_empty() short-circuits above so we
313    //            should not see -1 here).
314    //   Refusal: [-1, refusal_index_1_based]
315    let outcome: (i64, i64) = invocation
316        .invoke(conn)
317        .map_err(|e| StateError::Backend(e.to_string()))?;
318    match outcome {
319        (-1, refused) => {
320            let refused = usize::try_from(refused - 1)
321                .map_err(|_| StateError::Backend(format!("invalid Lua failure index {refused}")))?;
322            Ok(crate::TrySpendOutcome::Refused { index: refused })
323        }
324        (min_remaining, 0) => {
325            // -1 is the empty-KEYS sentinel (see the Lua); in practice
326            // unreachable here (guarded above) but honour it.
327            let remaining = if min_remaining < 0 {
328                u64::MAX
329            } else {
330                u64::try_from(min_remaining).unwrap_or(0)
331            };
332            Ok(crate::TrySpendOutcome::Committed {
333                post_commit_min_remaining: remaining,
334            })
335        }
336        (a, b) => Err(StateError::Backend(format!(
337            "TRY_SPEND_LUA returned unrecognised shape [{a}, {b}]"
338        ))),
339    }
340}

342impl StateStore for RedisStore {
343    fn counter_ttl_secs(&self) -> Option<u64> {
344        Some(BUDGET_COUNTER_TTL_SECS)
345    }
346
347    fn add(&self, key: &str, delta: u64) -> Result<u64, StateError> {
348        match &self.backend {
349            RedisBackend::Single(pool) => add_on(
350                &mut pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
351                key,
352                delta,
353            ),
354            RedisBackend::Cluster(pool) => add_on(
355                &mut *pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
356                key,
357                delta,
358            ),
359        }
360    }
361
362    fn get(&self, key: &str) -> Result<u64, StateError> {
363        match &self.backend {
364            RedisBackend::Single(pool) => get_on(
365                &mut pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
366                key,
367            ),
368            RedisBackend::Cluster(pool) => get_on(
369                &mut *pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
370                key,
371            ),
372        }
373    }
374
375    // `try_spend` uses the default trait impl (routes through
376    // `try_spend_many` with a 1-element slice). Removed the standalone
377    // wrapper in R68 — the default is exactly the same shape.
378
379    fn try_spend_many(&self, spends: &[Spend]) -> Result<crate::TrySpendOutcome, StateError> {
380        match &self.backend {
381            RedisBackend::Single(pool) => spend_many_on(
382                &mut pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
383                spends,
384            ),
385            RedisBackend::Cluster(pool) => spend_many_on(
386                &mut *pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
387                spends,
388            ),
389        }
390    }
391
392    fn remove(&self, key: &str) {
393        // Mirror the `refund` treatment
394        // — best-effort by contract, but a completely-silent Redis
395        // failure on the cleanup path leaves an operator with no
396        // signal that a session's stale key wasn't removed. The next
397        // recycled-session-id can then inherit the counter. Warn on
398        // both pool-get and DEL failure with structured fields.
399        let outcome: Result<(), redis::RedisError> = match &self.backend {
400            RedisBackend::Single(pool) => match pool.get() {
401                Ok(mut connection) => connection.del(key),
402                Err(error) => Err(redis::RedisError::from((
403                    redis::ErrorKind::IoError,
404                    "connection pool exhausted",
405                    error.to_string(),
406                ))),
407            },
408            RedisBackend::Cluster(pool) => match pool.get() {
409                Ok(mut connection) => connection.del(key),
410                Err(error) => Err(redis::RedisError::from((
411                    redis::ErrorKind::IoError,
412                    "connection pool exhausted",
413                    error.to_string(),
414                ))),
415            },
416        };
417        if let Err(error) = outcome {
418            let error_kind = error.kind();
419            tracing::warn!(
420                target: "av_state::redis",
421                kind = ?error_kind,
422                detail = %error,
423                "Redis remove failed silently; per-key cleanup was not applied — \
424                 a subsequent session recycling this key may inherit its counter \
425                 until the 24 h TTL expires"
426            );
427        }
428    }
429
430    /// Saturating refund via `DECRBY` + a MAX(0) clamp.
431    /// Best-effort — errors are silently swallowed so a Redis blip on
432    /// the compensation path can never turn a lost-race response into
433    /// a 5xx.
434    ///
435    /// NEVER resurrect a key that was already `DEL`'d
436    /// by a concurrent `remove_prefix`. The prior implementation
437    /// called `DECRBY` on the raw key: Redis initialises a missing
438    /// key to 0 first, so `DECRBY` returned `-amount` and the
439    /// `MAX(0)` clamp branch did `SET key 0` (no `EX`), producing
440    /// a permanent TTL-less key. Under the lost-claim-
441    /// plus-idle-close ordering (mcp_call sandbox-gate debit
442    /// races with the reconciler's clear_budget_state), the
443    /// refund path leaked one-to-three TTL-less keys per session
444    /// — attacker-choosable memory growth against the exact class
445    /// the `clear_budget_state` doc in reconciler.rs documents as
446    /// impossible. Fix: gate
447    /// the whole DECRBY on `EXISTS`. If the session was cleared,
448    /// the budget is already gone and there is nothing to
449    /// compensate; the refund is a silent no-op. If the session
450    /// is alive, we DECRBY-clamp AND refresh the counter TTL to
451    /// match `TRY_SPEND_LUA` (a plain DECRBY does not refresh).
452    fn refund(&self, key: &str, amount: u64) {
453        // Redis DECRBY takes i64. Cap at i64::MAX so a caller passing
454        // u64::MAX cannot silently wrap into a negative value.
455        let amount = i64::try_from(amount).unwrap_or(i64::MAX);
456        // Refund is best-effort by the trait contract, but
457        // its silent-swallow used to be COMPLETELY invisible — no log,
458        // no metric — so an operator seeing budget depletion during a
459        // Redis outage had no signal that compensation had been
460        // failing. Log the failed refund with structured fields so
461        // downstream aggregators (Vector → OTLP → SIEM) can alert on
462        // it. The response path still stays 200 OK so the caller
463        // never learns a compensation failure as a 5xx.
464        let outcome: Result<i64, redis::RedisError> = match &self.backend {
465            RedisBackend::Single(pool) => match pool.get() {
466                Ok(mut connection) => redis::Script::new(&REFUND_LUA)
467                    .key(key)
468                    .arg(amount)
469                    .invoke(&mut *connection),
470                Err(error) => Err(redis::RedisError::from((
471                    redis::ErrorKind::IoError,
472                    "connection pool exhausted",
473                    error.to_string(),
474                ))),
475            },
476            RedisBackend::Cluster(pool) => match pool.get() {
477                Ok(mut connection) => redis::Script::new(&REFUND_LUA)
478                    .key(key)
479                    .arg(amount)
480                    .invoke(&mut *connection),
481                Err(error) => Err(redis::RedisError::from((
482                    redis::ErrorKind::IoError,
483                    "connection pool exhausted",
484                    error.to_string(),
485                ))),
486            },
487        };
488        if let Err(error) = outcome {
489            let error_kind = error.kind();
490            tracing::warn!(
491                target: "av_state::redis",
492                kind = ?error_kind,
493                detail = %error,
494                "Redis refund failed silently; per-key compensation was not applied — \
495                 budget counter may remain over-charged until the 24 h TTL expires or a \
496                 successful spend/refund lands"
497            );
498        }
499    }
500
501    /// R66 F2: atomic multi-key refund via a single Lua script.
502    /// All keys share the caller's Redis Cluster hash tag (guaranteed
503    /// by `ActionBudget::session_prefix` which wraps its digest in
504    /// `{...}`), so cross-slot dispatch is impossible and the script
505    /// runs on ONE server-side EVAL. Per-key semantics match
506    /// `refund` exactly: EXISTS gate (never resurrect a
507    /// remove_prefix-cleared cell), DECRBY, clamp-at-zero via SET
508    /// with fresh TTL on underflow, EXPIRE refresh on non-underflow.
509    /// Best-effort like `refund` — Redis errors log-warn but never
510    /// propagate.
511    fn refund_many(&self, refunds: &[crate::Refund]) {
512        if refunds.is_empty() {
513            return;
514        }
515        let outcome: Result<i64, redis::RedisError> = match &self.backend {
516            RedisBackend::Single(pool) => match pool.get() {
517                Ok(mut connection) => {
518                    let script = redis::Script::new(&REFUND_MANY_LUA);
519                    let mut invocation = script.prepare_invoke();
520                    for r in refunds {
521                        invocation.key(&r.key);
522                        // Cap at i64::MAX so a caller passing u64::MAX
523                        // cannot silently wrap into a negative value.
524                        invocation.arg(i64::try_from(r.amount).unwrap_or(i64::MAX));
525                    }
526                    invocation.invoke(&mut *connection)
527                }
528                Err(error) => Err(redis::RedisError::from((
529                    redis::ErrorKind::IoError,
530                    "connection pool exhausted",
531                    error.to_string(),
532                ))),
533            },
534            RedisBackend::Cluster(pool) => match pool.get() {
535                Ok(mut connection) => {
536                    let script = redis::Script::new(&REFUND_MANY_LUA);
537                    let mut invocation = script.prepare_invoke();
538                    for r in refunds {
539                        invocation.key(&r.key);
540                        invocation.arg(i64::try_from(r.amount).unwrap_or(i64::MAX));
541                    }
542                    invocation.invoke(&mut *connection)
543                }
544                Err(error) => Err(redis::RedisError::from((
545                    redis::ErrorKind::IoError,
546                    "connection pool exhausted",
547                    error.to_string(),
548                ))),
549            },
550        };
551        if let Err(error) = outcome {
552            let error_kind = error.kind();
553            tracing::warn!(
554                target: "av_state::redis",
555                kind = ?error_kind,
556                detail = %error,
557                refunds = refunds.len(),
558                "Redis refund_many failed silently; per-key compensation was not applied — \
559                 budget counters may remain over-charged until the 24 h TTL expires or a \
560                 successful spend/refund lands"
561            );
562        }
563    }
564
565    /// Whole-session counter cleanup. Previously left as the
566    /// trait's default no-op on the assumption that the 24 h TTL made it
567    /// pure hygiene — but `SessionRegistry::get_or_open` recycles a
568    /// finalized session id into a fresh open session, which then spends
569    /// against the SAME hash-tagged keys. Against `InMemoryStore` (dev,
570    /// CI, and every budget test) the recycled incarnation starts from
571    /// zero; against Redis it silently inherited up to 24 h of the prior
572    /// incarnation's `tokens`/`total_calls`/`tool:*`/`payout` counters —
573    /// the exact cross-backend divergence class the
574    /// `add`/`try_spend_many` alignment closed.
575    ///
576    /// All of a session's keys share one cluster slot by construction
577    /// (`ActionBudget::session_prefix` wraps the digest in a `{hash-tag}`),
578    /// so a SCAN cursor + bounded DEL batches — routed by the prefix
579    /// pattern's hash-tag to the owning master in cluster mode — reach
580    /// every key on both single-node and cluster backends.
581    ///
582    /// SCAN, not KEYS. The first take used
583    /// `redis.call('KEYS', pattern)` inside a Lua script, which is
584    /// O(entire keyspace per node) AND blocks the Redis event loop
585    /// atomically for the whole scan+delete. On a shared Redis with high
586    /// session churn or attacker-chosen ids, every close stalled all
587    /// other Redis clients (including hot-path `try_spend_many` gates)
588    /// for the duration. SCAN is O(matches) amortized and yields between
589    /// batches, so per-session cleanup no longer blocks concurrent
590    /// traffic.
591    ///
592    /// `route_command` with an explicit slot for cluster
593    /// mode. An intermediate revision used
594    /// `Commands::scan_match` on the cluster connection, but bare
595    /// `SCAN` has no key argument — `RoutingInfo::for_routable` returns
596    /// `None`, and `ClusterConnection::request` treats that as an
597    /// immediate `UNROUTABLE_ERROR`. The iterator failed on
598    /// construction, the `Err(_) => return` arm swallowed it, and
599    /// cleanup silently did nothing on every cluster deployment.
600    /// Best-effort like `remove`/`refund`: any Redis blip on the
601    /// cleanup path must not fail the close.
602    fn remove_prefix(&self, prefix: &str) {
603        // `MATCH` glob treats `* ? [ ] \` as metacharacters. The prefix
604        // is `budget:{<32 hex>}:` today (no metacharacters), but escape
605        // defensively so a future prefix shape cannot over-match.
606        let mut pattern = String::with_capacity(prefix.len() + 1);
607        for c in prefix.chars() {
608            if matches!(c, '*' | '?' | '[' | ']' | '\\') {
609                pattern.push('\\');
610            }
611            pattern.push(c);
612        }
613        pattern.push('*');
614        match &self.backend {
615            RedisBackend::Single(pool) => {
616                if let Ok(mut connection) = pool.get() {
617                    scan_and_delete_single(&mut connection, &pattern);
618                }
619            }
620            RedisBackend::Cluster(pool) => {
621                if let Ok(mut connection) = pool.get() {
622                    // Route by the ORIGINAL (unescaped) prefix so the
623                    // hash-tag slot matches the one Redis derives from
624                    // the on-cluster keys. If the future adds a prefix
625                    // shape that contains any glob metachar, computing
626                    // the slot from the ESCAPED pattern would embed
627                    // `\` characters that never appear in the on-cluster
628                    // keys — SCAN would route to a slot no key lives on
629                    // and `remove_prefix` would silently no-op, letting
630                    // budget counters survive up to the 24 h TTL. Today's
631                    // prefix is metachar-free so escaping is a no-op and
632                    // this bug is latent; the fix protects against the
633                    // future shape change.
634                    scan_and_delete_cluster(&mut connection, &pattern, prefix);
635                }
636            }
637        }
638    }
639}

641/// Bounded SCAN+DEL loop against a single-node connection. Each SCAN
642/// call is O(COUNT) at the server; the loop terminates when the cursor
643/// returns to 0. DEL is batched so one round-trip retires up to
644/// `SCAN_COUNT` keys.
645fn scan_and_delete_single(conn: &mut redis::Connection, pattern: &str) {
646    const SCAN_COUNT: usize = 500;
647    let mut cursor: u64 = 0;
648    loop {
649        let scan: Result<(u64, Vec<String>), _> = redis::cmd("SCAN")
650            .arg(cursor)
651            .arg("MATCH")
652            .arg(pattern)
653            .arg("COUNT")
654            .arg(SCAN_COUNT)
655            .query(conn);
656        let (next, batch) = match scan {
657            Ok(pair) => pair,
658            Err(error) => {
659                // SCAN failing means we bail
660                // out mid-cleanup — the remaining keys survive with
661                // their TTLs, and a future session recycling this id
662                // (within 24 h) would inherit the leftover counters
663                // silently. Log with structured fields so an operator
664                // aggregating warns can spot chronic backend outages.
665                let error_kind = error.kind();
666                tracing::warn!(
667                    target: "av_state::redis",
668                    kind = ?error_kind,
669                    detail = %error,
670                    "Redis SCAN failed during remove_prefix; partial cleanup — surviving \
671                     keys will expire at the 24 h TTL or on the next successful \
672                     remove_prefix for the same prefix"
673                );
674                return;
675            }
676        };
677        if !batch.is_empty() {
678            let mut del = redis::cmd("DEL");
679            for key in &batch {
680                del.arg(key);
681            }
682            let outcome: Result<i64, _> = del.query(conn);
683            if let Err(error) = outcome {
684                let error_kind = error.kind();
685                tracing::warn!(
686                    target: "av_state::redis",
687                    kind = ?error_kind,
688                    detail = %error,
689                    batch_len = batch.len(),
690                    "Redis DEL batch failed during remove_prefix; these keys survive \
691                     until the 24 h TTL expires"
692                );
693            }
694        }
695        if next == 0 {
696            return;
697        }
698        cursor = next;
699    }
700}

702/// Cluster variant: bare `SCAN` has no key argument so
703/// `redis::cluster_routing::RoutingInfo::for_routable` returns `None`
704/// on it (the sync ClusterConnection then fails with
705/// `UNROUTABLE_ERROR`, silently deleting nothing — the
706/// regression this replaces). All matches share one hash-slot by
707/// construction (`ActionBudget::session_prefix` wraps the digest in
708/// `{hash-tag}`), so compute the slot from the prefix and route both
709/// SCAN and DEL through `route_command` to that specific master.
710///
711/// `pattern` carries the ESCAPED glob (metachars prefixed with `\`)
712/// for the SCAN MATCH argument. `route_prefix` carries the ORIGINAL
713/// unescaped prefix used SOLELY for slot computation — the hash-tag
714/// content Redis extracts from an on-cluster key is the RAW bytes,
715/// so a slot computed from an escaped pattern would miss on any
716/// prefix shape containing a glob metachar. Today's prefix is
717/// metachar-free so escape is a no-op; the separation protects
718/// against a future prefix shape change.
719fn scan_and_delete_cluster(conn: &mut redis::cluster::ClusterConnection, pattern: &str, route_prefix: &str) {
720    use redis::cluster_routing::{Route, RoutingInfo, SingleNodeRoutingInfo, SlotAddr};
721    const SCAN_COUNT: usize = 500;
722    let route_key = route_prefix.trim_end_matches('*');
723    let slot = redis::cluster_routing::get_slot(route_key.as_bytes());
724    let routing = RoutingInfo::SingleNode(SingleNodeRoutingInfo::SpecificNode(Route::new(
725        slot,
726        SlotAddr::Master,
727    )));
728    let mut cursor: u64 = 0;
729    loop {
730        let scan_cmd = {
731            let mut cmd = redis::cmd("SCAN");
732            cmd.arg(cursor)
733                .arg("MATCH")
734                .arg(pattern)
735                .arg("COUNT")
736                .arg(SCAN_COUNT);
737            cmd
738        };
739        let value = match conn.route_command(&scan_cmd, routing.clone()) {
740            Ok(value) => value,
741            Err(error) => {
742                // Parity with
743                // `scan_and_delete_single`. Cluster-mode SCAN failure
744                // was completely silent, so an operator watching
745                // `av_state::redis::warn` for cleanup problems saw
746                // nothing on cluster deployments even under Redis
747                // slowdown.
748                let error_kind = error.kind();
749                tracing::warn!(
750                    target: "av_state::redis",
751                    kind = ?error_kind,
752                    detail = %error,
753                    slot,
754                    "Redis cluster SCAN failed during remove_prefix; partial cleanup — \
755                     surviving keys will expire at the 24 h TTL or on the next successful \
756                     remove_prefix for the same prefix"
757                );
758                return;
759            }
760        };
761        let parsed: Result<(u64, Vec<String>), _> = redis::FromRedisValue::from_redis_value(&value);
762        let (next, batch) = match parsed {
763            Ok(pair) => pair,
764            Err(error) => {
765                let error_kind = error.kind();
766                tracing::warn!(
767                    target: "av_state::redis",
768                    kind = ?error_kind,
769                    detail = %error,
770                    "Redis cluster SCAN returned an unparsable response during remove_prefix; \
771                     aborting further cleanup for this prefix"
772                );
773                return;
774            }
775        };
776        if !batch.is_empty() {
777            let mut del = redis::cmd("DEL");
778            for key in &batch {
779                del.arg(key);
780            }
781            // Multi-key DEL is safe here because every match shares the
782            // slot we already computed above — route it explicitly so
783            // the driver does not need to inspect the keys again.
784            if let Err(error) = conn.route_command(&del, routing.clone()) {
785                let error_kind = error.kind();
786                tracing::warn!(
787                    target: "av_state::redis",
788                    kind = ?error_kind,
789                    detail = %error,
790                    slot,
791                    batch_len = batch.len(),
792                    "Redis cluster DEL batch failed during remove_prefix; these keys survive \
793                     until the 24 h TTL expires"
794                );
795            }
796        }
797        if next == 0 {
798            return;
799        }
800        cursor = next;
801    }
802}

831                    || script.contains(&format!("EXPIRE', key, {ttl}")),
